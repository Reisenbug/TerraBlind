using System;
using TerraBlind;

static class CatalogTests
{
    sealed class Condition : Terraria.GameContent.ItemDropRules.IItemDropRuleCondition
    { public string GetConditionDescription() => "test condition, not evaluated"; }
    sealed class Drop : Terraria.GameContent.ItemDropRules.IItemDropRule
    {
        readonly int item;
        public Drop(int item) { this.item = item; }
        public void ReportDroprates(System.Collections.Generic.List<Terraria.GameContent.ItemDropRules.DropRateInfo> drops,
            Terraria.GameContent.ItemDropRules.DropRateInfoChainFeed feed)
            => drops.Add(new() { itemId = item, stackMin = 1, stackMax = 2, dropRate = .25f,
                conditions = new() { new Condition() } });
    }
    public static class Sample
    {
        public const short AlphaTool = 7;
        public const int AlphaToolRare = 9, BetaTool = 8, Count = 9999;
        public static int RuntimeValue = 10;
        public const string TextValue = "text";
        public const int Many00 = 0, Many01 = 1, Many02 = 2, Many03 = 3, Many04 = 4, Many05 = 5,
            Many06 = 6, Many07 = 7, Many08 = 8, Many09 = 9, Many10 = 10, Many11 = 11, Many12 = 12;
    }

    static void Check(bool ok, string detail) { if (!ok) throw new Exception(detail); }

    public static void Run()
    {
        var result = AgentCatalog.Query(typeof(Sample), "ALPHA TOOL");
        Check(result["total"].GetValue<int>() == 1 && result["matches"][0]["type"].GetValue<int>() == 7
            && result["matches"][0]["exact"].GetValue<bool>(), "exact lookup included a different item's partial match");
        Check(AgentCatalog.Query(typeof(Sample), "Alpha")["total"].GetValue<int>() == 2,
            "fragment search lost related candidates");
        Check(AgentCatalog.Query(typeof(Sample), "beta_tool")["matches"][0]["type"].GetValue<int>() == 8, "catalog did not normalize identifiers");
        var byId = AgentCatalog.Query(typeof(Sample), "7");
        Check(byId["matches"].AsArray().Count == 2 && byId["matches"][0]["exact"].GetValue<bool>()
            && !AgentCatalog.Query(typeof(Sample), "9999")["found"].GetValue<bool>(), "numeric lookup lost aliases or included Count");
        var labeled = AgentCatalog.Query(typeof(Sample), "Placed Tool", id => id == 8 ? new[] { "Placed Tool" } : Array.Empty<string>());
        Check(labeled["matches"].AsArray().Count == 2 && labeled["matches"][0]["type"].GetValue<int>() == 8
            && labeled["matches"][0]["display_names"][0].GetValue<string>() == "Placed Tool", "catalog lost game display names or aliases");
        foreach (string name in new[] { "Count", "RuntimeValue", "TextValue", "unknown name" })
            Check(!AgentCatalog.Query(typeof(Sample), name)["found"].GetValue<bool>(), "catalog treated non-ID or unknown data as a match");
        var broad = AgentCatalog.Query(typeof(Sample), "Many");
        Check(broad["truncated"].GetValue<bool>() && broad["total"].GetValue<int>() == 13 && broad["matches"].AsArray().Count == 12,
            "catalog did not bound broad results");
        try { AgentCatalog.Query(typeof(Sample), " _ "); throw new Exception("catalog accepted empty identifier"); }
        catch (ArgumentException) { }
        var savedDrops = Terraria.Main.ItemDropsDB;
        int savedCount = Terraria.ModLoader.NPCLoader.NPCCount;
        try
        {
            Terraria.Main.ItemDropsDB = new();
            Terraria.ModLoader.NPCLoader.NPCCount = 4;
            Terraria.Main.ItemDropsDB.Rules[2] = new() { new Drop(77), new Drop(88) };
            Terraria.Main.ItemDropsDB.Global.Add(new Drop(99));
            var drops = AgentCatalog.NpcDropSources(77);
            Check(drops["total_rules"].GetValue<int>() == 1 && drops["sources"][0]["npc_type"].GetValue<int>() == 2
                && drops["sources"][0]["reported_drop_rate"].GetValue<float>() == .25f
                && drops["sources"][0]["conditions"][0]["description"].ToString().Contains("not evaluated"),
                "drop lookup lost item filtering, conditional rates or actual source identity");
            drops = AgentCatalog.NpcDropSources(99, 1);
            Check(drops["total_rules"].GetValue<int>() == 3 && drops["sources"].AsArray().Count == 1
                && drops["truncated"].GetValue<bool>(), "global drop rules were omitted or truncated silently");
            Check(AgentCatalog.NpcDropSources(100)["sources"].AsArray().Count == 0,
                "drop lookup invented an unregistered source");
            try { AgentCatalog.NpcDropSources(0); throw new Exception("drop lookup accepted invalid item"); }
            catch (ArgumentException) { }
        }
        finally { Terraria.Main.ItemDropsDB = savedDrops; Terraria.ModLoader.NPCLoader.NPCCount = savedCount; }
        Console.WriteLine("PASS: catalog resolves typed constants, normalizes names, ranks exact matches and bounds results without inventing unknown IDs");
    }
}
