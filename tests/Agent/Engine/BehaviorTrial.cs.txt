    async Task<JsonObject> Review(JsonObject wake, bool materials = false, CodexAgent session = null, string reportName = "behavior-review")
    {
        if (!ModLoader.TryGetMod("TerraBlind", out var mod)) throw new Exception("production mod missing");
        var schemas = JsonNode.Parse(Encoding.UTF8.GetString(mod.GetFileBytes("Agent/Tools.json"))).AsArray();
        AgentControlSession.DescribeClaims(schemas);
        AgentScript.DescribePrograms(schemas);
        var allowed = materials
            ? new[] { "game_api", "get_state", "npc_find", "inspect_terrain", "script_program", "script_progress", "inventory_review", "action_query", "gather_observe", "opportunity_assess", "npc_combat_observe", "item_info", "recipe" }
            : new[] { "game_api", "get_state", "npc_find", "inspect_terrain", "script_program" };
        var definitions = new JsonArray(schemas.Where(t => t["function"]["name"].ToString() is not ("run_script" or "ask"))
            .Select(t => (JsonNode)new JsonObject { ["name"] = t["function"]["name"].DeepClone(),
                ["summary"] = t["function"]["description"]?.ToString().Split('。')[0] }).ToArray());
        var entries = new JsonArray(schemas.Where(t => allowed.Contains(t["function"]["name"].ToString())).Select(t => t.DeepClone()).ToArray());
        int calls = 0;
        var replies = new JsonArray();
        var reviewCalls = new JsonArray();
        using var ownedAgent = session == null ? new CodexAgent() : null;
        var agent = session ?? ownedAgent;
        using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(materials ? 45 : 90));
        string request = materials
            ? "Inspect the failed material collection and save a JavaScript continuation named material_recovery. Do not execute it during this review. "
                + "Goal: preserve completed opportunistic chest/crystal collection and continue obtaining the recipe materials for Thorn Chakram. "
                + "Use current evidence to choose the next concrete recovery action; retain existing gather/navigation handles and progress. "
                + "Do not invent sources, equipment, healing or a safe route. Do not hardcode coordinates absent from observations. "
                + "The saved code must reobserve at execution, handle interruption, verify actual gains, and resume the same original gather when appropriate. "
                + "You may instead record a concrete unmet prerequisite if no justified action exists, but do not claim task success. "
                + "At most 8 tool calls; save source before your final reply.\nReview request:\n"
            : "Review and save a corrected revision of combat_loot_return. Preserve workflow behavior-trial and all completed step identifiers/handles. "
            + "Goal: move while attacking the nearby hostile, judge actual drops with Jev, collect a useful drop, then resume the original navigation handle. "
            + "Do not execute game actions now. Read the saved program and current terrain/state, then use script_program save with the current expected_revision. "
            + "The saved revision contains the full continuation but pauses at a pre-combat review gate. Use script_program patch to change only that gate and any concrete issue supported by observations. "
            + "Do not delete the continuation or save another immediate decision_point. Preserve the return contract: item_type,gained,original_handle,judgment,returned. "
            + "Handle interrupted actions and danger explicitly. Do not claim a kill or collection without evidence. Avoid walking off the current ledge. "
            + "At most 8 tool calls. Read current source via script_program get.\nReview request:\n";
        try
        {
        await agent.RunAsync(request + wake.ToJsonString(),
            "", Path.Combine(Main.SavePath, "TerraBlindAgent"), "gpt-6-luna",
            "You review and edit saved Terraria JS behavior programs. tool(name,args) inside source uses the game APIs below. "
            + "You may only query or save source during this review. No shell/files/network. Existing workflow step results are reused on resume.\n"
            + "During review get_state is a compact current-state view, not the full runtime snapshot.\n"
            + "The saved program will execute under this production runtime contract (execution is not authorized during this review):\n"
            + AgentScript.RuntimeInstructions + "\nThe following is an API purpose index, not parameter schemas. Before writing calls in saved source, use game_api with names to obtain exact schemas; reuse already obtained definitions.\n"
            + definitions.ToJsonString(new System.Text.Json.JsonSerializerOptions(System.Text.Json.JsonSerializerOptions.Default)
                { Encoder = System.Text.Encodings.Web.JavaScriptEncoder.UnsafeRelaxedJsonEscaping }), entries,
            async (name, args, token) =>
            {
                if (!allowed.Contains(name) || ++calls > 8) throw new InvalidOperationException("review tool budget or scope exceeded");
                if (materials && name == "script_program" && args["operation"]?.ToString() is not ("get" or "list")
                    && (args["name"]?.ToString() != "material_recovery" || args["operation"]?.ToString() is not ("save" or "patch")))
                    throw new InvalidOperationException("review may read programs but only edit material_recovery");
                if (name == "script_progress" && args["action"]?.ToString() is not (null or "" or "get"))
                    throw new InvalidOperationException("review may only read workflow progress");
                string Record(string value)
                {
                    reviewCalls.Add(new JsonObject { ["tool"] = name, ["args"] = args.DeepClone(),
                        ["result"] = value[..Math.Min(4000, value.Length)], ["truncated"] = value.Length > 4000,
                        ["result_chars"] = value.Length });
                    return value;
                }
                string output = await AgentTools.Execute(name, args, token);
                if (name != "get_state") return Record(output);
                var state = JsonNode.Parse(output);
                var compact = new JsonObject();
                foreach (string field in new[] { "player", "buffs", "enemies", "terrain", "healing_items", "health_dynamics", "task_plan", "permanent_max_health", "last_damage", "last_death", "control_overrides" })
                    if (state[field] != null) compact[field] = state[field].DeepClone();
                var watch = new JsonObject();
                foreach (string field in new[] { "events", "passive_events", "tile_types", "radius", "health_below", "pending" })
                    watch[field] = state["watch"]?[field]?.DeepClone();
                compact["watch"] = watch;
                compact["inventory_slots"] = new JsonArray(state["inventory_slots"].AsArray().Where(i => i["type"]?.GetValue<int>() > 0).Select(i => i.DeepClone()).ToArray());
                compact["action_tasks"] = new JsonArray(state["action_tasks"].AsArray().Select(a => (JsonNode)new JsonObject
                { ["handle"] = a["handle"]?.DeepClone(), ["tool"] = a["tool"]?.DeepClone(), ["args"] = a["args"]?.DeepClone(), ["status"] = a["status"]?.DeepClone() }).ToArray());
                var workflows = new JsonObject();
                foreach (var workflow in state["script_workflows"].AsObject())
                    workflows[workflow.Key] = new JsonObject { ["segment"] = workflow.Value["segment"]?.DeepClone(),
                        ["last_execution"] = workflow.Value["last_execution"]?.DeepClone(),
                        ["steps"] = new JsonArray(workflow.Value["steps"].AsArray().Select(step => (JsonNode)new JsonObject
                        { ["id"] = step["id"]?.DeepClone(), ["status"] = step["status"]?.DeepClone() }).ToArray()) };
                compact["script_workflows"] = workflows;
                Console.WriteLine($"MODEL_REVIEW_STATE raw_chars={output.Length} compact_chars={compact.ToJsonString().Length}");
                return Record(compact.ToJsonString());
            }, text => replies.Add(text), timeout.Token);
        }
        finally
        {
        File.WriteAllText(Path.Combine(Main.SavePath, reportName + ".json"), new JsonObject
        { ["model"] = "gpt-6-luna", ["calls"] = calls, ["tool_calls"] = reviewCalls,
            ["replies"] = replies, ["usage"] = agent.TokenUsage }.ToJsonString());
        }
        return await Call("script_program", new JsonObject { ["operation"] = "get", ["name"] = materials ? "material_recovery" : "combat_loot_return" });
    }
    public override void Load()
    {
        enabled = !Main.dedServ && Terraria.Program.LaunchParameters.ContainsKey("-agentbehaviorcheck");
        if (enabled) { On_Main.Update += Open; IL_Main.DoUpdate += Focus; }
        if (enabled && Environment.GetEnvironmentVariable("TB_MATERIAL_NAV_STUB") == "1")
        {
            materialNavigationHook = (original, x, y, mode) =>
            {
                if (!materialNavigationStub) { original(x, y, mode); return; }
                var player = Main.LocalPlayer;
                bool feetTarget = mode is RecedingNav.Mode.Snap or RecedingNav.Mode.Stand;
                var destination = new Vector2(x * 16 + 8 - player.width / 2f,
                    feetTarget ? (y + 1) * 16 - player.height : y * 16 + 8 - player.height / 2f);
                if (mode == RecedingNav.Mode.Stand)
                {
                    bool fits = false;
                    foreach (int offset in new[] { 0, -2, 2, -4, 4, -6, 6, -7, 7 })
                    {
                        var candidate = destination + new Vector2(offset, 0);
                        if (Collision.SolidCollision(candidate, player.width, player.height)) continue;
                        destination = candidate; fits = true; break;
                    }
                    if (!fits)
                    {
                        RecedingNav.Active = false; RecedingNav.LastStop = "test_stand_body_blocked";
                        Console.WriteLine($"MATERIAL_TEST_NAV_REJECT x={x} y={y} mode={mode} reason=body_blocked navigation=teleport_stub");
                        return;
                    }
                }
                player.Teleport(destination);
                player.velocity = Vector2.Zero;
                player.fallStart = player.fallStart2 = (int)(player.position.Y / 16f);
                if (feetTarget && (ActExecutor.OriginCy(player) != y || player.Bottom.Y != (y + 1) * 16))
                    throw new Exception("material navigation substitute violated feet-cell arrival contract");
                RecedingNav.Active = false; RecedingNav.LastStop = "done";
                Console.WriteLine($"MATERIAL_TEST_NAV x={x} y={y} mode={mode} center={player.Center} feet={player.Bottom.Y} body_blocked={Collision.SolidCollision(player.position, player.width, player.height)} navigation=teleport_stub");
            };
            MonoModHooks.Add(typeof(RecedingNav).GetMethod("Start", new[] { typeof(int), typeof(int), typeof(RecedingNav.Mode) }), materialNavigationHook);
        }
    }
    public override void Unload() { if (enabled) { On_Main.Update -= Open; IL_Main.DoUpdate -= Focus; } }
    static void Focus(ILContext il)
    {
        var cursor = new ILCursor(il);
        if (!cursor.TryGotoNext(MoveType.After, instruction => instruction.MatchStsfld<Main>("hasFocus"))) throw new Exception("focus hook missing");
        cursor.EmitDelegate<Action>(() => Main.hasFocus = true);
    }
    void Open(On_Main.orig_Update original, Main game, GameTime time)
    {
        original(game, time);
        if (opened || Main.menuMode != 0) return;
        opened = true;
        if (Terraria.Social.SocialAPI.Cloud != null) Terraria.Social.SocialAPI.Cloud.EnabledByDefault = false;
        WorldGen.clearWorld();
        WorldFile.GetAllMetadata(Path.Combine(Main.SavePath, "Worlds", "Checks.wld"), false).SetAsActive();
        var player = PlayerFileData.CreateAndSave(new Player { name = "JS behavior trial" });
        if (player.IsCloudSave || !player.Path.StartsWith(Main.SavePath, StringComparison.Ordinal)) throw new Exception("nonisolated player");
        Player.LoadPlayer(player.Path, false).SetAsActive(); WorldGen.playWorld();
    }
    Task<T> OnMain<T>(Func<T> body)
    {
        var done = new TaskCompletionSource<T>(TaskCreationOptions.RunContinuationsAsynchronously);
        queue.Enqueue(() => { try { done.SetResult(body()); } catch (Exception error) { done.SetException(error); } });
        return done.Task;
    }
    public override void PostUpdateEverything()
    {
        if (!enabled || finished) return;
        Main.soundVolume = Main.musicVolume = Main.ambientVolume = 0; Main.autoPause = false;
        while (queue.TryDequeue(out var work)) work();
        if (Main.gameMenu || !Main.LocalPlayer.active) return;
        frames++;
        if (!NaturalEncounters)
            for (int i = 0; i < Main.maxNPCs; i++)
                if (i != target || collectingStingers && Main.npc[i].type != NPCID.Hornet) Main.npc[i].active = false;
        var p = Main.LocalPlayer;
        if (samplingSources)
        {
            int x = ItemUseCoordinator.SnappedWx, y = ItemUseCoordinator.SnappedWy;
            bool hasTile = x >= 0 && y >= 0 && x < Main.maxTilesX && y < Main.maxTilesY && Main.tile[x, y].HasTile;
            sourceInputSamples.Add($"{Main.GameUpdateCount},{ItemUseCoordinator.IsActive},{ItemUseCoordinator.Outcome},{x},{y},{hasTile},{p.controlUseItem},{p.itemAnimation},{p.itemTime},{p.HeldItem.type}");
        }
        if (!NaturalEncounters && collectingStingers && frames % 60 == 0 && hornetsSpawned < 40
            && (target < 0 || !Main.npc[target].active) && p.inventory.Where(i => i.type == ItemID.Stinger).Sum(i => i.stack) < stingersRequired)
            SpawnMaterialHornet();
        if (p.controlUseItem && Math.Abs(p.velocity.X) + Math.Abs(p.velocity.Y) > 0.2f) movingAttackFrames++;
        if (frames == 60) trial = Task.Run(Run);
        if (frames % 120 == 0)
        {
            var npc = target >= 0 && target < Main.maxNPCs ? Main.npc[target] : null;
            Console.WriteLine($"BEHAVIOR_SAMPLE frame={frames} pos={p.position} hp={p.statLife} moving_attack={movingAttackFrames}"
                + $" attack={p.controlUseItem} target_active={npc?.active} target_hp={npc?.life} target_pos={npc?.position}"
                + $" clear_line={(npc != null && Collision.CanHitLine(p.Center, 4, 4, npc.Center, 4, 4))}");
        }
        if (trial?.IsCompleted == true || frames > FrameLimit)
        {
            finished = true;
            bool ok = trial?.IsCompletedSuccessfully == true;
            string detail = ok ? $"BEHAVIOR_COMPLETE moving_attack_frames={movingAttackFrames}" : trial?.Exception?.GetBaseException().ToString() ?? "behavior frame deadline";
            Console.WriteLine(detail);
            File.WriteAllText(Path.Combine(Main.SavePath, "agent-engine-result.json"), new JsonObject { ["ok"] = ok, ["detail"] = detail }.ToJsonString());
        }
    }
    static async Task<JsonObject> Call(string tool, JsonObject args)
    {
        var result = JsonNode.Parse(await AgentTools.Execute(tool, args, default)).AsObject();
        Console.WriteLine("BEHAVIOR_TOOL " + tool + " " + result.ToJsonString());
        return result;
    }
    bool SpawnMaterialHornet()
    {
        var p = Main.LocalPlayer;
        int selected = CombatLoadout.Select(p);
        if (selected >= 0)
        {
            var weapon = p.inventory[selected];
            Console.WriteLine($"MATERIAL_LOADOUT slot={selected} type={weapon.type} name={weapon.Name} use_ammo={weapon.useAmmo} has_ammo={p.HasAmmo(weapon)}");
        }
        foreach (int dx in new[] { 96, -96, 64, -64 })
            foreach (int dy in new[] { -64, 0, 64 })
            {
                var position = p.Center + new Vector2(dx, dy);
                if (Collision.SolidCollision(position - new Vector2(20), 40, 40)
                    || !Collision.CanHitLine(p.Center, 4, 4, position, 4, 4)) continue;
                target = NPC.NewNPC(new EntitySource_Misc("OpportunityMaterialTrial"), (int)position.X, (int)position.Y, NPCID.Hornet);
                hornetsSpawned++;
                Console.WriteLine($"MATERIAL_HORNET spawn={hornetsSpawned} who={target} hp={Main.npc[target].life} pos={Main.npc[target].position}");
                return true;
            }
        return false;
    }
    async Task Run()
    {
        if (Environment.GetEnvironmentVariable("TB_DROP_IDENTITY") == "1")
        {
            await OnMain(() => { foreach (var item in Main.item) item.active = false; return true; });
            var gather = new GameGather(new[] { new GatherTarget(ItemID.Stinger, 9, Array.Empty<string>(), Array.Empty<int>()) });
            var samples = new JsonArray();
            JsonObject previous = null;
            for (int i = 0; i < 3; i++)
            {
                int slot = await OnMain(() => Item.NewItem(new EntitySource_Misc("DropIdentityTrial"),
                    new Rectangle((int)Main.LocalPlayer.Center.X + 200, (int)Main.LocalPlayer.Center.Y, 16, 16), ItemID.Stinger));
                var candidate = (await gather.Observe(default))["candidates"].AsArray()
                    .First(c => c["kind"]?.ToString() == "drop" && c["who"]?.GetValue<int>() == slot).AsObject();
                var same = (await gather.Observe(default))["candidates"].AsArray()
                    .First(c => c["who"]?.GetValue<int>() == slot);
                if (candidate["id"].ToString() != same["id"].ToString()) throw new Exception("unchanged native drop changed identity");
                if (previous != null && (previous["who"].GetValue<int>() != slot
                    || previous["id"].ToString() == candidate["id"].ToString()
                    || (await gather.Act(previous, default))["outcome"]?.ToString() != "target_changed"))
                    throw new Exception("native reused drop slot inherited identity or accepted stale action");
                samples.Add(candidate.DeepClone()); previous = candidate;
                await OnMain(() => { Main.item[slot].active = false; return true; });
            }
            File.WriteAllText(Path.Combine(Main.SavePath, "drop-identity.json"), samples.ToJsonString());
            Console.WriteLine("DROP_IDENTITY_PASS native_spawns=3 reused_slot=true stale_action_rejected=true model_calls=0");
            return;
        }
        if (Environment.GetEnvironmentVariable("TB_COMBAT_TRACE") == "1")
        {
            await OnMain(() =>
            {
                for (int x = 95; x <= 115; x++)
                    for (int y = 95; y <= 105; y++) Main.tile[x, y].ClearEverything();
                var from = new Vector2(100 * 16 + 8, 100 * 16 + 8);
                var to = new Vector2(112 * 16 + 8, 100 * 16 + 8);
                var platform = Main.tile[105, 99]; platform.HasTile = true; platform.TileType = TileID.Platforms;
                var platformOnly = PvpTerrain.Trace(from, to);
                var wall = Main.tile[106, 100]; wall.HasTile = true; wall.TileType = TileID.Stone;
                var withWall = PvpTerrain.Trace(from, to);
                Main.tile[105, 99].ClearEverything();
                var withoutPlatform = PvpTerrain.Trace(from, to);
                Main.tile[106, 100].ClearEverything();
                var cleared = PvpTerrain.Trace(from, to);
                File.WriteAllText(Path.Combine(Main.SavePath, "combat-trace.json"), new JsonObject
                { ["platform_only"] = platformOnly, ["wall_and_platform"] = withWall,
                    ["removed_platform"] = withoutPlatform, ["removed_wall"] = cleared }.ToJsonString());
                if (platformOnly["clear"]?.GetValue<bool>() != true || cleared["clear"]?.GetValue<bool>() != true
                    || withWall["blocker"]?["x"]?.GetValue<int>() != 106 || withWall["blocker"]?["y"]?.GetValue<int>() != 100
                    || withoutPlatform["clear"]?.GetValue<bool>() != false)
                    throw new Exception("combat trace mistook nearby platform for the native line obstruction");
                return true;
            });
            Console.WriteLine("COMBAT_TRACE_PASS native_collision=true platform_not_blocker=true model_calls=0");
            return;
        }
        if (Environment.GetEnvironmentVariable("TB_INVENTORY_MANAGEMENT") == "1")
        {
            await OnMain(() =>
            {
                NativeAgent.Stop(); Halt.All(); Manual.On = false;
                Config.I.AgentEnabled = true; Config.I.AiClient = false;
                var p = Main.LocalPlayer;
                foreach (var item in p.inventory) item.TurnToAir();
                foreach (var item in p.armor) item.TurnToAir();
                Main.mouseItem.TurnToAir(); p.itemTime = p.itemAnimation = 0;
                p.inventory[0].SetDefaults(ItemID.WoodHelmet);
                p.inventory[1].SetDefaults(ItemID.CloudinaBottle);
                p.inventory[2].SetDefaults(ItemID.StoneBlock); p.inventory[2].stack = 17; p.inventory[2].favorited = true;
                return true;
            });
            var overview = await Call("inventory_review", new JsonObject());
            JsonNode Inventory(int slot) => overview["inventory"].AsArray().First(i => i["slot"].GetValue<int>() == slot).DeepClone();
            foreach (var (inventory, equipment) in new[] { (0, 0), (1, 3) })
            {
                var equipped = await Call("equip_item", new JsonObject { ["inventory_slot"] = inventory, ["equipment_slot"] = equipment,
                    ["expected_inventory"] = Inventory(inventory), ["expected_equipment"] = overview["equipment"][equipment].DeepClone() });
                if (equipped["ok"]?.GetValue<bool>() != true) throw new Exception("native equipment rejected: " + equipped);
            }
            var dropArgs = new JsonObject { ["slot"] = 2, ["expected"] = Inventory(2) };
            var favorite = await Call("inventory_drop", dropArgs);
            if (favorite["error"]?.ToString() != "favorited_item") throw new Exception("native favorite was discarded");
            await OnMain(() => { Main.LocalPlayer.inventory[2].favorited = false; return true; });
            dropArgs["expected"]["stack"] = 18;
            var stale = await Call("inventory_drop", dropArgs);
            if (stale["error"]?.ToString() != "stale_inventory") throw new Exception("native stale inventory was discarded");
            dropArgs["expected"]["stack"] = 17;
            var dropped = await Call("inventory_drop", dropArgs);
            var verified = await OnMain(() =>
            {
                var p = Main.LocalPlayer;
                int who = dropped["world_item"]?.GetValue<int>() ?? -1;
                return dropped["ok"]?.GetValue<bool>() == true && p.inventory[2].IsAir && who >= 0 && who < Main.maxItems
                    && Main.item[who].active && Main.item[who].type == ItemID.StoneBlock && Main.item[who].stack == 17
                    && p.armor[0].type == ItemID.WoodHelmet && p.statDefense > 0
                    && p.armor[3].type == ItemID.CloudinaBottle && p.GetJumpState(ExtraJump.CloudInABottle).Enabled
                    && Main.mouseItem.IsAir;
            });
            File.WriteAllText(Path.Combine(Main.SavePath, "inventory-management.json"), new JsonObject
                { ["review"] = overview, ["favorite"] = favorite, ["stale"] = stale, ["drop"] = dropped, ["verified"] = verified }.ToJsonString());
            if (!verified) throw new Exception("native inventory/equipment effects unverified");
            Console.WriteLine("INVENTORY_MANAGEMENT_PASS native_equipment=true native_drop=true protected=true model_calls=0");
            return;
        }
        if (Environment.GetEnvironmentVariable("TB_PROJECTILE_ONLY_DEFENSE") == "1")
        {
            await OnMain(() =>
            {
                var p = Main.LocalPlayer;
                int cx = (int)(p.Center.X / 16), cy = (int)(p.Center.Y / 16);
                for (int x = cx - 12; x <= cx + 12; x++)
                    for (int y = cy - 12; y <= cy + 12; y++) Main.tile[x, y].ClearEverything();
                foreach (var npc in Main.npc) npc.active = false;
                float oldGravity = p.gravity;
                p.velocity = Vector2.Zero; p.gravity = 0;
                int shot = Projectile.NewProjectile(p.GetSource_Misc("projectile-only-defense-test"),
                    p.Center + new Vector2(0, -120), new Vector2(0, 5), ProjectileID.Stinger, 10, 0);
                Main.projectile[shot].hostile = true; Main.projectile[shot].friendly = false;
                var policy = (AgentControlPolicy)typeof(AgentTools).GetField("Control",
                    System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.NonPublic).GetValue(null);
                string veto = null;
                try
                {
                    FightCoordinator.Start(20);
                    FightCoordinator.Tick(p);
                    bool moved = p.controlLeft || p.controlRight;
                    if (!moved || p.controlUseItem) throw new Exception("native incoming projectile was ignored without an NPC target");
                    veto = policy.Deny(new[] { ControlBehavior.Dodge }, ControlLifetime.Task, null, "test movement veto");
                    FightCoordinator.Tick(p);
                    if (p.controlLeft || p.controlRight || p.controlJump)
                        throw new Exception("native projectile defense overrode movement veto");
                    Console.WriteLine("PROJECTILE_ONLY_DEFENSE_PASS native_inputs=true no_npc=true veto_respected=true");
                }
                finally
                {
                    if (veto != null) policy.Release(veto);
                    FightCoordinator.Stop(); Main.projectile[shot].active = false; p.gravity = oldGravity;
                }
                return true;
            });
            return;
        }
        if (Environment.GetEnvironmentVariable("TB_SOURCE_DATABASE_PROBE") == "1")
        {
            await OnMain(() =>
            {
                var timer = Stopwatch.StartNew();
                var sources = new JsonArray();
                for (int type = 1; type < NPCID.Count; type++)
                {
                    var drops = new List<Terraria.GameContent.ItemDropRules.DropRateInfo>();
                    foreach (var rule in Main.ItemDropsDB.GetRulesForNPCID(type, true))
                        rule.ReportDroprates(drops, new Terraria.GameContent.ItemDropRules.DropRateInfoChainFeed(1f));
                    foreach (var drop in drops.Where(d => d.itemId == ItemID.Stinger))

                                .map(a=>a.result?.state?.pos??(a.tool==='nav_to'?{x:a.args.x,y:a.args.y}:null))
                                .filter(p=>p&&Number.isFinite(p.x)&&Number.isFinite(p.y));
                              const nearest=state.enemies.slice().sort((a,b)=>Math.hypot(a.pos.x-state.player.pos.x,a.pos.y-state.player.pos.y)
                                -Math.hypot(b.pos.x-state.player.pos.x,b.pos.y-state.player.pos.y))[0];
                              if(nearest) {
                                const nearby=tool('npc_combat_observe',{npc:nearest.who,npc_type:nearest.type});
                                if(!nearby.error) for(const p of nearby.terrain.stance_probes)
                                  if(p.body_fits) anchors.push({x:p.support_x,y:p.support_y-1});
                              }
                              anchors.sort((a,b)=>distanceToEnemy(b)-distanceToEnemy(a));
                              recovery.retreat_candidates=[];
                              for(const anchor of anchors) {
                                const enemyDistance=distanceToEnemy(anchor);
                                if(Math.hypot(anchor.x+.5-state.player.pos.x/16-state.player.width/32,
                                  anchor.y+.5-state.player.pos.y/16-state.player.height/32)>60) {
                                  recovery.retreat_candidates.push({anchor,rejected:'outside_observation_range'});continue;
                                }
                                if(enemyDistance<20){recovery.retreat_candidates.push({anchor,enemyDistance,rejected:'enemy_nearby'});continue;}
                                const facts=tool('opportunity_assess',anchor);
                                if(facts.error||!Array.isArray(facts.hazards)) {
                                  recovery.retreat_candidates.push({anchor,enemyDistance,rejected:'observation_failed',error:facts.error,detail:facts.detail});
                                  continue;
                                }
                                recovery.retreat_candidates.push({anchor,enemyDistance,hazards:facts.hazards});
                                const cx=anchor.x*16+8,cy=(anchor.y+1)*16-state.player.height/2;
                                const overlaps=h=>h.x*16<cx+state.player.width/2&&(h.x+1)*16>cx-state.player.width/2
                                  &&h.y*16<cy+state.player.height/2&&(h.y+1)*16>cy-state.player.height/2;
                                if(!facts.hazards.some(h=>h.kind!=='contact_damage'||h.on_direct_corridor||overlaps(h))) {
                                  retreat={anchor,facts};break;
                                }
                              }
                              } finally {
                                const released=tool('action_control',{handle:guard.handle,command:'cancel'});
                                recovery.defense={handle:guard.handle,status:released.status,hp:released.observation?.hp,pos:released.observation?.pos};
                              }
                              state=tool('get_state');
                              if(state.player.hp<=0||state.player.suffocating)
                                tool('decision_point',{reason:'Recovery observation ended in danger; retain original gather and defense evidence'});
                              if(retreat) {
                                tool('script_events',{wait_ms:0});
                                const policy={events:state.watch.events,passive_events:state.watch.passive_events,
                                  tile_types:state.watch.tile_types,radius:state.watch.radius,health_below:state.watch.health_below};
                                tool('watch',{...policy,passive_events:[...new Set([...policy.passive_events,'low_health'])]});
                                let moved;
                                try { moved=tool('nav_to',retreat.anchor); }
                                finally { tool('watch',policy); }
                                recovery.retreat={anchor:retreat.anchor,hazards:retreat.facts.hazards,
                                  result:{done:moved.done,outcome:moved.outcome,handle:moved.handle??moved.managed_handle,state:moved.state,error:moved.error}};
                                step('combat-retreat:'+action.attempt,()=>recovery.retreat);
                                if(recovery.retreat.result.done!==true)
                                  tool('decision_point',{reason:'Recovery retreat interrupted; retain original gather and inspect navigation handle'});
                                state=tool('get_state');
                                recovery.rest=[];
                                let previousHp=state.player.hp;
                                const damageTick=state.last_damage?.tick;
                                for(let sample=0;sample<20&&state.player.hp<state.watch.health_below;sample++) {
                                  const position={x:state.player.pos.x/16,y:state.player.pos.y/16};
                                  if(state.player.suffocating||state.player.hp<=40||state.last_damage?.tick!==damageTick
                                    ||(state.player.hp<previousHp&&state.health_dynamics?.regen_direction!=='damage')||distanceToEnemy(position)<20) break;
                                  previousHp=state.player.hp;
                                  tool('script_events',{wait_ms:1000});
                                  state=tool('get_state');
                                  recovery.rest.push({hp:state.player.hp,pos:state.player.pos,health_dynamics:state.health_dynamics,
                                    last_damage:state.last_damage,buffs:state.buffs});
                                }
                              }
                            }
                            recovery.hp_after=state.player.hp;
                            step('combat-recovery:'+action.attempt,()=>recovery);
                            if(state.player.hp<state.watch.health_below||state.player.suffocating)
                              tool('decision_point',{reason:'Material combat needs a different recovery or equipment plan; health remains below the configured threshold: '+JSON.stringify(recovery)});
                            if(['combat_obstructed','combat_out_of_range'].includes(action.result?.outcome)) restoreShotLine();
                            const before=step('stinger-baseline',()=>{throw new Error('original material baseline missing');});
                            step('combat-review:'+action.attempt,()=>({handle:action.handle,hp:state.player.hp,
                              reason:'Recovery checked against actual inventory and health; retain the same material target'}));
                            tool('script_events',{wait_ms:0});
                            tool('action_control',{handle:action.handle,command:'resume'});
                            for(let i=0;i<20;i++) {
                              const observed=tool('script_events',{wait_ms:1000});
                              if(observed.events.some(e=>['player_died','low_health','suffocating'].includes(e.kind)))
                                tool('decision_point',{reason:'Revised combat health limit reached; do not resume blindly'});
                              const current=tool('action_query',{handle:action.handle});
                              if(current.status==='completed'||current.status==='failed') break;
                              if(current.status==='paused') tool('action_control',{handle:action.handle,command:'resume'});
                            }
                            const after=tool('get_state').inventory_slots.filter(i=>i.type===209).reduce((n,i)=>n+i.stack,0);
                            const final=tool('action_query',{handle:action.handle});
                            return {before,after,gained:after-before,resumed_handle:action.handle,
                              action:{handle:final.handle,status:final.status,attempt:final.attempt,
                                result:{outcome:final.result?.outcome,targets:final.result?.targets,last_result:final.result?.last_result}}};
                            """ });
                }
                await OnMain(() => { collectingStingers = false; return true; });
                int actualStingers = await OnMain(() => Main.LocalPlayer.inventory.Where(i => i.type == ItemID.Stinger).Sum(i => i.stack));
                combat["verified_inventory"] = new JsonObject { ["item_type"] = ItemID.Stinger,
                    ["actual"] = actualStingers, ["required"] = stingersRequired };
                File.WriteAllText(Path.Combine(Main.SavePath, "stinger-result.json"), combat.ToJsonString());
                if (actualStingers < stingersRequired)
                    throw new Exception("recipe stinger quantity not fulfilled; inspect stinger-result.json");
                Console.WriteLine($"OPPORTUNITY_STINGER_PROGRESS verified_recipe_stingers=true spawned_hornets={hornetsSpawned}");
            }
            var verification = await Call("task_step", new JsonObject { ["id"] = "materials", ["action"] = "verify" });
            File.WriteAllText(Path.Combine(Main.SavePath, "material-plan-verification.json"), verification.ToJsonString());
            bool allMaterials = Environment.GetEnvironmentVariable("TB_OPPORTUNITY_STINGERS") == "1";
            if (verification["verified"]?.GetValue<bool>() != allMaterials || verification["current_step"]?.ToString() != (allMaterials ? "crafted" : "materials")
                || verification["steps"]?[1]?["id"]?.ToString() != "crafted")
                throw new Exception("partial materials advanced or lost the preserved crafting plan");
            Console.WriteLine($"OPPORTUNITY_RECIPE_PENDING materials_complete={allMaterials} crafting_goal_preserved=true");
            if (allMaterials)
            {
                var crafted = await Call("run_script", new JsonObject { ["workflow"] = "material-crafting", ["event_delivery"] = "script",
                    ["code"] = """
                        function recipe(){return tool('recipe',{name:'ThornChakram'}).recipes.find(r=>r.internal==='ThornChakram');}
                        let current=recipe();
                        if(!current.can_craft_now) {
                          const candidates=[];
                          for(const station of current.stations)
                            candidates.push(...tool('find_tiles',{name:station.tile,max_dist:2000,n:5}).tiles);
                          if(!candidates.length) tool('decision_point',{reason:'Materials verified but crafting station unavailable; preserve crafting step',evidence:{stations:current.stations}});
                          const station=candidates[0];
                          const arrival=tool('nav_to',{x:station.x,y:station.y,reach:true});
                          if(arrival.done!==true) tool('decision_point',{reason:'Station approach unfinished',evidence:{station,arrival}});
                          current=recipe();
                        }
                        if(!current.can_craft_now) tool('decision_point',{reason:'Station arrival did not make recipe craftable',evidence:{recipe:current}});
                        const made=tool('craft',{name:current.internal});
                        if(made.error||made.ok===false) throw new Error(JSON.stringify(made));
                        const verified=tool('task_step',{id:'crafted',action:'verify'});
                        const inventory=tool('get_state').inventory_slots;
                        return {verified:verified.verified,plan_status:verified.status,
                          crystals:inventory.filter(i=>i.type===29).reduce((n,i)=>n+i.stack,0),
                          weapon:inventory.filter(i=>i.type===191).reduce((n,i)=>n+i.stack,0)};
                        """ });
                File.WriteAllText(Path.Combine(Main.SavePath, "crafted-result.json"), crafted.ToJsonString());
                if (crafted["value"]?["verified"]?.GetValue<bool>() != true
                    || crafted["value"]?["crystals"]?.GetValue<int>() is not >= 1
                    || crafted["value"]?["weapon"]?.GetValue<int>() is not >= 1)
                    throw new Exception("final crafting or preserved crystal unverified; inspect crafted-result.json");
                Console.WriteLine("OPPORTUNITY_CRAFTED_VERIFIED native_inventory=true model_calls=0");
            }
        }
    }
