namespace Engine;

public sealed class BehaviorTrial : ModSystem
{
    readonly ConcurrentQueue<Action> queue = new();
    bool enabled, opened, finished;
    int frames, movingAttackFrames, originX, floor, target = -1;
    Task trial;
    bool Natural => Environment.GetEnvironmentVariable("TB_BEHAVIOR_NATURAL") == "1";
    async Task<JsonObject> Review(JsonObject wake, string proposedCode)
    {
        if (!ModLoader.TryGetMod("TerraBlind", out var mod)) throw new Exception("production mod missing");
        var schemas = JsonNode.Parse(Encoding.UTF8.GetString(mod.GetFileBytes("Agent/Tools.json"))).AsArray();
        var allowed = new[] { "get_state", "npc_find", "inspect_terrain", "script_program" };
        var programTools = allowed.Concat(new[] { "watch", "fight", "action_start", "action_control", "action_query", "script_events",
            "decision_point", "jev_judge", "nearby_drops", "gather", "script_progress" }).ToArray();
        var definitions = new JsonArray(schemas.Where(t => programTools.Contains(t["function"]["name"].ToString())).Select(t => t["function"].DeepClone()).ToArray());
        var entries = new JsonArray(schemas.Where(t => allowed.Contains(t["function"]["name"].ToString())).Select(t => t.DeepClone()).ToArray());
        int calls = 0;
        var replies = new JsonArray();
        using var agent = new CodexAgent();
        using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(90));
        await agent.RunAsync("Review and save a corrected revision of combat_loot_return. Preserve workflow behavior-trial and all completed step identifiers/handles. "
            + "Goal: move while attacking the nearby hostile, judge actual drops with Jev, collect a useful drop, then resume the original navigation handle. "
            + "Do not execute game actions now. Read the saved program and current terrain/state, then use script_program save with the current expected_revision. "
            + "The saved revision currently ends at a review request. Replace that request with the executable candidate continuation below, correcting it as needed. "
            + "Saving another immediate decision_point without the combat/collection/return logic is NOT a solution. Preserve the candidate return contract: item_type,gained,original_handle,judgment,returned. "
            + "Handle interrupted actions and danger explicitly. Do not claim a kill or collection without evidence. Avoid walking off the current ledge. "
            + "At most 8 tool calls. Existing candidate continuation (editable):\n" + proposedCode + "\nReview request:\n" + wake.ToJsonString(),
            "", Path.Combine(Main.SavePath, "TerraBlindAgent"), "gpt-6-luna",
            "You review and edit saved Terraria JS behavior programs. tool(name,args) inside source uses the game APIs below. "
            + "You may only query or save source during this review. No shell/files/network. Existing workflow step results are reused on resume.\n"
            + "During review get_state is a compact current-state view, not the full runtime snapshot.\n" + definitions.ToJsonString(), entries,
            async (name, args, token) =>
            {
                if (!allowed.Contains(name) || ++calls > 8) throw new InvalidOperationException("review tool budget or scope exceeded");
                string output = await AgentTools.Execute(name, args, token);
                if (name != "get_state") return output;
                var state = JsonNode.Parse(output);
                var compact = new JsonObject();
                foreach (string field in new[] { "player", "permanent_max_health", "last_damage", "last_death", "control_overrides" })
                    if (state[field] != null) compact[field] = state[field].DeepClone();
                compact["inventory_slots"] = new JsonArray(state["inventory_slots"].AsArray().Where(i => i["type"]?.GetValue<int>() > 0).Select(i => i.DeepClone()).ToArray());
                compact["action_tasks"] = new JsonArray(state["action_tasks"].AsArray().Select(a => (JsonNode)new JsonObject
                { ["handle"] = a["handle"]?.DeepClone(), ["tool"] = a["tool"]?.DeepClone(), ["args"] = a["args"]?.DeepClone(), ["status"] = a["status"]?.DeepClone() }).ToArray());
                compact["script_workflows"] = state["script_workflows"]?.DeepClone();
                return compact.ToJsonString();
            }, text => replies.Add(text), timeout.Token);
        File.WriteAllText(Path.Combine(Main.SavePath, "behavior-review.json"), new JsonObject
        { ["model"] = "gpt-6-luna", ["calls"] = calls, ["replies"] = replies, ["usage"] = agent.TokenUsage }.ToJsonString());
        return await Call("script_program", new JsonObject { ["operation"] = "get", ["name"] = "combat_loot_return" });
    }
    public override void Load()
    {
        enabled = !Main.dedServ && Terraria.Program.LaunchParameters.ContainsKey("-agentbehaviorcheck");
        if (enabled) { On_Main.Update += Open; IL_Main.DoUpdate += Focus; }
    }
    public override void Unload() { if (enabled) { On_Main.Update -= Open; IL_Main.DoUpdate -= Focus; } }
    static void Focus(ILContext il)
    {
        var cursor = new ILCursor(il);
        if (!cursor.TryGotoNext(MoveType.After, instruction => instruction.MatchStsfld<Main>("hasFocus"))) throw new Exception("focus hook missing");
        cursor.EmitDelegate<Action>(() => Main.hasFocus = true);
    }
    void Open(On_Main.orig_Update original, Main game, GameTime time)
    {
        original(game, time);
        if (opened || Main.menuMode != 0) return;
        opened = true;
        if (Terraria.Social.SocialAPI.Cloud != null) Terraria.Social.SocialAPI.Cloud.EnabledByDefault = false;
        WorldGen.clearWorld();
        WorldFile.GetAllMetadata(Path.Combine(Main.SavePath, "Worlds", "Checks.wld"), false).SetAsActive();
        var player = PlayerFileData.CreateAndSave(new Player { name = "JS behavior trial" });
        if (player.IsCloudSave || !player.Path.StartsWith(Main.SavePath, StringComparison.Ordinal)) throw new Exception("nonisolated player");
        Player.LoadPlayer(player.Path, false).SetAsActive(); WorldGen.playWorld();
    }
    Task<T> OnMain<T>(Func<T> body)
    {
        var done = new TaskCompletionSource<T>(TaskCreationOptions.RunContinuationsAsynchronously);
        queue.Enqueue(() => { try { done.SetResult(body()); } catch (Exception error) { done.SetException(error); } });
        return done.Task;
    }
    public override void PostUpdateEverything()
    {
        if (!enabled || finished) return;
        Main.soundVolume = Main.musicVolume = Main.ambientVolume = 0; Main.autoPause = false;
        while (queue.TryDequeue(out var work)) work();
        if (Main.gameMenu || !Main.LocalPlayer.active) return;
        frames++;
        for (int i = 0; i < Main.maxNPCs; i++) if (i != target) Main.npc[i].active = false;
        var p = Main.LocalPlayer;
        if (p.controlUseItem && Math.Abs(p.velocity.X) + Math.Abs(p.velocity.Y) > 0.2f) movingAttackFrames++;
        if (frames == 60) trial = Task.Run(Run);
        if (frames % 120 == 0)
        {
            var npc = target >= 0 && target < Main.maxNPCs ? Main.npc[target] : null;
            Console.WriteLine($"BEHAVIOR_SAMPLE frame={frames} pos={p.position} hp={p.statLife} moving_attack={movingAttackFrames}"
                + $" attack={p.controlUseItem} target_active={npc?.active} target_hp={npc?.life} target_pos={npc?.position}"
                + $" clear_line={(npc != null && Collision.CanHitLine(p.Center, 4, 4, npc.Center, 4, 4))}");
        }
        if (trial?.IsCompleted == true || frames > 7200)
        {
            finished = true;
            bool ok = trial?.IsCompletedSuccessfully == true;
            string detail = ok ? $"BEHAVIOR_COMPLETE moving_attack_frames={movingAttackFrames}" : trial?.Exception?.GetBaseException().ToString() ?? "behavior frame deadline";
            Console.WriteLine(detail);
            File.WriteAllText(Path.Combine(Main.SavePath, "agent-engine-result.json"), new JsonObject { ["ok"] = ok, ["detail"] = detail }.ToJsonString());
        }
    }
    static async Task<JsonObject> Call(string tool, JsonObject args)
    {
        var result = JsonNode.Parse(await AgentTools.Execute(tool, args, default)).AsObject();
        Console.WriteLine("BEHAVIOR_TOOL " + tool + " " + result.ToJsonString());
        return result;
    }
    async Task Run()
    {
        await OnMain(() =>
        {
            NativeAgent.Stop(); Halt.All(); Config.I.AgentEnabled = true; Config.I.AiClient = false; Manual.On = false;
            Combat.Enabled = Dodge.Enabled = false; AgentPermissions.Reset();
            var p = Main.LocalPlayer; originX = (int)p.Center.X / 16; floor = (int)p.Bottom.Y / 16 + 3;
            if (Natural)
            {
                bool found = false;
                for (int y = (int)Main.rockLayer; y < Main.maxTilesY - 250 && !found; y += 3)
                for (int x = 100; x < Main.maxTilesX - 100 && !found; x += 3)
                {
                    if (!WorldGen.SolidTile(x, y) || Main.tile[x, y].TileType != TileID.JungleGrass) continue;
                    if (Collision.SolidCollision(new Vector2((x - 3) * 16, (y - 4) * 16), 8 * 16, 4 * 16)
                        || Collision.SolidCollision(new Vector2((x + 8) * 16, (y - 6) * 16), 40, 40)) continue;
                    if (Main.tile[x, y - 1].LiquidAmount > 0 || !WorldGen.SolidTile(x + 3, y)) continue;
                    originX = x; floor = y; found = true;
                }
                if (!found) throw new Exception("no suitable native jungle cave found; terrain was not altered");
                Console.WriteLine($"BEHAVIOR_NATURAL x={originX} y={floor} fixture_tile_edits=0");
            }
            else for (int x = originX - 30; x <= originX + 35; x++)
            {
                for (int y = floor - 15; y < floor; y++) Main.tile[x, y].ClearEverything();
                Main.tile[x, floor].ResetToType(TileID.Stone);
            }
            foreach (var item in p.inventory) item.TurnToAir();
            p.inventory[0].SetDefaults(ItemID.ThrowingKnife); p.inventory[0].stack = 500;
            p.ConsumedLifeCrystals = 15; p.statLife = p.statLifeMax = p.statLifeMax2 = 400;
            p.position = new Vector2(originX * 16, floor * 16 - p.height); p.velocity = Vector2.Zero;
            p.fallStart = p.fallStart2 = (int)p.position.Y / 16;
            target = NPC.NewNPC(new EntitySource_Misc("BehaviorTrial"), (int)p.Center.X + (Natural ? 128 : 75),
                (int)p.Bottom.Y - (Natural ? 80 : 0), Natural ? NPCID.Hornet : NPCID.BlueSlime);
            if (!Natural) Main.npc[target].life = Main.npc[target].lifeMax = 120;
            return true;
        });
        string code = """
            tool('watch',{events:['nearby_npc_killed','nearby_drop'],radius:40});
            const original = step('original',()=>tool('action_start',{tool:'nav_to',args:{x:GOAL_X,y:GOAL_Y}}));
            step('pause-original',()=>tool('action_control',{handle:original.handle,command:'pause'}));
            function chooseMovement() {
              const terrain=tool('inspect_terrain',{radius:12});
              const enemies=tool('npc_find',{type:NPC_TYPE});
              const answer=tool('jev_judge',{state:{terrain,enemies},questions:{movement:{type:'choice',
                instructions:'Choose movement for the next one second of throwing-knife combat. NPC cell and terrain origin use tile coordinates. Read player_bounds and rows. Do not walk off a ledge into a lower pocket, liquids, or a wall. Prefer auto when the current area permits attack or when terrain is uncertain. A horizontal move is useful only if it improves the firing position without falling into a worse pocket.',
                criteria:{auto:'Let the existing controller dodge threats and restore a blocked firing line.',left:'Move left while aiming and attacking for one second.',right:'Move right while aiming and attacking for one second.'}}}});
              if(answer.error) tool('decision_point',{reason:'Movement judgment failed: '+JSON.stringify(answer)});
              return answer.answers.movement.choice;
            }
            const fight = step('combat',()=>tool('action_start',{tool:'fight',args:{npc_types:[NPC_TYPE],max_dist:40,seconds:60}}));
            const initialMovement=chooseMovement();
            tool('action_control',{handle:fight.handle,command:'pause'});
            tool('action_control',{handle:fight.handle,command:'resume',args:{npc_types:[NPC_TYPE],max_dist:40,seconds:60,movement:initialMovement}});
            let death=false, previousLife=null, judgments=1, resetMovement=true;
            for(let i=0;i<30&&!death;i++) {
              const notice=tool('script_events',{wait_ms:1000});
              death=notice.events.some(e=>e.kind==='nearby_npc_killed');
              if(death) break;
              if(resetMovement) {
                tool('action_control',{handle:fight.handle,command:'pause'});
                tool('action_control',{handle:fight.handle,command:'resume',args:{npc_types:[NPC_TYPE],max_dist:40,seconds:60,movement:'auto'}});
                resetMovement=false;
              } else if(notice.paused_action) tool('action_control',{handle:notice.paused_action,command:'resume'});
              if(i%6===5) {
                const enemies=tool('npc_find',{type:NPC_TYPE}).npcs;
                const life=enemies.reduce((sum,n)=>sum+n.life,0);
                if(previousLife!==null&&life>=previousLife&&enemies.length) {
                  if(judgments>=3) tool('decision_point',{reason:'No combat progress after local adjustments; inspect terrain and preserved combat handle instead of waiting indefinitely.'});
                  const movement=chooseMovement(); judgments++;
                  tool('action_control',{handle:fight.handle,command:'pause'});
                  tool('action_control',{handle:fight.handle,command:'resume',args:{npc_types:[NPC_TYPE],max_dist:40,seconds:60,movement}});
                  resetMovement=true;
                }
                previousLife=life;
              }
            }
            if(!death) tool('decision_point',{reason:'No verified kill after bounded local combat; inspect the current target and terrain, then modify this program without restarting the original goal.'});
            if(fight) tool('action_control',{handle:fight.handle,command:'cancel'});
            let drops=[];
            for(let i=0;i<5&&!drops.length;i++) {
              drops=tool('nearby_drops',{radius:40}).drops.filter(d=>d.item_type!==279);
              if(!drops.length) tool('script_events',{wait_ms:250});
            }
            if(!drops.length) throw new Error('no visible native drop to judge');
            const terrain=tool('inspect_terrain',{radius:12});
            const decision=tool('jev_judge',{state:{goal:'Collect a useful dropped material or money then return to the original destination',drops,terrain},
              questions:{pickup:{type:'noul',instructions:'Is collecting one of these drops useful for the stated goal? This is usefulness, not a claim that the route is safe.'}}});
            if(decision.error) throw new Error(JSON.stringify(decision));
            if(decision.answers.pickup.noul<0.7) {tool('decision_point',{reason:'drop judgment needs review'});}
            const itemType=drops.find(d=>d.item_type===209||d.item_type===23)?.item_type??drops[0].item_type;
            const before=tool('get_state').inventory_slots.filter(i=>i.type===itemType).reduce((n,i)=>n+i.stack,0);
            let collected=tool('gather',{targets:[{item_type:itemType,count:before+1}]});
            for(let i=0;i<20&&collected.handle;i++) {
              const notice=tool('script_events',{wait_ms:0});
              if([...(collected.events??[]),...notice.events].some(e=>['player_died','low_health','suffocating'].includes(e.kind)))
                tool('decision_point',{reason:'Danger while collecting; inspect current state before resuming the preserved gather handle.'});
              let status=tool('action_query',{handle:collected.handle,wait_ms:1000});
              if(status.status==='completed') {collected=status;break;}
              if(status.status==='paused') tool('action_control',{handle:collected.handle,command:'resume'});
              else if(status.status!=='running') throw new Error(JSON.stringify(status));
              collected={handle:collected.handle};
            }
            const after=tool('get_state').inventory_slots.filter(i=>i.type===itemType).reduce((n,i)=>n+i.stack,0);
            if(after<=before) throw new Error('drop not verified in inventory: '+JSON.stringify(collected));
            tool('action_control',{handle:original.handle,command:'resume'});
            let status;
            for(let i=0;i<25;i++) {
              status=tool('action_query',{handle:original.handle,wait_ms:1000});
              if(status.status!=='running') break;
            }
            if(status.status!=='completed') throw new Error('original goal not resumed: '+JSON.stringify(status));
            return {item_type:itemType,gained:after-before,original_handle:original.handle,judgment:decision,returned:true};
            """.Replace("GOAL_X", (originX + (Natural ? 3 : 18)).ToString()).Replace("GOAL_Y", (floor - 1).ToString())
                .Replace("NPC_TYPE", (Natural ? NPCID.Hornet : NPCID.BlueSlime).ToString());
        string reviewCode = code[..code.IndexOf("function chooseMovement", StringComparison.Ordinal)]
            + "tool('decision_point',{reason:'Review local terrain before selecting combat movement; retain original action handle.'});";
        var saved = await Call("script_program", new JsonObject { ["operation"] = "save", ["name"] = "combat_loot_return", ["expected_revision"] = 0, ["code"] = reviewCode });
        var review = await Call("run_script", new JsonObject { ["program"] = "combat_loot_return", ["expected_revision"] = saved["revision"].DeepClone(),
            ["workflow"] = "behavior-trial", ["event_delivery"] = "script" });
        if (review["value"]?["status"]?.ToString() != "yield" || review["value"]?["wake"] == null)
            throw new Exception("review did not yield: " + review.ToJsonString());
        if (Environment.GetEnvironmentVariable("TB_BEHAVIOR_REVIEW") == "1") saved = await Review(review, code);
        else saved = await Call("script_program", new JsonObject { ["operation"] = "save", ["name"] = "combat_loot_return", ["expected_revision"] = saved["revision"].DeepClone(), ["code"] = code });
        var result = await Call("run_script", new JsonObject { ["program"] = "combat_loot_return", ["expected_revision"] = saved["revision"].DeepClone(),
            ["workflow"] = "behavior-trial", ["event_delivery"] = "script" });
        if (result["error"] != null || result["value"]?["returned"]?.GetValue<bool>() != true) throw new Exception("behavior did not complete: " + result.ToJsonString());
        var finalState = await Call("get_state", new JsonObject());
        await OnMain(() =>
        {
            if (Main.LocalPlayer.dead || movingAttackFrames == 0 || result["value"]?["gained"]?.GetValue<int>() <= 0)
                throw new Exception("native movement/combat/loot evidence missing");
            var originalActions = finalState["action_tasks"].AsArray().Where(a => a["tool"]?.ToString() == "nav_to").ToArray();
            if (originalActions.Length != 1 || originalActions[0]["handle"]?.ToString() != result["value"]?["original_handle"]?.ToString()
                || result["program"]?["revision"]?.GetValue<int>() < 2)
                throw new Exception("edited program replayed the original task or did not use revision 2");
            return true;
        });
    }
}
