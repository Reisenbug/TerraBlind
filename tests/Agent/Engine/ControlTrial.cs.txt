namespace Engine;

public sealed class ControlTrial : ModSystem
{
    readonly ConcurrentQueue<Action> queue = new();
    bool enabled, started, opened, finished;
    int frames, sandX, floor, sandPickAttempts;
    int interactionNpc = -1;
    Task trial;

    public override void Load()
    {
        enabled = !Main.dedServ && Terraria.Program.LaunchParameters.ContainsKey("-agentcontrolcheck");
        if (enabled && Environment.GetEnvironmentVariable("TB_NATIVE_PRIMITIVES") == "1")
            PrimitiveBindingProbe.Binding = KeybindLoader.RegisterKeybind(Mod, "PrimitiveInputProbe", "F10");
        if (enabled) { IL_Main.DoUpdate += Focus; On_Main.Update += OpenWorld; On_Player.PickTile += ObservePick; }
    }
    public override void Unload() { if (enabled) { IL_Main.DoUpdate -= Focus; On_Main.Update -= OpenWorld; On_Player.PickTile -= ObservePick; } }
    void ObservePick(On_Player.orig_PickTile original, Player player, int x, int y, int power)
    {
        if (started && player == Main.LocalPlayer && x >= sandX && x < sandX + 4 && y >= floor - 7 && y < floor)
        {
            sandPickAttempts++;
            Console.WriteLine($"CONTROL_SAND_PICK frame={frames} target=({x},{y}) player={player.position} power={power} attempt={sandPickAttempts}");
        }
        original(player, x, y, power);
    }
    void OpenWorld(On_Main.orig_Update original, Main game, GameTime time)
    {
        original(game, time);
        if (opened || Main.menuMode != 0) return;
        opened = true;
        if (Terraria.Social.SocialAPI.Cloud != null) Terraria.Social.SocialAPI.Cloud.EnabledByDefault = false;
        WorldGen.clearWorld();
        WorldFile.GetAllMetadata(Path.Combine(Main.SavePath, "Worlds", "Checks.wld"), false).SetAsActive();
        var file = PlayerFileData.CreateAndSave(new Player { name = "Control trial" });
        Check(!file.IsCloudSave && file.Path.StartsWith(Main.SavePath, StringComparison.Ordinal), "player_not_isolated");
        Player.LoadPlayer(file.Path, false).SetAsActive();
        WorldGen.playWorld();
    }
    static void Focus(ILContext il)
    {
        var cursor = new ILCursor(il);
        if (!cursor.TryGotoNext(MoveType.After, instruction => instruction.MatchStsfld<Main>("hasFocus")))
            throw new InvalidOperationException("focus boundary missing");
        cursor.EmitDelegate<Action>(() => Main.hasFocus = true);
    }
    Task<T> OnMain<T>(Func<T> body)
    {
        var result = new TaskCompletionSource<T>(TaskCreationOptions.RunContinuationsAsynchronously);
        queue.Enqueue(() => { try { result.SetResult(body()); } catch (Exception error) { result.SetException(error); } });
        return result.Task;
    }
    static void Check(bool valid, string message) { if (!valid) throw new Exception(message); }
    static async Task<JsonObject> Call(string tool, JsonObject args)
    {
        var result = JsonNode.Parse(await AgentTools.Execute(tool, args, default)).AsObject();
        Console.WriteLine("CONTROL_TOOL " + tool + " " + result.ToJsonString());
        return result;
    }
    async Task DelayFrames(int count)
    {
        int until = await OnMain(() => frames + count);
        while (await OnMain(() => frames < until)) await Task.Delay(10);
    }

    public override void PostUpdateEverything()
    {
        if (!enabled || finished) return;
        Main.soundVolume = Main.musicVolume = Main.ambientVolume = 0;
        Main.autoPause = false;
        while (queue.TryDequeue(out var work)) work();
        if (Main.gameMenu || !Main.LocalPlayer.active) return;
        frames++;
        foreach (var npc in Main.npc)
            if (npc.whoAmI != interactionNpc) npc.active = false;
        if (!started && frames >= 60)
        {
            started = true;
            trial = Task.Run(Run);
        }
        if (frames % 60 == 0)
        {
            var p = Main.LocalPlayer;
            Console.WriteLine($"CONTROL_SAMPLE frame={frames} pos={p.position} hp={p.statLife} dead={p.dead} nav={RecedingNav.Active} act={ActExecutor.IsActive} use={p.controlUseItem}");
        }
        if (trial?.IsCompleted == true)
        {
            if (trial.IsFaulted) Console.WriteLine("CONTROL_TAKEOVER_FAILED " + trial.Exception.GetBaseException());
            else Console.WriteLine("CONTROL_TAKEOVER_COMPLETE native_engine=true model_calls=0");
            File.WriteAllText(Path.Combine(Main.SavePath, "agent-engine-result.json"),
                new JsonObject { ["ok"] = !trial.IsFaulted, ["detail"] = trial.Exception?.GetBaseException().ToString() ?? "control takeover verified" }.ToJsonString());
            finished = true;
        }
    }

    async Task Run()
    {
        if (Environment.GetEnvironmentVariable("TB_SOCIAL_INTERACTIONS") == "1")
        {
            await SocialInteractions();
            return;
        }
        if (Environment.GetEnvironmentVariable("TB_CONSTRUCTION_GUARD") == "1")
        {
            await ConstructionAndGuard();
            return;
        }
        string containerMode = Environment.GetEnvironmentVariable("TB_CONTAINER_CHAIN");
        if (containerMode is "script" or "live")
        {
            await ContainerChain(containerMode == "live");
            return;
        }
        if (Environment.GetEnvironmentVariable("TB_NATIVE_PRIMITIVES") == "1")
        {
            await NativePrimitives();
            return;
        }
        await OnMain(() =>
        {
            Config.I.AgentEnabled = true; Config.I.AiClient = false;
            NativeAgent.Stop(); Halt.All(); Manual.On = false;
            Combat.Enabled = Dodge.Enabled = false;
            AgentPermissions.Reset();
            var p = Main.LocalPlayer;
            p.ConsumedLifeCrystals = 15; p.statLife = p.statLifeMax = p.statLifeMax2 = 400;
            foreach (var item in p.inventory) item.TurnToAir();
            p.inventory[0].SetDefaults(ItemID.CopperPickaxe);
            p.inventory[1].SetDefaults(ItemID.Starfury);
            p.inventory[2].SetDefaults(ItemID.DirtBlock); p.inventory[2].stack = 999;
            p.inventory[3].SetDefaults(ItemID.Wood); p.inventory[3].stack = 50;
            p.selectedItem = 0;
            int x = (int)p.Center.X / 16;
            floor = (int)p.Bottom.Y / 16 + 3; sandX = x + 4;
            for (int tx = x - 15; tx <= x + 18; tx++)
            {
                for (int ty = floor - 12; ty < floor; ty++)
                { Main.tile[tx, ty].ClearTile(); Main.tile[tx, ty].LiquidAmount = 0; }
                Main.tile[tx, floor].ResetToType(TileID.Stone);
            }
            for (int tx = sandX; tx <= sandX + 3; tx++)
                for (int ty = floor - 7; ty < floor; ty++) Main.tile[tx, ty].ResetToType(TileID.Sand);
            p.position = new Vector2(x * 16, floor * 16 - p.height); p.velocity = Vector2.Zero;
            p.fallStart = p.fallStart2 = (int)p.position.Y / 16;
            Console.WriteLine($"CONTROL_FIXTURE sand=({sandX},{floor - 7})..({sandX + 3},{floor - 1}) player={p.position}");
            Console.WriteLine($"CONTROL_MATERIALS platform_slot={NavCoordinator.FindPlatformSlot(p)} wood={p.inventory[3].stack} dirt={p.inventory[2].stack}");
            return true;
        });
        var action = await Call("action_start", new JsonObject { ["tool"] = "nav_to",
            ["args"] = new JsonObject { ["x"] = sandX + 8, ["y"] = floor - 1 } });
        Check(action["handle"] != null, "navigation did not start");
        string handle = action["handle"].ToString();
        int excavationDeadline = await OnMain(() => frames + 900);
        while (await OnMain(() => sandPickAttempts == 0 && frames < excavationDeadline && !Main.LocalPlayer.dead))
            await DelayFrames(1);
        await OnMain(() =>
        {
            Check(sandPickAttempts > 0, "no real sand excavation occurred; takeover scenario not reproduced");
            Check(!Main.LocalPlayer.dead, "player died before takeover");
            return true;
        });
        var claim = await Call("control", new JsonObject { ["operation"] = "claim",
            ["channels"] = new JsonArray("movement", "item_use"), ["scope"] = "task", ["reason"] = "stop excavating under sand" });
        Check(claim["status"]?.ToString() == "ready", "takeover was not ready");
        string claimId = claim["id"].ToString();
        int sandBefore = await OnMain(CountSand);
        Vector2 before = await OnMain(() => Main.LocalPlayer.position);
        await OnMain(() =>
        {
            Combat.Enabled = Dodge.Enabled = true;
            RecedingNav.Start(sandX + 8, floor - 1);
            StateSpacePlanner.ExecuteAsync(sandX + 8, floor - 1);
            FightCoordinator.Start(60);
            return true;
        });
        await DelayFrames(90);
        await OnMain(() =>
        {
            Check(!RecedingNav.Active && !StateSpacePlanner.ExecRunning, "late/default navigation regained control");
            Check(CountSand() == sandBefore, "sand was excavated after takeover");
            Check(Vector2.Distance(before, Main.LocalPlayer.position) < 4, "default controls moved the player");
            Check(!Main.LocalPlayer.controlUseItem, "default attack wrote use input");
            return true;
        });
        var moved = await Call("act", new JsonObject { ["control_claim"] = claimId,
            ["steps"] = JsonNode.Parse("[{\"keys\":[\"left\"],\"until\":{\"frames\":30}}]") });
        Check(moved["error"] == null && moved["outcome"]?.ToString() == "done", "claimed movement did not finish");
        await OnMain(() =>
        { Check(Main.LocalPlayer.position.X < before.X - 30, "claimed action failed to move left"); return true; });
        var localVeto = await Call("control", new JsonObject { ["operation"] = "deny", ["scope"] = "task",
            ["behaviors"] = new JsonArray("mining"), ["reason"] = "do not undermine this sand wall",
            ["region"] = new JsonObject { ["x"] = sandX, ["y"] = floor - 7, ["width"] = 4, ["height"] = 7 } });
        Check(localVeto["status"]?.ToString() == "ready", "local excavation veto not ready");
        var crafted = await Call("craft_into_inventory", new JsonObject { ["name"] = "WoodPlatform", ["amount"] = 20, ["control_claim"] = claimId });
        Check(crafted["ok"]?.ToString() == "true", "could not prepare platforms for the alternate route");
        await OnMain(() =>
        { Check(NavCoordinator.FindPlatformSlot(Main.LocalPlayer) >= 0, "crafted platforms absent from real inventory"); return true; });
        await Call("control", new JsonObject { ["operation"] = "release", ["id"] = claimId });
        int picksBeforeResume = await OnMain(() => sandPickAttempts);
        var retried = await Call("nav_to", new JsonObject { ["x"] = sandX + 8, ["y"] = floor - 1 }).WaitAsync(TimeSpan.FromSeconds(60));
        Check(retried["done"]?.ToString() == "true", "navigation failed to reach the original destination after local veto");
        await OnMain(() =>
        {
            var p = Main.LocalPlayer;
            Check(AgentTools.ControlAllows(ControlBehavior.Navigation, ControlChannels.All), "local veto disabled all navigation");
            Check(sandPickAttempts == picksBeforeResume && CountSand() == sandBefore,
                "resumed navigation repeated rejected excavation");
            Check(!p.dead && p.Center.X > (sandX + 4) * 16, "player did not cross the sand wall alive");
            return true;
        });
        Console.WriteLine($"CONTROL_ASSERTIONS paused_handle={handle} sand_unchanged={sandBefore} explicit_move=true native_restart_rejected=true");
        Console.WriteLine("CONTROL_REGION_ASSERTIONS original_destination_reached=true rejected_excavation_not_repeated=true navigation_enabled=true");
    }

    async Task SocialInteractions()
    {
        int x = await OnMain(() =>
        {
            Config.I.AgentEnabled = true; Config.I.AiClient = false;
            NativeAgent.Stop(); AgentTools.NewTask(); Halt.All(); Manual.On = false;
            var p = Main.LocalPlayer;
            int left = (int)p.Center.X / 16;
            floor = (int)p.Bottom.Y / 16 + 5;
            for (int tx = left - 8; tx <= left + 80; tx++)
            for (int ty = floor - 8; ty <= floor + 16; ty++)
            {
                var tile = Main.tile[tx, ty];
                tile.ClearEverything();
                if (ty >= floor && !(tx >= left + 47 && tx <= left + 76 && ty < floor + 14))
                    tile.ResetToType(TileID.Stone);
                else if (ty >= floor) tile.LiquidAmount = byte.MaxValue;
            }
            foreach (int origin in new[] { left + 2, left + 28 })
            {
                for (int dx = 0; dx < 3; dx++)
                for (int dy = 0; dy < 4; dy++)
                {
                    var tile = Main.tile[origin + dx, floor - 4 + dy];
                    tile.ResetToType(TileID.TeleportationPylon);
                    tile.TileFrameX = (short)(8 * 54 + dx * 18);
                    tile.TileFrameY = (short)(dy * 18);
                }
                TETeleportationPylon.Place(origin, floor - 4);
            }
            p.position = new Vector2(left * 16, floor * 16 - p.height);
            p.velocity = Vector2.Zero;
            p.fallStart = p.fallStart2 = (int)p.position.Y / 16;
            Main.mouseItem.TurnToAir();
            return left;
        });
        await DelayFrames(8);
        var pylons = await Call("pylon_list", new JsonObject());
        Check(pylons["near_pylon"]?.GetValue<bool>() == true && pylons["pylons"].AsArray().Count(p =>
            p["x"]?.GetValue<int>() is int px && (px == x + 2 || px == x + 28)) == 2, "native_pylons_not_registered");
        var teleported = await Call("pylon_use", new JsonObject { ["x"] = x + 28, ["y"] = floor - 4 });
        Check(teleported["outcome"]?.ToString() == "arrived"
            && await OnMain(() => Math.Abs(Main.LocalPlayer.Center.X / 16 - (x + 28)) < 3), "native_pylon_arrival_not_observed");

        long coinsBefore = await OnMain(() =>
        {
            var p = Main.LocalPlayer;
            interactionNpc = NPC.NewNPC(new EntitySource_Misc("reforge-fixture"), (x + 30) * 16, floor * 16, NPCID.GoblinTinkerer);
            Check(interactionNpc < Main.maxNPCs, "goblin_not_spawned");
            p.SetTalkNPC(interactionNpc);
            Main.InReforgeMenu = true; Main.playerInventory = true;
            p.inventory[10].SetDefaults(ItemID.Aglet);
            p.inventory[51].SetDefaults(ItemID.GoldCoin); p.inventory[51].stack = 50;
            return Utils.CoinsCount(out _, p.inventory);
        });
        async Task Click(string section, int index)
        {
            var args = new JsonObject { ["section"] = section, ["slot"] = index };
            var state = await Call("slot_state", args);
            args["button"] = "left"; args["expected_item"] = state["item"].DeepClone();
            args["expected_cursor"] = state["cursor"].DeepClone();
            Check((await Call("slot_click", args))["outcome"]?.ToString() == "changed", "native_reforge_slot_click_failed");
        }
        await Click("inventory", 10);
        await Click("reforge", 0);
        var quote = await Call("reforge", new JsonObject { ["action"] = "quote" });
        int cost = quote["cost_copper"]?.GetValue<int>() ?? 0;
        Check(cost > 0, "native_reforge_price_missing");
        var args = new JsonObject { ["action"] = "roll", ["expected_item"] = quote["item"].DeepClone(), ["max_cost"] = cost - 1 };
        var expensive = await Call("reforge", args);
        Check(expensive["error"]?.ToString() == "reforge_budget_exceeded"
            && await OnMain(() => Utils.CoinsCount(out _, Main.LocalPlayer.inventory) == coinsBefore), "reforge_spent_above_budget");
        args["max_cost"] = cost;
        var reforged = await Call("reforge", args);
        Check(reforged["spent_copper"]?.GetValue<int>() == cost && await OnMain(() =>
            Main.reforgeItem.prefix > 0 && Utils.CoinsCount(out _, Main.LocalPlayer.inventory) == coinsBefore - cost), "native_reforge_not_charged");
        await Click("reforge", 0);
        await Click("inventory", 10);
        await OnMain(() =>
        {
            var p = Main.LocalPlayer;
            p.SetTalkNPC(-1); Main.InReforgeMenu = Main.playerInventory = false;
            interactionNpc = -1;
            p.position = new Vector2((x + 44) * 16, floor * 16 - p.height);
            p.velocity = Vector2.Zero;
            p.fallStart = p.fallStart2 = (int)p.position.Y / 16;
            p.inventory[0].SetDefaults(ItemID.GoldenFishingRod);
            p.inventory[1].SetDefaults(ItemID.MasterBait); p.inventory[1].stack = 50;
            return true;
        });
        var cast = await Call("fish", new JsonObject { ["action"] = "cast", ["slot"] = 0, ["x"] = x + 55, ["y"] = floor + 1 });
        Check(cast["outcome"]?.ToString() == "cast", "native_fishing_cast_not_observed");
        var reeled = await Call("fish", new JsonObject { ["action"] = "wait_reel", ["slot"] = 0, ["wait_seconds"] = 65 });
        Check(reeled["outcome"]?.ToString() == "reeled", "native_bite_or_reel_not_observed");
        Console.WriteLine("SOCIAL_INTERACTIONS_COMPLETE pylon_arrived=true reforge_spent=" + cost + " native_bite_reeled=true model_calls=0");
        File.WriteAllText(Path.Combine(Main.SavePath, "social-interactions.json"), new JsonObject
            { ["pylon"] = teleported, ["reforge"] = reforged, ["cast"] = cast, ["reel"] = reeled }.ToJsonString());
    }

    async Task ConstructionAndGuard()
    {
        int x = await OnMain(() =>
        {
            Config.I.AgentEnabled = true; Config.I.AiClient = false;
            NativeAgent.Stop(); AgentTools.NewTask(); Halt.All(); Manual.On = false;
            Combat.Enabled = Dodge.Enabled = false;
            var p = Main.LocalPlayer;
            int left = (int)p.Center.X / 16;
            floor = (int)p.Bottom.Y / 16 + 3;
            for (int tx = left - 9; tx <= left + 23; tx++)
            for (int ty = floor - 7; ty <= floor + 2; ty++)
            {
                Main.tile[tx, ty].ClearEverything();
                if (ty >= floor) Main.tile[tx, ty].ResetToType(TileID.Stone);
            }
            p.position = new Vector2(left * 16, floor * 16 - p.height);
            p.velocity = Vector2.Zero;
            p.fallStart = p.fallStart2 = (int)p.position.Y / 16;
            foreach (var item in p.inventory) item.TurnToAir();
            p.inventory[0].SetDefaults(ItemID.CopperPickaxe);
            p.inventory[1].SetDefaults(ItemID.WoodenHammer);
            p.inventory[2].SetDefaults(ItemID.Wood); p.inventory[2].stack = 30;
            Main.tile[left - 4, floor - 1].ResetToType(TileID.WoodBlock);
            Main.tile[left - 3, floor - 1].ResetToType(TileID.WoodBlock);
            var half = Main.tile[left - 3, floor - 1];
            half.IsHalfBlock = true;
            return left;
        });
        var pattern = await Call("block_region", new JsonObject { ["x"] = x - 4, ["y"] = floor - 1, ["width"] = 2, ["height"] = 1 });
        Check(pattern["cells"]?[0]?["shape"]?.ToString() == "full" && pattern["cells"]?[1]?["shape"]?.ToString() == "half", "capture_wrong_shapes");
        var saved = await Call("script_program", new JsonObject { ["operation"] = "save", ["name"] = "test_half_block_pattern",
            ["expected_revision"] = 0,
            ["code"] = "return skill('build_block_pattern',0,{x:input.x,y:input.y,pattern:" + pattern.ToJsonString() + "});" });
        int revision = saved["revision"]?.GetValue<int>() ?? 0;
        Check(revision > 0, "pattern_not_saved");
        for (int i = 0; i < 2; i++)
        {
            int destination = x + 3;
            if (i > 0)
            {
                destination = x + 9;
                await OnMain(() =>
                {
                    Main.LocalPlayer.position.X = (destination - 3) * 16;
                    Main.LocalPlayer.velocity = Vector2.Zero;
                    return true;
                });
            }
            var built = await Call("run_script", new JsonObject { ["code"] = $"return skill('test_half_block_pattern',{revision},"
                + new JsonObject { ["x"] = destination, ["y"] = floor - 1 }.ToJsonString() + ");" });
            Check(built["value"]?["verified"]?.GetValue<bool>() == true, "pattern_build_failed: " + built);
            Check(await OnMain(() => Main.tile[destination, floor - 1].TileType == TileID.WoodBlock
                && !Main.tile[destination, floor - 1].IsHalfBlock && Main.tile[destination + 1, floor - 1].IsHalfBlock),
                "pattern_world_geometry_mismatch");
        }
        Console.WriteLine("BLOCK_PATTERN_COMPLETE saved=true copies=2 full_and_half=true");

        int plateX = x + 15, plateY = floor - 1, triggers = 0;
        void ObserveSwitch(On_Wiring.orig_HitSwitch original, int tileX, int tileY)
        {
            if (tileX == plateX && tileY == plateY) triggers++;
            original(tileX, tileY);
        }
        await OnMain(() =>
        {
            WorldGen.PlaceTile(plateX, plateY, TileID.PressurePlates, mute: true, forced: true);
            Check(Main.tile[plateX, plateY].HasTile && Main.tile[plateX, plateY].TileType == TileID.PressurePlates, "plate_missing");
            var plate = Main.tile[plateX, plateY];
            plate.RedWire = true;
            On_Wiring.HitSwitch += ObserveSwitch;
            return true;
        });
        try
        {
            var result = await Call("run_script", new JsonObject { ["code"] = "return skill('travel_disarm_plates',0,"
                + new JsonObject { ["x"] = x + 20, ["y"] = plateY, ["allow_removal"] = true }.ToJsonString() + ");" });
            var actual = await OnMain(() => new JsonObject { ["x"] = (int)(Main.LocalPlayer.Center.X / 16),
                ["triggers"] = triggers, ["plate_removed"] = !Main.tile[plateX, plateY].HasTile });
            File.WriteAllText(Path.Combine(Main.SavePath, "construction-guard.json"), new JsonObject
                { ["travel"] = result.DeepClone(), ["actual"] = actual, ["copies"] = 2 }.ToJsonString());
            Check(result["value"]?["done"]?.GetValue<bool>() == true && actual["x"].GetValue<int>() >= x + 19
                && actual["plate_removed"].GetValue<bool>() && triggers == 0, "plate_crossing_failed: " + actual.ToJsonString());
            Console.WriteLine("PRESSURE_PLATE_GUARD_COMPLETE triggers=0 plate_removed=true real_navigation=true");
        }
        finally
        {
            await OnMain(() => { On_Wiring.HitSwitch -= ObserveSwitch; return true; });
        }
    }

    async Task ContainerChain(bool live)
    {
        var chest = await OnMain(() =>
        {
            Config.I.AgentEnabled = true;
            Config.I.AgentFreezeWhileThinking = false;
            NativeAgent.Stop(); AgentTools.NewTask(); Halt.All(); Manual.On = false;
            var p = Main.LocalPlayer;
            foreach (var item in p.inventory) item.TurnToAir();
            Main.mouseItem.TurnToAir();
            int x = (int)p.Center.X / 16;
            int y = (int)p.Bottom.Y / 16 + 3;
            for (int tx = x - 4; tx <= x + 8; tx++)
            for (int ty = y - 5; ty <= y; ty++)
            {
                Main.tile[tx, ty].ClearEverything();
                if (ty == y) WorldGen.PlaceTile(tx, ty, TileID.Stone, mute: true, forced: true);
            }
            p.position = new Vector2(x * 16, y * 16 - p.height);
            p.velocity = Vector2.Zero;
            p.fallStart = p.fallStart2 = (int)p.position.Y / 16;
            int id = WorldGen.PlaceChest(x + 3, y - 1, TileID.Containers, false, 0);
            Check(id >= 0, "container_chain_fixture_placement_failed");
            var fixture = Main.chest[id];
            fixture.item[0].SetDefaults(ItemID.Wood);
            fixture.item[0].stack = 10;
            fixture.item[1].SetDefaults(ItemID.Torch);
            fixture.item[1].stack = 5;
            return (Id: id, fixture.x, fixture.y);
        });
        JsonObject Snapshot()
        {
            var p = Main.LocalPlayer;
            var contents = Main.chest[chest.Id].item;
            return new JsonObject
            {
                ["wood_in_chest"] = contents.Where(i => i.type == ItemID.Wood).Sum(i => i.stack),
                ["torches_in_chest"] = contents[1].type == ItemID.Torch ? contents[1].stack : 0,
                ["workbench_in_fifth_hotbar_slot"] = p.inventory[4].type == ItemID.WorkBench ? p.inventory[4].stack : 0,
                ["cursor_type"] = Main.mouseItem.type, ["cursor_empty"] = Main.mouseItem.IsAir,
                ["open_container"] = p.chest
            };
        }
        bool Complete(JsonObject state) => state["wood_in_chest"].GetValue<int>() == 0
            && state["torches_in_chest"].GetValue<int>() == 5
            && state["workbench_in_fifth_hotbar_slot"].GetValue<int>() == 1 && state["cursor_empty"].GetValue<bool>();
        var before = await OnMain(Snapshot);
        var clock = System.Diagnostics.Stopwatch.StartNew();
        var timeline = new JsonArray();
        JsonObject result = null;
        if (live)
        {
            await OnMain(() =>
            {
                Config.I.AgentUseCodex = true;
                Config.I.AgentCodexModel = "gpt-6.2-sol";
                NativeAgent.Submit("打开面前的箱子，取10个木头，火把留在箱里。做一个工作台，放到第5个快捷栏槽位。");
                return true;
            });
            JsonObject last = null;
            while (clock.Elapsed.TotalSeconds < 75)
            {
                var sample = await OnMain(() => new JsonObject { ["state"] = Snapshot(), ["status"] = NativeAgent.Status });
                if (!JsonNode.DeepEquals(last, sample))
                    timeline.Add(new JsonObject { ["elapsed_ms"] = clock.ElapsedMilliseconds, ["sample"] = sample.DeepClone() });
                last = sample;
                if (sample["status"]?.ToString() is "idle" or "failed" or "waiting for answer") break;
                await Task.Delay(50);
            }
            await OnMain(() => { NativeAgent.Stop(); return true; });
        }
        else
        {
            string script = $$"""
                function checked(name, args) {
                  const result = tool(name, args);
                  if (('ok' in result && result.ok === false) || ('error' in result && result.error)) throw new Error(name + ': ' + JSON.stringify(result));
                  return result;
                }
                function click(section, slot, expectedContainer) {
                  const address = {section, slot};
                  const observed = checked('slot_state', address);
                  const args = {section, slot, button:'left', expected_item:observed.item, expected_cursor:observed.cursor};
                  if (section === 'container') args.expected_container = expectedContainer;
                  const clicked = checked('slot_click', args);
                  if (clicked.outcome !== 'changed') throw new Error('slot did not change: ' + section + '/' + slot);
                }
                const opened = checked('interact', {x:{{chest.x}}, y:{{chest.y}}});
                if (opened.outcome !== 'opened' || opened.chest !== {{chest.Id}}) throw new Error('fixture chest did not open');
                click('container', 0, opened.chest);
                click('inventory', 10);
                const recipes = checked('craft_recipes', {item_type:{{ItemID.WorkBench}}}).recipes;
                if (!recipes.length) throw new Error('work bench recipe not available');
                checked('craft_recipe', {index:recipes[0].index, expected_type:{{ItemID.WorkBench}}});
                click('inventory', 4);
                return {ok:true};
                """;
            result = await Call("run_script", new JsonObject { ["code"] = script });
        }
        var after = await OnMain(Snapshot);
        var report = new JsonObject { ["mode"] = live ? "live" : "script", ["elapsed_ms"] = clock.ElapsedMilliseconds,
            ["before"] = before, ["after"] = after, ["script"] = result, ["timeline"] = timeline };
        File.WriteAllText(Path.Combine(Main.SavePath, "container-chain.json"), report.ToJsonString());
        Check(Complete(after) && (live || result["ok"]?.GetValue<bool>() == true
            && result["value"]?["ok"]?.GetValue<bool>() == true), "container_chain_failed: " + after.ToJsonString());
        Console.WriteLine("CONTAINER_CHAIN_COMPLETE " + report.ToJsonString());
    }

    async Task NativePrimitives()
    {
        await OnMain(() =>
        {
            Config.I.AgentEnabled = true; Config.I.AiClient = false;
            Manual.On = false;
            AgentTools.NewTask();
            var player = Main.LocalPlayer;
            player.inventory[7].SetDefaults(ItemID.Wood);
            player.inventory[7].stack = 10;
            player.inventory[12].TurnToAir();
            Main.mouseItem.TurnToAir();
            return true;
        });
        var inventory = await Call("game_input", new JsonObject { ["action"] = "inventory", ["enabled"] = true });
        Check(inventory["inventory_open"]?.GetValue<bool>() == true, "primitive_inventory_did_not_open");
        var selected = await Call("game_input", new JsonObject { ["action"] = "hotbar", ["index"] = 7 });
        Check(selected["selected_slot"]?.GetValue<int>() == 7, "primitive_hotbar_slot_not_selected");
        var recipes = await Call("craft_recipes", new JsonObject { ["item_type"] = (int)ItemID.WorkBench });
        Check(recipes["recipes"] is JsonArray { Count: > 0 }, "primitive_workbench_recipe_not_found");
        var crafted = await Call("craft_recipe", new JsonObject
        {
            ["index"] = recipes["recipes"][0]["index"].DeepClone(), ["expected_type"] = (int)ItemID.WorkBench
        });
        Check(crafted["produced"]?.GetValue<int>() == 1 && await OnMain(() => Main.mouseItem.type == ItemID.WorkBench
            && Main.LocalPlayer.inventory[7].IsAir), "primitive_native_craft_cursor_or_materials_incorrect");
        var args = new JsonObject { ["section"] = "inventory", ["slot"] = 12 };
        var state = await Call("slot_state", args);
        args["button"] = "left";
        args["expected_item"] = state["item"].DeepClone();
        args["expected_cursor"] = state["cursor"].DeepClone();
        var placed = await Call("slot_click", args);
        Check(placed["outcome"]?.ToString() == "changed" && await OnMain(() => Main.mouseItem.IsAir
            && Main.LocalPlayer.inventory[12].type == ItemID.WorkBench), "primitive_slot_click_did_not_transfer_crafted_item");
        var bindings = await Call("mod_input_list", new JsonObject { ["mod"] = Mod.Name });
        string key = Mod.Name + "/PrimitiveInputProbe";
        Check(bindings["keys"].AsArray().Any(item => item.ToString() == key), "primitive_registered_mod_binding_missing");
        var dispatched = await Call("mod_input", new JsonObject { ["key"] = key, ["frames"] = 3 });
        Check(dispatched["outcome"]?.ToString() == "dispatched" && await OnMain(() =>
            PrimitiveBindingProbe.Pressed == 1 && PrimitiveBindingProbe.Held == 3 && PrimitiveBindingProbe.Released == 1),
            "primitive_mod_trigger_did_not_deliver_press_hold_release");
        Console.WriteLine("NATIVE_PRIMITIVES_COMPLETE native_engine=true model_calls=0");
    }

    int CountSand()
    {
        int count = 0;
        for (int x = sandX; x <= sandX + 3; x++)
        for (int y = floor - 7; y < floor; y++)
            if (Main.tile[x, y].HasTile && Main.tile[x, y].TileType == TileID.Sand) count++;
        return count;
    }
}

public sealed class PrimitiveBindingProbe : ModPlayer
{
    internal static ModKeybind Binding;
    internal static int Pressed, Held, Released;

    public override void ProcessTriggers(Terraria.GameInput.TriggersSet triggersSet)
    {
        if (Binding == null) return;
        if (Binding.JustPressed) Pressed++;
        if (Binding.Current) Held++;
        if (Binding.JustReleased) Released++;
    }
}
