public sealed class HornetTrial : ModSystem
{
    public static bool Enabled => Terraria.Program.LaunchParameters.ContainsKey("-agenthornetcheck");
    bool Obstacle => Terraria.Program.LaunchParameters.ContainsKey("-agenthornetwall");
    public static int Target = -1;
    public static bool Killed;
    readonly JsonArray rounds = new();
    readonly HashSet<(int, int)> hostileProjectiles = new();
    int x, floor, tick, spawnedAt, hpBefore, knivesBefore, shots, moves, damageTaken, previousHp;
    int blockedFrames, blockedAttackFrames;
    bool started, finished;
    double distance;
    Vector2 previousPosition;

    public override void PostUpdateEverything()
    {
        if (!Enabled || finished || Main.gameMenu || !Main.LocalPlayer.active) return;
        try
        {
            var p = Main.LocalPlayer;
            if (!started)
            {
                started = true;
                Config.I.AgentEnabled = true; Config.I.AiClient = false;
                AgentPermissions.Attack = AgentPermissions.Dodge = true;
                Combat.Enabled = Dodge.Enabled = false;
                x = (int)p.Center.X / 16; floor = (int)p.Bottom.Y / 16;
                for (int tx = x - 22; tx <= x + 22; tx++)
                for (int ty = floor - 13; ty <= floor + 2; ty++)
                {
                    var tile = Main.tile[tx, ty]; tile.ClearEverything();
                    if (tx == x - 22 || tx == x + 22 || ty == floor - 13 || ty >= floor)
                    { tile.HasTile = true; tile.TileType = TileID.Stone; }
                }
                foreach (var npc in Main.npc) npc.active = false;
                if (Obstacle)
                    for (int tx = x + 7; tx <= x + 8; tx++)
                    for (int ty = floor - 5; ty < floor; ty++)
                    { var tile = Main.tile[tx, ty]; tile.HasTile = true; tile.TileType = TileID.Stone; }
                foreach (var projectile in Main.projectile) projectile.active = false;
                foreach (var item in Main.item) item.active = false;
                foreach (var item in p.inventory) item.TurnToAir();
                p.inventory[0].SetDefaults(ItemID.ThrowingKnife); p.inventory[0].stack = 500;
                p.ConsumedLifeCrystals = 10; p.statLifeMax = p.statLife = 300;
                p.position = new Vector2(x * 16, floor * 16 - p.height); p.velocity = Vector2.Zero;
                p.immune = false; p.immuneTime = 0;
                Main.playerInventory = false;
                previousHp = p.statLife; previousPosition = p.position;
                Console.WriteLine($"HORNET_TRIAL_START arena=42x12 knives=500 hp=300 immunity=false sequential=3 model_calls=0 obstacle={Obstacle}");
                Spawn(p); return;
            }
            tick++;
            distance += Vector2.Distance(previousPosition, p.position); previousPosition = p.position;
            if (p.controlLeft || p.controlRight || p.controlJump) moves++;
            damageTaken += Math.Max(0, previousHp - p.statLife); previousHp = p.statLife;
            foreach (var projectile in Main.projectile)
                if (projectile.active && projectile.hostile) hostileProjectiles.Add((projectile.whoAmI, projectile.identity));
            shots = Math.Max(shots, knivesBefore - p.inventory.Where(i => i.type == ItemID.ThrowingKnife).Sum(i => i.stack));
            if (p.dead) { Record(p, "player_dead"); Finish(false, "player_dead"); return; }
            if (p.Center.X < (x - 22) * 16 || p.Center.X > (x + 22) * 16 || p.Center.Y < (floor - 13) * 16)
            { Record(p, "left_arena"); Finish(false, "left_arena"); return; }
            int elapsed = tick - spawnedAt;
            if (elapsed >= 120 && Main.npc[Target].active && !Collision.CanHitLine(p.Center, 4, 4, Main.npc[Target].Center, 4, 4))
            { blockedFrames++; if (p.controlUseItem) blockedAttackFrames++; }
            if (elapsed == 120) FightCoordinator.Start(60, new[] { (int)NPCID.Hornet });
            if (Killed)
            {
                Record(p, "killed"); FightCoordinator.Stop();
                if (rounds.Count == 3) { Finish(true, "three_hornets_killed"); return; }
                Spawn(p); return;
            }
            if (!Main.npc[Target].active) { Record(p, "target_lost"); Finish(false, "target_lost"); return; }
            if (elapsed >= 1800) { Record(p, "timeout"); Finish(false, "timeout"); }
        }
        catch (Exception error) { Finish(false, error.ToString()); }
    }

    void Spawn(Player p)
    {
        int side = rounds.Count % 2 == 0 ? 1 : -1;
        Target = NPC.NewNPC(new EntitySource_Misc("HornetTrial"), (x + side * 17) * 16, (floor - 6) * 16, NPCID.Hornet);
        if (Target >= Main.maxNPCs) throw new InvalidOperationException("hornet_spawn_failed");
        Killed = false; spawnedAt = tick; hpBefore = p.statLife;
        knivesBefore = p.inventory.Where(i => i.type == ItemID.ThrowingKnife).Sum(i => i.stack);
        shots = moves = damageTaken = blockedFrames = blockedAttackFrames = 0; hostileProjectiles.Clear(); distance = 0; previousPosition = p.position; previousHp = p.statLife;
        Console.WriteLine($"HORNET_SPAWN round={rounds.Count + 1} id={Target} hp={Main.npc[Target].life} side={side}");
    }

    void Record(Player p, string outcome)
    {
        var result = new JsonObject { ["round"] = rounds.Count + 1, ["outcome"] = outcome,
            ["frames"] = tick - spawnedAt, ["hp_before"] = hpBefore, ["hp_after"] = p.statLife,
            ["damage_taken"] = damageTaken, ["knives_used"] = shots, ["movement_frames"] = moves,
            ["hostile_projectiles"] = hostileProjectiles.Count,
            ["line_blocked_frames"] = blockedFrames, ["blocked_attack_frames"] = blockedAttackFrames,
            ["distance_pixels"] = distance, ["target_hp"] = Main.npc[Target].life };
        rounds.Add(result); Console.WriteLine("HORNET_ROUND " + result.ToJsonString());
    }

    void Finish(bool ok, string outcome)
    {
        finished = true; FightCoordinator.Stop();
        File.WriteAllText(Path.Combine(Main.SavePath, "agent-engine-result.json"), new JsonObject
        { ["ok"] = ok, ["detail"] = "HORNET_TRIAL_COMPLETE " + outcome, ["rounds"] = rounds,
            ["model_calls"] = 0, ["immunity"] = false, ["arena_cells"] = "42x12", ["obstacle"] = Obstacle }.ToJsonString());
    }
}

public sealed class HornetTrialNpcs : GlobalNPC
{
    public override void EditSpawnRate(Player player, ref int spawnRate, ref int maxSpawns)
    { if (HornetTrial.Enabled) { spawnRate = int.MaxValue; maxSpawns = 0; } }
    public override void OnKill(NPC npc)
    { if (HornetTrial.Enabled && npc.whoAmI == HornetTrial.Target && npc.type == NPCID.Hornet) HornetTrial.Killed = true; }
}

public sealed class Driver : ModSystem
{
    bool enabled, opened;
    bool liveRunning;
    DateTime liveStart;
    int phase, ticks;
    int heartX, heartY;
    int expectedCrystals;
    JsonObject gatherRequest;
    Task<string> pending;
    readonly JsonArray results = new();
    delegate void OriginalStart(int x, int y, RecedingNav.Mode mode);
    delegate void StartHook(OriginalStart original, int x, int y, RecedingNav.Mode mode);
    StartHook navigationHook;
    bool WorkflowLive => Terraria.Program.LaunchParameters.ContainsKey("-agentworkflowcheck");
    bool PlanningReadOnly => Terraria.Program.LaunchParameters.ContainsKey("-agentplanningcheck");
    bool HealthLive => WorkflowLive || Terraria.Program.LaunchParameters.ContainsKey("-agenthealthcheck");
    bool hazardApplied;
    int hazardUntil, hazardStock;
    bool ScriptCheck => Terraria.Program.LaunchParameters.ContainsKey("-agentscriptcheck");
    bool ModelScript => File.Exists(Path.Combine(Main.SavePath, "model-script.json"));
    bool ContinuationCheck => Terraria.Program.LaunchParameters.ContainsKey("-agentcontinuationcheck");
    string actionHandle;
    int stockAtPause;
    readonly List<int> modelAnnouncements = new();
    delegate void OriginalSay(string text, bool bot);
    delegate void SayHook(OriginalSay original, string text, bool bot);
    SayHook sayHook;
    bool TeleportCheck => Terraria.Program.LaunchParameters.ContainsKey("-agentteleportcheck");
    int teleportCheckUntil;

    public override void Load()
    {
        enabled = !Main.dedServ && Terraria.Program.LaunchParameters.ContainsKey("-agentenginecheck");
        if (!enabled) return;
        string root = Path.GetFullPath(Main.SavePath);
        if (!root.StartsWith("/tmp/") && !root.StartsWith("/private/tmp/")) throw new InvalidOperationException("temporary_root_required");
        On_Main.Update += Update;
        IL_Main.DoUpdate += Focus;
        if (HealthLive || ScriptCheck)
        {
            navigationHook = NavigateWithoutPlanning;
            MonoModHooks.Add(typeof(RecedingNav).GetMethod("Start", new[] { typeof(int), typeof(int), typeof(RecedingNav.Mode) }), navigationHook);
        }
        if (ModelScript)
        {
            sayHook = (original, text, bot) =>
            {
                if (text.Trim() == "叮") modelAnnouncements.Add(Main.LocalPlayer.inventory
                    .Where(item => item.type == ItemID.LifeCrystal).Sum(item => item.stack));
                original(text, bot);
            };
            MonoModHooks.Add(typeof(AgentChat).GetMethod("Say"), sayHook);
        }
    }

    static void NavigateWithoutPlanning(OriginalStart original, int x, int y, RecedingNav.Mode mode)
    {
        var player = Main.LocalPlayer;
        int previousFall = player.fallStart;
        player.Teleport(new Vector2(x * 16 + 8 - player.width / 2f, y * 16 + 8 - player.height / 2f));
        player.velocity = Vector2.Zero;
        player.fallStart = player.fallStart2 = (int)(player.position.Y / 16f);
        RecedingNav.Active = false;
        RecedingNav.LastStop = "done";
        Console.WriteLine($"AGENT_TEST_NAV x={x} y={y} mode={mode} landed={player.position} fall_before={previousFall} fall_after={player.fallStart} navigation=teleport_stub");
    }

    static string TileStamp(int x, int y) => string.Join(";", Enumerable.Range(-2, 5).SelectMany(dx =>
        Enumerable.Range(-2, 5).Select(dy =>
        {
            var tile = Main.tile[x + dx, y + dy];
            return $"{tile.HasTile},{tile.TileType},{tile.TileFrameX},{tile.TileFrameY},{tile.WallType},{tile.LiquidAmount},{tile.LiquidType}";
        })));

    static void Focus(ILContext il)
    {
        var cursor = new ILCursor(il);
        if (!cursor.TryGotoNext(MoveType.After, instruction => instruction.MatchStsfld<Main>("hasFocus")))
            throw new InvalidOperationException("focus_boundary_missing");
        cursor.EmitDelegate<Action>(() => Main.hasFocus = true);
    }

    void Update(On_Main.orig_Update original, Main game, GameTime time)
    {
        original(game, time);
        if (opened || Main.menuMode != 0) return;
        opened = true;
        if (Terraria.Social.SocialAPI.Cloud != null) Terraria.Social.SocialAPI.Cloud.EnabledByDefault = false;
        WorldGen.clearWorld();
        WorldFile.GetAllMetadata(Path.Combine(Main.SavePath, "Worlds", "Checks.wld"), false).SetAsActive();
        var file = PlayerFileData.CreateAndSave(new Player { name = "Agent engine check" });
        if (file.IsCloudSave || !file.Path.StartsWith(Main.SavePath, StringComparison.Ordinal)) throw new InvalidOperationException("player_not_isolated");
        Player.LoadPlayer(file.Path, false).SetAsActive(); Main.autoPause = false;
        WorldGen.playWorld();
    }

    static void Check(bool ok, string message) { if (!ok) throw new InvalidOperationException(message); }
    void Call(string tool, JsonObject args) => pending = AgentTools.Execute(tool, args, default);
    static JsonObject Step(string id) => new() { ["id"] = id, ["action"] = "verify" };

    static void KillQueen()
    {
        int id = NPC.NewNPC(new EntitySource_Misc("AgentChecks"), (int)Main.LocalPlayer.Center.X + 500,
            (int)Main.LocalPlayer.Center.Y, NPCID.QueenBee);
        Check(id < Main.maxNPCs, "npc_spawn_failed");
        Main.npc[id].StrikeInstantKill();
    }

    public override void PostUpdateEverything()
    {
        if (!enabled || Main.gameMenu || !Main.LocalPlayer.active) return;
        if (Terraria.Program.LaunchParameters.ContainsKey("-agenthornetcheck")) return;
        try
        {
            if (liveRunning)
            {
                if (HealthLive) { Main.LocalPlayer.immune = true; Main.LocalPlayer.immuneTime = 60; }
                if (WorkflowLive)
                {
                    int count = Main.LocalPlayer.inventory.Where(item => item.type == ItemID.LifeCrystal).Sum(item => item.stack);
                    if (!hazardApplied && count >= 3 && count < 10)
                    {
                        hazardApplied = true; hazardUntil = ticks + 60; hazardStock = count;
                        Console.WriteLine($"AGENT_WORKFLOW_HAZARD stock={count} hp=80 frames=60");
                    }
                    if (hazardApplied) Main.LocalPlayer.statLife = ticks < hazardUntil ? 80 : 100;
                }
                if (pending != null)
                {
                    if (!pending.IsCompleted) return;
                    var state = JsonNode.Parse(pending.GetAwaiter().GetResult());
                    pending = null;
                    NativeAgent.Stop();
                    File.WriteAllText(Path.Combine(Main.SavePath, "agent-live-result.json"), new JsonObject
                    { ["elapsed_seconds"] = (DateTime.UtcNow - liveStart).TotalSeconds,
                        ["state"] = state, ["controlled_engine_checks_passed"] = !HealthLive,
                        ["navigation"] = HealthLive ? "teleport_stub" : "real",
                        ["health_goal_met"] = state["permanent_max_health"]?.GetValue<int>() >= 300,
                        ["health_goal_verified"] = state["permanent_max_health"]?.GetValue<int>() >= 300
                            && state["task_plan"]?["status"]?.ToString() == "completed" }.ToJsonString());
                    if (WorkflowLive)
                    {
                        int inventory = Main.LocalPlayer.inventory.Where(item => item.type == ItemID.LifeCrystal).Sum(item => item.stack);
                        var actions = state["action_tasks"].AsArray();
                        bool resumed = actions.Count == 1 && actions[0]["attempt"]?.GetValue<int>() == 2
                            && actions[0]["status"]?.ToString() == "completed"
                            && actions[0]["previous_attempt"]?["status"]?.ToString() == "paused"
                            && actions[0]["previous_attempt"]?["observation"]?["hp"]?.GetValue<int>() < 90;
                        bool segmented = state["script_workflows"].AsObject().Any(w => w.Value["segment"]?.GetValue<int>() >= 2);
                        results.Add(new JsonObject { ["hazard_stock"] = hazardStock, ["inventory"] = inventory,
                            ["resumed_same_handle"] = resumed, ["segmented"] = segmented, ["state"] = state.DeepClone() });
                        Finish(hazardApplied && inventory == 10 && resumed && segmented,
                            "AGENT_WORKFLOW_LIVE_RESULT inventory=" + inventory + " resumed=" + resumed + " segmented=" + segmented);
                        return;
                    }
                    Finish(true, "AGENT_LIVE_CAPTURED bounded_probe=true goal_success_requires_state_verification=true");
                    return;
                }
                double seconds = (DateTime.UtcNow - liveStart).TotalSeconds;
                if (++ticks % 120 == 0) Console.WriteLine($"AGENT_LIVE_PROGRESS seconds={seconds:0.0} status={NativeAgent.Status}");
                if (seconds >= 180 || seconds >= 10 && NativeAgent.Status is "failed" or "paused" or "idle" or "waiting for answer")
                    pending = AgentTools.State(default);
                return;
            }
            if (++ticks > (PlanningReadOnly ? 18000 : 7200)) throw new TimeoutException("engine_check_frame_limit");
            if (!TeleportCheck) { Main.LocalPlayer.immune = true; Main.LocalPlayer.immuneTime = 60; }
            if (pending != null && !pending.IsCompleted) return;
            JsonObject result = pending == null ? null : JsonNode.Parse(pending.GetAwaiter().GetResult()).AsObject();
            pending = null;
            Console.WriteLine($"AGENT_ENGINE_PHASE phase={phase} frame={ticks}");
            if (result != null) results.Add(new JsonObject { ["phase"] = phase, ["result"] = result.DeepClone() });
            var player = Main.LocalPlayer;
            switch (phase++)
            {
                case 0:
                    Config.I.AgentEnabled = true; Config.I.AiClient = false;
                    if (PlanningReadOnly)
                    {
                        Concessions.Enabled = false;
                        Combat.Enabled = Dodge.Enabled = false;
                        player.ConsumedLifeCrystals = 0; player.statLifeMax = player.statLife = 100;
                        foreach (var item in player.inventory) item.TurnToAir();
                        player.inventory[0].SetDefaults(ItemID.CopperShortsword);
                        player.inventory[1].SetDefaults(ItemID.CopperPickaxe);
                        player.inventory[2].SetDefaults(ItemID.CopperAxe);
                        phase = 28; pending = PlanningTrial.Run();
                        break;
                    }
                    if (TeleportCheck)
                    {
                        AgentPermissions.Attack = AgentPermissions.Dodge = false;
                        foreach (var npc in Main.npc) npc.active = false;
                        foreach (var shot in Main.projectile) shot.active = false;
                        var sources = JsonNode.Parse(HttpServerSystem.Dispatch("/find_tiles", "{\"name\":\"Heart\",\"n\":1,\"max_dist\":5000}").Body);
                        Check(sources["tiles"]?.AsArray().Count == 1, "existing_crystal_required_for_teleport_check");
                        heartX = sources["tiles"][0]["x"].GetValue<int>(); heartY = sources["tiles"][0]["y"].GetValue<int>();
                        string beforeTeleportTiles = TileStamp(heartX, heartY);
                        player.statLifeMax = player.statLife = 100;
                        player.immune = false; player.immuneTime = 0; player.noFallDmg = false;
                        player.velocity.Y = 20;
                        NavigateWithoutPlanning(null, heartX, heartY, RecedingNav.Mode.Snap);
                        Check(RecedingNav.LastStop == "done" && player.velocity == Vector2.Zero
                            && player.fallStart == (int)(player.position.Y / 16) && player.fallStart2 == player.fallStart,
                            "teleport_did_not_reset_fall_state");
                        Check(player.Center == new Vector2(heartX * 16 + 8, heartY * 16 + 8), "teleport_did_not_use_exact_target");
                        Check(TileStamp(heartX, heartY) == beforeTeleportTiles, "test_teleport_modified_terrain");
                        teleportCheckUntil = ticks + 90; phase = 24;
                        break;
                    }
                    if (ScriptCheck)
                    {
                        AgentPermissions.Attack = AgentPermissions.Dodge = false;
                        player.ConsumedLifeCrystals = 0; player.statLifeMax = player.statLife = 100;
                        foreach (var item in player.inventory) item.TurnToAir();
                        player.inventory[0].SetDefaults(ItemID.CopperPickaxe);
                        foreach (var item in Main.item) item.active = false;
                        Main.playerInventory = false; PrepareResources(player);
                        if (ContinuationCheck)
                        {
                            phase = 26;
                            Call("run_script", new JsonObject { ["workflow"] = "engine-continuation", ["seconds"] = 90, ["code"] = """
                                const action=step('start',()=>tool('action_start',{tool:'gather',args:{targets:[{item_type:29,count:10,tile_names:['Heart']}]}}));
                                for(let i=0;i<80;i++) {
                                    const q=tool('action_query',{handle:action.handle,wait_ms:200});
                                    const count=q.observation.inventory_slots.filter(x=>x.type===29).reduce((n,x)=>n+x.stack,0);
                                    if(count>=3 && q.status==='running') {
                                        const paused=step('pause',()=>tool('action_control',{handle:action.handle,command:'pause'}));
                                        if(paused.status!=='paused') throw new Error('not paused');
                                        return {status:'yield',handle:action.handle,stock:count};
                                    }
                                    if(q.status!=='running') throw new Error('ended before pause: '+JSON.stringify(q));
                                }
                                throw new Error('pause deadline');
                                """ });
                            break;
                        }
                        if (ModelScript)
                        {
                            var events = JsonNode.Parse(File.ReadAllText(Path.Combine(Main.SavePath, "model-script.json"))).AsArray();
                            string code = events.Single(e => e?["tool"]?.ToString() == "run_script")["args"]["code"].ToString();
                            results.Add(new JsonObject { ["model_code"] = code, ["placed_crystals"] = 10,
                                ["initial_crystals"] = 0, ["navigation"] = "teleport_stub" });
                            phase = 25;
                            Call("run_script", new JsonObject { ["seconds"] = 120, ["code"] = code });
                            break;
                        }
                        phase = 22;
                        Call("run_script", new JsonObject { ["seconds"] = 90, ["code"] = $$"""
                            tool('task_plan', {goal:'script real-engine check',replace:true,steps:[
                              {id:'health',description:'collect and use crystals',conditions:[{kind:'permanent_health',count:300}]}]});
                            const gathered = tool('gather', {targets:[{item_type:{{ItemID.LifeCrystal}},count:10,tile_names:['Heart']}]});
                            if (!gathered.ok) return {error:'collection_failed',gathered};
                            let state = tool('get_state');
                            while (state.permanent_max_health < 300) {
                              const slot = state.inventory_slots.find(i => i.type === {{ItemID.LifeCrystal}} && i.stack > 0);
                              if (!slot) return {error:'missing_crystal',hp:state.permanent_max_health};
                              const used = tool('use_item',{slot:slot.slot,x:Math.floor(state.player.pos.x/16),y:Math.floor(state.player.pos.y/16),duration_ticks:45});
                              if (used.error) return {error:'use_failed',used};
                              const next = tool('get_state');
                              if (next.permanent_max_health <= state.permanent_max_health) throw new Error('no health progress');
                              state = next;
                            }
                            const verified = tool('task_step',{id:'health',action:'verify'});
                            return {hp:state.permanent_max_health,verified:verified.verified};
                            """ });
                        break;
                    }
                    if (HealthLive)
                    {
                        AgentPermissions.Attack = AgentPermissions.Dodge = false;
                        player.ConsumedLifeCrystals = 0; player.statLifeMax = player.statLife = 100;
                        for (int i = 0; i < player.inventory.Length; i++) player.inventory[i].TurnToAir();
                        player.inventory[0].SetDefaults(ItemID.CopperShortsword);
                        player.inventory[1].SetDefaults(ItemID.CopperPickaxe);
                        player.inventory[2].SetDefaults(ItemID.CopperAxe);
                        Main.playerInventory = false;
                        for (int i = 0; i < Main.maxItems; i++) Main.item[i].active = false;
                        Config.I.AgentUseCodex = true;
                        if (WorkflowLive)
                        {
                            PrepareResources(player);
                            Config.I.AgentCodexModel = "gpt-6-astra";
                            NativeAgent.Submit("用宿主workflow与可中断后台任务收集10颗生命水晶，不使用。已查证物品29、来源方块Heart。采集期间持续观察最新血量；低于90就立刻暂停任务，保存进度并return status=yield交回模型。测试环境稍后会恢复血量；下一段确认血量恢复后，恢复同一个句柄继续收集到10，不能重新启动任务或重复已完成步骤。完成后实际验证库存，返回status=done。请自己编写代码实现，不只给计划。");
                        }
                        else NativeAgent.Submit("把永久生命上限提升到300。请建立任务计划，按实际状态执行并验证结果，合理的中间步骤自己决定。");
                        liveRunning = true; liveStart = DateTime.UtcNow;
                        Console.WriteLine("AGENT_HEALTH_LIVE_STARTED initial_hp=100 resources=existing_world terrain_setup=false max_seconds=180 navigation=teleport_stub");
                        break;
                    }
                    if (Terraria.Program.LaunchParameters.ContainsKey("-agentbatchcheck"))
                    {
                        AgentPermissions.Attack = AgentPermissions.Dodge = false;
                        for (int i = 0; i < player.inventory.Length; i++) player.inventory[i].TurnToAir();
                        player.inventory[0].SetDefaults(ItemID.CopperPickaxe);
                        Main.playerInventory = false;
                        for (int i = 0; i < Main.maxItems; i++) Main.item[i].active = false;
                        PrepareResources(player);
                        expectedCrystals = 10;
                        phase = 21;
                        Console.WriteLine("AGENT_BATCH_STARTED placed_hearts=10 initial_crystals=0 model_calls=0 navigation=teleport_stub");
                        pending = BatchGather(expectedCrystals);
                        break;
                    }
                    AgentPermissions.Attack = AgentPermissions.Dodge = false;
                    player.ConsumedLifeCrystals = 0; player.statLifeMax = player.statLife = 100;
                    for (int i = 0; i < player.inventory.Length; i++) player.inventory[i].TurnToAir();
                    player.inventory[0].SetDefaults(ItemID.LifeCrystal);
                    player.inventory[1].SetDefaults(ItemID.JungleSpores); player.inventory[1].stack = 12;
                    player.inventory[2].SetDefaults(ItemID.Stinger); player.inventory[2].stack = 9;
                    int x = (int)player.Center.X / 16, y = (int)player.Bottom.Y / 16;
                    for (int tx = x - 24; tx <= x + 24; tx++)
                    for (int ty = y - 5; ty <= y + 2; ty++)
                    {
                        var tile = Main.tile[tx, ty];
                        tile.ClearEverything();
                        if (ty >= y) { tile.HasTile = true; tile.TileType = TileID.Dirt; }
                    }
                    player.position.Y = y * 16 - player.height; player.velocity = Vector2.Zero;
                    WorldGen.PlaceObject(x + 2, y - 1, TileID.Anvils);
                    Check(Main.tile[x + 2, y - 1].TileType == TileID.Anvils, "anvil_fixture_missing");
                    Main.playerInventory = true;
                    KillQueen();

        catch (Exception error) { Finish(false, error.ToString()); }
    }

    static async Task<string> BatchGather(int wanted)
    {
        var targets = new[] { new GatherTarget(ItemID.LifeCrystal, wanted, new[] { "Heart" }, Array.Empty<int>()) };
        return (await Gather.Run(targets, 90, new ArrivedGather(targets), default)).ToJsonString();
    }

    sealed class ArrivedGather : IGatherWorld
    {
        readonly GameGather world;
        public ArrivedGather(GatherTarget[] targets) => world = new GameGather(targets);
        public Task<JsonObject> Observe(CancellationToken token) => world.Observe(token);
        public async Task<JsonObject> Act(JsonObject candidate, CancellationToken token)
        {
            var arrived = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
            Main.QueueMainThreadAction(() =>
            {
                if (token.IsCancellationRequested) { arrived.TrySetCanceled(token); return; }
                try
                {
                    var player = Main.LocalPlayer;
                    player.position.X = candidate["x"].GetValue<int>() * 16 - player.width / 2;
                    player.velocity = Vector2.Zero;
                    Console.WriteLine($"AGENT_BATCH_ARRIVED x={candidate["x"]} y={candidate["y"]} navigation=teleport_stub");
                    arrived.TrySetResult(true);
                }
                catch (Exception error) { arrived.TrySetException(error); }
            });
            await arrived.Task.WaitAsync(token);
            return await world.Act(candidate, token);
        }
        public Task Stop(CancellationToken token) => world.Stop(token);
    }

    static void PrepareResources(Player player)
    {
        int x = (int)player.Center.X / 16, y = (int)player.Bottom.Y / 16;
        for (int tx = x - 24; tx <= x + 24; tx++)
        for (int ty = y - 5; ty <= y + 2; ty++)
        {
            var tile = Main.tile[tx, ty];
            tile.ClearEverything();
            if (ty >= y) { tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        WorldGen.PlaceObject(x - 3, y - 1, TileID.Anvils);
        for (int i = 0; i < 10; i++)
        {
            int crystalX = x + 2 + i * 2;
            WorldGen.PlaceObject(crystalX, y - 1, TileID.Heart);
            Check(Main.tile[crystalX, y - 1].HasTile && Main.tile[crystalX, y - 1].TileType == TileID.Heart,
                "resource_fixture_missing");
        }
    }

    void Finish(bool ok, string detail)
    {
        File.WriteAllText(Path.Combine(Main.SavePath, "agent-engine-result.json"), new JsonObject
        { ["ok"] = ok, ["detail"] = detail, ["frames"] = ticks, ["results"] = results }.ToJsonString());
        Console.WriteLine(detail);
        enabled = false;
    }

    public override void Unload()
    {
        if (!enabled) return;
        On_Main.Update -= Update; IL_Main.DoUpdate -= Focus;
    }
