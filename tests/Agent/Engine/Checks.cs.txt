    static IEnumerable<JsonObject> LearningReceipts(JsonNode value)
    {
        if (value is JsonObject obj)
        {
            if (obj["observation"] is JsonObject receipt) yield return receipt;
            foreach (var pair in obj)
                foreach (var nested in LearningReceipts(pair.Value)) yield return nested;
        }
        if (value is JsonArray array)
            foreach (var entry in array)
                foreach (var nested in LearningReceipts(entry)) yield return nested;
    }

    static JsonNode LearningView(JsonNode value)
    {
        if (value is JsonObject obj)
        {
            if (obj["slot"]?.ToString() == "0" && obj["name"] != null)
            {
                obj["name"] = "Experimental weapon"; obj.Remove("id"); obj.Remove("type");
            }
            if (obj["held_item"] is JsonObject held)
            {
                held["name"] = "Experimental weapon"; held.Remove("id"); held.Remove("type");
            }
            if (obj["identity"] != null && obj["transitions"] != null) obj.Remove("type");
            foreach (var pair in obj) LearningView(pair.Value);
        }
        else if (value is JsonArray array) foreach (var entry in array) LearningView(entry);
        return value;
    }

        void Setup(StateSpacePlanner.ExecStep step, string phase)
        {
            MineCoordinator.Stop(); SettleAt.Stop();
            Set("_ssSteps", new List<StateSpacePlanner.ExecStep> { step });
            Set("_ssStepIdx", 0); Set("_ssDispatched", true); Set("_ssPrevStep", step);
            Set("_stepTimeoutTicks", long.MaxValue); Set("_stepHardTicks", long.MaxValue);
            Set("_execDone", false); Set("_execFailCode", null); Phase(phase);
        }
        void Tick() => type.GetMethod("TickSteps", hidden).Invoke(null, null);
        try
        {
            if (StateSpacePlanner.ExecRunning || MineCoordinator.IsActive || SkillExecutor.IsActive || BridgeBuilder.IsRunning || SettleAt.IsRunning)
                throw new InvalidOperationException("navigation contract requires idle executors");
            Set("_rolling", false);
            foreach (string outcome in new[] { "out_of_box", "out_of_reach", "stalled", "unbreakable", "no_tiles", "done" })
            {
                var dig = new StateSpacePlanner.ExecStep { Dig = true, TargetCx = 20, TargetCy = 20 };
                Setup(dig, "Mining");
                player.velocity = new Vector2(0, outcome == "done" ? 0 : -3);
                MineCoordinator.Outcome = outcome;
                Tick();
                Console.WriteLine($"MINING_CONTRACT outcome={outcome} steps_active={StateSpacePlanner.StepsActive} done={StateSpacePlanner.ExecDone} fail={StateSpacePlanner.ExecFailCode}");
                if (StateSpacePlanner.StepsActive || (outcome == "done") != StateSpacePlanner.ExecDone
                    || outcome != "done" && StateSpacePlanner.ExecFailCode != "mining_" + outcome)
                    throw new InvalidOperationException("failed mining retained or advanced its stale plan: " + outcome);
            }
            Setup(new StateSpacePlanner.ExecStep { Dig = true }, "Mining");
            player.velocity = new Vector2(0, -3); MineCoordinator.Outcome = "done"; Tick();
            if (!StateSpacePlanner.StepsActive || StateSpacePlanner.ExecDone)
                throw new InvalidOperationException("successful excavation advanced before actual landing");
            foreach (string outcome in new[] { "running", "timeout", "done" })
            {
                Setup(new StateSpacePlanner.ExecStep { Dig = true, TargetCx = 20, TargetCy = 20 }, "Aligning");
                player.velocity = Vector2.Zero; MineCoordinator.Outcome = "out_of_box"; SettleAt.Outcome = outcome;
                typeof(SettleAt).GetField("_running", hidden).SetValue(null, outcome == "running");
                Tick();
                if (StateSpacePlanner.ExecDone || StateSpacePlanner.StepsActive != (outcome != "timeout")
                    || MineCoordinator.IsActive != (outcome == "done")
                    || outcome != "timeout" && (int)type.GetField("_ssStepIdx", hidden).GetValue(null) != 0)
                    throw new InvalidOperationException("alignment phase advanced or read stale mining outcome: " + outcome);
                Console.WriteLine($"ALIGNMENT_CONTRACT outcome={outcome} mine_started={MineCoordinator.IsActive} steps_active={StateSpacePlanner.StepsActive}");
            }
            foreach (string outcome in new[] { "out_of_reach", "done" })
            {
                Setup(new StateSpacePlanner.ExecStep { Pillar = true }, "ClearingHeadroom");
                player.velocity = new Vector2(0, -3); MineCoordinator.Outcome = outcome; Tick();
                if (StateSpacePlanner.ExecDone || StateSpacePlanner.StepsActive != (outcome == "done")
                    || outcome == "done" && type.GetField("_stepPhase", hidden).GetValue(null).ToString() != "HeadroomReady")
                    throw new InvalidOperationException("headroom excavation was mistaken for a completed pillar: " + outcome);
                Console.WriteLine($"HEADROOM_CONTRACT outcome={outcome} steps_active={StateSpacePlanner.StepsActive}");
            }
            Console.WriteLine("NAVIGATION_MINING_CONTRACTS_COMPLETE checks=12 gameplay_loop=false model_calls=0");
        }
        finally
        {
            MineCoordinator.Stop(); SettleAt.Stop();
            foreach (var entry in saved) Set(entry.Key, entry.Value);
            player.velocity = velocity; MineCoordinator.Outcome = previousOutcome; SettleAt.Outcome = previousSettle;
        }
    }

public sealed class NavigationFieldProbe : ModSystem
{
    bool finished;
    public override void OnWorldLoad() => PostUpdateWorld();
    public override void PostUpdateWorld()
    {
        if (finished || !Terraria.Program.LaunchParameters.ContainsKey("-navfieldcheck")) return;
        finished = true;
        try
        {
            Main.myPlayer = 0;
            Main.player[0] = new Player { active = true, width = 20, height = 42 };
            var player = Main.player[0];
            player.inventory[1].SetDefaults(ItemID.CopperPickaxe);
            player.inventory[3].SetDefaults(ItemID.Wood); player.inventory[3].stack = 9999;
            var policy = (AgentControlPolicy)typeof(AgentTools).GetField("Control", BindingFlags.Static | BindingFlags.NonPublic).GetValue(null);
            policy.Reset();
            var results = new JsonArray();
            foreach (var scene in new[] { (1100, 252, 1180, 450), (1058, 695, 1480, 870), (1100, 252, 1130, 326) })
            {
                var (sx, sy, gx, gy) = scene;
                player.position = new Vector2(sx * 16, (sy + 1) * 16 - player.height);
                var field = MazeWand.BuildField(gx, gy, sx, sy, bigMargin: true);
                var cells = new JsonArray();
                for (int x = gx - 1; x <= gx + 1; x++)
                for (int y = gy - 3; y <= gy + 1; y++)
                {
                    var tile = Main.tile[x, y];
                    cells.Add(new JsonObject { ["x"] = x, ["y"] = y, ["has_tile"] = tile.HasTile,
                        ["type"] = tile.TileType, ["wall"] = tile.WallType, ["liquid"] = tile.LiquidAmount,
                        ["liquid_type"] = tile.LiquidType, ["solid"] = tile.HasTile && Main.tileSolid[tile.TileType],
                        ["can_mine"] = DigTable.MineableWith(x, y, 35), ["lava_safe"] = LavaSafety.CanMine(x, y),
                        ["navigation_forbidden"] = policy.ForbiddenAt(ControlBehavior.Navigation, x, y) });
                }
                var costs = new JsonArray();
                foreach (var direction in new[] { (1, 0), (-1, 0), (0, 1), (0, -1) })
                    costs.Add(MazeWand.StepCostPublic(gx, gy, gx + direction.Item1, gy + direction.Item2));
                var mainQueue = (System.Collections.Concurrent.ConcurrentQueue<Action>)typeof(NativeAgent)
                    .GetField("MainQueue", BindingFlags.Static | BindingFlags.NonPublic).GetValue(null);
                var frameQueue = (System.Collections.Concurrent.ConcurrentQueue<Action>)typeof(NativeAgent)
                    .GetField("FrameQueue", BindingFlags.Static | BindingFlags.NonPublic).GetValue(null);
                void Pump()
                {
                    int frames = frameQueue.Count;
                    for (int i = 0; i < frames && frameQueue.TryDequeue(out var frame); i++) frame();
                    while (mainQueue.TryDequeue(out var action)) action();
                    Thread.Sleep(1);
                }
                using var cancel = new CancellationTokenSource();
                var timer = System.Diagnostics.Stopwatch.StartNew();
                var navigation = (Task<JsonObject>)typeof(AgentTools).GetMethod("Navigate", BindingFlags.Static | BindingFlags.NonPublic)
                    .Invoke(null, new object[] { new JsonObject { ["x"] = gx, ["y"] = gy }, cancel.Token, 0 });
                while (!navigation.IsCompleted && timer.Elapsed.TotalSeconds < 10)
                {
                    Pump();
                    if (field.Count > 1 && RecedingNav.Active && MazeWand.PeekFieldOrNull(gx, gy) != null
                        && timer.ElapsedMilliseconds >= 100)
                        break;
                }
                JsonObject outcome;
                if (field.Count == 1)
                {
                    if (!navigation.IsCompleted) throw new Exception("isolated target did not return within ten seconds");
                    outcome = navigation.GetAwaiter().GetResult();
                    if (outcome["error"]?.ToString() != "navigation_goal_isolated" || RecedingNav.Active)
                        throw new Exception("isolated target failed to stop through the real navigation API: " + outcome);
                }
                else
                {
                    if (navigation.IsCompleted || !RecedingNav.Active) throw new Exception("connected route was prematurely stopped");
                    outcome = new JsonObject { ["connected_route_running"] = true };
                }
                double elapsed = timer.Elapsed.TotalMilliseconds;
                cancel.Cancel();
                while (!navigation.IsCompleted && timer.Elapsed.TotalSeconds < 12) Pump();
                if (!navigation.IsCompleted) throw new Exception("navigation did not finish cancellation");
                RecedingNav.Stop();
                results.Add(new JsonObject { ["start"] = new JsonArray(sx, sy), ["goal"] = new JsonArray(gx, gy),
                    ["pick_power"] = MazeWand.BestPickPower(), ["field_cells"] = field.Count,
                    ["start_in_field"] = field.ContainsKey((sx, sy)), ["goal_step_costs"] = costs, ["goal_tiles"] = cells,
                    ["navigation_api"] = outcome, ["navigation_api_ms"] = elapsed });
            }
            File.WriteAllText(Path.Combine(Main.SavePath, "nav-field-probe.json"), results.ToJsonString());
            Console.WriteLine("NAV_FIELD_PROBE_COMPLETE native_world=true model_calls=0 scenes=" + results.Count);
            Environment.Exit(0);
        }
        catch (Exception error)

public sealed class HornetTrial : ModSystem
{
    public static bool Enabled => Terraria.Program.LaunchParameters.ContainsKey("-agenthornetcheck");
    bool Obstacle => Terraria.Program.LaunchParameters.ContainsKey("-agenthornetwall");
    public static int Target = -1;
    public static bool Killed;
    readonly JsonArray rounds = new();
    readonly HashSet<(int, int)> hostileProjectiles = new();
    int x, floor, tick, spawnedAt, hpBefore, knivesBefore, shots, moves, damageTaken, previousHp;
    int blockedFrames, blockedAttackFrames;
    bool started, finished;
    double distance;
    Vector2 previousPosition;

    public override void PostUpdateEverything()
    {
        if (!Enabled || finished || Main.gameMenu || !Main.LocalPlayer.active) return;
        try
        {
            var p = Main.LocalPlayer;
            if (!started)
            {
                started = true;
                Config.I.AgentEnabled = true; Config.I.AiClient = false;
                AgentPermissions.Attack = AgentPermissions.Dodge = true;
                Combat.Enabled = Dodge.Enabled = false;
                x = (int)p.Center.X / 16; floor = (int)p.Bottom.Y / 16;
                for (int tx = x - 22; tx <= x + 22; tx++)
                for (int ty = floor - 13; ty <= floor + 2; ty++)
                {
                    var tile = Main.tile[tx, ty]; tile.ClearEverything();
                    if (tx == x - 22 || tx == x + 22 || ty == floor - 13 || ty >= floor)
                    { tile.HasTile = true; tile.TileType = TileID.Stone; }
                }
                foreach (var npc in Main.npc) npc.active = false;
                if (Obstacle)
                    for (int tx = x + 7; tx <= x + 8; tx++)
                    for (int ty = floor - 5; ty < floor; ty++)
                    { var tile = Main.tile[tx, ty]; tile.HasTile = true; tile.TileType = TileID.Stone; }
                foreach (var projectile in Main.projectile) projectile.active = false;
                foreach (var item in Main.item) item.active = false;
                foreach (var item in p.inventory) item.TurnToAir();
                p.inventory[0].SetDefaults(ItemID.ThrowingKnife); p.inventory[0].stack = 500;
                p.ConsumedLifeCrystals = 10; p.statLifeMax = p.statLife = 300;
                p.position = new Vector2(x * 16, floor * 16 - p.height); p.velocity = Vector2.Zero;
                p.immune = false; p.immuneTime = 0;
                Main.playerInventory = false;
                previousHp = p.statLife; previousPosition = p.position;
                Console.WriteLine($"HORNET_TRIAL_START arena=42x12 knives=500 hp=300 immunity=false sequential=3 model_calls=0 obstacle={Obstacle}");
                Spawn(p); return;
            }
            tick++;
            distance += Vector2.Distance(previousPosition, p.position); previousPosition = p.position;
            if (p.controlLeft || p.controlRight || p.controlJump) moves++;
            damageTaken += Math.Max(0, previousHp - p.statLife); previousHp = p.statLife;
            foreach (var projectile in Main.projectile)
                if (projectile.active && projectile.hostile) hostileProjectiles.Add((projectile.whoAmI, projectile.identity));
            shots = Math.Max(shots, knivesBefore - p.inventory.Where(i => i.type == ItemID.ThrowingKnife).Sum(i => i.stack));
            if (p.dead) { Record(p, "player_dead"); Finish(false, "player_dead"); return; }
            if (p.Center.X < (x - 22) * 16 || p.Center.X > (x + 22) * 16 || p.Center.Y < (floor - 13) * 16)
            { Record(p, "left_arena"); Finish(false, "left_arena"); return; }
            int elapsed = tick - spawnedAt;
            if (elapsed >= 120 && Main.npc[Target].active && !Collision.CanHitLine(p.Center, 4, 4, Main.npc[Target].Center, 4, 4))
            { blockedFrames++; if (p.controlUseItem) blockedAttackFrames++; }
            if (elapsed == 120) FightCoordinator.Start(60, new[] { (int)NPCID.Hornet });
            if (Killed)
            {
                Record(p, "killed"); FightCoordinator.Stop();
                if (rounds.Count == 3) { Finish(true, "three_hornets_killed"); return; }
                Spawn(p); return;
            }
            if (!Main.npc[Target].active) { Record(p, "target_lost"); Finish(false, "target_lost"); return; }
            if (elapsed >= 1800) { Record(p, "timeout"); Finish(false, "timeout"); }
        }
        catch (Exception error) { Finish(false, error.ToString()); }
    }

    void Spawn(Player p)
    {
        int side = rounds.Count % 2 == 0 ? 1 : -1;
        Target = NPC.NewNPC(new EntitySource_Misc("HornetTrial"), (x + side * 17) * 16, (floor - 6) * 16, NPCID.Hornet);
        if (Target >= Main.maxNPCs) throw new InvalidOperationException("hornet_spawn_failed");
        Killed = false; spawnedAt = tick; hpBefore = p.statLife;
        knivesBefore = p.inventory.Where(i => i.type == ItemID.ThrowingKnife).Sum(i => i.stack);
        shots = moves = damageTaken = blockedFrames = blockedAttackFrames = 0; hostileProjectiles.Clear(); distance = 0; previousPosition = p.position; previousHp = p.statLife;
        Console.WriteLine($"HORNET_SPAWN round={rounds.Count + 1} id={Target} hp={Main.npc[Target].life} side={side}");
    }

    void Record(Player p, string outcome)
    {
        var result = new JsonObject { ["round"] = rounds.Count + 1, ["outcome"] = outcome,
            ["frames"] = tick - spawnedAt, ["hp_before"] = hpBefore, ["hp_after"] = p.statLife,
            ["damage_taken"] = damageTaken, ["knives_used"] = shots, ["movement_frames"] = moves,
            ["hostile_projectiles"] = hostileProjectiles.Count,
            ["line_blocked_frames"] = blockedFrames, ["blocked_attack_frames"] = blockedAttackFrames,
            ["distance_pixels"] = distance, ["target_hp"] = Main.npc[Target].life };
        rounds.Add(result); Console.WriteLine("HORNET_ROUND " + result.ToJsonString());
    }

    void Finish(bool ok, string outcome)
    {
        finished = true; FightCoordinator.Stop();
        File.WriteAllText(Path.Combine(Main.SavePath, "agent-engine-result.json"), new JsonObject
        { ["ok"] = ok, ["detail"] = "HORNET_TRIAL_COMPLETE " + outcome, ["rounds"] = rounds,
            ["model_calls"] = 0, ["immunity"] = false, ["arena_cells"] = "42x12", ["obstacle"] = Obstacle }.ToJsonString());
    }
}

    async Task<string> DescentRouteProbe()
    {
        var timer = System.Diagnostics.Stopwatch.StartNew();
        var first = AgentTools.Execute("descent_route", new JsonObject { ["name"] = "jungle", ["limit"] = 2 }, default);
        long previous = timer.ElapsedMilliseconds, maxGap = 0;
        int probes = 0;
        while (!first.IsCompleted)
        {
            await Task.Delay(50);
            await AgentTools.Execute("get_state", new JsonObject { ["fields"] = new JsonArray("permanent_max_health") }, default);
            long now = timer.ElapsedMilliseconds;
            maxGap = Math.Max(maxGap, now - previous);
            previous = now;
            probes++;
        }
        var route = JsonNode.Parse(await first);
        long coldMs = timer.ElapsedMilliseconds;
        timer.Restart();
        var page = JsonNode.Parse(await AgentTools.Execute("descent_route",
            new JsonObject { ["name"] = "jungle", ["offset"] = 1, ["limit"] = 1 }, default));
        bool same = route?["line"]?.AsArray().Count == 2 && page?["line"]?.AsArray().Count == 1
            && JsonNode.DeepEquals(route["line"][1], page["line"][0])
            && JsonNode.DeepEquals(route["entrance"], page["entrance"]) && JsonNode.DeepEquals(route["line_len"], page["line_len"]);
        var receipt = new JsonObject { ["ok"] = same && probes > 2 && maxGap < 1000,
            ["cold_ms"] = coldMs, ["max_probe_gap_ms"] = maxGap, ["probes"] = probes,
            ["page_ms"] = timer.ElapsedMilliseconds, ["pages_match"] = same };
        return receipt.ToJsonString();
    }

                            $"AGENT_EARLY_FALL_CHECK hp={player.statLife} web_at_frame={webPlacedAt} fall_start={player.fallStart} bottom={player.Bottom.Y} model_calls=0");
                    }
                    phase = 92; break;
                case 48:
                    Check(result["found"]?.GetValue<bool>() == true && result["local_biome_tiles"].GetValue<int>() >= SceneMetrics.JungleTileThreshold,
                        "no_verified_jungle_surface");
                    heartX = result["x"].GetValue<int>(); heartY = result["y"].GetValue<int>();
                    for (int sy = 0; sy <= heartY; sy++)
                    {
                        var sky = Main.tile[heartX, sy];
                        Check(!sky.HasUnactuatedTile || !Main.tileSolid[sky.TileType] || Main.tileSolidTop[sky.TileType], "entrance_is_a_covered_pocket");
                    }
                    player.Teleport(new Vector2(heartX * 16 + 8 - player.width / 2f, (heartY + 1) * 16 - player.height));
                    player.velocity = Vector2.Zero; player.fallStart = player.fallStart2 = (int)(player.position.Y / 16);
                    interactionWait = ticks + 90; break;
                case 49:
                    if (ticks < interactionWait) { phase = 49; break; }
                    Check(player.ZoneJungle, "native_player_did_not_recognize_selected_jungle");
                    results.Add(new JsonObject { ["native_zone_jungle"] = player.ZoneJungle, ["x"] = heartX, ["y"] = heartY });
                    Finish(true, "AGENT_BIOME_COMPLETE sky_exposed=true native_zone_jungle=true model_calls=0"); break;
                case 29:
                    Check(result["ok"]?.GetValue<bool>() == true && player.armor[0].type == ItemID.WoodHelmet && player.inventory[0].IsAir,
                        "helmet_not_equipped");
                    Call("task_step", new JsonObject { ["id"] = "wear", ["action"] = "verify" }); phase = 47;
                    break;
                case 30:
                    Check(result["error"]?.ToString() == "incompatible_equipment" && player.inventory[1].type == ItemID.WoodBreastplate
                        && player.armor[0].type == ItemID.WoodHelmet, "wrong_armor_slot_changed_items");
                    var staleEquip = EquipRequest(1, 1); staleEquip["expected_inventory"]["stack"] = 99;
                    Call("equip_item", staleEquip);
                    break;
                case 31:
                    Check(result["error"]?.ToString() == "stale_equipment", "stale_equipment_was_accepted");
                    Call("equip_item", EquipRequest(2, 3));
                    break;
                case 32:
                    Check(result["ok"]?.GetValue<bool>() == true && player.armor[3].type == ItemID.CloudinaBottle,
                        "accessory_not_equipped");
                    Call("equip_item", EquipRequest(3, 4));
                    break;
                case 33:
                    Check(result["error"]?.ToString() == "incompatible_equipment" && player.inventory[3].type == ItemID.CloudinaBottle,
                        "duplicate_accessory_was_accepted");
                    Check(player.statDefense > 0 && player.GetJumpState(ExtraJump.CloudInABottle).Enabled,
                        "equipment_did_not_affect_real_player");
                    Call("equip_item", EquipRequest(3, 9));
                    break;
                case 34:
                    Check(result["error"]?.ToString() == "equipment_slot_locked", "locked_accessory_slot_was_accepted");
                    Call("equip_item", EquipRequest(4, 3));
                    break;
                case 35:
                    Check(result["ok"]?.GetValue<bool>() == true && player.armor[3].IsAir && player.inventory[4].type == ItemID.CloudinaBottle
                        && Main.mouseItem.IsAir, "unequip_lost_or_duplicated_items");
                    Main.dayTime = true; Main.time = 1000;
                    chatNpc = NPC.NewNPC(new EntitySource_Misc("AgentInteractions"), (int)player.Center.X + 40, (int)player.Bottom.Y, NPCID.OldMan);
                    Check(chatNpc < Main.maxNPCs, "old_man_fixture_missing");
                    Main.npc[chatNpc].aiStyle = -1;
                    Call("npc_chat", new JsonObject { ["action"] = "open", ["npc"] = chatNpc, ["npc_type"] = (int)NPCID.OldMan });
                    break;
                case 36:
                    Check(result["open"]?.GetValue<bool>() == true && result["buttons_ready"]?.GetValue<bool>() == true,
                        "native_dialogue_did_not_open");
                    Check(!player.GetJumpState(ExtraJump.CloudInABottle).Enabled, "unequipped_accessory_effect_persisted");
                    Call("npc_chat", new JsonObject { ["action"] = "choose", ["npc"] = chatNpc, ["npc_type"] = (int)NPCID.OldMan,
                        ["button"] = 1, ["expected_text"] = "not-current" });
                    break;
                case 37:
                    Check(result["error"]?.ToString() == "dialogue_option_changed" && !NPC.AnyNPCs(NPCID.SkeletronHead),
                        "daytime_or_stale_button_bypassed_native_rules");
                    Main.dayTime = false; Main.time = 1000;
                    interactionWait = ticks + 5; phase = 38;
                    break;
                case 38:
                    if (ticks < interactionWait) { phase = 38; break; }
                    var dialogue = NpcDialogue.Snapshot();
                    Check(dialogue["buttons"].AsArray().Count > 0, "night_button_missing");
                    chatLabel = dialogue["buttons"][0]["text"].ToString();
                    ChatHookProbe.Block = true;
                    Call("npc_chat", ChooseButton());
                    break;
                case 39:
                    Check(result["ok"]?.GetValue<bool>() == true && ChatHookProbe.PreCalls == 1 && ChatHookProbe.PostCalls == 0
                        && !NPC.AnyNPCs(NPCID.SkeletronHead), "mod_chat_veto_was_bypassed");
                    ChatHookProbe.Block = false;
                    Call("npc_chat", ChooseButton());
                    break;
                case 40:
                    Check(result["ok"]?.GetValue<bool>() == true && ChatHookProbe.PreCalls == 2 && ChatHookProbe.PostCalls == 1,
                        "native_chat_hooks_not_dispatched_once");
                    Check(Main.npc.Count(n => n.active && n.type == NPCID.SkeletronHead) == 1, "native_button_did_not_summon_one_boss");
                    results.Add(new JsonObject { ["defense"] = (int)player.statDefense, ["button"] = chatLabel,
                        ["pre_hooks"] = ChatHookProbe.PreCalls, ["post_hooks"] = ChatHookProbe.PostCalls,
                        ["skeletron_count"] = 1, ["cursor_empty"] = Main.mouseItem.IsAir });
                    foreach (var npc in Main.npc.Where(n => n.active && (n.boss || n.type == NPCID.SkeletronHand))) npc.aiStyle = -1;
                    chatNpc = NPC.NewNPC(new EntitySource_Misc("AgentInteractions"), (int)player.Center.X + 2000, (int)player.Bottom.Y, NPCID.Merchant);
                    Call("npc_chat", new JsonObject { ["action"] = "open", ["npc"] = chatNpc, ["npc_type"] = (int)NPCID.Merchant });
                    break;
                case 41:
                    Check(result["error"]?.ToString() == "tool_exception" && result["detail"]?.ToString() == "npc_out_of_reach",
                        "remote_dialogue_bypassed_range");
                    Main.npc[chatNpc].position = player.position - new Vector2(40, 0);
                    Main.npc[chatNpc].aiStyle = -1;
                    Call("npc_chat", new JsonObject { ["action"] = "open", ["npc"] = chatNpc, ["npc_type"] = (int)NPCID.Merchant });
                    break;
                case 42:
                    Check(result["open"]?.GetValue<bool>() == true && result["buttons"].AsArray().Count > 0,
                        "merchant_dialogue_not_available");
                    Call("npc_chat", new JsonObject { ["action"] = "choose", ["npc"] = chatNpc, ["npc_type"] = (int)NPCID.Merchant,
                        ["button"] = 1, ["expected_text"] = result["buttons"][0]["text"].ToString() });
                    break;
                case 43:
                    Check(result["ok"]?.GetValue<bool>() == true && Main.npcShop > 0, "generic_shop_button_not_dispatched");
                    Call("npc_chat", new JsonObject { ["action"] = "close" });
                    break;
                case 44:
                    Check(player.talkNPC == -1 && result["open"]?.GetValue<bool>() == false, "dialogue_close_failed");
                    Finish(true, "AGENT_INTERACTIONS_COMPLETE checks=19 model_calls=0 equipped_verified=true equipment_effects=true native_dialogue=true boss_summoned=1 merchant_shop=true range_guard=true");
                    break;
                case 45:
                    Call("task_step", new JsonObject { ["id"] = "wear", ["action"] = "verify" });
                    break;
                case 46:
                    Check(result["verified"]?.GetValue<bool>() == false, "carried_helmet_falsely_verified_as_worn");
                    phase = 29; Call("equip_item", EquipRequest(0, 0));
                    break;
                case 47:
                    Check(result["verified"]?.GetValue<bool>() == true, "actually_worn_helmet_not_verified");
                    phase = 30; Call("equip_item", EquipRequest(1, 0));
                    break;
                case 28:
                    if (result["experiment"]?.ToString() == "crossing_discovery")
                    {
                        Finish(result["ok"]?.GetValue<bool>() == true, "AGENT_CROSSING_TRIAL_COMPLETE");
                        break;
                    }
                    Finish(result["ok"]?.GetValue<bool>() == true,
                        "AGENT_PLANNING_READONLY queries=" + result["queries"] + " blocked=" + result["blocked_mutators"] + " game_actions=0");
                    break;
                case 26:
                    Check(result["ok"]?.GetValue<bool>() == true && result["value"]?["status"]?.ToString() == "yield",
                        "first_segment_did_not_yield_after_pause");
                    actionHandle = result["value"]["handle"].ToString();
                    stockAtPause = player.inventory.Where(item => item.type == ItemID.LifeCrystal).Sum(item => item.stack);
                    Check(stockAtPause >= 3 && stockAtPause < 10 && !RecedingNav.Active && !ItemUseCoordinator.IsActive,
                        "pause_lost_partial_inventory_or_left_inputs_active");
                    phase = 27;
                    Call("run_script", new JsonObject { ["workflow"] = "engine-continuation", ["seconds"] = 90, ["code"] = """
                        const action=step('start',()=>{throw new Error('completed action start replayed')});
                        const resumed=step('resume',()=>tool('action_control',{handle:action.handle,command:'resume'}));
                        if(resumed.handle!==action.handle || resumed.attempt!==2) throw new Error('handle changed');
                        for(let i=0;i<80;i++) {
                            const q=tool('action_query',{handle:action.handle,wait_ms:200});
                            if(q.status==='running') continue;
                            if(q.status!=='completed') throw new Error(JSON.stringify(q));
                            const count=q.observation.inventory_slots.filter(x=>x.type===29).reduce((n,x)=>n+x.stack,0);
                            if(count!==10) throw new Error('inventory mismatch');
                            step('finish',()=>{tool('say',{text:'continuation-complete'});return {count}});
                            return {status:'done',handle:action.handle,count,attempt:q.attempt,previous:q.previous_attempt};
                        }
                        throw new Error('resume deadline');
                        """ });
                    break;
                case 27:
                    Check(result["ok"]?.GetValue<bool>() == true && result["value"]?["status"]?.ToString() == "done"
                        && result["value"]?["handle"]?.ToString() == actionHandle
                        && result["value"]?["previous"]?["status"]?.ToString() == "paused"
                        && player.inventory.Where(item => item.type == ItemID.LifeCrystal).Sum(item => item.stack) == 10,
                        "continuation_did_not_resume_original_intent");
                    results.Add(new JsonObject { ["stock_at_pause"] = stockAtPause, ["final_inventory"] = 10,
                        ["same_handle"] = actionHandle, ["attempt"] = 2 });
                    Finish(true, "AGENT_CONTINUATION_COMPLETE checks=2 model_calls=0 initial=0 final=10 same_handle=true attempt=2 navigation=teleport_stub");
                    break;
                case 25:
                    int crystals = player.inventory.Where(item => item.type == ItemID.LifeCrystal).Sum(item => item.stack);
                    results.Add(new JsonObject { ["actual_crystals"] = crystals, ["permanent_health"] = player.statLifeMax,
                        ["announcements"] = new JsonArray(modelAnnouncements.Select(n => (JsonNode)JsonValue.Create(n)).ToArray()) });
                    Check(result["ok"]?.GetValue<bool>() == true && result["value"]?["completed"]?.GetValue<int>() == 10
                        && crystals == 10 && player.statLifeMax == 100, "model_script_collection_not_verified");
                    Check(modelAnnouncements.SequenceEqual(Enumerable.Range(1, 10)), "model_script_per_item_order_wrong");
                    Finish(true, "AGENT_MODEL_SCRIPT_COMPLETE checks=2 model_calls=0 placed_crystals=10 inventory=10 announcements=10 hp=100 navigation=teleport_stub");
                    break;
                case 24:
                    if (ticks < teleportCheckUntil) { phase = 24; break; }
                    Check(!player.dead && player.statLife == 100 && !player.noFallDmg,
                        "teleport_caused_fall_damage");
                    var nearest = JsonNode.Parse(HttpServerSystem.Dispatch("/find_tiles", "{\"name\":\"Heart\",\"n\":1,\"max_dist\":5000}").Body);
                    var many = JsonNode.Parse(HttpServerSystem.Dispatch("/find_tiles", "{\"name\":\"Heart\",\"n\":100,\"max_dist\":5000}").Body);
                    var candidates = many["tiles"].AsArray();
                    Check(nearest["tiles"][0]["dist"].GetValue<int>() == candidates[0]["dist"].GetValue<int>(), "nearest_query_disagrees_with_sorted_candidates");
                    for (int i = 1; i < candidates.Count; i++)
                        Check(candidates[i - 1]["dist"].GetValue<int>() <= candidates[i]["dist"].GetValue<int>(), "tile_scan_not_sorted_by_reported_distance");
                    results.Add(new JsonObject { ["hp_after_90_frames"] = player.statLife, ["target_x"] = heartX, ["target_y"] = heartY,
                        ["nearest"] = nearest, ["candidate_count"] = candidates.Count });
                    Finish(true, "AGENT_TELEPORT_COMPLETE checks=4 model_calls=0 hp=100 fall_protection=false terrain_unchanged=true target=exact");
                    break;
            }
        }
        catch (Exception error) { Finish(false, error.ToString()); }
    }

    static async Task<string> BatchGather(int wanted)
    {
        var targets = new[] { new GatherTarget(ItemID.LifeCrystal, wanted, new[] { "Heart" }, Array.Empty<int>()) };
        return (await Gather.Run(targets, 90, new ArrivedGather(targets), default)).ToJsonString();
    }

    static JsonObject EquipRequest(int source, int target)
    {
        JsonObject State(Item item) => new() { ["type"] = item.type, ["prefix"] = item.prefix, ["stack"] = item.stack };
        return new JsonObject { ["inventory_slot"] = source, ["equipment_slot"] = target,
            ["expected_inventory"] = State(Main.LocalPlayer.inventory[source]), ["expected_equipment"] = State(Main.LocalPlayer.armor[target]) };
    }

    JsonObject ChooseButton() => new() { ["action"] = "choose", ["npc"] = chatNpc, ["npc_type"] = (int)NPCID.OldMan,
        ["button"] = 1, ["expected_text"] = chatLabel };

    sealed class ArrivedGather : IGatherWorld
    {
        readonly GameGather world;
        public ArrivedGather(GatherTarget[] targets) => world = new GameGather(targets);
        public Task<JsonObject> Observe(CancellationToken token) => world.Observe(token);
        public async Task<JsonObject> Act(JsonObject candidate, CancellationToken token)
        {
            var arrived = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
            Main.QueueMainThreadAction(() =>
            {
                if (token.IsCancellationRequested) { arrived.TrySetCanceled(token); return; }
                try
                {
                    var player = Main.LocalPlayer;
                    player.position.X = candidate["x"].GetValue<int>() * 16 - player.width / 2;
                    player.velocity = Vector2.Zero;
                    Console.WriteLine($"AGENT_BATCH_ARRIVED x={candidate["x"]} y={candidate["y"]} navigation=teleport_stub");
                    arrived.TrySetResult(true);
                }
                catch (Exception error) { arrived.TrySetException(error); }
            });
            await arrived.Task.WaitAsync(token);
            return await world.Act(candidate, token);
        }
        public Task Stop(CancellationToken token) => world.Stop(token);
    }

    static void PrepareResources(Player player)
    {
        int x = (int)player.Center.X / 16, y = (int)player.Bottom.Y / 16;
        for (int tx = x - 24; tx <= x + 24; tx++)
        for (int ty = y - 5; ty <= y + 2; ty++)
        {
            var tile = Main.tile[tx, ty];
            tile.ClearEverything();
            if (ty >= y) { tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        WorldGen.PlaceObject(x - 3, y - 1, TileID.Anvils);
        for (int i = 0; i < 10; i++)
        {
            int crystalX = x + 2 + i * 2;
            WorldGen.PlaceObject(crystalX, y - 1, TileID.Heart);
            Check(Main.tile[crystalX, y - 1].HasTile && Main.tile[crystalX, y - 1].TileType == TileID.Heart,
                "resource_fixture_missing");
        }
    }

    void Finish(bool ok, string detail)
    {
        File.WriteAllText(Path.Combine(Main.SavePath, "agent-engine-result.json"), new JsonObject
        { ["ok"] = ok, ["detail"] = detail, ["frames"] = ticks, ["results"] = results }.ToJsonString());
        Console.WriteLine(detail);
        enabled = false;
    }

    public override void Unload()
    {
        if (!enabled) return;
        On_Main.Update -= Update; IL_Main.DoUpdate -= Focus;
    }
