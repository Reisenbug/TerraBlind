static class PlanningTrial
{
    static readonly HashSet<string> ReadTools = new()
    { "get_state", "boss_status", "equipment_list", "npc_dialogue_status", "game_catalog", "item_info", "recipe", "find_tiles", "npc_find", "find_biome", "tile_names", "wiki_search", "wiki_page" };

    public static async Task<string> Run()
    {
        if (Environment.GetEnvironmentVariable("TB_TERRAIN_ASCII") != null) return await AsciiTerrainTrial.Run();
        if (Environment.GetEnvironmentVariable("TB_RAIL_SKILL") != null) return await RailSkillTrial.Run();
        if (Environment.GetEnvironmentVariable("TB_CROSSING_DISCOVERY") == "1") return await CrossingDiscovery.Run();
        if (!ModLoader.TryGetMod("TerraBlind", out var mod)) throw new InvalidOperationException("production_mod_missing");
        var schemas = JsonNode.Parse(Encoding.UTF8.GetString(mod.GetFileBytes("Agent/Tools.json"))).AsArray();
        var entries = new JsonArray(schemas.Where(t => ReadTools.Contains(t["function"]["name"].ToString())
            || t["function"]["name"].ToString() == "run_script").Select(t => t.DeepClone()).ToArray());
        string system = AgentScript.Instructions(Encoding.UTF8.GetString(mod.GetFileBytes("Agent/System.txt")), schemas)
            + "\n本轮是只读规划，开放的查询工具可直接调用或在run_script内调用。只允许读取，不得移动、采集、使用、开战、修改许可或登记计划。"
            + "最多12次查询；没有执行过的查询不能伪称有结果。最终只输出JSON，包含first_action:{tool,args,reason,expected_result,fallback}，equipment_plan、readiness、boss_control、questions、evidence。"
            + "first_action必须是现在可以尝试的游戏行动，不是继续查询；仅说明，不实际调用。args按真实API填写，坐标须有查询来源。";
        var events = new JsonArray();
        var replies = new List<string>();
        int queries = 0, blocked = 0;
        void Validate(string name, JsonObject args)
        {
            if (!ReadTools.Contains(name)) { blocked++; throw new InvalidOperationException("read_only: " + name); }
            var schema = schemas.First(t => t["function"]["name"].ToString() == name)["function"];
            foreach (var key in schema["parameters"]?["required"]?.AsArray() ?? new JsonArray())
                if (args[key.ToString()] == null) throw new ArgumentException("missing query argument: " + key);
        }
        async Task<string> Query(string name, JsonObject args, CancellationToken token)
        {
            var item = new JsonObject { ["tool"] = name, ["args"] = args.DeepClone() };
            events.Add(item);
            string result;
            if (name == "run_script") result = (await AgentScript.Run(args["code"].ToString(), 30, Validate, Query,
                _ => Task.CompletedTask, token)).ToJsonString();
            else
            {
                Validate(name, args);
                if (++queries > 12) return "{\"error\":\"planning_query_budget\",\"detail\":\"请根据已有事实提出下一步，不继续查询。\"}";
                result = await AgentTools.Execute(name, args, token);
            }
            item["result"] = result;
            return result;
        }
        var initial = JsonNode.Parse(await AgentTools.State(default));
        string input = "自行寻找装备，合理搭配，击败骷髅王。\n先只读查询并规划，暂不执行行动。常规路线、装备选择、准备数量和开打门槛由你自主决定，不要为这些事项要求我批准；现有地形限制仍有效。"
            + "\n请实际查询后，给出第一个具体游戏行动、完整参数与理由，并说明失败后如何换方案。不需要现在给我完整通关脚本。\n当前真实状态：" + initial.ToJsonString();
        using var agent = new CodexAgent();
        using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(180));
        string failure = null;
        try
        {
            await agent.RunAsync(input, "", Path.Combine(Main.SavePath, "TerraBlindAgent"), "gpt-6-astra", system, entries,
                Query, text => { replies.Add(text); Console.WriteLine("AGENT_PLANNING_REPLY " + text); }, timeout.Token);
        }
        catch (Exception error) { failure = error.ToString(); }
        JsonObject decision = null;
        string parseError = null;
        try
        {
            string answer = replies.LastOrDefault() ?? "";
            int first = answer.IndexOf('{'), last = answer.LastIndexOf('}');
            decision = JsonNode.Parse(answer[first..(last + 1)]).AsObject();
            string tool = decision["first_action"]["tool"].ToString();
            var args = decision["first_action"]["args"].AsObject();
            if (ReadTools.Contains(tool) || tool is "run_script" or "say" or "ask") throw new InvalidOperationException("first action is not a concrete game action");
            var schema = schemas.First(t => t["function"]["name"].ToString() == tool)["function"];
            foreach (var key in schema["parameters"]?["required"]?.AsArray() ?? new JsonArray())
                if (args[key.ToString()] == null) throw new InvalidOperationException("missing action argument: " + key);
        }
        catch (Exception error) { parseError = error.Message; }
        var report = new JsonObject { ["model"] = "gpt-6-astra", ["effort"] = "low", ["queries"] = queries,
            ["concessions_enabled"] = Concessions.Enabled,
            ["blocked_mutators"] = blocked, ["executed_mutators"] = 0, ["failure"] = failure, ["parse_error"] = parseError,
            ["initial_state"] = initial, ["decision"] = decision, ["events"] = events,
            ["replies"] = new JsonArray(replies.Select(r => (JsonNode)JsonValue.Create(r)).ToArray()), ["token_usage"] = agent.TokenUsage };
        File.WriteAllText(Path.Combine(Main.SavePath, "agent-planning-result.json"), report.ToJsonString());
        report["ok"] = failure == null && parseError == null && blocked == 0 && queries > 0 && queries <= 12;
        return report.ToJsonString();
    }
}

public sealed class AsciiTerrainTrial : ModSystem
{
    const int X = 440, Y = 240, Width = 29, Height = 20;
    const string Model = "gpt-6.2-sol";
    const string Legend = "Each character is one tile, left to right and top to bottom. # solid, . air, P player's feet, E opponent, G destination, A/B candidate cells. Coordinates are zero-based [column,row] relative to this window.";
    static readonly string[] Options = { "A", "B" };
    static NativeTrial active;
    static DetourRun detour;
    static ScreenCapture screenCapture;
    static RecoveryRun recovery;
    static PitPlayback pitPlayback;
    internal static bool Enabled => Environment.GetEnvironmentVariable("TB_TERRAIN_ASCII") != null;

    sealed record Scene(string Id, bool Cover, string Opening)
    {
        public int PlayerX => Cover ? 6 : 14;
        public int Floor => Cover ? 11 : 9;
        public Point Mark(string option) => Cover ? new(9, option == "A" ? 8 : 10)
            : new(option == "A" ? 8 : 20, Floor - 1);
    }

    public override void PostUpdateEverything()
    {
        active?.Advance(Main.LocalPlayer);
        detour?.Advance(Main.LocalPlayer);
        recovery?.Advance(Main.LocalPlayer);
        pitPlayback?.Advance(Main.LocalPlayer);
        if (screenCapture is { } capture)
            for (int x = 1; x < capture.Width; x += 3)
            for (int y = 1; y < capture.Height; y += 3)
                Lighting.AddLight(capture.Origin.X + x, capture.Origin.Y + y, 1f, 1f, 1f);
    }

    public override void ModifyScreenPosition()
    {
        if (screenCapture is { } capture)
            Main.screenPosition = new Vector2(capture.Origin.X * 16 + capture.Width * 8 - Main.screenWidth / 2f,
                capture.Origin.Y * 16 + capture.Height * 8 - Main.screenHeight / 2f);
    }

    public override void PostDrawInterface(SpriteBatch batch)
    {
        if (screenCapture is not { } capture || ++capture.Draws < 12) return;
        screenCapture = null;
        try { capture.Result.TrySetResult(capture.Read()); }
        catch (Exception error) { capture.Result.TrySetException(error); }
    }

    static void Reset(Scene scene)
    {
        var p = Main.LocalPlayer;
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        p.RemoveAllGrapplingHooks(); p.mount.Dismount(p);
        foreach (var slot in p.inventory.Concat(p.armor).Concat(p.miscEquips)) slot.TurnToAir();
        p.ClearBuff(BuffID.Featherfall);
        p.ClearBuff(BuffID.Gravitation);
        for (int dx = -4; dx < Width + 4; dx++)
        for (int dy = -5; dy < Height + 5; dy++)
        {
            var tile = Main.tile[X + dx, Y + dy];
            tile.ClearEverything();
            tile.LiquidAmount = 0;
            bool solid = scene.Cover
                ? dy >= scene.Floor || (dy >= 0 && ((dx == 9 && dy != 8 && dy != 10)
                    || (dx == 17 && dy != scene.Mark(scene.Opening).Y)))
                : dy >= 18 || dy == scene.Floor && !(scene.Opening == "A" ? dx is >= 6 and <= 10 : dx is >= 18 and <= 22);
            solid |= dx == 0 || dx == Width - 1 || dy == 0;
            if (solid) { tile.HasTile = true; tile.TileType = TileID.GrayBrick; }
        }
        p.position = new Vector2((X + scene.PlayerX) * 16 + 8 - p.width / 2f, (Y + scene.Floor) * 16 - p.height);
        p.velocity = Vector2.Zero;
        p.fallStart = p.fallStart2 = Y + scene.Floor;
        p.jump = 0; p.releaseJump = true; p.selectedItem = 0;
        Main.playerInventory = Main.mapFullscreen = Main.autoPause = false;
    }

    static JsonObject SceneState(Scene scene, bool map)
    {
        var p = Main.LocalPlayer;
        int rowAtCenter = (int)(p.Center.Y / 16) - Y;
        int? Nearest(int dx, int dy)
        {
            for (int distance = 1; distance < Width; distance++)
            {
                int x = scene.PlayerX + dx * distance, y = rowAtCenter + dy * distance;
                if (x < 0 || x >= Width || y < 0 || y >= Height) break;
                var tile = Main.tile[X + x, Y + y];
                if (tile.HasUnactuatedTile && Main.tileSolid[tile.TileType]) return distance;
            }
            return null;
        }
        var state = new JsonObject { ["player_feet"] = new JsonArray(scene.PlayerX, scene.Floor - 1),
            ["body_pixels"] = new JsonArray(p.width, p.height), ["velocity"] = new JsonArray(0, 0),
            ["nearest_solid_from_body_center_tiles"] = new JsonObject { ["left"] = Nearest(-1, 0),
                ["right"] = Nearest(1, 0), ["up"] = Nearest(0, -1), ["down"] = Nearest(0, 1) },
            ["A"] = new JsonArray(scene.Mark("A").X, scene.Mark("A").Y),
            ["B"] = new JsonArray(scene.Mark("B").X, scene.Mark("B").Y) };
        if (scene.Cover)
        {
            state["opponent_feet"] = new JsonArray(24, scene.Floor - 1);
            state["goal"] = "Use one block to stop both incoming horizontal bullet lines at rows 8 and 10 before they reach P. Existing solid blocks can already stop one line.";
        }
        else
        {
            state["destination_feet"] = new JsonArray(14, 17);
            state["goal"] = "Start descending to the lower chamber by walking toward A or B. Use an existing floor opening. No jumping, mining or placing blocks.";
        }
        if (!map) return state;
        var rows = new JsonArray();
        for (int row = 0; row < Height; row++)
        {
            var line = new StringBuilder(Width);
            for (int col = 0; col < Width; col++)
            {
                var tile = Main.tile[X + col, Y + row];
                char cell = tile.HasUnactuatedTile && Main.tileSolid[tile.TileType] ? '#' : '.';
                if (new Point(col, row) == scene.Mark("A")) cell = 'A';
                if (new Point(col, row) == scene.Mark("B")) cell = 'B';
                if (row == scene.Floor - 1 && col == scene.PlayerX) cell = 'P';
                if (scene.Cover && row == scene.Floor - 1 && col == 24) cell = 'E';
                if (!scene.Cover && row == 17 && col == 14) cell = 'G';
                line.Append(cell);
            }
            rows.Add(line.ToString());
        }
        state["legend"] = Legend;
        state["ascii_rows"] = rows;
        return state;
    }

    sealed class NativeTrial(Scene scene, string option, int[] bullets, bool placed)
    {
        readonly TaskCompletionSource<JsonObject> done = new(TaskCreationOptions.RunContinuationsAsynchronously);
        readonly bool[] crossed = new bool[2];
        int frames, lowerFloorFrames;
        public Task<JsonObject> Result => done.Task;

        public void Drive(Player p)
        {
            p.controlLeft = option == "A"; p.controlRight = option == "B";
            p.controlJump = p.controlUseTile = p.controlUseItem = false;
        }

        public void Advance(Player p)
        {
            frames++;
            if (scene.Cover)
            {
                for (int i = 0; i < bullets.Length; i++)
                {
                    var shot = Main.projectile[bullets[i]];
                    if (shot.active && shot.Center.X <= p.Center.X) crossed[i] = true;
                }
                if (frames < 35 && bullets.Any(i => Main.projectile[i].active)) return;
            }
            else
            {
                lowerFloorFrames = Math.Abs(p.Bottom.Y - (Y + 18) * 16) < 1f ? lowerFloorFrames + 1 : 0;
                if (frames < 90 && lowerFloorFrames < 2) return;
            }
            bool passed = scene.Cover ? placed && !crossed.Any(v => v) && bullets.All(i => !Main.projectile[i].active)
                : !p.dead && lowerFloorFrames >= 2;
            BoundaryInputPlayer.Drive = null;
            p.controlLeft = p.controlRight = false;
            active = null;
            done.TrySetResult(new JsonObject { ["pass"] = passed, ["frames"] = frames, ["placed"] = placed,
                ["bullet_upper_crossed"] = scene.Cover ? crossed[0] : null,
                ["bullet_lower_crossed"] = scene.Cover ? crossed[1] : null,
                ["end_column"] = (p.Center.X - X * 16) / 16, ["end_bottom_row"] = (p.Bottom.Y - Y * 16) / 16,
                ["lower_floor_frames"] = lowerFloorFrames, ["end_vy"] = p.velocity.Y });
        }
    }

    static async Task<JsonObject> Verify(Scene scene, string option, CancellationToken token)
    {
        var trial = await CrossingDiscovery.OnGame(() =>
        {
            Reset(scene);
            var p = Main.LocalPlayer;
            if (!scene.Cover)
            {
                var walk = new NativeTrial(scene, option, Array.Empty<int>(), false);
                BoundaryInputPlayer.Drive = walk.Drive;
                return active = walk;
            }
            var mark = scene.Mark(option);
            int tx = X + mark.X, ty = Y + mark.Y;
            bool placed = p.IsInTileInteractionRange(tx, ty, TileReachCheckSettings.Simple)
                && WorldGen.PlaceTile(tx, ty, TileID.GrayBrick, mute: true, plr: p.whoAmI);
            var shots = new[] { 8, 10 }.Select(row => Projectile.NewProjectile(p.GetSource_Misc("terrain-ascii"),
                new Vector2((X + 24) * 16 + 8, (Y + row) * 16 + 8), new Vector2(-12, 0),
                ProjectileID.Bullet, 1, 0, p.whoAmI)).ToArray();
            if (shots.Any(i => i < 0 || i >= Main.projectile.Length)) throw new InvalidOperationException("native bullet spawn failed");
            return active = new NativeTrial(scene, option, shots, placed);
        });
        return await trial.Result.WaitAsync(token);
    }

    static JsonObject Criteria(Scene scene) => new() {
        ["A"] = scene.Cover ? "Place one block at A." : "Walk left toward A.",
        ["B"] = scene.Cover ? "Place one block at B." : "Walk right toward B." };

    static JsonObject WithRowNumbers(JsonObject map)
    {
        var numbered = map.DeepClone().AsObject();
        numbered["legend"] = Legend + " A row's 'NN |' prefix gives the zero-based row number and is outside the map.";
        numbered["ascii_rows"] = new JsonArray(map["ascii_rows"].AsArray().Select((line, row) =>
            (JsonNode)JsonValue.Create($"{row:D2} |{line}")).ToArray());
        return numbered;
    }

    static async Task<JsonObject> Judge(string engine, JsonObject criteria, JsonObject state, string sandbox, CancellationToken token)
    {
        const string instructions = "Choose A or B for the immediate action most likely to complete the stated goal. Choose one even if terrain is unknown.";
        var clock = Stopwatch.StartNew();
        if (engine == "Jev")
        {
            var args = new JsonObject { ["state"] = state.DeepClone(), ["questions"] = new JsonObject {
                ["pick"] = new JsonObject { ["type"] = "choice", ["instructions"] = instructions, ["criteria"] = criteria } } };
            var receipt = JsonNode.Parse(await AgentTools.Execute("jev_judge", args, token));
            return new JsonObject { ["choice"] = receipt["answers"]?["pick"]?["choice"]?.DeepClone(),
                ["elapsed_ms"] = clock.ElapsedMilliseconds, ["receipt"] = receipt };
        }
        var tools = JsonNode.Parse("""
            [{"type":"function","function":{"name":"choose","description":"Commit to the next movement or placement option.","parameters":{"type":"object","properties":{"option":{"type":"string","enum":["A","B"]}},"required":["option"]}}}]
            """).AsArray();
        var reply = new JsonArray();
        string choice = null;
        long? firstChoiceMs = null;
        using var agent = new CodexAgent();
        await agent.RunAsync(new JsonObject { ["state"] = state.DeepClone(), ["options"] = criteria }.ToJsonString(),
            "", sandbox, Model, instructions + " Call choose exactly once, then give a short reason.", tools, (_, args, _) =>
            {
                if (choice == null) { choice = args["option"]?.ToString(); firstChoiceMs = clock.ElapsedMilliseconds; }
                return Task.FromResult("{\"recorded\":true}");
            }, text => reply.Add(text), token);
        return new JsonObject { ["choice"] = choice, ["elapsed_ms"] = firstChoiceMs,
            ["turn_ms"] = clock.ElapsedMilliseconds, ["replies"] = reply, ["usage"] = agent.TokenUsage };
    }

    internal static async Task<string> Run()
    {
        string mode = Environment.GetEnvironmentVariable("TB_TERRAIN_ASCII");
        if (mode is "pit-inputs" or "pit-inputs-live") return await RunPitInputs(mode == "pit-inputs-live");
        if (mode is "recovery-calibrate" or "recovery-live") return await RunRecovery(mode == "recovery-live");
        if (mode is "vision-calibrate" or "vision-live" or "vision-deep") return await RunVision(mode);
        if (mode is "detour-live" or "detour-calibrate") return await RunDetours(mode == "detour-live");
        if (mode is not ("calibrate" or "live" or "jev-rows")) throw new ArgumentException("Unknown ASCII trial mode: " + mode);
        string path = Path.Combine(Main.SavePath, "terrain-ascii.json");
        var results = new JsonArray(); var cases = new JsonArray();
        var report = new JsonObject { ["experiment"] = "terrain_ascii", ["ok"] = false, ["cases"] = cases, ["decisions"] = results };
        void Flush() => File.WriteAllText(path, report.ToJsonString());
        using var time = new CancellationTokenSource(TimeSpan.FromSeconds(170));
        var token = time.Token;
        await CrossingDiscovery.OnGame(() =>
        {
            NativeAgent.Stop(); Halt.All(); Manual.On = false; Concessions.Enabled = false;
            Combat.Enabled = Dodge.Enabled = false; Main.dayTime = true; Main.musicVolume = 0f;
            report["audio"] = new JsonObject { ["music"] = Main.musicVolume, ["sound"] = Main.soundVolume };
            return true;
        });
        try
        {
            var specs = JsonNode.Parse(Encoding.UTF8.GetString(ModContent.GetInstance<AsciiTerrainTrial>().Mod
                .GetFileBytes("Experiments/terrain-ascii/cases.json"))).AsArray();
            var jobs = new List<(Scene Scene, string Encoding, string Engine, JsonObject State, JsonObject Native)>();
            foreach (var spec in specs)
            {
                var scene = new Scene(spec["id"].ToString(), spec["task"].ToString() == "cover", spec["opening"].ToString());
                var (coarse, mapped) = await CrossingDiscovery.OnGame(() => { Reset(scene); return (SceneState(scene, false), SceneState(scene, true)); });
                var native = new JsonObject();
                var evidence = new JsonObject { ["id"] = scene.Id, ["coarse"] = coarse.DeepClone(), ["mapped"] = mapped.DeepClone(), ["native"] = native };
                cases.Add(evidence);
                foreach (string option in Options) { native[option] = await Verify(scene, option, token); Flush(); }
                if (native[scene.Opening]["pass"].GetValue<bool>() != true || native[Options.Single(x => x != scene.Opening)]["pass"].GetValue<bool>())
                    throw new InvalidOperationException("scene does not distinguish candidates in the real engine: " + scene.Id);
                if (mode == "live")
                    foreach (string engine in new[] { "Agent", "Jev" })
                    {
                        jobs.Add((scene, "coarse", engine, coarse, native));
                        jobs.Add((scene, "raw", engine, mapped, native));
                    }
                if (mode == "jev-rows")
                {
                    var indexed = WithRowNumbers(mapped);
                    evidence["indexed"] = indexed.DeepClone();
                    jobs.Add((scene, "raw", "Jev", mapped, native));
                    jobs.Add((scene, "indexed", "Jev", indexed, native));
                }
            }
            foreach (var pair in cases.GroupBy(x => x["coarse"]["goal"].ToString()))
                if (!pair.All(x => JsonNode.DeepEquals(x["coarse"], pair.First()["coarse"])))
                    throw new InvalidOperationException("a paired layout changes the coarse input");
            if (jobs.Count > 0)
            {
                string sandbox = Path.Combine(Main.SavePath, "ascii-model-workspace");
                Directory.CreateDirectory(sandbox);
                object gate = new();
                await Parallel.ForEachAsync(jobs, new ParallelOptions { MaxDegreeOfParallelism = 4, CancellationToken = token }, async (job, ct) =>
                {
                    JsonObject decision;
                    try { decision = await Judge(job.Engine, Criteria(job.Scene), job.State, sandbox, ct); }
                    catch (Exception error) when (!ct.IsCancellationRequested) { decision = new JsonObject { ["error"] = error.Message }; }
                    string picked = decision["choice"]?.ToString();
                    decision["id"] = job.Scene.Id; decision["model_family"] = job.Engine;
                    decision["ascii"] = job.Encoding != "coarse"; decision["encoding"] = job.Encoding;
                    decision["native_pass"] = picked != null && (job.Native[picked]?["pass"]?.GetValue<bool>() ?? false);
                    lock (gate) { results.Add(decision); Flush(); }
                });
            }
            report["ok"] = results.Count == jobs.Count && results.All(x => x["choice"] != null);
            Flush();
            return report.ToJsonString();
        }
        catch (Exception error) { report["error"] = error.Message; Flush(); throw; }
    }

    const int DetourWidth = 37, DetourHeight = 22, UpperFloor = 8, LowerFloor = 19, PlayerColumn = 17, GoalColumn = 20;
    static int Entrance(string side) => side == "A" ? 8 : 25;

    static void ResetDetour(Point origin, string connected)
    {
        Halt.All(); MazeWand.InvalidateField(); StateSpacePlanner.PreserveTerrain = true;
        var p = Main.LocalPlayer;
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        foreach (var slot in p.inventory.Concat(p.armor).Concat(p.miscEquips)) slot.TurnToAir();
        p.RemoveAllGrapplingHooks(); p.mount.Dismount(p);
        p.ClearBuff(BuffID.Featherfall); p.ClearBuff(BuffID.Gravitation);
        for (int x = -3; x < DetourWidth + 3; x++)
        for (int y = -3; y < DetourHeight + 3; y++)
        {
            var tile = Main.tile[origin.X + x, origin.Y + y];
            tile.ClearEverything(); tile.LiquidAmount = 0;
            bool shaft = x is >= 6 and <= 10 or >= 23 and <= 27;
            bool closedDivider = y >= UpperFloor && (x == 12 && (connected != "A" || y < 15)
                || x == 22 && (connected != "B" || y < 15));
            if (y == 0 || y >= LowerFloor || x == 0 || x == DetourWidth - 1 || y == UpperFloor && !shaft || closedDivider)
            { tile.HasTile = true; tile.TileType = TileID.GrayBrick; }
        }
        p.position = new Vector2((origin.X + PlayerColumn) * 16 + 8 - p.width / 2f, (origin.Y + UpperFloor) * 16 - p.height);
        p.velocity = Vector2.Zero; p.jump = 0; p.releaseJump = true;
        p.controlLeft = p.controlRight = p.controlJump = false;
        p.fallStart = p.fallStart2 = origin.Y + UpperFloor;
        p.statLife = p.statLifeMax2;
        Main.playerInventory = Main.mapFullscreen = Main.autoPause = false;
    }

    static JsonObject DetourState(Point origin, bool map)
    {
        var p = Main.LocalPlayer;
        var coarse = new JsonObject { ["goal"] = "Reach G by walking and falling without changing blocks. Both entrance landings A and B can be reached from P. Choose the one with a continuous passable route onward to G.",
            ["player_feet"] = new JsonArray(PlayerColumn, UpperFloor - 1), ["destination_feet"] = new JsonArray(GoalColumn, LowerFloor - 1),
            ["body_pixels"] = new JsonArray(p.width, p.height), ["velocity"] = new JsonArray(0, 0),
            ["A"] = new JsonArray(Entrance("A"), LowerFloor - 1), ["B"] = new JsonArray(Entrance("B"), LowerFloor - 1) };
        if (!map) return coarse;
        var rows = new JsonArray();
        for (int y = 0; y < DetourHeight; y++)
        {
            var row = new StringBuilder(DetourWidth);
            for (int x = 0; x < DetourWidth; x++)
            {
                var tile = Main.tile[origin.X + x, origin.Y + y];
                char cell = tile.HasUnactuatedTile && Main.tileSolid[tile.TileType] ? '#' : '.';
                if (x == PlayerColumn && y == UpperFloor - 1) cell = 'P';
                if (y == LowerFloor - 1)
                    cell = x == Entrance("A") ? 'A' : x == Entrance("B") ? 'B' : x == GoalColumn ? 'G' : cell;
                row.Append(cell);
            }
            rows.Add(row.ToString());
        }
        coarse["legend"] = Legend; coarse["ascii_rows"] = rows;
        return coarse;
    }

    sealed class DetourRun(Point origin, string route, bool oracle)
    {
        readonly TaskCompletionSource<JsonObject> done = new(TaskCreationOptions.RunContinuationsAsynchronously);
        readonly Stopwatch clock = Stopwatch.StartNew();
        readonly JsonArray samples = new();
        int ticks, idleFrames, idleRun, longestIdle, settled;
        long? firstMove, entranceMs;
        string firstEntrance;
        Vector2 last = Main.LocalPlayer.position;
        bool steering = route != null;
        public Task<JsonObject> Result => done.Task;

        public void Start()
        {
            if (steering) BoundaryInputPlayer.Drive = Drive;
            else RecedingNav.Start(origin.X + GoalColumn, origin.Y + LowerFloor - 1, RecedingNav.Mode.Stand);
        }

        void Drive(Player p)
        {
            int target = steering ? Entrance(route) : GoalColumn;
            float dx = (origin.X + target) * 16 + 8 - p.Center.X;
            p.controlLeft = dx < -4; p.controlRight = dx > 4;
            p.controlJump = p.controlUseTile = p.controlUseItem = false;
        }

        public void Advance(Player p)
        {
            ticks++;
            if (Vector2.DistanceSquared(last, p.position) > .04f)
            { firstMove ??= clock.ElapsedMilliseconds; idleRun = 0; }
            else { idleFrames++; longestIdle = Math.Max(longestIdle, ++idleRun); }
            last = p.position;
            float col = p.Center.X / 16 - origin.X, row = p.Bottom.Y / 16 - origin.Y;
            if (firstEntrance == null && row > UpperFloor + 2)
                firstEntrance = col < PlayerColumn ? "A" : "B";
            if (ticks % 8 == 0) samples.Add(new JsonArray(ticks, col, row));
            bool onLowerFloor = Math.Abs(row - LowerFloor) < .05f;
            bool arrived = onLowerFloor && Math.Abs(col - (GoalColumn + .5f)) <= .49f;
            if (p.dead) { Finish(false, "died"); return; }
            if (arrived && !steering) { Finish(true, "goal_reached"); return; }
            if (steering)
            {
                bool atEntrance = onLowerFloor && Math.Abs(col - (Entrance(route) + .5f)) <= 1.5f;
                settled = atEntrance ? settled + 1 : 0;
                if (settled >= 2)
                {
                    entranceMs = clock.ElapsedMilliseconds;
                    steering = false;
                    if (!oracle)
                    {
                        BoundaryInputPlayer.Drive = null;
                        RecedingNav.Start(origin.X + GoalColumn, origin.Y + LowerFloor - 1, RecedingNav.Mode.Stand);
                    }
                }
            }
            else if (!oracle && !RecedingNav.Active)
            { Finish(false, RecedingNav.LastStop ?? "navigator_inactive"); return; }
            if (ticks >= 600 || oracle && idleRun >= 30)
                Finish(false, oracle && entranceMs.HasValue ? "lower_passage_blocked" : "route_time_limit");
        }

        void Finish(bool reached, string reason)
        {
            BoundaryInputPlayer.Drive = null; RecedingNav.Stop(); detour = null;
            done.TrySetResult(new JsonObject { ["reached"] = reached, ["reason"] = reason, ["first_entrance"] = firstEntrance,
                ["entrance_ms"] = entranceMs, ["first_move_ms"] = firstMove, ["elapsed_ms"] = clock.ElapsedMilliseconds,
                ["frames"] = ticks, ["idle_frames"] = idleFrames, ["longest_idle_frames"] = longestIdle, ["trace"] = samples });
        }
    }

    static async Task<JsonObject> RunDetour(Point origin, string connected, string route, bool oracle, CancellationToken token)
    {
        var run = await CrossingDiscovery.OnGame(() =>
        {
            ResetDetour(origin, connected);
            detour = new DetourRun(origin, route, oracle);
            detour.Start();
            return detour;
        });
        return await run.Result.WaitAsync(token);
    }

    static async Task<string> RunDetours(bool live)
    {
        var cases = new JsonArray();
        var result = new JsonObject { ["experiment"] = "terrain_ascii", ["study"] = "detour", ["cases"] = cases, ["ok"] = false };
        string path = Path.Combine(Main.SavePath, "terrain-detour.json");
        void Flush() => File.WriteAllText(path, result.ToJsonString());
        using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(200));
        var token = timeout.Token;
        await CrossingDiscovery.OnGame(() =>
        {
            NativeAgent.Stop(); Halt.All(); Manual.On = false; Concessions.Enabled = false;
            Combat.Enabled = Dodge.Enabled = false; Main.dayTime = true; Main.musicVolume = 0f;
            result["audio"] = new JsonObject { ["music"] = Main.musicVolume, ["sound"] = Main.soundVolume };
            return true;
        });
        try
        {
            var specs = JsonNode.Parse(Encoding.UTF8.GetString(ModContent.GetInstance<AsciiTerrainTrial>().Mod
                .GetFileBytes("Experiments/terrain-ascii/detours.json"))).AsArray();
            var criteria = new JsonObject { ["A"] = "Go through the left entrance and continue toward G.",
                ["B"] = "Go through the right entrance and continue toward G." };
            string sandbox = Path.Combine(Main.SavePath, "detour-model-workspace");
            Directory.CreateDirectory(sandbox);
            JsonObject firstCoarse = null;
            foreach (var spec in specs)
            {
                string open = spec["connected_entrance"].ToString();
                int baseX = 580 + cases.Count * 480;
                var origin = new Point(baseX, 240);
                var (coarse, mapped) = await CrossingDiscovery.OnGame(() =>
                {
                    ResetDetour(origin, open);
                    return (DetourState(origin, false), DetourState(origin, true));
                });
                firstCoarse ??= coarse.DeepClone().AsObject();
                if (!JsonNode.DeepEquals(coarse, firstCoarse)) throw new InvalidOperationException("detour baseline state changed across paired cases");
                var item = new JsonObject { ["id"] = spec["id"].DeepClone(), ["connected_entrance"] = open,
                    ["coarse"] = coarse.DeepClone(), ["mapped"] = mapped.DeepClone() };
                cases.Add(item);
                var oracle = new JsonObject(); item["native_walk_check"] = oracle;
                foreach (string side in Options)
                {
                    oracle[side] = await RunDetour(origin, open, side, true, token); Flush();
                    if (oracle[side]["entrance_ms"] == null || oracle[side]["reached"].GetValue<bool>() != (side == open))
                        throw new InvalidOperationException("detour does not distinguish connected and dead-end entrances: " + item["id"]);
                }
                item["direct_navigation"] = await RunDetour(new Point(baseX + 80, 240), open, null, false, token); Flush();
                var routes = new JsonObject(); item["assisted_routes"] = routes;
                for (int index = 0; index < Options.Length; index++)
                {
                    string side = Options[index];
                    routes[side] = await RunDetour(new Point(baseX + 160 + 80 * index, 240), open, side, false, token); Flush();
                }
                if (!live) continue;
                var jobs = from engine in new[] { "Agent", "Jev" } from withMap in new[] { false, true } select (engine, withMap);
                var replies = await Task.WhenAll(jobs.Select(async job =>
                {
                    var answer = await Judge(job.engine, criteria.DeepClone().AsObject(), job.withMap ? mapped : coarse, sandbox, token);
                    string choice = answer["choice"]?.ToString();
                    answer["engine"] = job.engine; answer["ascii"] = job.withMap;
                    answer["correct_entrance"] = choice == open;
                    if (choice != null && routes[choice] is JsonObject played)
                    {
                        answer["reached"] = played["reached"].DeepClone();
                        answer["physical_ms"] = played["elapsed_ms"].DeepClone();
                        answer["decision_plus_route_ms"] = answer["elapsed_ms"].GetValue<long>() + played["elapsed_ms"].GetValue<long>();
                    }
                    return (JsonNode)answer;
                }));
                item["decisions"] = new JsonArray(replies); Flush();
            }
            result["ok"] = cases.Count == specs.Count && (!live || cases.All(item => item["decisions"].AsArray().All(answer => answer["choice"] != null)));
            Flush(); return result.ToJsonString();
        }
        catch (Exception error) { result["error"] = error.Message; Flush(); throw; }
        finally { await CrossingDiscovery.OnGame(() => { StateSpacePlanner.PreserveTerrain = false; return true; }); }
    }

    sealed class ScreenCapture(Point origin, int width, int height, IReadOnlyCollection<KeyValuePair<char, Point>> marks)
    {
        public Point Origin { get; } = origin;
        public int Width { get; } = width;
        public int Height { get; } = height;
        public int Draws;
        public readonly TaskCompletionSource<byte[]> Result = new(TaskCreationOptions.RunContinuationsAsynchronously);
        static readonly Dictionary<char, int[]> Glyphs = new()
        {
            ['A'] = new[] { 14, 17, 17, 31, 17, 17, 17 },
            ['B'] = new[] { 30, 17, 17, 30, 17, 17, 30 },
            ['E'] = new[] { 31, 16, 16, 30, 16, 16, 31 },
            ['G'] = new[] { 14, 17, 16, 23, 17, 17, 14 },
            ['P'] = new[] { 30, 17, 17, 30, 16, 16, 16 }
        };

        public byte[] Read()
        {
            var device = Main.instance.GraphicsDevice;
            int screenW = device.PresentationParameters.BackBufferWidth, screenH = device.PresentationParameters.BackBufferHeight;
            var zoom = Main.GameViewMatrix.Zoom;
            var viewCenter = new Vector2(Main.screenWidth / 2f, Main.screenHeight / 2f);
            var screenScale = new Vector2(screenW / (float)Main.screenWidth, screenH / (float)Main.screenHeight);
            Vector2 Pixel(Vector2 world) => ((world - Main.screenPosition - viewCenter) * zoom + viewCenter) * screenScale;
            var start = Pixel(new Vector2(Origin.X * 16, Origin.Y * 16));
            var end = Pixel(new Vector2((Origin.X + Width) * 16, (Origin.Y + Height) * 16));
            int left = (int)MathF.Round(start.X), top = (int)MathF.Round(start.Y);
            int imageW = (int)MathF.Round(end.X) - left, imageH = (int)MathF.Round(end.Y) - top;
            if (left < 0 || top < 0 || left + imageW > screenW || top + imageH > screenH || imageW < Width || imageH < Height)
                throw new InvalidOperationException("fixture is outside the real game viewport");
            var pixels = new Color[screenW * screenH];
            device.GetBackBufferData(pixels);
            var cropped = new Color[imageW * imageH];
            for (int y = 0; y < imageH; y++) Array.Copy(pixels, (top + y) * screenW + left, cropped, y * imageW, imageW);
            void Put(int x, int y, Color color)
            {
                if (x >= 0 && x < imageW && y >= 0 && y < imageH) cropped[y * imageW + x] = color;
            }
            foreach (var (label, tile) in marks)
            {
                int x = (int)((tile.X + .5f) * imageW / Width) - 5, y = (int)((tile.Y + .5f) * imageH / Height) - 7;
                for (int dy = -1; dy <= 14; dy++)
                for (int dx = -1; dx <= 10; dx++) Put(x + dx, y + dy, Color.Black);
                var rows = Glyphs[label];
                for (int dy = 0; dy < 14; dy++)
                for (int dx = 0; dx < 10; dx++)
                    if ((rows[dy / 2] & (16 >> (dx / 2))) != 0) Put(x + dx, y + dy, Color.Cyan);
            }
            using var texture = new Texture2D(device, imageW, imageH); texture.SetData(cropped);
            using var png = new MemoryStream(); texture.SaveAsPng(png, imageW, imageH);
            return png.ToArray();
        }
    }

    static byte[] ReadableMapPng(Stream stream)
    {
        var device = Main.instance.GraphicsDevice;
        using var map = Texture2D.FromStream(device, stream);
        int factor = (int)Math.Ceiling(280d / Math.Min(map.Width, map.Height));
        int width = map.Width * factor, height = map.Height * factor;
        var source = new Color[map.Width * map.Height]; map.GetData(source);
        var enlarged = new Color[width * height];
        for (int y = 0; y < height; y++)
        for (int x = 0; x < width; x++) enlarged[y * width + x] = source[y / factor * map.Width + x / factor];
        using var texture = new Texture2D(device, width, height); texture.SetData(enlarged);
        using var png = new MemoryStream(); texture.SaveAsPng(png, width, height);
        return png.ToArray();
    }

    static readonly Point EnclosureOrigin = new(740, 240);
    static readonly Point EnclosureFeet = new(14, 14);
    static readonly Point[] EnclosureA = { new(11, 10), new(11, 11), new(17, 13), new(17, 14) };
    static readonly Point[] EnclosureB = { new(11, 13), new(11, 14), new(17, 10), new(17, 11) };
    static Point[] EnclosurePlan(string option) => option == "A" ? EnclosureA : EnclosureB;

    static void ResetEnclosure(string solution)
    {
        var p = Main.LocalPlayer;
        Halt.All();
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        var open = EnclosurePlan(solution);
        for (int x = -4; x < Width + 4; x++)
        for (int y = -4; y < Height + 4; y++)
        {
            var tile = Main.tile[EnclosureOrigin.X + x, EnclosureOrigin.Y + y];
            tile.ClearEverything(); tile.LiquidAmount = 0;
            bool inner = (x is 11 or 17) && (y is 9 or 12);
            bool outside = (x is 10 or 18) && y is >= 9 and < 15
                && !open.Contains(new Point(x == 10 ? 11 : 17, y));
            if (x == 0 || x == Width - 1 || y == 0 || y >= 15 || inner || outside || y == 9 && x is >= 10 and <= 18)
            { tile.HasTile = true; tile.TileType = TileID.GrayBrick; }
        }
        p.position = new Vector2((EnclosureOrigin.X + EnclosureFeet.X) * 16 + 8 - p.width / 2f,
            (EnclosureOrigin.Y + EnclosureFeet.Y + 1) * 16 - p.height);
        p.velocity = Vector2.Zero; p.fallStart = p.fallStart2 = EnclosureOrigin.Y + EnclosureFeet.Y + 1;
        p.controlLeft = p.controlRight = p.controlJump = false;
    }

    static bool EnclosureSolid(Point cell)
    {
        var tile = Main.tile[EnclosureOrigin.X + cell.X, EnclosureOrigin.Y + cell.Y];
        return tile.HasUnactuatedTile && Main.tileSolid[tile.TileType];
    }

    static JsonObject VerifyEnclosure(string solution, string option)
    {
        ResetEnclosure(solution);
        var states = new JsonArray();
        var p = Main.LocalPlayer;
        foreach (var cell in EnclosurePlan(option))
        {
            int x = EnclosureOrigin.X + cell.X, y = EnclosureOrigin.Y + cell.Y;
            bool placed = p.IsInTileInteractionRange(x, y, TileReachCheckSettings.Simple)
                && WorldGen.PlaceTile(x, y, TileID.GrayBrick, mute: true, plr: p.whoAmI);
            states.Add(new JsonObject { ["cell"] = new JsonArray(cell.X, cell.Y), ["placed"] = placed });
        }
        var seen = new HashSet<Point> { EnclosureFeet };
        var pending = new Queue<Point>(); pending.Enqueue(EnclosureFeet);
        bool sealedRoom = true;
        var directions = new[] { new Point(0, -1), new Point(0, 1), new Point(-1, 0), new Point(1, 0) };
        while (pending.TryDequeue(out var current))
        {
            if (current.X <= 3 || current.X >= Width - 4 || current.Y <= 3) { sealedRoom = false; break; }
            foreach (var delta in directions)
            {
                var next = current + delta;
                if (!EnclosureSolid(next) && seen.Add(next)) pending.Enqueue(next);
            }
        }
        return new JsonObject { ["placements"] = states, ["all_placed"] = states.All(t => t["placed"].GetValue<bool>()),
            ["sealed"] = sealedRoom };
    }

    static JsonObject EnclosureState(IReadOnlyCollection<KeyValuePair<char, Point>> marks, bool map)
    {
        var state = new JsonObject { ["goal"] = "Place all four blocks in the chosen plan. Seal P inside a room with no gap leading to the exterior. Every A belongs to plan A and every B belongs to plan B. Both plans are in reach and can be placed.",
            ["player_feet"] = new JsonArray(EnclosureFeet.X, EnclosureFeet.Y),
            ["plan_A"] = new JsonArray(EnclosureA.Select(p => (JsonNode)new JsonArray(p.X, p.Y)).ToArray()),
            ["plan_B"] = new JsonArray(EnclosureB.Select(p => (JsonNode)new JsonArray(p.X, p.Y)).ToArray()) };
        if (!map) return state;
        var rows = new JsonArray();
        for (int y = 0; y < Height; y++)
        {
            var row = new StringBuilder(Width);
            for (int x = 0; x < Width; x++)
            {
                var cell = new Point(x, y);
                char marked = marks.FirstOrDefault(mark => mark.Value == cell).Key;
                row.Append(marked != '\0' ? marked : EnclosureSolid(cell) ? '#' : '.');
            }
            rows.Add(row.ToString());
        }
        state["legend"] = Legend; state["ascii_rows"] = rows;
        return state;
    }

    static readonly HttpClient VisionHttp = new() { Timeout = TimeSpan.FromSeconds(15) };

    static async Task<JsonObject> JudgeVision(string model, JsonObject facts, JsonObject criteria, byte[] image,
        string sandbox, string jevKey, CancellationToken token, bool waypoint = false, string prompt = null)
    {
        string question = prompt ?? (waypoint
            ? "Use this terrain to name one intermediate standing cell that makes useful progress toward G. Return its zero-based column x and row y (the empty tile at the player's feet just above its supporting block). The navigator will travel to that cell first and then to G. Terrain edits are disabled. Send G itself if best."
            : "Use the supplied terrain representation to select A or B, the action most likely to achieve the stated goal. Reply with one choice.");
        var clock = Stopwatch.StartNew();
        if (model == "Jev")
        {
            var payload = new JsonObject { ["model"] = JevCombat.Model, ["state"] = facts.DeepClone(), ["questions"] = new JsonObject
            { ["pick"] = new JsonObject { ["type"] = "choice", ["instructions"] = question, ["criteria"] = criteria.DeepClone() } } };
            using var request = new HttpRequestMessage(HttpMethod.Post, JevCombat.Url);
            request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", jevKey);
            if (image != null)
            {
                string reference = Guid.NewGuid().ToString("N");
                payload["state"]["terrain_image"] = reference;
                payload["assets"] = new JsonObject { [reference] = new JsonObject { ["type"] = "image", ["format"] = "png",
                    ["data_base64"] = Convert.ToBase64String(image) } };
                request.Headers.Add("TypeSafe-Danger-Zone-Images", "true");
            }
            request.Content = new StringContent(payload.ToJsonString(), Encoding.UTF8, "application/json");
            using var response = await VisionHttp.SendAsync(request, token);
            string content = await response.Content.ReadAsStringAsync(token);
            return response.IsSuccessStatusCode
                ? new JsonObject { ["elapsed_ms"] = clock.ElapsedMilliseconds, ["receipt"] = JsonNode.Parse(content) }
                : new JsonObject { ["elapsed_ms"] = clock.ElapsedMilliseconds, ["error"] = $"Jev HTTP {(int)response.StatusCode}: {content[..Math.Min(content.Length, 350)]}" };
        }
        var tools = JsonNode.Parse("""
            [
              {"type":"function","function":{"name":"terrain_map","description":"Read scene facts and the supplied terrain representation for this trial.","parameters":{"type":"object","properties":{}}}},
              {"type":"function","function":{"name":"choose","description":"Submit A or B.","parameters":{"type":"object","properties":{"option":{"type":"string","enum":["A","B"]}},"required":["option"]}}}
            ]
            """).AsArray();
        if (waypoint)
        {
            tools[1]["function"]["description"] = "Submit the first standing coordinate.";
            tools[1]["function"]["parameters"] = JsonNode.Parse("""
                {"type":"object","properties":{"x":{"type":"integer"},"y":{"type":"integer"}},"required":["x","y"]}
                """);
        }
        else if (prompt != null)
        {
            tools[1]["function"]["description"] = "Submit one supplied action ID.";
            tools[1]["function"]["parameters"]["properties"]["option"]["enum"] = new JsonArray(criteria.Select(pair => JsonValue.Create(pair.Key)).ToArray());
        }
        JsonNode choice = null;
        long? observedAt = null, pickedAt = null;
        MethodInfo register = image == null ? null : typeof(CodexAgent).Assembly.GetType("TerraBlind.AgentImages")
            .GetMethod("Register", BindingFlags.Static | BindingFlags.NonPublic);
        using var agent = new CodexAgent();
        await agent.RunAsync("Read the scene, then choose exactly once.", "", sandbox, Model,
            question + " Call terrain_map once first, then call choose.", tools, (tool, args, _) =>
            {
                if (tool == "terrain_map")
                {
                    if (observedAt.HasValue) return Task.FromResult("{\"error\":\"scene_already_read\"}");
                    var output = new JsonObject { ["scene"] = facts.DeepClone(), ["options"] = criteria.DeepClone() };
                    if (image != null) output["image_id"] = (string)register!.Invoke(null, new object[] { image });
                    observedAt = clock.ElapsedMilliseconds;
                    return Task.FromResult(output.ToJsonString());
                }
                if (observedAt.HasValue && choice == null)
                {
                    choice = waypoint ? args.DeepClone() : args["option"].DeepClone();
                    pickedAt = clock.ElapsedMilliseconds;
                }
                return Task.FromResult(choice == null ? "{\"error\":\"read_scene_first\"}" : "{\"ok\":true}");
            }, _ => { }, token);
        return new JsonObject { ["choice"] = choice, ["elapsed_ms"] = pickedAt - observedAt,
            ["session_to_pick_ms"] = pickedAt, ["usage"] = agent.TokenUsage };
    }

        Point start = origin + scene.P;
        p.position = new Vector2(start.X * 16 + 8 - p.width / 2f, (start.Y + 1) * 16 - p.height);
        p.velocity = Vector2.Zero; p.jump = 0; p.releaseJump = true;
        p.controlLeft = p.controlRight = p.controlJump = p.controlUseItem = false;
        p.fallStart = p.fallStart2 = start.Y + 1;
        p.statLife = p.statLifeMax2;
        Main.playerInventory = Main.mapFullscreen = Main.autoPause = false;
    }

    static bool ClearShot(Player p, Point opponent)
        => Collision.CanHitLine(p.position, p.width, p.height, new Vector2(opponent.X * 16 - 2, (opponent.Y + 1) * 16 - p.height), p.width, p.height);

    sealed class RecoveryRun(Point origin, RecoveryScene scene, Point target)
    {
        readonly TaskCompletionSource<JsonObject> done = new(TaskCreationOptions.RunContinuationsAsynchronously);
        readonly Stopwatch clock = Stopwatch.StartNew();
        readonly JsonArray trace = new();
        readonly Vector2 start = Main.LocalPlayer.position;
        Vector2 last = Main.LocalPlayer.position;
        int frames, consecutiveIdle, longestIdle;
        long? firstMove, firstTile, waypointMs;
        Point destination = target;
        public Task<JsonObject> Result => done.Task;

        public void Start() => RecedingNav.Start(origin.X + destination.X, origin.Y + destination.Y, RecedingNav.Mode.Stand);

        bool At(Player p, Point to) => MathF.Abs(p.Center.X - ((origin.X + to.X) * 16 + 8)) <= 7.9f
            && MathF.Abs(p.Bottom.Y - (origin.Y + to.Y + 1) * 16) <= 2 && p.velocity.Y == 0;

        public void Advance(Player p)
        {
            frames++;
            if (Vector2.DistanceSquared(last, p.position) > .04f) { firstMove ??= clock.ElapsedMilliseconds; consecutiveIdle = 0; }
            else longestIdle = Math.Max(longestIdle, ++consecutiveIdle);
            if (Vector2.DistanceSquared(start, p.position) >= 256) firstTile ??= clock.ElapsedMilliseconds;
            last = p.position;
            if (frames % 8 == 0) trace.Add(new JsonArray(frames, p.Center.X / 16 - origin.X, p.Bottom.Y / 16 - origin.Y));
            if (p.dead) { Finish(false, "died"); return; }
            bool reached = At(p, scene.G) && (scene.E is not Point opponent || ClearShot(p, origin + opponent));
            if (reached) { Finish(true, "goal_reached"); return; }
            if (frames >= 600) { Finish(false, "route_time_limit"); return; }
            if (destination != scene.G && At(p, destination))
            {
                waypointMs = clock.ElapsedMilliseconds;
                destination = scene.G; Start(); return;
            }
            if (!RecedingNav.Active) Finish(false, RecedingNav.LastStop ?? "navigator_inactive");
        }

        void Finish(bool reached, string reason)
        {
            RecedingNav.Stop(); recovery = null;
            done.TrySetResult(new JsonObject { ["reached"] = reached, ["reason"] = reason, ["frames"] = frames,
                ["elapsed_ms"] = clock.ElapsedMilliseconds, ["first_move_ms"] = firstMove, ["first_tile_ms"] = firstTile,
                ["waypoint_ms"] = waypointMs, ["longest_idle_frames"] = longestIdle, ["trace"] = trace });
        }
    }

    static async Task<JsonObject> PlayRecovery(Point origin, RecoveryScene scene, Point target, CancellationToken token)
    {
        var result = await CrossingDiscovery.OnGame(() =>
        {
            ResetRecovery(origin, scene);
            if (target.X <= 0 || target.X >= RecoveryWidth - 1 || target.Y <= 2 || target.Y >= 20
                || scene.Solid(target.X, target.Y) || scene.Solid(target.X, target.Y - 1)
                || scene.Solid(target.X, target.Y - 2) || !scene.Solid(target.X, target.Y + 1))
                return Task.FromResult(new JsonObject { ["reached"] = false, ["reason"] = "target_not_a_standing_cell" });
            recovery = new RecoveryRun(origin, scene, target); recovery.Start();
            return recovery.Result;
        });
        return await result.WaitAsync(token);
    }

    static readonly MethodInfo ObserveWorker = typeof(VanillaNavigationDemo).GetMethod("Observe", BindingFlags.Instance | BindingFlags.NonPublic);
    static Task<JsonObject> PitForecast(JsonObject request, CancellationToken token) =>
        (Task<JsonObject>)ObserveWorker.Invoke(VanillaNavigationDemo.Instance, new object[] { request, token });

    static JsonArray PitKeys(IEnumerable<int> keys) => new(keys.Select(key => JsonValue.Create(key)).ToArray());
    static float Sample(JsonNode frame, string key) => frame[key].GetValue<float>();
    static int[] ReadPitKeys(JsonNode keys) => keys.AsArray().Select(key => key.GetValue<int>()).ToArray();

    static Task<JsonObject> ForecastPitInput(JsonObject snapshot, int[] keys, CancellationToken token) => PitForecast(
        new JsonObject { ["command"] = "native_replay", ["snapshot"] = snapshot.DeepClone(), ["controls"] = PitKeys(keys),
            ["capture_end"] = true, ["detailed"] = false, ["stop_on_damage"] = true }, token);

    static int[] PitPrefix(int direction, int run, int jump)
        => Enumerable.Repeat(direction, run).Concat(Enumerable.Repeat(direction | 4, jump))
            .Concat(Enumerable.Repeat(0, 52 - run - jump)).ToArray();

    sealed record PitOption(string Id, string Action, int[] Keys, float X, float Y);

    static async Task<(PitOption[] Options, JsonObject Evidence)> PitOptions(JsonObject snapshot, Point origin, int width, int height,
        CancellationToken token)
    {
        var proposals = (from dir in new[] { 1, 2 }
            from moves in new[] { (run: 24, jump: 0), (run: 0, jump: 18), (run: 8, jump: 18), (run: 16, jump: 18) }
            select (name: dir == 1 ? "left" : "right", keys: PitPrefix(dir, moves.run, moves.jump), moves)).ToArray();
        var clock = Stopwatch.StartNew();
        var batch = await PitForecast(new JsonObject { ["command"] = "combat_replays", ["snapshot"] = snapshot.DeepClone(),
            ["branches"] = new JsonArray(proposals.Select((option, i) => (JsonNode)new JsonObject {
                ["id"] = $"move{i + 1}", ["controls"] = PitKeys(option.keys) }).ToArray()) }, token);
        var evidence = new JsonArray();
        var options = new List<PitOption>();
        foreach (var branch in batch["branches"].AsArray())
        {
            string id = branch["id"].ToString();
            var proposal = proposals[int.Parse(id[4..]) - 1];
            var frames = branch["frames"].AsArray();
            var end = frames.Last();
            float x = (Sample(end, "x") + width / 2f) / 16 - origin.X;
            float y = (Sample(end, "y") + height) / 16 - origin.Y;
            bool safe = branch["outcome"].ToString() == "complete" && frames.Count == proposal.keys.Length;
            bool grounded = end["grounded"].GetValue<bool>() && Sample(end, "vy") == 0;
            float travel = MathF.Abs(Sample(end, "x") - Sample(frames.First(), "x"))
                + MathF.Abs(Sample(end, "y") - Sample(frames.First(), "y"));
            string action = $"Walk {proposal.name} for {proposal.moves.run} frames, then hold {proposal.name}+jump for {proposal.moves.jump} frames, then release both.";
            evidence.Add(new JsonObject { ["id"] = id, ["action"] = action, ["controls"] = PitKeys(proposal.keys),
                ["foot_x"] = x, ["foot_floor_y"] = y, ["grounded"] = grounded, ["safe"] = safe, ["travel_px"] = travel });
            if (safe && grounded && travel >= 16 && !options.Any(p => Math.Abs(p.X - x) < .25f && Math.Abs(p.Y - y) < .01f))
                options.Add(new PitOption(id, action, proposal.keys, x, y));
        }
        return (options.ToArray(), new JsonObject { ["wall_ms"] = clock.ElapsedMilliseconds,
            ["worker_ms"] = batch["total_ms"].DeepClone(), ["trials"] = evidence,
            ["eligible"] = new JsonArray(options.Select(option => JsonValue.Create(option.Id)).ToArray()) });
    }

    sealed class PitPlayback(Point origin, RecoveryScene scene, int[] keys, JsonArray predicted)
    {
        readonly Stopwatch clock = Stopwatch.StartNew();
        readonly TaskCompletionSource<JsonObject> done = new(TaskCreationOptions.RunContinuationsAsynchronously);
        readonly JsonArray trace = new();
        readonly int hp = Main.LocalPlayer.statLife;
        readonly Vector2 start = Main.LocalPlayer.position;
        int frame, inputFrames;
        bool escaped;
        float error;
        long? firstTile;
        public Task<JsonObject> Result => done.Task;
        public void Drive(Player p)
        {
            VanillaPrediction.ApplyInput(p, new VanillaPrediction.Input((VanillaKeys)keys[frame]));
            inputFrames++;
        }
        public void Advance(Player p)
        {
            if (inputFrames == frame) return;
            var expected = predicted[frame];
            error = Math.Max(error, Vector2.Distance(p.position, new Vector2(Sample(expected, "x"), Sample(expected, "y"))));
            frame++;
            if (Vector2.DistanceSquared(start, p.position) >= 256) firstTile ??= clock.ElapsedMilliseconds;
            float x = p.Bottom.X / 16 - origin.X, y = p.Bottom.Y / 16 - origin.Y;
            if (frame % 4 == 0) trace.Add(new JsonArray(frame, x, y, p.velocity.X, p.velocity.Y));
            bool support = p.velocity.Y == 0 && Collision.SolidCollision(p.position + Vector2.UnitY, p.width, p.height);
            float forwardX = scene.Mirror ? RecoveryWidth - x : x;
            escaped |= support && forwardX is >= 14 and <= 34 && Math.Abs(y - 11) < .1f;
            bool reached = support && Math.Abs(x - (scene.G.X + .5f)) < .75f && Math.Abs(y - (scene.G.Y + 1)) < .1f;
            if (frame < keys.Length && !p.dead && p.statLife >= hp && !reached) return;
            pitPlayback = null; BoundaryInputPlayer.Drive = null;
            VanillaPrediction.ApplyInput(p, new VanillaPrediction.Input(VanillaKeys.None));
            done.SetResult(new JsonObject { ["escaped"] = escaped && p.statLife >= hp, ["reached"] = reached && p.statLife >= hp,
                ["frames"] = frame, ["first_tile_ms"] = firstTile, ["elapsed_ms"] = clock.ElapsedMilliseconds,
                ["damage"] = hp - p.statLife, ["max_prediction_error_px"] = error, ["trace"] = trace });
        }
    }

    static Rectangle PitRegion(Point origin, RecoveryScene scene, int left, int right, int y)
    {
        if (scene.Mirror) (left, right) = (RecoveryWidth * 16 - right, RecoveryWidth * 16 - left);
        return new Rectangle(origin.X * 16 + left, (origin.Y + y) * 16, right - left, 0);
    }

    static async Task<JsonObject> RunPitRoute(Point origin, RecoveryScene scene, JsonObject snapshot, PitOption option,
        CancellationToken token)
    {
        var clock = Stopwatch.StartNew();
        var segments = new JsonArray();
        var controls = new List<int>();
        var result = new JsonObject { ["option"] = option?.Id, ["segments"] = segments };
        var next = snapshot;
        if (option != null)
        {
            var forecast = await ForecastPitInput(next, option.Keys, token);
            if (forecast["end_snapshot"] is not JsonObject end) throw new InvalidOperationException("confirmed pit input failed to replay");
            controls.AddRange(option.Keys);
            next = end;
        }
        foreach (var region in new[] {
            PitRegion(origin, scene, 12 * 16 + 10, 13 * 16 + 4, 17),
            PitRegion(origin, scene, 10 * 16, 11 * 16 + 6, 14),
            PitRegion(origin, scene, 15 * 16, 18 * 16, 11),
            PitRegion(origin, scene, 29 * 16, 30 * 16, 11) })
        {
            var part = await PitForecast(new JsonObject { ["command"] = "region_search", ["snapshot"] = next.DeepClone(),
                ["region"] = new JsonArray(region.X, region.Y, region.Width, region.Height), ["horizon"] = 84,
                ["beam"] = 12, ["quantum"] = 6, ["trace_layers"] = false }, token);
            string status = part["status"].ToString();
            var keys = ReadPitKeys(part["controls"]);
            segments.Add(new JsonObject { ["status"] = status, ["frames"] = keys.Length,
                ["remaining_px"] = part["distance"]?.DeepClone(), ["worker_ms"] = part["elapsed_ms"]?.DeepClone() });
            controls.AddRange(keys);
            if (status != "reached") break;
            next = part["end_snapshot"].AsObject();
        }
        result["planning_ms"] = clock.ElapsedMilliseconds;
        result["controls"] = PitKeys(controls);
        if (controls.Count == 0) return result;
        var combined = await ForecastPitInput(snapshot, controls.ToArray(), token);
        var predicted = combined["frames"].AsArray();
        result["predicted_outcome"] = combined["outcome"].DeepClone();
        if (predicted.Count != controls.Count) return result;
        var playback = await CrossingDiscovery.OnGame(() =>
        {
            ResetRecovery(origin, scene);
            pitPlayback = new PitPlayback(origin, scene, controls.ToArray(), predicted);
            BoundaryInputPlayer.Drive = pitPlayback.Drive;
            return pitPlayback;
        });
        result["physical"] = await playback.Result.WaitAsync(token);
        return result;
    }

                    || physical[candidate]["reached"].GetValue<bool>() != (candidate == side)))
                    throw new InvalidOperationException("detour calibration failed for " + id);
                var (coarse, ascii) = await CrossingDiscovery.OnGame(() =>
                { ResetDetour(origin, side); return (DetourState(origin, false), DetourState(origin, true)); });
                var marks = new Dictionary<char, Point> { ['P'] = new(PlayerColumn, UpperFloor - 1), ['G'] = new(GoalColumn, LowerFloor - 1),
                    ['A'] = new(Entrance("A"), LowerFloor - 1), ['B'] = new(Entrance("B"), LowerFloor - 1) };
                var (map, game) = await CaptureVision(origin, DetourWidth, DetourHeight, marks, token);
                Add(id, coarse, ascii, new JsonObject { ["A"] = "Go through the left entrance.", ["B"] = "Go through the right entrance." }, side, map, game, physical);
            }
            if (deep)
                foreach (string side in Options)
                {
                    var marks = EnclosureA.Select(cell => new KeyValuePair<char, Point>('A', cell))
                        .Concat(EnclosureB.Select(cell => new KeyValuePair<char, Point>('B', cell)))
                        .Append(new KeyValuePair<char, Point>('P', EnclosureFeet)).ToArray();
                    var (physical, coarse, ascii) = await CrossingDiscovery.OnGame(() =>
                    {
                        var checkedPlans = new JsonObject();
                        foreach (string option in Options) checkedPlans[option] = VerifyEnclosure(side, option);
                        if (Options.Any(option => checkedPlans[option]["all_placed"].GetValue<bool>() != true
                            || checkedPlans[option]["sealed"].GetValue<bool>() != (option == side)))
                            throw new InvalidOperationException("enclosure calibration failed: " + side + " " + checkedPlans);
                        ResetEnclosure(side);
                        return (checkedPlans, EnclosureState(marks, false), EnclosureState(marks, true));
                    });
                    var (map, game) = await CaptureVision(EnclosureOrigin, Width, Height, marks, token);
                    Add("enclosure-" + side, coarse, ascii, new JsonObject { ["A"] = "Place the four A blocks.", ["B"] = "Place the four B blocks." }, side, map, game, physical);
                }

            void Add(string id, JsonObject coarse, JsonObject ascii, JsonObject criteria, string side,
                byte[] map, byte[] game, JsonObject physical)
            {
                string mapPath = Path.Combine(root, $"{id}-map.png"), gamePath = Path.Combine(root, $"{id}-game.png");
                File.WriteAllBytes(mapPath, map); File.WriteAllBytes(gamePath, game);
                cases.Add(new JsonObject { ["id"] = id, ["ascii"] = ascii.DeepClone(), ["coarse"] = coarse.DeepClone(),
                    ["map_png"] = mapPath, ["game_png"] = gamePath, ["correct"] = side, ["native"] = physical });
                scenarios.Add(new VisionCase(id, coarse, ascii, criteria, side, map, game));
                Flush();
            }

            if (live)
            {
                string sandbox = Path.Combine(root, "model-workspace"); Directory.CreateDirectory(sandbox);
                var jobs = from scene in scenarios from model in new[] { "Agent", "Jev" }
                    from format in new[] { "ASCII", "map", "game" }
                    where !deep || scene.Id.StartsWith("enclosure-", StringComparison.Ordinal) || model == "Jev" && format == "map"
                    select (scene, model, format);
                object gate = new();
                await Parallel.ForEachAsync(jobs, new ParallelOptions { MaxDegreeOfParallelism = 4, CancellationToken = token }, async (job, ct) =>
                {
                    byte[] image = job.format == "map" ? job.scene.TerrainPng : job.format == "game" ? job.scene.GamePng : null;
                    var facts = (image == null ? job.scene.Ascii : job.scene.Coarse).DeepClone().AsObject();
                    if (image != null) facts["image_legend"] = "P player, E opponent, G goal, A/B candidate cells. Same tile window as the coordinate facts.";
                    JsonObject answer;
                    try { answer = await JudgeVision(job.model, facts, job.scene.Criteria, image, sandbox, jevKey, ct); }
                    catch (Exception error) when (!ct.IsCancellationRequested) { answer = new JsonObject { ["error"] = error.Message }; }
                    string choice = answer["choice"]?.ToString() ?? answer["receipt"]?["answers"]?["pick"]?["choice"]?.ToString();
                    answer["id"] = job.scene.Id; answer["engine"] = job.model; answer["format"] = job.format;
                    answer["choice"] = choice; answer["native_pass"] = choice == job.scene.Correct;
                    lock (gate) { selections.Add(answer); Flush(); }
                });
            }
            output["ok"] = cases.Count == (deep ? 6 : 4)
                && (!live || selections.Count == (deep ? 16 : 24) && selections.All(n => n["choice"] != null));
            Flush(); return output.ToJsonString();
        }
        catch (Exception error) { output["error"] = error.Message; Flush(); throw; }
        finally { await CrossingDiscovery.OnGame(() => { StateSpacePlanner.PreserveTerrain = false; return true; }); }
    }
}

public sealed class CrossingDiscovery : ModSystem
{
    readonly record struct Layout(int X, int Y, int Direction, int Gap, int Delay);
    sealed record Step(string[] Keys, int Frames, string UntilSignal);
    static readonly Layout Practice = new(230, 210, 1, 13, 7);
    static readonly Layout Transfer = new(320, 252, -1, 15, 25);
    const string SkillName = "learned_crossing";
    const int RepairBudget = 8;
    static readonly ConcurrentQueue<Action> Queue = new();
    static CrossingTrial active;

    public override void PreUpdatePlayers()
    {
        while (Queue.TryDequeue(out var action)) action();
    }

    public override void PostUpdatePlayers() => active?.Sample(Main.LocalPlayer);

    internal static Task<T> OnGame<T>(Func<T> action)
    {
        var result = new TaskCompletionSource<T>(TaskCreationOptions.RunContinuationsAsynchronously);
        Queue.Enqueue(() =>
        {
            try { result.SetResult(action()); }
            catch (Exception error) { result.SetException(error); }
        });
        return result.Task;
    }

    static JsonObject Describe(Layout layout)
    {
        var p = Main.LocalPlayer;
        return new JsonObject { ["x"] = p.Center.X, ["bottom"] = p.Bottom.Y, ["vx"] = p.velocity.X,
            ["device_x"] = layout.X * 16 + 8, ["signal"] = active?.Signal ?? "dark",
            ["landing_x"] = (layout.X + layout.Direction * (layout.Gap + 1)) * 16 + 8,
            ["landing_y"] = layout.Y * 16 };
    }

    static void Reset(Layout layout)
    {
        var p = Main.LocalPlayer;
        for (int dx = -14; dx <= 48; dx++)
        for (int dy = -14; dy <= 14; dy++)
        {
            var tile = Main.tile[layout.X + dx * layout.Direction, layout.Y + dy];
            tile.ClearEverything();
            if (dy >= 10 || dy >= 0 && (dx <= 0 || dx >= layout.Gap))
            {
                tile.HasTile = true;
                tile.TileType = TileID.GrayBrick;
            }
        }
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var shot in Main.projectile) shot.active = false;
        foreach (var item in p.armor) item.TurnToAir();
        foreach (var item in p.inventory) item.TurnToAir();
        foreach (var item in p.miscEquips) item.TurnToAir();
        p.ClearBuff(BuffID.Featherfall);
        p.ClearBuff(BuffID.Gravitation);
        p.position = new Vector2(layout.X * 16 + 8 - p.width * .5f, layout.Y * 16 - p.height);
        p.velocity = Vector2.Zero;
        p.fallStart = p.fallStart2 = (int)(p.position.Y / 16);
        p.jump = 0; p.releaseJump = true; p.controlJump = false;
    }

    sealed class CrossingTrial(Layout layout, Step[] steps, string id)
    {
        readonly TaskCompletionSource<JsonObject> done = new(TaskCreationOptions.RunContinuationsAsynchronously);
        readonly JsonArray changes = new(), samples = new();
        int frames, stage, stageFrames, interactionFrame = -1, impulseFrame = -1;
        bool lastJump, lastInteract;
        string previousSignal = "dark";
        public Task<JsonObject> Result => done.Task;
        public string Signal
        {
            get
            {
                if (impulseFrame >= 0) return "spent";
                int age = frames - interactionFrame;
                return interactionFrame < 0 || age >= layout.Delay + 5 ? "dark" : age >= layout.Delay ? "white" : "amber";
            }
        }

        public void Drive(Player p)
        {
            while (stage < steps.Length && (stageFrames >= steps[stage].Frames
                    || steps[stage].UntilSignal != null && steps[stage].UntilSignal == Signal))
            { stage++; stageFrames = 0; }
            string[] keys = stage < steps.Length ? steps[stage].Keys : Array.Empty<string>();
            p.controlLeft = keys.Contains("left");
            p.controlRight = keys.Contains("right");
            p.controlJump = keys.Contains("jump");
            p.controlUseTile = keys.Contains("interact");
            p.controlUseItem = p.controlHook = p.controlUp = p.controlDown = false;
            stageFrames++;
        }

        public void ObserveInput(Player p)
        {
            bool inReach = Math.Abs(p.Center.X - (layout.X * 16 + 8)) <= 26 && Math.Abs(p.Bottom.Y - layout.Y * 16) <= 44;
            if (p.controlUseTile && !lastInteract && inReach && Signal == "dark")
                interactionFrame = frames;
            if (p.controlJump && !lastJump && inReach && Signal == "white" && p.controlLeft != p.controlRight)
            {
                impulseFrame = frames;
                p.velocity = new Vector2(p.controlLeft ? -13 : 13, -10);
                p.jump = 0;
            }
            if (Signal != previousSignal)
            {
                changes.Add(new JsonObject { ["frame"] = frames, ["signal"] = Signal });
                previousSignal = Signal;
            }
            lastInteract = p.controlUseTile; lastJump = p.controlJump;
        }

        public void Sample(Player p)
        {
            if (frames % 4 == 0)
                samples.Add(new JsonObject { ["frame"] = frames, ["x"] = p.Center.X, ["bottom"] = p.Bottom.Y,
                    ["vx"] = p.velocity.X, ["vy"] = p.velocity.Y });
            frames++;
            bool reached = (p.Center.X - (layout.X * 16 + 8)) * layout.Direction >= (layout.Gap + 1) * 16
                && Math.Abs(p.Bottom.Y - layout.Y * 16) <= 2 && p.velocity.Y == 0;
            if (!reached && frames < 150 && !p.dead) return;
            active = null; BoundaryInputPlayer.Drive = BoundaryInputPlayer.ObserveInput = null;
            p.controlLeft = p.controlRight = p.controlJump = p.controlUseTile = false;
            done.SetResult(new JsonObject { ["id"] = id, ["reached"] = reached, ["frames"] = frames,
                ["impulse_frame"] = impulseFrame < 0 ? null : JsonValue.Create(impulseFrame),
                ["signal_changes"] = changes, ["samples"] = samples, ["end"] = Describe(layout) });
        }
    }

    static readonly string[] ControlKeys = { "left", "right", "jump", "interact" };

    static Step[] ParseSteps(JsonObject args)
    {
        var steps = args["steps"].AsArray();
        if (steps.Count is < 1 or > 10) throw new ArgumentException("steps must contain 1–10 entries");
        return steps.Select(step =>
        {
            int frames = step["frames"].GetValue<int>();
            string signal = step["until_signal"]?.GetValue<string>();
            var keys = step["keys"].AsArray().Select(key => key.GetValue<string>()).ToArray();
            if (frames is < 1 or > 150 || keys.Any(key => !ControlKeys.Contains(key))
                || signal is not (null or "dark" or "amber" or "white" or "spent"))
                throw new ArgumentException("invalid trial frames, keys or signal selector");
            return new Step(keys, frames, signal);
        }).ToArray();
    }

    static async Task<JsonObject> Attempt(Layout layout, Step[] steps, string id, CancellationToken token)
    {
        var trial = await OnGame(() =>
        {
            if (active != null) throw new InvalidOperationException("another sequence is active");
            Reset(layout);
            var next = new CrossingTrial(layout, steps, id);
            BoundaryInputPlayer.Drive = next.Drive;
            BoundaryInputPlayer.ObserveInput = next.ObserveInput;
            active = next;
            return next;
        });
        return await trial.Result.WaitAsync(token);
    }

    static async Task Learn(string directory, JsonObject report, Action flush, CancellationToken token)
    {
        const string prompt = "Reach the far landing and save a reusable way to do it. A normal running jump has already failed.";
        const string system = """
            Use the supplied game tools. The device exposes a colored indicator and accepts nearby interactions.
            Each trial starts the isolated avatar at the same position and advances the actual game.
            Valid keys are left, right, jump, interact. An empty key list releases all keys.
            A step runs for its frames limit or ends early when its optional until_signal is observed.
            Read trial outcomes and test meaningful alternatives. Budget: 12 trials.
            Save a reusable JavaScript function body with save_crossing after a verified success.
            input is {mode,scene}. In mode='probe', return {applicable:boolean} with no actions.
            In mode='execute', use scene to choose the direction, call tool('trial',{steps:[...]}) exactly once,
            and return that result. The scene has x, bottom, device_x, landing_x, landing_y and signal.
            Another layout will use the saved source without your help. Base it on observable state.
            Use only the provided game tools and observations. Report the experimental evidence you used.
            """;
        var tools = JsonNode.Parse("""
            [
              {"type":"function","function":{"name":"observe","description":"Read the current practice scene and available input keys.","parameters":{"type":"object","properties":{}}}},
              {"type":"function","function":{"name":"trial","description":"Reset to the starting position, apply an input sequence and report actual motion, indicator transitions and landing.","parameters":{"type":"object","properties":{"steps":{"type":"array","items":{"type":"object","properties":{"keys":{"type":"array","items":{"type":"string"}},"frames":{"type":"integer"},"until_signal":{"type":"string"}},"required":["keys","frames"]}}},"required":["steps"]}}},
              {"type":"function","function":{"name":"save_crossing","description":"Save reusable JavaScript referencing successful trials.","parameters":{"type":"object","properties":{"code":{"type":"string"},"evidence":{"type":"array","items":{"type":"string"}}},"required":["code","evidence"]}}}
            ]
            """).AsArray();
        var library = new ScriptLibrary(Path.Combine(directory, "programs"));
        var trials = report["trials"].AsArray();
        var replies = new JsonArray();
        report["prompt"] = prompt; report["replies"] = replies;
        await OnGame(() => { Reset(Practice); return true; });
        async Task<string> Execute(string tool, JsonObject args, CancellationToken caller)
        {
            switch (tool)
            {
                case "observe":
                    return (await OnGame(() => Describe(Practice))).ToJsonString();
                case "trial":
                    if (trials.Count >= 12) return """{"error":"trial_budget_exhausted"}""";
                    var trial = await Attempt(Practice, ParseSteps(args), $"trial-{trials.Count + 1}", caller);
                    trial["steps"] = args["steps"].DeepClone();
                    trials.Add(trial); flush();
                    return trial.ToJsonString();
                case "save_crossing":
                    var evidence = args["evidence"].AsArray();
                    if (evidence.Count == 0 || !evidence.All(id => trials.Any(t => t["id"].ToString() == id.ToString()))
                        || !trials.Any(t => t["reached"].GetValue<bool>() && evidence.Any(id => id.ToString() == t["id"].ToString())))
                        return """{"error":"cite_a_successful_practice_trial"}""";
                    var saved = library.Save(SkillName, args["code"].GetValue<string>(), library.Get(SkillName)["revision"].GetValue<int>());
                    if (saved["error"] == null)
                    {
                        report["skill"] = saved.DeepClone();
                        report["skill_evidence"] = evidence.DeepClone();
                        flush();
                    }
                    return saved.ToJsonString();
                default:
                    return """{"error":"unknown_tool"}""";
            }
        }
        const string model = "gpt-6.1-sol";
        using var agent = new CodexAgent();
        string sandbox = Path.Combine(directory, "agent-workspace");
        Directory.CreateDirectory(sandbox);
        report["model"] = model;
        try
        {
            await agent.RunAsync(prompt, "", sandbox, model, system, tools, Execute,
                text => { replies.Add(text); flush(); }, token);
        }
        finally { report["token_usage"] = agent.TokenUsage; flush(); }

        var savedSkill = library.Get(SkillName);
        if (savedSkill["found"]?.GetValue<bool>() != true || trials.Count < 2)
            throw new InvalidOperationException("discovery requires a saved program and multiple actual trials");
        report["source_sha256"] = SourceHash(savedSkill["code"].ToString());
        report["transfer_model_calls"] = 0;
        if (!await Replay(Practice, "practice_replay", library, report, flush, token))
            throw new InvalidOperationException("saved skill did not reach the practice landing");
        var transferBaseline = await Attempt(Transfer, new[] { new Step(new[] { "left", "jump" }, 120, null) }, "transfer-walk-jump", token);
        report["transfer_baseline"] = transferBaseline; flush();
        if (transferBaseline["reached"].GetValue<bool>()) throw new InvalidOperationException("transfer gap accepts ordinary jump");
        if (!await Replay(Transfer, "transfer", library, report, flush, token))
            throw new InvalidOperationException("saved skill did not reach the transfer landing");
    }

    static string SourceHash(string source) => Convert.ToHexString(
        System.Security.Cryptography.SHA256.HashData(Encoding.UTF8.GetBytes(source)));

    static async Task<bool> Replay(Layout layout, string phase, ScriptLibrary library, JsonObject report,
        Action flush, CancellationToken token, JsonObject candidate = null)
    {
        var entry = candidate ?? library.Get(SkillName);
        int revision = entry["revision"].GetValue<int>();
        var programs = candidate == null ? library.Callable() : new JsonArray(candidate.DeepClone());
        var scene = await OnGame(() => { Reset(layout); return Describe(layout); });
        JsonObject observed = null;
        async Task<string> RunTool(string tool, JsonObject args, CancellationToken caller)
        {
            if (tool != "trial" || observed != null) throw new InvalidOperationException("only one physical replay is permitted");
            observed = await Attempt(layout, ParseSteps(args), phase, caller);
            observed["steps"] = args["steps"].DeepClone();
            report[phase] = observed;
            return observed.ToJsonString();
        }
        string Invoke(string mode) => $"return skill('{SkillName}',{revision},"
            + new JsonObject { ["mode"] = mode, ["scene"] = scene.DeepClone() }.ToJsonString() + ");";
        var applicability = await AgentScript.Run(Invoke("probe"), 10,
            (_, _) => throw new InvalidOperationException("applicability must be read-only"), RunTool,
            _ => Task.CompletedTask, token, programs: programs);
        if (applicability["value"] is not JsonObject probe || probe["applicable"] is not JsonValue applicable
            || !applicable.TryGetValue<bool>(out bool applies) || !applies || observed != null)
        {
            report[phase + "_script"] = applicability;
            flush();
            return false;
        }
        var execution = await AgentScript.Run(Invoke("execute"), 10, (tool, _) =>
            { if (tool != "trial") throw new InvalidOperationException("only trial is permitted"); }, RunTool,
            _ => Task.CompletedTask, token, programs: programs);
        if (candidate == null) library.RecordCalls(execution["skill_calls"].AsArray(), phase);
        report[phase + "_script"] = execution;
        flush();
        return observed?["reached"]?.GetValue<bool>() == true && execution["ok"]?.GetValue<bool>() == true
            && execution["skill_calls"].AsArray().Any(call => call["name"].ToString() == SkillName
                && call["revision"].GetValue<int>() == revision);
    }

    static JsonObject ReplayEvidence(JsonObject result, string phase)
    {
        var evidence = new JsonObject { ["scene"] = phase };
        foreach (string field in new[] { "steps", "signal_changes", "impulse_frame", "frames", "reached", "end" })
            evidence[field] = result[phase]?[field]?.DeepClone();
        var script = result[phase + "_script"];
        if (evidence["reached"] == null || script?["ok"]?.GetValue<bool>() != true)
            evidence["script"] = new JsonObject { ["error"] = script?["error"]?.DeepClone(),
                ["detail"] = script?["detail"]?.DeepClone(), ["value"] = script?["value"]?.DeepClone() };
        return evidence;
    }

    static async Task Repair(string from, string directory, JsonObject report, Action flush, CancellationToken token)
    {
        var previous = new ScriptLibrary(Path.Combine(from, "programs")).Get(SkillName, includeRuntime: false);
        var library = new ScriptLibrary(Path.Combine(directory, "programs"));
        if (previous["revision"]?.GetValue<int>() != 1 || library.Get(SkillName)["found"].GetValue<bool>())
            throw new InvalidOperationException("repair requires an existing revision 1 and a fresh output directory");
        var seed = library.Save(SkillName, previous["code"].ToString(), 0);
        if (seed["revision"]?.GetValue<int>() != 1) throw new InvalidOperationException("unable to import skill: " + seed);
        report["original_source_sha256"] = SourceHash(seed["code"].ToString());
        report["repair_from"] = from;
        if (!await Replay(Practice, "original_practice", library, report, flush, token)
            || await Replay(Transfer, "original_transfer", library, report, flush, token))
            throw new InvalidOperationException("original skill must pass practice and fail transfer");

        var known = new List<(Layout Layout, string Name)> { (Practice, "practice"), (Transfer, "transfer") };
        var repairAttempts = new JsonArray();
        var replies = new JsonArray();
        report["repairs"] = repairAttempts;
        report["replies"] = replies;
        var input = new JsonObject { ["skill"] = seed.DeepClone(),
            ["practice"] = ReplayEvidence(report, "original_practice"), ["failure"] = ReplayEvidence(report, "original_transfer") };
        var tools = JsonNode.Parse("""
            [{"type":"function","function":{"name":"revise_crossing","description":"Evaluate a replacement JavaScript function body with the real engine. Returns failed replay evidence for the next revision. A passing revision is saved.","parameters":{"type":"object","properties":{"code":{"type":"string"}},"required":["code"]}}}]
            """).AsArray();
        async Task<string> Execute(string tool, JsonObject args, CancellationToken caller)
        {
            if (tool != "revise_crossing") return """{"error":"unknown_tool"}""";
            if (report["skill"] != null) return """{"status":"already_verified"}""";
            if (repairAttempts.Count >= RepairBudget) return """{"error":"repair_budget_exhausted"}""";
            string code = args["code"].GetValue<string>();
            var candidate = new JsonObject { ["name"] = SkillName, ["revision"] = 2, ["code"] = code };
            var run = new JsonObject { ["candidate"] = candidate.DeepClone(), ["source_sha256"] = SourceHash(code) };
            repairAttempts.Add(run);
            foreach (var check in known)
            {
                if (!await Replay(check.Layout, check.Name, library, run, flush, caller, candidate))
                    return ReplayEvidence(run, check.Name).ToJsonString();
            }
            int index = known.Count - 2;
            var unseen = new Layout(460 + index * 70, 280, index % 2 == 0 ? 1 : -1, 16, 41 + index * 5);
            string phase = $"holdout-{index + 1}";
            if (!await Replay(unseen, phase, library, run, flush, caller, candidate))
            {
                known.Add((unseen, phase));
                return ReplayEvidence(run, phase).ToJsonString();
            }
            var saved = library.Save(SkillName, code, 1);
            if (saved["revision"]?.GetValue<int>() != 2) return saved.ToJsonString();
            if (!await Replay(unseen, "saved_holdout", library, report, flush, caller))
                throw new InvalidOperationException("persisted skill did not reproduce the candidate landing");
            report["skill"] = saved.DeepClone();
            report["source_sha256"] = SourceHash(code);
            report["holdout_model_calls"] = 0;
            flush();
            return new JsonObject { ["status"] = "verified", ["revision"] = 2,
                ["holdout"] = ReplayEvidence(report, "saved_holdout") }.ToJsonString();
        }

        const string system = """
            Repair your saved crossing skill using the provided successful and failed real engine traces.
            Submit the revised synchronous JavaScript function body through revise_crossing. If it fails, use its feedback and revise again.
            The script input is {mode,scene}. For probe return {applicable:boolean} without actions. For execute make one synchronous tool('trial',{steps:[...]}) call.
            Each step takes keys and a frames limit (1–150). If until_signal is used, the step ends on that signal or when the frame limit expires.
            A run has at most 150 frames. Use only the supplied tools and observations. Continue until the host reports verified.
            """;
        string sandbox = Path.Combine(directory, "agent-workspace");
        Directory.CreateDirectory(sandbox);
        const string model = "gpt-6.1-sol";
        report["model"] = model;
        using var agent = new CodexAgent();
        try
        {
            await agent.RunAsync(input.ToJsonString(), "", sandbox, model, system, tools, Execute,
                text => { replies.Add(text); flush(); }, token);
        }
        finally { report["token_usage"] = agent.TokenUsage; flush(); }
        if (report["skill"] == null) throw new InvalidOperationException("no skill revision passed the repair and holdout layouts");
    }

    internal static async Task<string> Run()
    {
        string directory = Path.Combine(Main.SavePath, "crossing-discovery");
        Directory.CreateDirectory(directory);
        var report = new JsonObject { ["ok"] = false, ["experiment"] = "crossing_discovery", ["trials"] = new JsonArray() };
        string path = Path.Combine(directory, "report.json");
        void Flush() => File.WriteAllText(path, report.ToJsonString());
        string repairFrom = Environment.GetEnvironmentVariable("TB_CROSSING_REPAIR_FROM");
        using var limit = new CancellationTokenSource(TimeSpan.FromSeconds(repairFrom == null ? 160 : 240));
        CancellationToken token = limit.Token;
        await OnGame(() =>
        {
            NativeAgent.Stop(); Halt.All(); Manual.On = false; Concessions.Enabled = false;
            Combat.Enabled = Dodge.Enabled = false;
            Main.dayTime = true;
            return true;
        });
        try
        {
            var baseline = await Attempt(Practice, new[] { new Step(new[] { "right", "jump" }, 120, null) }, "walk-jump", token);
            report["baseline"] = baseline; Flush();
            if (baseline["reached"].GetValue<bool>()) throw new InvalidOperationException("practice gap accepts ordinary jump");
            if (Environment.GetEnvironmentVariable("TB_CROSSING_CALIBRATE") == "1")
            {
                foreach (var layout in new[] { Practice, Transfer })
                {
                    string side = layout.Direction > 0 ? "right" : "left";
                    var check = await Attempt(layout, new[] { new Step(new[] { "interact" }, 1, null),
                        new Step(Array.Empty<string>(), 60, "white"), new Step(new[] { side, "jump" }, 1, null),
                        new Step(new[] { side }, 100, null) }, "calibration-" + side, token);
                    report["trials"].AsArray().Add(check); Flush();
                    if (!check["reached"].GetValue<bool>()) throw new InvalidOperationException("calibration did not cross " + side);
                }
                report["ok"] = true; Flush();
                Console.WriteLine("CROSSING_CALIBRATION_PASS native_frames=true model_calls=0 " + path);
                return report.ToJsonString();
            }
            if (Environment.GetEnvironmentVariable("TB_CROSSING_MODEL") != "1")
                throw new InvalidOperationException("Set TB_CROSSING_MODEL=1 to run the bounded live discovery turn.");
            if (repairFrom == null) await Learn(directory, report, Flush, token);
            else await Repair(repairFrom, directory, report, Flush, token);
            report["ok"] = true; Flush();
            Console.WriteLine("CROSSING_DISCOVERY_PASS " + path);
            return report.ToJsonString();
        }
        catch (Exception error)
        {
            report["error"] = error.ToString(); Flush();
            throw;
        }
    }
}

public sealed class RailSkillTrial : ModSystem
{
    const int Floor = 220;
    const int StartX = 520;
    const int Stock = 9999;
    const string Objective = """
        为了可能的那点惊喜，我会描述为：
        前置知识：泰拉瑞亚无法凭空放置物块，但是可以凭空放置一些东西再放，比如分线盒。
        铁轨加速。
        要求动作：1，可交互范围内，向上放置分线盒后放置铁轨，进行一次铁轨加速。
        2，在空中连续执行5次1而0次失败。
        3，使用羽落药水，执行1的同时全程按住w，直到达到最高点。
        4，在3的基础上，向右凭空分线盒+铁轨加速。
        5，在4的基础上，连续5次向右凭空分线盒+铁轨加速。
        这是一个人类由于反应力和手指灵活限制，不是很能执行的一个帅气操作。但是是代码的甜点。可以试试。
        给与：9999分线盒，9999矿车轨道，9999羽落药水。
        环境：平地，上方无任何遮挡。
        """;
    static TaskCompletionSource<bool> recordingStart;
    static RailRun active;
    public static bool Recording => Environment.GetEnvironmentVariable("TB_RAIL_SKILL") == "record";

    static Projectile Hook(Player p)
    {
        int index = p.grappling[0];
        if (index < 0 || index >= Main.projectile.Length) return null;
        var hook = Main.projectile[index];
        return hook.active && hook.owner == p.whoAmI && hook.type == ProjectileID.TrackHook ? hook : null;
    }

    public override void PreUpdatePlayers()
    {
        if (recordingStart != null && !Main.drawingPlayerChat && Keyboard.GetState().IsKeyDown(Keys.L))
        {
            foreach (string line in Objective.Split('\n')) AgentChat.Display(line, 120, 200, 255);
            recordingStart.TrySetResult(true);
            recordingStart = null;
        }
        active?.Before(Main.LocalPlayer);
    }
    public override void PostUpdatePlayers() => active?.After(Main.LocalPlayer);

    static void Prepare(Player p, int startX)
    {
        Halt.All(); FrameProgram.Clear();
        for (int x = 300; x < 820; x++)
        for (int y = 25; y <= Floor + 4; y++)
        {
            var tile = Main.tile[x, y];
            tile.ClearEverything();
            if (y < Floor) continue;
            tile.HasTile = true; tile.TileType = TileID.GrayBrick;
        }
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var item in p.inventory) item.TurnToAir();
        foreach (var item in p.armor) item.TurnToAir();
        foreach (var item in p.miscEquips) item.TurnToAir();
        p.inventory[0].SetDefaults(ItemID.WirePipe);
        p.inventory[1].SetDefaults(ItemID.MinecartTrack);
        p.inventory[2].SetDefaults(ItemID.FeatherfallPotion);
        for (int slot = 0; slot < 3; slot++) p.inventory[slot].stack = Stock;
        p.ClearBuff(BuffID.Featherfall);
        p.RemoveAllGrapplingHooks();
        p.mount.Dismount(p);
        p.Teleport(new Vector2(startX * 16 + 8 - p.width / 2f, Floor * 16 - p.height), 0);
        p.velocity = Vector2.Zero;
        p.fallStart = p.fallStart2 = Floor;
        p.statLife = p.statLifeMax2;
        p.itemAnimation = p.itemTime = p.jump = 0;
        p.selectedItem = 0; p.releaseJump = p.releaseUseItem = true;
        p.controlJump = p.controlUp = p.controlUseItem = p.controlUseTile = false;
        Main.playerInventory = Main.mapFullscreen = Main.autoPause = false;
        Main.mouseItem.TurnToAir();
    }

    sealed class RailRun(bool calibration)
    {
        readonly TaskCompletionSource<JsonObject> done = new(TaskCreationOptions.RunContinuationsAsynchronously);
        readonly JsonArray clicks = new(), boosts = new(), frames = new();
        int frame, pendingFrame, boxFrame, jumpFrame = -1, featherApex = -1, upMissing, groundTouches;
        int boxes, tracks, lastBoxFrame;
        bool seventhBoxStarted;
        float previousVy, peakY = float.MaxValue;
        JsonObject pending;
        public Task<JsonObject> Result => done.Task;

        public void Before(Player p)
        {
            var hook = Hook(p);
            FrameProgram.Set(new JsonObject { ["data"] = new JsonObject
            {
                ["floor_y"] = Floor, ["boosts"] = boosts.Count, ["apex_reached"] = featherApex >= 0,
                ["featherfall"] = p.HasBuff(BuffID.Featherfall), ["release_jump"] = p.releaseJump,
                ["hook"] = hook == null ? null : new JsonObject { ["x"] = hook.Center.X, ["y"] = hook.Center.Y,
                    ["touching"] = p.Hitbox.Intersects(hook.Hitbox) },
                ["track_tile"] = TileID.MinecartTrack, ["box_tile"] = TileID.WirePipe
            } });
        }

        public void ApplyClick(Player p)
        {
            if (p.controlJump && Hook(p) != null && pending != null && jumpFrame < 0) jumpFrame = frame;
            var click = FrameProgram.Snapshot()["memory"]?["click"] as JsonObject;
            if (click?["tick"]?.GetValue<ulong>() != Main.GameUpdateCount) return;
            if (pending != null) { Finish(p, "right_click_while_hook_pending"); return; }
            int x = click["x"].GetValue<int>(), y = click["y"].GetValue<int>();
            bool track = WorldGen.InWorld(x, y) && Main.tile[x, y].HasUnactuatedTile && Main.tile[x, y].TileType == TileID.MinecartTrack;
            bool box = track && new[] { (x-1,y), (x+1,y), (x,y-1), (x,y+1) }.Any(c =>
                Main.tile[c.Item1, c.Item2].HasUnactuatedTile && Main.tile[c.Item1, c.Item2].TileType == TileID.WirePipe);
            bool reach = p.IsInTileInteractionRange(x, y, TileReachCheckSettings.Simple);
            pending = new JsonObject { ["frame"] = frame, ["x"] = x, ["y"] = y, ["track"] = track,
                ["adjacent_box"] = box, ["reach"] = reach, ["vx_before"] = p.velocity.X, ["vy_before"] = p.velocity.Y,
                ["delta_x"] = x * 16 + 8 - p.Center.X, ["delta_y"] = y * 16 + 8 - p.Center.Y,
                ["boxes_used"] = Stock - p.inventory[0].stack, ["tracks_used"] = Stock - p.inventory[1].stack };
            clicks.Add(pending);
            if (!track || !box || !reach || y >= Floor - 1) { Finish(p, "invalid_track_target"); return; }
            if (clicks.Take(clicks.Count - 1).Any(c => c["x"].GetValue<int>() == x && c["y"].GetValue<int>() == y)
                || Stock - p.inventory[0].stack < clicks.Count || Stock - p.inventory[1].stack < clicks.Count)
            { Finish(p, "track_or_junction_box_not_new"); return; }
            var before = (p.tileInteractAttempted, p.releaseUseTile, p.tileInteractionHappened, Main.mouseRightRelease);
            try
            {
                p.tileInteractAttempted = p.releaseUseTile = Main.mouseRightRelease = true;
                p.tileInteractionHappened = false;
                p.TileInteractionsCheck(x, y);
                pending["native_handled"] = p.tileInteractionHappened;
            }
            finally { (p.tileInteractAttempted, p.releaseUseTile, p.tileInteractionHappened, Main.mouseRightRelease) = before; }
            pendingFrame = frame; jumpFrame = -1; boxFrame = lastBoxFrame;
        }

        public void After(Player p)
        {
            bool feather = p.HasBuff(BuffID.Featherfall), grounded = p.Bottom.Y >= Floor * 16 - 1;
            int newBoxes = Stock - p.inventory[0].stack, newTracks = Stock - p.inventory[1].stack;
            bool placedBox = newBoxes != boxes, placedTrack = newTracks != tracks;
            if (placedBox)
            {
                lastBoxFrame = frame;
                if (boosts.Count == 6) seventhBoxStarted = true;
            }
            boxes = newBoxes; tracks = newTracks;
            if (seventhBoxStarted && featherApex < 0 && !p.controlUp) upMissing++;
            if (boosts.Count > 0 && grounded) groundTouches++;
            peakY = Math.Min(peakY, p.Center.Y);
            if (boosts.Count == 7 && featherApex < 0 && previousVy < 0 && p.velocity.Y >= 0) featherApex = frame;
            previousVy = p.velocity.Y;
            if (frame % 4 == 0 || pending != null || placedBox || placedTrack)
                frames.Add(new JsonObject { ["f"] = frame, ["x"] = p.Center.X, ["y"] = p.Center.Y, ["vx"] = p.velocity.X,
                    ["vy"] = p.velocity.Y, ["hook"] = Hook(p) != null, ["up"] = p.controlUp, ["cart"] = p.mount.Active,
                    ["feather"] = feather, ["ground"] = grounded, ["boxes"] = boxes, ["tracks"] = tracks,
                    ["slot"] = p.selectedItem, ["item_time"] = p.itemTime });
            if (p.dead || p.mount.Active) { Finish(p, p.dead ? "died" : "boarded_cart"); return; }
            if (groundTouches > 0) { Finish(p, "landed_mid_chain"); return; }
            if (pending != null && jumpFrame >= 0 && Hook(p) == null)
            {
                bool right = boosts.Count >= 7;
                float speed = right ? p.velocity.X : -p.velocity.Y;
                var boost = new JsonObject { ["jump_frame"] = jumpFrame, ["click_frame"] = pendingFrame, ["box_frame"] = boxFrame,
                    ["direction"] = right ? "right" : "up", ["vx"] = p.velocity.X, ["vy"] = p.velocity.Y,
                    ["x"] = p.Center.X, ["y"] = p.Center.Y, ["feather"] = feather, ["up"] = p.controlUp,
                    ["speed_along_target"] = speed, ["after_apex"] = featherApex >= 0,
                    ["track_x"] = pending["x"].DeepClone(), ["track_y"] = pending["y"].DeepClone() };
                bool supportAhead = right ? pending["delta_x"].GetValue<float>() > 0 : pending["delta_y"].GetValue<float>() < 0;
                if (!supportAhead || speed < 4 || boosts.Count < 6 && feather || boosts.Count >= 6 && !feather
                    || right && featherApex < 0)
                { pending["failed_boost"] = boost; Finish(p, "boost_conditions_failed"); return; }
                boosts.Add(boost); pending = null;
                if (calibration || boosts.Count == 13)
                {
                    bool continuous = calibration || groundTouches == 0 && upMissing == 0 && featherApex >= 0;
                    Finish(p, continuous ? null : "sequence_continuity_failed"); return;
                }
            }
            if (pending != null && frame - pendingFrame > 24) { Finish(p, "hook_not_cancelled"); return; }
            if (frame++ >= (calibration ? 180 : 1500) || !FrameProgram.Active)
                Finish(p, FrameProgram.Active ? "frame_limit" : "program_ended_early");
        }

        void Finish(Player p, string failure)
        {
            var result = new JsonObject { ["ok"] = failure == null, ["failure"] = failure, ["frame"] = frame,
                ["inputs"] = FrameProgram.Snapshot(), ["boosts"] = boosts, ["clicks"] = clicks, ["trace"] = frames,
                ["ground_touches"] = groundTouches, ["up_missing_frames"] = upMissing, ["apex_frame"] = featherApex,
                ["peak_height_px"] = Floor * 16 - p.height / 2f - peakY, ["boxes_remaining"] = p.inventory[0].stack,
                ["tracks_remaining"] = p.inventory[1].stack, ["potions_remaining"] = p.inventory[2].stack };
            active = null; BoundaryInputPlayer.Drive = null; FrameProgram.Clear();
            done.TrySetResult(result);
        }
    }

    static async Task<JsonObject> RunProgram(string code, bool calibration, CancellationToken token, int startX = StartX)
    {
        var trial = await CrossingDiscovery.OnGame(() =>
        {
            var p = Main.LocalPlayer;
            Prepare(p, startX);
            var run = new RailRun(calibration);
            FrameProgram.Set(new JsonObject { ["code"] = code });
            active = run; BoundaryInputPlayer.Drive = run.ApplyClick;
            return run;
        });
        return await trial.Result.WaitAsync(token);
    }

    const string Calibration = """
        function tick(s) {
          const p=s.player, d=s.data, m=s.memory;
          m.target ??= {x:Math.floor((p.x+p.width/2)/16), y:Math.floor((p.y+p.height/2)/16)-4};
          const t=m.target, b={x:t.x+1,y:t.y};
          const has=(c,type)=>s.tiles.some(a=>a.x===c.x&&a.y===c.y&&a.type===type);
          delete m.click;
          if(d.hook && d.release_jump && !d.hook.touching) return {keys:['jump'],memory:m};
          if (!has(b,d.box_tile)) return {slot:0,use:true,aim:b,memory:m};
          if (!has(t,d.track_tile)) return {slot:1,use:true,aim:t,memory:m};
          if(!m.clicked) { m.clicked=true; m.click={...t,tick:s.tick}; }
          return {memory:m};
        }
        """;

    internal static async Task<string> Run()
    {
        string mode = Environment.GetEnvironmentVariable("TB_RAIL_SKILL");
        string folder = Path.Combine(Main.SavePath, "rail-skill");
        Directory.CreateDirectory(folder);
        var report = new JsonObject { ["experiment"] = "rail_skill", ["ok"] = false };
        void Flush() => File.WriteAllText(Path.Combine(folder, "report.json"), report.ToJsonString());
        using var limit = new CancellationTokenSource();
        if (!Recording) limit.CancelAfter(TimeSpan.FromSeconds(mode == "live" ? 360 : 60));
        var token = limit.Token;
        await CrossingDiscovery.OnGame(() =>
        {
            NativeAgent.Stop(); Halt.All(); Manual.On = false; Concessions.Enabled = false;
            Combat.Enabled = Dodge.Enabled = false;
            Main.dayTime = true;
            return true;
        });
        try
        {
            if (Recording)
            {
                report["phase"] = "waiting_for_l";
                report["spawn_disabled"] = true;
                Flush();
                var start = await CrossingDiscovery.OnGame(() =>
                {
                    Prepare(Main.LocalPlayer, StartX);
                    AgentChat.Display("铁轨加速录制已就绪，开始录屏后按 L。", 255, 215, 130);
                    Console.WriteLine("AGENT_RAIL_RECORD_READY");
                    recordingStart = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
                    return recordingStart.Task;
                });
                await start;
                limit.CancelAfter(TimeSpan.FromSeconds(360));
                report["phase"] = "running";
                Flush();
                Console.WriteLine("AGENT_RAIL_RECORD_STARTED");
                await Learn(folder, report, Flush, token);
                report["ok"] = report["skill"] != null;
                report["phase"] = "finished";
                Flush();
                return report.ToJsonString();
            }
            if (mode == "replay")
            {
                string path = Environment.GetEnvironmentVariable("TB_RAIL_POLICY_FILE");
                string code = File.ReadAllText(path);
                report["replay_start_x"] = StartX + 37;
                report["policy_file"] = path;
                report["replay"] = await RunProgram(code, false, token, StartX + 37);
                report["ok"] = report["replay"]["ok"].GetValue<bool>();
                report["model_calls"] = 0;
                Flush();
                return report.ToJsonString();
            }
            var calibration = await RunProgram(Calibration, true, token);
            report["calibration"] = calibration; Flush();
            if (calibration["ok"].GetValue<bool>() && mode == "live") await Learn(folder, report, Flush, token);
            report["ok"] = mode == "calibrate" ? calibration["ok"].GetValue<bool>() : report["skill"] != null;
            Flush();
            return report.ToJsonString();
        }
        catch (Exception error) { report["error"] = error.Message; Flush(); throw; }
    }

    static async Task Learn(string folder, JsonObject report, Action flush, CancellationToken token)
    {
        const string system = """
            Implement the requested single continuous sequence as a synchronous JavaScript function tick(s) with native game inputs. Submit it to run_policy, inspect real trace on failure, revise and retry. The tool resets to an empty flat arena each attempt. It saves the exact JS only after all stages pass. Use only the supplied tool. Up to eight real attempts.
            Rail boost as the player previously described: right-click the track and let the native track hook pull you. Jump before boarding to preserve the velocity. Briefly tell the player in Simplified Chinese what you are testing and what the observed results changed.
            s.player:{x,y,width,height,vx,vy,item_animation,item_time,release_use,selected_slot}; x,y are top-left pixels. s.tiles contains nearby existing tiles {x,y,type,solid,reachable}, in 16px tile coordinates. s.inventory contains {slot,type,stack,create_tile}; slots 0/1/2 are junction box/track/potion. s.tick is the world frame. s.memory is previously returned memory (use for phases and locked target cells). s.data has floor_y (tile row), boosts (host-counted verified cancels), featherfall, apex_reached, release_jump, track_tile, box_tile, hook (null or {x,y,touching} in pixels).
            Return {keys:string[],use:boolean,slot:number,aim:{x,y},memory:{...}}. Keys are pressed names only, e.g. ['up','jump']. Aim x/y and click x/y are integer TILE coordinates, in the same units as s.tiles. Use places/consumes the selected native item; changing slots waits until animations end. One right click: return memory.click={x,y,tick:s.tick} only on the desired frame. Jump is native input. The engine supplies all physics; you specify placements and timing. World y grows down. The host requires each click's own newly placed adjacent box and track, fresh upward or rightward boost with the requested buff, no grounding mid-chain and W continuously from the seventh box placement to apex. It finishes after 13 verified boosts. Do not return done until success.
            """;
        var tools = JsonNode.Parse("""
            [{"type":"function","function":{"name":"run_policy","description":"Execute your tick function in real Terraria and return the host's observed placements, launches, and tail trace.","parameters":{"type":"object","properties":{"code":{"type":"string"}},"required":["code"]}}}]
            """).AsArray();
        var attempts = new JsonArray(); var replies = new JsonArray();
        report["model"] = "gpt-6.2-sol"; report["attempts"] = attempts; report["replies"] = replies;
        report["objective"] = Objective;
        async Task<string> Execute(string tool, JsonObject args, CancellationToken caller)
        {
            if (report["skill"] != null) return """{"status":"already_verified"}""";
            if (attempts.Count >= 8) return """{"error":"attempt_budget_exhausted"}""";
            string code = args["code"].GetValue<string>();
            var attempt = new JsonObject { ["code"] = code };
            attempts.Add(attempt); flush();
            if (Recording) AgentChat.Say($"实测第 {attempts.Count} 版脚本。", bot: true);
            JsonObject result;
            try { result = await RunProgram(code, false, caller); }
            catch (ArgumentException error) { result = new JsonObject { ["ok"] = false, ["failure"] = error.Message }; }
            attempt["result"] = result; flush();
            if (result["ok"].GetValue<bool>())
            {
                string path = Path.Combine(folder, "learned_rail_policy.js");
                File.WriteAllText(path, code);
                report["skill"] = path;
                flush();
            }
            var feedback = result.DeepClone().AsObject();
            if (feedback["trace"] is JsonArray trace)
                feedback["trace"] = new JsonArray(trace.TakeLast(12).Select(entry => entry.DeepClone()).ToArray());
            return feedback.ToJsonString();
        }
        string sandbox = Path.Combine(folder, "agent-workspace");
        Directory.CreateDirectory(sandbox);
        using var agent = new CodexAgent();
        try
        {
            await agent.RunAsync(Objective, "", sandbox, report["model"].ToString(), system, tools, Execute,
                text => { replies.Add(text); if (Recording) AgentChat.Say(text); flush(); }, token);
        }
        finally { report["token_usage"] = agent.TokenUsage; flush(); }
    }
}
