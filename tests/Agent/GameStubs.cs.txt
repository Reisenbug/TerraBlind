        {
            transferred = numToTransfer ?? Math.Min(source.stack, destination.maxStack - destination.stack);

        }

        {
            var transfer = source.Clone();
            transfer.stack = Math.Min(limit, source.stack);
            source.stack -= transfer.stack;
            if (source.stack == 0) source.TurnToAir();
            return transfer;
        }

        public static System.Text.Json.Nodes.JsonObject InteractAt(int x, int y)
        {
            var result = System.Text.Json.Nodes.JsonNode.Parse(Dispatch("/interact",
                new System.Text.Json.Nodes.JsonObject { ["tile_x"] = x, ["tile_y"] = y }.ToJsonString()).Body).AsObject();
            result["x"] ??= x; result["y"] ??= y;
            result["world_id"] ??= Terraria.Main.worldID;
            return result;
        }
