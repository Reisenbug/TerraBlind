using System;
using System.Linq;
using System.Text.Json.Nodes;
using TerraBlind;
using Game = Terraria.Main;

static class InventoryTests
{
    public static void Run()
    {
        var self = Game.LocalPlayer;
        var old0 = self.inventory[0]; var old12 = self.inventory[12];
        int net = Game.netMode;
        var oldItems = Game.item;
        var oldTrash = self.trashItem;
        try
        {
            Game.netMode = Terraria.ID.NetmodeID.MultiplayerClient;
            self.inventory[0] = new Terraria.Item { type = 1, prefix = 2, stack = 1, Name = "weapon" };
            self.inventory[12] = new Terraria.Item { type = 3, stack = 20, Name = "blocks" };
            var args = new JsonObject { ["src"] = 12, ["dst"] = 0,
                ["expected_src"] = new JsonObject { ["type"] = 3, ["prefix"] = 0, ["stack"] = 20 },
                ["expected_dst"] = new JsonObject { ["type"] = 1, ["prefix"] = 2, ["stack"] = 1 } };
            self.itemAnimation = 1;
            if (PvpInventory.Swap(args)["error"]?.ToString() != "inventory_busy") throw new Exception("inventory swapped during animation");
            self.itemAnimation = 0;
            int sent = Terraria.NetMessage.Details.Count;
            var result = PvpInventory.Swap(args);
            if (result["ok"] == null || self.inventory[0].type != 3 || self.inventory[12].type != 1
                || !Terraria.NetMessage.Details.Skip(sent).Select(message => message.Slot).SequenceEqual(new float[] { 12, 0 }))
                throw new Exception("inventory swap lost an item or did not synchronize both actual slots");
            if (PvpInventory.Swap(args)["error"]?.ToString() != "stale_inventory") throw new Exception("stale expected items were swapped again");
            if (PvpInventory.Snapshot()[0]["prefix"].GetValue<int>() != 0) throw new Exception("snapshot omitted exact prefix");
            self.inventory[12] = new Terraria.Item { type = 3, stack = 20, favorited = true, material = true };
            var review = PvpInventory.Review();
            if (!review["inventory"].AsArray().Any(i => i["slot"].GetValue<int>() == 12 && i["favorited"].GetValue<bool>()
                && i["material"].GetValue<bool>()) || review["equipment"].AsArray().Count != 10)
                throw new Exception("inventory review omitted preservation or equipment facts");
            var drop = new JsonObject { ["slot"] = 12,
                ["expected"] = new JsonObject { ["type"] = 3, ["prefix"] = 0, ["stack"] = 20 } };
            if (PvpInventory.Drop(drop)["error"]?.ToString() != "favorited_item") throw new Exception("favorite was discarded");
            self.inventory[12].favorited = false;
            drop["expected"]["stack"] = 19;
            if (PvpInventory.Drop(drop)["error"]?.ToString() != "stale_inventory") throw new Exception("stale slot was discarded");
            drop["expected"]["stack"] = 20;
            Game.item = Enumerable.Range(0, Game.maxItems).Select(_ => new Terraria.Item { active = true }).ToArray();
            if (PvpInventory.Drop(drop)["error"]?.ToString() != "world_item_capacity" || self.inventory[12].stack != 20)
                throw new Exception("full world item array lost inventory");
            Game.item[0].active = false;
            var original = self.inventory[12];
            var dropped = PvpInventory.Drop(drop);
            if (dropped["ok"]?.GetValue<bool>() != true || !self.inventory[12].IsAir || !ReferenceEquals(Game.item[0], original)
                || Game.item[0].stack != 20 || dropped["world_item"].GetValue<int>() != 0)
                throw new Exception("drop lost the item instance or reported an unverified world slot");
            self.inventory[12] = new Terraria.Item { type = 3, stack = 20, favorited = true };
            self.trashItem = new Terraria.Item { type = 4, prefix = 1, stack = 7 };
            var trash = (JsonObject)drop.DeepClone();
            trash["expected_trash"] = PvpInventory.Review()["trash"].DeepClone();
            if (PvpInventory.Trash(trash)["error"]?.ToString() != "favorited_item")
                throw new Exception("favorite was trashed");
            self.inventory[12].favorited = false;
            self.trashItem.stack = 8;
            if (PvpInventory.Trash(trash)["error"]?.ToString() != "stale_trash" || self.inventory[12].stack != 20)
                throw new Exception("unobserved trash was overwritten");
            trash["expected_trash"]["stack"] = 8;
            int worldCount = Game.item.Count(i => i.active);
            original = self.inventory[12];
            var trashed = PvpInventory.Trash(trash);
            if (trashed["outcome"]?.ToString() != "trashed" || !self.inventory[12].IsAir
                || !ReferenceEquals(self.trashItem, original) || self.trashItem.stack != 20
                || trashed["replaced"]["stack"].GetValue<int>() != 8 || Game.item.Count(i => i.active) != worldCount)
                throw new Exception("trash failed to retain the stack or created a ground drop");
        }
        finally
        {
            self.itemAnimation = 0; self.inventory[0] = old0; self.inventory[12] = old12; Game.netMode = net; Game.item = oldItems;
            self.trashItem = oldTrash;
        }
        Console.WriteLine("PASS: inventory swap validates busy/stale states, preserves both items and emits two slot syncs");
    }
}
