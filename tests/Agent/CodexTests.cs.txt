    static void ContextChanges()
    {
        var context = new AgentContext();
        var state = JsonNode.Parse("""
            {"player":{"hp":240},"inventory_slots":[{"type":191,"stack":1}],
             "task_plan":{"revision":3,"status":"active","steps":[{"id":"walk","workflow":"walk","status":"active","script":"move()","on_failure":"repair()","conditions":[{"kind":"position","x":10,"y":20}]}]},
             "script_workflows":{"walk":{"revision":5,"running":true,"steps":[{"id":"first","status":"completed","value":{"large":"old result"}}]}},
             "action_tasks":[{"handle":"old","status":"completed","args":{"code":"large old code"}},
                 {"handle":"live","status":"running","tool":"nav_to","args":{"x":10,"y":20}}],
             "equipment":{"duplicate_inventory":"large"},"last_death":{"npc_type":222}}
            """).AsObject();
        var first = context.Capture(state);
        Check(first["mode"].ToString() == "snapshot" && first["state"]["task_plan"]["steps"][0]["script"] == null
            && first["state"]["task_plan"]["steps"][0]["on_failure"] == null
            && first["state"]["task_plan"]["steps"][0]["on_failure_present"]?.GetValue<bool>() == true
            && first["state"]["script_workflows"]["walk"]["steps"][0]["value"] == null
            && first["state"]["action_tasks"].AsArray().Count == 1 && first["state"]["equipment"] == null,
            "automatic context repeated completed effects, program source or duplicate inventory");
        Check(first["state"]["action_tasks"][0]["handle"].ToString() == "live"
            && first["state"]["task_plan"]["steps"][0]["workflow"]?.ToString() == "walk"
            && first["state"]["last_death"]["npc_type"].GetValue<int>() == 222,
            "compact context lost live ownership or actual death evidence");
        state["player"]["hp"] = 70; state["last_death"] = null;
        var changed = context.Capture(state);
        Check(changed["mode"].ToString() == "delta" && changed["state"]["player"]["hp"].GetValue<int>() == 70
            && changed["state"]["inventory_slots"] == null && changed["state"]["task_plan"] == null
            && changed["state"].AsObject().ContainsKey("last_death") && changed["state"]["last_death"] == null,
            "delta resent stable sections or retained a cleared danger fact");
        Check(state["task_plan"]["steps"][0]["script"].ToString() == "move()",
            "model context altered host-owned task source");
        context.Reset();
        Check(context.Capture(state)["mode"].ToString() == "snapshot", "compaction did not restore the current host snapshot");
        Console.WriteLine("PASS: model context emits bounded summaries and changed sections while retaining live evidence");
        if (Environment.GetEnvironmentVariable("TB_CONTEXT_REPLAY") is string recording)
        {
            context.Reset();
            long before = 0, after = 0, next = 0; int samples = 0;
            foreach (string line in File.ReadLines(recording))
            {
                var sample = JsonNode.Parse(line);
                long time = sample["elapsed_ms"].GetValue<long>();
                if (time < next) continue;
                next = time + 15000;
                var full = sample["state"].AsObject();
                before += full.ToJsonString(AgentTools.ModelJson).Length;
                after += context.Capture(full).ToJsonString(AgentTools.ModelJson).Length;
                samples++;
            }
            Console.WriteLine($"CONTEXT_REPLAY samples={samples} original_chars={before} delta_chars={after}");
        }
    }

    static async Task NativeExecutionSteering(bool script = false, bool preparedReview = false, bool rejectSteer = false, bool preparationOnly = false)
    {
        var native = new NativeAgent();
        var server = new Server { HoldFinalReply = true, RejectSteer = rejectSteer };
        var handler = HttpServerSystem.Handler;
        bool fail = false;
        int starts = 0;
        string handle = null;
        Config.I.AgentUseCodex = true;
        native.Load();
        typeof(NativeAgent).GetField("_codex", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic)
            .SetValue(native, new CodexAgent((_, _) => server.Connect()));
        HttpServerSystem.Handler = (path, _) =>
        {
            if (path == "/nav_recede") starts++;
            return (200, path == "/nav_recede_done" ? fail ? new JsonObject { ["done"] = false,
                ["status"] = "failed", ["reason"] = "fixture_blocked", ["detail"] = new string('x', 10000) }.ToJsonString()
                : "{\"done\":false}" : "{}");
        };
        server.AfterTool = calls =>
        {
            if (calls != 1) return;
            _ = NativeAgent.OnMain(() =>
            {
                AgentTools.Plan.Create(JsonNode.Parse("""
                    {"goal":"prepare during a delayed reply","mode":"dag","steps":[
                     {"id":"travel","description":"keep moving while preparing","depends_on":[],
                      "script":"return step('move',()=>{const r=tool('nav_to',{x:900,y:100});if(!r.done)throw new Error('blocked');return r;});",
                      "conditions":[{"kind":"permanent_health","count":9999}]}]}
                    """).AsObject(), AgentTaskEvidence.Measure);
                if (preparationOnly)
                {
                    var addition = JsonNode.Parse("""
                        {"expected_revision":0,"reason":"future action is still a template lookup","add_nodes":[
                         {"id":"collect","description":"execute the prepared collection","depends_on":["travel"],
                          "script":"return tool('script_program',{operation:'get',name:'use_life_crystals'});",
                          "conditions":[{"kind":"permanent_health","count":9999}]}]}
                        """).AsObject();
                    addition["expected_revision"] = AgentTools.Plan.Snapshot()["revision"].DeepClone();
                    AgentTools.Plan.Revise(addition, AgentTaskEvidence.Measure);
                }
                ActionTasks.NodeContext.Value = "travel";
                try { handle = AgentTools.Actions.Start(script ? "run_script" : "nav_to", script
                    ? new JsonObject { ["task_node"] = "travel" } : new JsonObject { ["x"] = 900, ["y"] = 100 }, default)["handle"].ToString(); }
                finally { ActionTasks.NodeContext.Value = null; }
                return true;
            }, default);
        };
        async Task Until(Func<bool> condition)
        {
            var deadline = DateTime.UtcNow.AddSeconds(3);
            while (!condition())
            {
                Check(DateTime.UtcNow < deadline, "active execution was not delivered before the delayed reply finished");
                native.PostUpdateEverything(); await Task.Delay(5);
            }
        }
        try
        {
            NativeAgent.Submit("prepare the next operation while this one executes");
            await Until(() => server.Steers.Count >= 1 && starts == 1);
            Check(server.Inputs.Count == 1 && server.Interrupts == 0 && AgentTools.Actions.ActiveHandle == handle,
                "preparation steering started another turn, paused or replaced the route");
            Check(server.Steers[0]["input"][0]["text"].ToString().Contains("prepare_successor"),
                "action start did not request preparation in the current reasoning turn");
            var quiet = DateTime.UtcNow.AddMilliseconds(700);
            while (DateTime.UtcNow < quiet) { native.PostUpdateEverything(); await Task.Delay(5); }
            Check(rejectSteer || server.Steers.Count == 1, "unchanged action repeatedly steered the same delayed reply");
            if (preparedReview)
            {
                server.FinishReply();
                if (preparationOnly)
                {
                    await Until(() => server.Inputs.Count == 2);
                    Check(server.Inputs[1].Contains("collect") && AgentTools.Actions.ActiveHandle == handle && starts == 1,
                        "preparation of a read-only successor waited for arrival or replaced navigation");
                    server.FinishReply();
                    quiet = DateTime.UtcNow.AddMilliseconds(700);
                    while (DateTime.UtcNow < quiet) { native.PostUpdateEverything(); await Task.Delay(5); }
                    Check(server.Inputs.Count == 2, "unchanged preparation-only source caused an unbounded background review loop");
                    Console.WriteLine("PASS: preparation-only successors receive a bounded review while the original action runs");
                    return;
                }
                if (rejectSteer)
                {
                    await Until(() => server.Inputs.Count == 2);
                    Check(AgentTools.Actions.ActiveHandle == handle && starts == 1,
                        "fallback preparation replaced the action after steering was rejected");
                    Console.WriteLine("PASS: rejected preparation steering retains the normal background preparation turn");
                    return;
                }
                quiet = DateTime.UtcNow.AddMilliseconds(700);
                while (DateTime.UtcNow < quiet) { native.PostUpdateEverything(); await Task.Delay(5); }
                Check(server.Inputs.Count == 1 && AgentTools.Actions.ActiveHandle == handle && starts == 1,
                    "accepted in-turn preparation was repeated in another model turn");
                AgentTools.Watches.Emit("low_health", "new-health-after-preparation", new JsonObject { ["hp"] = 20 });
                await Until(() => server.Inputs.Count == 2);
                Check(server.Inputs[1].Contains("new-health-after-preparation") && AgentTools.Actions.ActiveHandle == handle && starts == 1,
                    "preparation deduplication suppressed a new health observation or replaced the running action");
                Console.WriteLine("PASS: accepted preparation avoids a duplicate turn while new health evidence still wakes the model");
                return;
            }
            fail = true;
            await Until(() => server.Steers.Count == 2);
            Check(server.Steers[1]["input"][0]["text"].ToString().Contains("execution_gap")
                && server.Inputs.Count == 1 && server.Interrupts == 0,
                "failed execution waited for final text and a new reasoning turn before reporting its gap");
            string failureNotice = server.Steers[1]["input"][0]["text"].ToString();
            Check(failureNotice.Contains("fixture_blocked") && failureNotice.Contains("900") && failureNotice.Length < 2500,
                "execution notice omitted the actual failed tool and target, or resent an oversized result: " + failureNotice[..Math.Min(500, failureNotice.Length)]);
            Console.WriteLine("PASS: active-turn execution steering prepares successors and delivers uncovered failure before final reply");
        }
        finally
        {
            NativeAgent.Stop(); native.OnWorldUnload(); native.Unload();
            Config.I.AgentUseCodex = false; HttpServerSystem.Handler = handler;
            AgentTools.Plan.Reset(); AgentTools.Watches.Reset();
        }
    }

    static async Task NativeBackgroundReview()
    {
        var native = new NativeAgent();
        var server = new Server();
        var originalPosition = Terraria.Main.LocalPlayer.position;
        Config.I.AgentUseCodex = true;
        native.Load();
        typeof(NativeAgent).GetField("_codex", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic)
            .SetValue(native, new CodexAgent((_, _) => server.Connect()));
        string handle = null;
        Terraria.Main.LocalPlayer.position = new(1600, 1600);
        HttpServerSystem.Handler = (_, _) => (200, "{\"done\":false}");
        server.AfterTool = calls =>
        {
            if (calls == 1) _ = NativeAgent.OnMain(() =>
            {
                handle = AgentTools.Actions.Start("nav_to", new JsonObject { ["x"] = 900, ["y"] = 100 }, default)["handle"].ToString();
                return true;
            }, default);
        };
        try
        {
            NativeAgent.Submit("/goal observe while navigation remains active");
            var movement = System.Diagnostics.Stopwatch.StartNew();
            void AdvanceMovement() => Terraria.Main.LocalPlayer.position.X = 1600 + (float)movement.Elapsed.TotalSeconds * 50;
            var deadline = DateTime.UtcNow.AddSeconds(25);

    sealed class Server
    {
        readonly Lines _input = new();
        public int Threads, Interrupts;
        int _turn;
        public bool Stopped, LoggedIn = true, FailTurn, EmitCompaction, HoldFinalReply, RejectSteer;
        public string ToolName = "get_state";
        public Action<int> AfterTool;
        public JsonObject ThreadArgs;
        public JsonObject Goal;
        public int GoalSets;
        public int RepeatedToolCalls = 1;
        public readonly List<JsonObject> ToolResponses = new();
        public readonly List<string> Inputs = new();
        public readonly List<JsonObject> Steers = new();
        public void Drop() => _input.End();
        public CodexConnection Connect() => new(_input, new Writer(Receive), () => { Stopped = true; _input.End(); });
        void Send(JsonObject message) => _input.Add(message.ToJsonString());
        void Reply(JsonNode id, JsonObject result) => Send(new JsonObject { ["id"] = id.DeepClone(), ["result"] = result });
        void Completed(string status) => Send(new JsonObject { ["method"] = "turn/completed", ["params"] = new JsonObject
        { ["threadId"] = "thread", ["turn"] = new JsonObject { ["id"] = "turn" + _turn, ["status"] = status } } });
        public void FinishReply()
        {
            var completed = new JsonObject { ["method"] = "item/completed", ["params"] = new JsonObject
            { ["threadId"] = "thread", ["turnId"] = "turn" + _turn, ["item"] = new JsonObject
            { ["id"] = "reply" + _turn, ["type"] = "agentMessage", ["text"] = "完成了" } } };
            Send(completed); Send(completed); Completed("completed");
        }
        void Receive(string line)
        {
            var message = JsonNode.Parse(line).AsObject();
            string method = message["method"]?.ToString();
            if (method == "initialized") return;
            if (method == "initialize") Reply(message["id"], new JsonObject());
            else if (method == "account/read") Reply(message["id"], new JsonObject { ["account"] = LoggedIn ? new JsonObject { ["type"] = "chatgpt" } : null });
            else if (method == "config/read") Reply(message["id"], JsonNode.Parse("{\"config\":{\"mcp_servers\":{\"unrelated\":{}}}}").AsObject());
            else if (method == "thread/start")
            { Threads++; ThreadArgs = message["params"].AsObject(); Reply(message["id"], JsonNode.Parse("{\"thread\":{\"id\":\"thread\"}}").AsObject()); }
            else if (method == "thread/goal/set")
            {
                GoalSets++; Goal ??= new JsonObject();
                foreach (var pair in message["params"].AsObject())
                    if (pair.Key is "objective" or "status" or "tokenBudget") Goal[pair.Key] = pair.Value?.DeepClone();
                Reply(message["id"], new JsonObject { ["goal"] = Goal.DeepClone() });
            }
            else if (method == "thread/goal/get") Reply(message["id"], new JsonObject { ["goal"] = Goal?.DeepClone() });
            else if (method == "turn/start")
            {
                Inputs.Add(message["params"]["input"][0]["text"].ToString());
                _turn++;
                Reply(message["id"], new JsonObject { ["turn"] = new JsonObject { ["id"] = "turn" + _turn } });
                if (FailTurn) { Completed("failed"); return; }
                if (EmitCompaction) Send(new JsonObject { ["method"] = "item/completed", ["params"] = new JsonObject
                    { ["threadId"] = "thread", ["turnId"] = "turn" + _turn, ["item"] = new JsonObject { ["type"] = "contextCompaction" } } });
                Send(new JsonObject { ["method"] = "item/tool/call", ["id"] = "tool" + _turn, ["params"] = new JsonObject
                { ["threadId"] = "thread", ["turnId"] = "turn" + _turn, ["callId"] = "call" + _turn, ["tool"] = ToolName, ["arguments"] = new JsonObject() } });
            }
            else if (method == "turn/interrupt") { Interrupts++; Reply(message["id"], new JsonObject()); Completed("interrupted"); }
            else if (method == "turn/steer")
            {
                Steers.Add(message["params"].AsObject());
                if (RejectSteer) Send(new JsonObject { ["id"] = message["id"].DeepClone(), ["error"] = new JsonObject { ["message"] = "fixture steering rejected" } });
                else Reply(message["id"], new JsonObject { ["turnId"] = "turn" + _turn });
            }
            else if (method == null)
            {
                ToolResponses.Add(message["result"].AsObject());
                AfterTool?.Invoke(ToolResponses.Count);
                var usage = new JsonObject { ["method"] = "thread/tokenUsage/updated", ["params"] = new JsonObject
                    { ["threadId"] = "thread", ["turnId"] = "turn" + _turn, ["tokenUsage"] = new JsonObject
                        { ["total"] = new JsonObject { ["totalTokens"] = _turn * 100 } } } };
                Send(usage); Send(usage);
                if (ToolResponses.Count < RepeatedToolCalls)
                {
                    Send(new JsonObject { ["method"] = "item/tool/call", ["id"] = "repeat" + ToolResponses.Count,
                        ["params"] = new JsonObject { ["threadId"] = "thread", ["turnId"] = "turn" + _turn,
                            ["tool"] = ToolName, ["arguments"] = new JsonObject() } });
                    return;
                }
                if (HoldFinalReply) return;
                FinishReply();
            }
        }
    }
    sealed class Lines : TextReader
    {
        readonly Channel<string> _lines = Channel.CreateUnbounded<string>();
        public void Add(string line) => _lines.Writer.TryWrite(line);
        public void End() => _lines.Writer.TryComplete();
        public override ValueTask<string> ReadLineAsync(CancellationToken token) => _lines.Reader.ReadAsync(token);
    }
    sealed class Writer : TextWriter
    {
        readonly Action<string> _send;
        public Writer(Action<string> send) => _send = send;
        public override Encoding Encoding => Encoding.UTF8;
        public override Task WriteLineAsync(ReadOnlyMemory<char> value, CancellationToken token = default)
        { token.ThrowIfCancellationRequested(); _send(value.ToString()); return Task.CompletedTask; }
    }
