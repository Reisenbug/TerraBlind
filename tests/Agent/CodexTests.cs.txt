    static readonly JsonArray Tools = JsonNode.Parse("""

    static void Check(bool value, string message) { if (!value) throw new Exception(message); }

    public static async Task Run()
    {
        await GoalProtocol();
        await NativeGoalProtocol();
        var launch = new System.Diagnostics.ProcessStartInfo();
        launch.Environment.Clear();
        CodexAgent.ApplySystemProxy(launch, new System.Net.WebProxy("http://127.0.0.1:12345"));
        Check(launch.Environment["HTTPS_PROXY"] == "http://127.0.0.1:12345/", "system proxy was not forwarded to Codex");
        launch.Environment["HTTPS_PROXY"] = "http://explicit:8080";
        CodexAgent.ApplySystemProxy(launch, new System.Net.WebProxy("http://127.0.0.1:12346"));
        Check(launch.Environment["HTTPS_PROXY"] == "http://explicit:8080", "explicit proxy was overwritten");
        var server = new Server();
        using var agent = new CodexAgent((_, _) => server.Connect());
        int calls = 0;
        var spoken = new List<string>();
        Task<string> Execute(string name, JsonObject args, CancellationToken token)
        { calls++; return Task.FromResult("{\"wood\":23}"); }
        await agent.RunAsync("检查背包", "mock", "/tmp", "", "game instructions", Tools, Execute, spoken.Add, default);
        await agent.RunAsync("继续", "mock", "/tmp", "", "game instructions", Tools, Execute, spoken.Add, default);
        Check(calls == 2 && server.Threads == 1 && spoken.Count == 2, "Codex did not reuse thread or deduplicate replies");
        Check(agent.TokenUsage["total"]["totalTokens"].GetValue<int>() == 200 && agent.UsageUpdates == 2 && agent.TurnsStarted == 2,
            "Codex usage telemetry did not preserve cumulative token usage across turns");
        Check(server.ThreadArgs["baseInstructions"].ToString() == "game instructions" && server.ThreadArgs["environments"].AsArray().Count == 0
            && server.ThreadArgs["config"]["mcp_servers.unrelated.enabled"].GetValue<bool>() == false,
            "game thread did not override instructions/environments/MCP");
        string gameInstructions = server.ThreadArgs["developerInstructions"].ToString();
        Check(gameInstructions.Contains("script_program") && gameInstructions.Contains("run_script")
            && !gameInstructions.Contains("不要开发代码") && gameInstructions.Contains("不要修改模组源码"),
            "game thread forbids its own behavior scripts or permits editing mod source");
        Check(server.ThreadArgs["dynamicTools"][0]["type"].ToString() == "function" && server.ToolResponses.All(r => r["success"].GetValue<bool>()),
            "dynamic tool registration/result contract changed");
        Console.WriteLine("PASS: Codex protocol registers game tools, isolates thread settings, reuses context and deduplicates replies");

        server.ToolName = "unknown";
        await agent.RunAsync("未知工具", "mock", "/tmp", "", "game instructions", Tools, Execute, spoken.Add, default);
        Check(calls == 2 && !server.ToolResponses.Last()["success"].GetValue<bool>(), "unknown Codex tool executed");
        server.ToolName = "ask";
        await agent.RunAsync("缺少参数", "mock", "/tmp", "", "game instructions", Tools, Execute, spoken.Add, default);
        Check(calls == 2 && !server.ToolResponses.Last()["success"].GetValue<bool>(), "missing required Codex argument executed");
        server.ToolName = "get_state";
        await agent.RunAsync("数组结果", "mock", "/tmp", "", "game instructions", Tools,
            (_, _, _) => Task.FromResult("[1,2]"), spoken.Add, default);
        Check(server.ToolResponses.Last()["success"].GetValue<bool>(), "JSON array tool result was rejected");
        foreach (string rejected in new[] { "{\"ok\":false}", "{\"accepted\":false}", "{\"error\":\"blocked\"}" })
        {
            await agent.RunAsync("被拒绝的操作", "mock", "/tmp", "", "game instructions", Tools,
                (_, _, _) => Task.FromResult(rejected), spoken.Add, default);
            Check(!server.ToolResponses.Last()["success"].GetValue<bool>(), "Codex marked rejected tool output successful: " + rejected);
            Check(server.ToolResponses.Last()["contentItems"][0]["text"].ToString() == rejected, "Codex lost rejection details");
        }
        Console.WriteLine("PASS: Codex rejects unknown/malformed calls and accepts non-object tool output");
        var unicodeResult = new JsonObject { ["name"] = "生命水晶", ["code"] = "const text = '\\u4e2d';\nreturn text;", ["hp"] = 60 };
        await agent.RunAsync("保留中文与代码", "mock", "/tmp", "", "game instructions", Tools,
            (_, _, _) => Task.FromResult(unicodeResult.ToJsonString()), spoken.Add, default);
        string unicodeText = server.ToolResponses.Last()["contentItems"][0]["text"].ToString();
        Check(unicodeText.Contains("生命水晶") && JsonNode.DeepEquals(JsonNode.Parse(unicodeText), unicodeResult),
            "model result serialization expands Chinese or changes literal code escapes");

        using var cancel = new CancellationTokenSource();
        var entered = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
        var running = agent.RunAsync("长任务", "mock", "/tmp", "", "game instructions", Tools,
            async (_, _, token) => { entered.SetResult(true); await Task.Delay(Timeout.Infinite, token); return "unreachable"; }, spoken.Add, cancel.Token);
        await entered.Task.WaitAsync(TimeSpan.FromSeconds(2));
        int turnsBeforeDanger = agent.TurnsStarted;

            && server.Steers.Count == 1 && server.Steers[0]["expectedTurnId"].ToString().StartsWith("turn")
            && server.Steers[0]["threadId"].ToString() == "thread" && agent.TurnsStarted == turnsBeforeDanger,
            "live danger did not steer the current turn or replaced the task");
        cancel.Cancel();
        try { await running; throw new Exception("Codex cancellation ignored"); } catch (OperationCanceledException) { }
        Check(server.Interrupts == 1 && !server.Stopped, "Codex did not interrupt while preserving reusable session");
        Check(!await agent.Steer("stale danger", default) && server.Steers.Count == 1,
            "ended turn accepted stale danger steering");
        await agent.RunAsync("新任务", "mock", "/tmp", "", "game instructions", Tools, Execute, spoken.Add, default);
        Check(server.Threads == 1 && calls == 3, "new Codex task failed after interruption");
        Console.WriteLine("PASS: Codex interrupts blocked tool calls and accepts a replacement task in the same thread");

        var loggedOut = new Server { LoggedIn = false };
        using var noLogin = new CodexAgent((_, _) => loggedOut.Connect());
        try
        {
            await noLogin.RunAsync("hello", "mock", "/tmp", "", "system", Tools, Execute, spoken.Add, default);
            throw new Exception("missing Codex login ignored");
        }
        catch (InvalidOperationException ex) { Check(ex.Message.Contains("login"), "login failure is not actionable"); }
        Check(loggedOut.Stopped, "failed Codex startup leaked process");
        var broken = new Server { FailTurn = true };
        using var failed = new CodexAgent((_, _) => broken.Connect());
        try
        {
            await failed.RunAsync("hello", "mock", "/tmp", "", "system", Tools, Execute, spoken.Add, default);
            throw new Exception("failed Codex turn ignored");
        }
        catch (InvalidOperationException ex) { Check(ex.Message.Contains("failed"), "failed turn reported as success"); }
        Check(broken.Stopped, "failed Codex turn leaked process");
        var disconnected = new Server();
        using var connection = disconnected.Connect();
        var pending = connection.Request("no_reply", new JsonObject(), default);
        disconnected.Drop();
        try { await pending.WaitAsync(TimeSpan.FromSeconds(2)); throw new Exception("Codex EOF did not fail pending request"); }
        catch (IOException) { }
        Console.WriteLine("PASS: Codex login/turn failures report errors and dispose owned connection");

        var native = new NativeAgent();
        var hosted = new Server();
        Config.I.AgentUseCodex = true; Config.I.AgentApiKey = "";
        native.Load();
        typeof(NativeAgent).GetField("_codex", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic)
            .SetValue(native, new CodexAgent((_, _) => hosted.Connect()));
        int previous = AgentChat.Messages.Count;
        try
        {
            NativeAgent.Submit("看看背包");
            Check(AgentChat.Messages.Count == previous + 1 && NativeAgent.Status == "queued", "task submission did not immediately acknowledge");
            var deadline = DateTime.UtcNow.AddSeconds(5);
            while (AgentChat.Messages.Count == previous || NativeAgent.Status != "idle")

                Check(DateTime.UtcNow < deadline, "NativeAgent Codex dispatch timed out");
                native.PostUpdateEverything(); await Task.Delay(5);
            }

            Check(hosted.ThreadArgs["model"].ToString() == "gpt-6-luna", "game Agent did not select Luna when model was unset");
            native.OnWorldUnload();
            Check(hosted.Stopped, "world unload did not dispose Codex process");
        }
        finally { native.Unload(); Config.I.AgentUseCodex = false; Config.I.AgentApiKey = "test-key"; }
        Console.WriteLine("PASS: NativeAgent routes through Codex without API key and closes connection on world unload");
        native = new NativeAgent(); hosted = new Server();
        var self = Terraria.Main.LocalPlayer; var opponent = Terraria.Main.player[1];
        self.dead = opponent.dead = false; self.active = opponent.active = self.hostile = opponent.hostile = true;
        self.team = opponent.team = 0;
        self.inventory[0] = new Terraria.Item { type = 99, stack = 1, damage = 20, useStyle = 1, noMelee = true, shoot = 1, shootSpeed = 10 };
        Config.I.AgentUseCodex = Config.I.AiClient = true;
        Terraria.Main.netMode = Terraria.ID.NetmodeID.MultiplayerClient;
        native.Load();
        typeof(NativeAgent).GetField("_codex", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic)
            .SetValue(native, new CodexAgent((_, _) => hosted.Connect()));
        hosted.AfterTool = count =>
        {
            _ = NativeAgent.OnMain(() =>
            {
                if (count == 1)
                {
                    if (!PvpDuel.Start(1, 30, 0, 600, "持续交战", out string why,
                        (_, _) => Task.FromResult(new PvpTactics.Decision("m1", "ranged")))) throw new Exception(why);
                }
                return true;
            }, default);
        };
        try
        {
            NativeAgent.Submit("持续交战并复盘");
            async Task WaitParked(int turns)
            {
                var until = DateTime.UtcNow.AddSeconds(5);
                while (hosted.ToolResponses.Count < turns || NativeAgent.Status != "等待战斗指令（不调用模型）")
                {
                    Check(DateTime.UtcNow < until, "combat Agent failed to park after turn " + turns);
                    native.PostUpdateEverything(); Terraria.Main.GameUpdateCount++;
                    await Task.Delay(5);
                }
            }
            await WaitParked(1);
            Terraria.Main.GameUpdateCount += 60 * 60 * 60;
            for (int i = 0; i < 70; i++) { native.PostUpdateEverything(); await Task.Delay(5); }
            Check(hosted.ToolResponses.Count == 1 && PvpDuel.Active, "elapsed game time caused a periodic model request");
            NativeAgent.Submit("别追近战");
            await WaitParked(2);
            Terraria.Collision.Clear = false;
            for (int i = 0; i < 181; i++) native.PostUpdateEverything();
            await WaitParked(3);
            for (int i = 0; i < 100; i++) { native.PostUpdateEverything(); await Task.Delay(5); }
            Check(hosted.ToolResponses.Count == 3, "persistent obstruction repeatedly requested automatic Agent help");
            Terraria.Collision.Clear = true;
            PvpDuel.Stop();
            var deadline = DateTime.UtcNow.AddSeconds(8);
            while (hosted.ToolResponses.Count < 2 || NativeAgent.Status != "idle")
            {
                Check(DateTime.UtcNow < deadline, "supervised combat timeout status=" + NativeAgent.Status + " responses=" + hosted.ToolResponses.Count
                    + " active=" + PvpDuel.Active + " outcome=" + PvpDuel.Outcome + " messages=" + string.Join(";", AgentChat.Messages.TakeLast(5)));

                if (PvpDuel.Active) PvpDuel.Tick(self);
                await Task.Delay(5);
            }
            Check(hosted.Threads == 1 && !PvpDuel.Active, "combat supervision created a second Codex thread or ignored stop");
            var review = JsonNode.Parse(hosted.Inputs[1].Split("战况：")[1]);
            Check(review["player_messages"][0].GetValue<string>() == "别追近战", "queued player follow-up was lost before combat review");
        }
        finally
        {
            native.OnWorldUnload(); native.Unload();
            Config.I.AgentUseCodex = Config.I.AiClient = false;
            Terraria.Main.netMode = Terraria.ID.NetmodeID.SinglePlayer;
        }
        Console.WriteLine("PASS: combat makes no periodic model calls, handles player follow-ups, permits one recovery and respects stop on the same Codex thread");
    }

    static async Task NativeGoalProtocol()
    {
        var native = new NativeAgent();
        var server = new Server();
        Config.I.AgentUseCodex = true;
        native.Load();
        typeof(NativeAgent).GetField("_codex", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic)
            .SetValue(native, new CodexAgent((_, _) => server.Connect()));
        try
        {
            NativeAgent.Submit("/goal do not trust an unverified final reply");
            async Task WaitCalls(int calls)
            {
                var deadline = DateTime.UtcNow.AddSeconds(20);
                while (server.ToolResponses.Count < calls)
                {
                    Check(DateTime.UtcNow < deadline, "unverified native goal stopped continuing");
                    native.PostUpdateEverything(); await Task.Delay(5);
                }
            }
            await WaitCalls(3);
            Check(server.Goal["status"].ToString() == "active" && server.Threads == 1,
                "model final reply completed missing game conditions or failed to preserve goal");
            Check(NativeAgent.Status != "needs_repair" && NativeAgent.GoalState["host_tokens_used"].GetValue<long>() >= 300,
                "goal stopped at the old no-progress bound or reset usage");
            NativeAgent.Submit("stop");
            int calls = server.ToolResponses.Count;
            for (int i = 0; i < 30; i++) { native.PostUpdateEverything(); await Task.Delay(5); }
            Check(server.ToolResponses.Count == calls, "explicit stop failed to cancel goal continuation");
        }
        finally { native.OnWorldUnload(); native.Unload(); Config.I.AgentUseCodex = false; }
        Console.WriteLine("PASS: unverified game goal continues beyond no-progress turns on the same thread until explicit stop");
    }

    static async Task GoalProtocol()
    {
        var server = new Server();
        using var agent = new CodexAgent((_, _) => server.Connect());
        var options = new JsonObject { ["objective"] = "collect and verify", ["request"] = "initial" };
        Task Run() => agent.RunAsync("work", "mock", "/tmp", "", "game", Tools,
            (_, _, _) => Task.FromResult("{\"stock\":10}"), _ => { }, default, options);
        await Run();
        Check(server.GoalSets == 1 && server.ThreadArgs["ephemeral"].GetValue<bool>() == false
            && server.ThreadArgs["config"]["features.goals"].GetValue<bool>() == false,
            "goal session was not persistent or left model-side completion/dispatch uncontrolled");
        var rejected = await agent.CompleteGoal(_ => Task.FromResult(new JsonObject { ["verified"] = false }), default);
        Check(rejected["error"]?.ToString() == "goal_not_verified" && server.Goal["status"].ToString() == "active",
            "model ending or failed verification completed native goal");
        await Run();
        Check(server.GoalSets == 1 && agent.GoalState["host_tokens_used"].GetValue<long>() == 200,
            "continuation reset objective, goal or usage");
        await agent.SuspendGoal("paused", default);
        options["request"] = "explicit-resume";
        await Run();
        Check(agent.GoalState["host_tokens_used"].GetValue<long>() == 300, "resume reset token accounting");
        var passed = await agent.CompleteGoal(_ => Task.FromResult(new JsonObject { ["verified"] = true }), default);
        Check(passed["verified"].GetValue<bool>() && server.Goal["status"].ToString() == "complete",
            "verified goal was not committed through native goal/set");
        Console.WriteLine("PASS: native goal RPC persists across turns, rejects unverified completion and preserves resume accounting");

        var limited = new Server();
        using var budget = new CodexAgent((_, _) => limited.Connect());
        try
        {
            await budget.RunAsync("work", "mock", "/tmp", "", "game", Tools,
                (_, _, _) => Task.FromResult("{}"), _ => { }, default,
                new JsonObject { ["objective"] = "bounded", ["request"] = "budget", ["tokenBudget"] = 50L });
            throw new Exception("goal token limit was ignored");
        }
        catch (GoalBudgetExceededException) { }
        Check(limited.Interrupts == 1 && limited.Goal["status"].ToString() == "budgetLimited",
            "budget did not interrupt or falsely completed goal");
        Console.WriteLine("PASS: actual token telemetry interrupts a budget-limited goal without completing it");
        var repeated = new Server { RepeatedToolCalls = 33 };
        using var bounded = new CodexAgent((_, _) => repeated.Connect());
        int executions = 0;
        var loopOptions = new JsonObject { ["objective"] = "bounded failures", ["request"] = "retry" };
        await bounded.RunAsync("retry", "mock", "/tmp", "", "game", Tools,
            (_, _, _) => { executions++; return Task.FromResult("{\"error\":\"no_progress\"}"); }, _ => { }, default, loopOptions);
        Check(executions == 32 && repeated.Goal["status"].ToString() == "active" && !repeated.Stopped && repeated.Interrupts == 1
            && repeated.ToolResponses.Last()["contentItems"][0]["text"].ToString().Contains("host_turn_limit"),
            "turn limit executed the rejected call, stopped the goal or disposed its connection");
        repeated.RepeatedToolCalls = repeated.ToolResponses.Count + 1;
        await bounded.RunAsync("continue", "mock", "/tmp", "", "game", Tools,
            (_, _, _) => { executions++; return Task.FromResult("{}"); }, _ => { }, default, loopOptions);
        Check(executions == 33 && repeated.Threads == 1 && repeated.Goal["status"].ToString() == "active",
            "next turn failed to continue on the same goal and thread");
        Console.WriteLine("PASS: managed goal keeps the per-turn effect bound and continues on the same thread after budget handoff");
    }

    public static async Task GoalSmoke()
    {
        using var agent = new CodexAgent();
        using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(120));
        string cwd = Path.Combine(Path.GetTempPath(), "terrablind-goal-protocol-" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(cwd);
        var events = new JsonArray();
        await agent.RunAsync("Read fixture state once, then finish this turn.", "", cwd, "gpt-6-astra",
            "Only use get_state once and report its value. The game host controls the goal lifecycle.", Tools,
            (name, args, _) => { events.Add(new JsonObject { ["tool"] = name, ["args"] = args.DeepClone() }); return Task.FromResult("{\"fixture_stock\":10}"); },
            text => events.Add(new JsonObject { ["text"] = text }), timeout.Token,
            new JsonObject { ["objective"] = "Read fixture state; completion requires host verification.", ["request"] = "smoke" });
        Check(events.Any(e => e["tool"]?.ToString() == "get_state"), "real model did not invoke fixture tool");
        var before = agent.GoalState;
        var denied = await agent.CompleteGoal(_ => Task.FromResult(new JsonObject { ["verified"] = false }), timeout.Token);
        Check(denied["error"]?.ToString() == "goal_not_verified" && agent.GoalState["native"]["status"].ToString() == "active", "native goal completed without host evidence");
        var completed = await agent.CompleteGoal(_ => Task.FromResult(new JsonObject { ["verified"] = true, ["fixture_stock"] = 10 }), timeout.Token);
        Check(agent.GoalState["native"]["status"].ToString() == "complete", "native goal did not complete");
        File.WriteAllText(Path.Combine(cwd, "result.json"), new JsonObject { ["before"] = before,
            ["rejected"] = denied, ["completed"] = completed, ["events"] = events, ["usage"] = agent.TokenUsage }.ToJsonString());
        Console.WriteLine("GOAL_PROTOCOL_SMOKE_COMPLETE actual_codex=true mock_world=true report=" + Path.Combine(cwd, "result.json"));
    }

    public static async Task GoalControlSmoke()
    {
        string root = Path.Combine(Path.GetTempPath(), "terrablind-goal-controls-" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(root);
        var results = new JsonObject();
        using var deadline = new CancellationTokenSource(TimeSpan.FromSeconds(180));
        using (var stopped = new CodexAgent())
        using (var cancel = CancellationTokenSource.CreateLinkedTokenSource(deadline.Token))
        {
            var entered = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
            int effects = 0;
            var running = stopped.RunAsync("Read get_state once now.", "", Path.Combine(root, "stop"), "gpt-6-astra",
                "Only call the supplied get_state tool once. Do not guess the state.", Tools,
                async (_, _, token) => { effects++; entered.TrySetResult(true); await Task.Delay(Timeout.Infinite, token); return "unreachable"; },
                _ => { }, cancel.Token, new JsonObject { ["objective"] = "read fixture", ["request"] = "stop" });
            await entered.Task.WaitAsync(deadline.Token);
            cancel.Cancel();
            try { await running; throw new Exception("real Codex stop failed"); } catch (OperationCanceledException) { }
            await Task.Delay(500, deadline.Token);
            Check(effects == 1 && stopped.TurnsStarted == 1 && stopped.GoalState["native"]["status"].ToString() == "paused",
                "real stop leaked effects, requests or active native goal");
            results["stop"] = new JsonObject { ["goal"] = stopped.GoalState, ["effects"] = effects, ["turns"] = stopped.TurnsStarted };
        }
        using (var limited = new CodexAgent())
        {
            try
            {
                await limited.RunAsync("Read get_state once.", "", Path.Combine(root, "budget"), "gpt-6-astra", "Use get_state once.", Tools,
                    (_, _, _) => Task.FromResult("{\"fixture\":true}"), _ => { }, deadline.Token,
                    new JsonObject { ["objective"] = "bounded read", ["request"] = "budget", ["tokenBudget"] = 1L });
                throw new Exception("real token telemetry failed to stop budget goal");
            }
            catch (GoalBudgetExceededException) { }
            Check(limited.GoalState["native"]["status"].ToString() == "budgetLimited" && limited.TurnsStarted == 1,
                "real budget ended as complete or continued another turn");
            results["budget"] = new JsonObject { ["goal"] = limited.GoalState, ["usage"] = limited.TokenUsage, ["turns"] = limited.TurnsStarted };
        }
        File.WriteAllText(Path.Combine(root, "result.json"), results.ToJsonString());
        Console.WriteLine("GOAL_CONTROL_SMOKE_COMPLETE actual_codex=true mock_world=true checks=2 report=" + Path.Combine(root, "result.json"));
    }

            .Where(t => fullTools || names.Contains(t["function"]["name"].ToString())).Select(t => t.DeepClone()).ToArray());
        string system = File.ReadAllText(Path.Combine(root, "Agent/System.txt"))
            + "\n本次为隔离的脚本组合验收，仅提供所列工具；无需建立任务计划。资源来源已查证：LifeCrystal物品29，方块Heart。";

        Directory.CreateDirectory(output);
        string model = fullTools ? string.IsNullOrEmpty(selectedModel) ? "gpt-6-luna" : selectedModel : "";
        var entryTools = fullTools ? AgentScript.EntryTools(tools) : tools;
        if (fullTools) system = AgentScript.Instructions(system, tools);
        Console.WriteLine($"SCRIPT_MODEL artifacts={output} mock_world=true full_tools={fullTools} model={model} effort=low");
        foreach (bool perItem in new[] { false, true })
        {
            using var agent = new CodexAgent();
            using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(120));
            int stock = 0, scripts = 0, actionsInScript = 0, depth = 0, turns = 0;
            var announcements = new List<int>();
            var events = new JsonArray();
            void Validate(string name, JsonObject args)
            {
                var definition = tools.FirstOrDefault(t => t["function"]["name"].ToString() == name)?["function"];
                if (definition == null) throw new ArgumentException("工具不可用: " + name);
                foreach (var required in definition["parameters"]?["required"]?.AsArray() ?? new JsonArray())
                    if (!args.ContainsKey(required.ToString())) throw new ArgumentException("缺少参数: " + required);
            }
            async Task<string> Execute(string name, JsonObject args, CancellationToken token)
            {
                Validate(name, args);
                if (++turns > 100) throw new InvalidOperationException("验收调用预算已耗尽");
                var entry = new JsonObject { ["tool"] = name, ["args"] = args.DeepClone(), ["inside_script"] = depth > 0 };
                events.Add(entry);
                string result;
                if (name == "run_script")
                {
                    scripts++; depth++;
                    try { result = (await AgentScript.Run(args["code"].ToString(), 30, Validate, Execute,
                        _ => Task.CompletedTask, token)).ToJsonString(); }
                    finally { depth--; }
                }
                else if (name == "game_api") result = AgentScript.DescribeApi(tools, args).ToJsonString();
                else if (name == "get_state") result = new JsonObject
                {
                    ["inventory_slots"] = new JsonArray(new JsonObject { ["slot"] = 0, ["type"] = 29, ["prefix"] = 0,
                        ["stack"] = stock, ["name"] = "Life Crystal" }), ["permanent_max_health"] = 100
                }.ToJsonString();
                else if (name == "gather")
                {
                    var target = args["targets"].AsArray().Single();
                    Check(target["item_type"].GetValue<int>() == 29, "model gathered the wrong item");
                    int wanted = target["count"].GetValue<int>();
                    Check(wanted >= stock && wanted <= 10, "model exceeded the requested collection");
                    if (depth > 0) actionsInScript++;
                    stock = wanted;
                    result = new JsonObject { ["ok"] = true, ["outcome"] = "enough", ["targets"] = new JsonArray(
                        new JsonObject { ["item_type"] = 29, ["wanted"] = wanted, ["actual"] = stock, ["met"] = true }) }.ToJsonString();
                }
                else if (name == "say")
                {
                    if (args["text"].ToString().Trim() == "叮") announcements.Add(stock);
                    result = "{\"ok\":true}";
                }
                else if (name == "track") result = "{\"outcome\":\"tracking\"}";
                else if (name == "game_catalog") result = "{\"matches\":[{\"type\":29,\"internal\":\"LifeCrystal\",\"exact\":true}]}";
                else result = "{\"error\":\"not_simulated\",\"detail\":\"此模拟世界只执行库存查询、gather、say及脚本；其他工具不产生游戏效果。\"}";
                entry["result"] = result;
                return result;
            }
            string goal = perItem ? "收集10颗生命水晶，每收集一颗立刻通过say只说一个字：叮。然后才收下一颗。不要使用水晶。"
                : "收集10颗生命水晶放在背包里，不要使用。";
            try
            {
                await agent.RunAsync(goal, "", root, model, system, entryTools, Execute,
                    reply => events.Add(new JsonObject { ["reply"] = reply }), timeout.Token);
                Check(stock == 10, "model did not reach the requested inventory");
                Check(!perItem || scripts > 0 && actionsInScript > 0, "model did not compose per-item behavior through a script");
                Check(!perItem || announcements.SequenceEqual(Enumerable.Range(1, 10)), "model lost per-item collection/behavior order");
                Console.WriteLine($"PASS: model execution per_item={perItem} stock={stock} scripts={scripts} mock_world=true");
            }
            finally { File.WriteAllText(Path.Combine(output, perItem ? "per-item.json" : "collect.json"), events.ToJsonString()); }
        }
        Console.WriteLine("SCRIPT_MODEL_COMPLETE checks=2 mock_world=true");
    }

    public static async Task Smoke()
    {
        var launch = new System.Diagnostics.ProcessStartInfo();
        CodexAgent.ApplySystemProxy(launch, System.Net.Http.HttpClient.DefaultProxy);
        Console.WriteLine("CODEX_SMOKE https_proxy_configured=" + launch.Environment.ContainsKey("HTTPS_PROXY"));
        using var agent = new CodexAgent();
        using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(120));
        int calls = 0;
        var replies = new List<string>();
        Task<string> Execute(string name, JsonObject args, CancellationToken token)
        {
            Check(name == "get_state", "smoke attempted unexpected tool: " + name);
            calls++; Console.WriteLine("CODEX_SMOKE tool=get_state mock_world=true");
            return Task.FromResult("{\"equipment\":{\"items\":[{\"name\":\"Wood\",\"stack\":23}]}}");
        }
        string cwd = Path.Combine(Path.GetTempPath(), "TerraBlindCodexSmoke");
        const string system = "你是 Terraria 队友。只使用提供的游戏工具，只简短回答，不使用其他能力。";
        await agent.RunAsync("必须调用 get_state 查看背包，然后只回答木材数量。", "", cwd, "", system, Tools, Execute,
            text => { replies.Add(text); Console.WriteLine("CODEX_SMOKE reply=" + text); }, timeout.Token);
        Check(calls == 1 && replies.Last().Contains("23"), "real Codex tool roundtrip did not return mock inventory");
        await agent.RunAsync("刚才木材有多少？根据已有对话回答，不再调用工具。", "", cwd, "", system, Tools, Execute,
            text => { replies.Add(text); Console.WriteLine("CODEX_SMOKE reply=" + text); }, timeout.Token);
        Check(calls == 1 && replies.Last().Contains("23"), "real Codex did not retain conversation context");
        using var interrupt = CancellationTokenSource.CreateLinkedTokenSource(timeout.Token);
        var entered = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
        var running = agent.RunAsync("再次调用 get_state 读取背包。", "", cwd, "", system, Tools,
            async (name, args, token) =>
            {
                Check(name == "get_state", "unexpected tool in interrupt probe");
                entered.TrySetResult(true); await Task.Delay(Timeout.Infinite, token); return "unreachable";
            }, replies.Add, interrupt.Token);
        await entered.Task.WaitAsync(timeout.Token); interrupt.Cancel();
        try { await running; throw new Exception("real Codex cancellation ignored"); } catch (OperationCanceledException) { }
        Console.WriteLine("CODEX_SMOKE interrupted=true");
        await agent.RunAsync("刚才任务取消了。根据已有对话，只回答之前读到的木材数量，不再调用工具。", "", cwd, "", system, Tools, Execute,
            text => { replies.Add(text); Console.WriteLine("CODEX_SMOKE reply=" + text); }, timeout.Token);
        Check(calls == 1 && replies.Last().Contains("23"), "real Codex lost context after interruption");
        Console.WriteLine("CODEX_SMOKE_COMPLETE checks=4 mock_world=true");
    }

    sealed class Server
    {
        readonly Lines _input = new();
        public int Threads, Interrupts;
        int _turn;
        public bool Stopped, LoggedIn = true, FailTurn;
        public string ToolName = "get_state";
        public Action<int> AfterTool;
        public JsonObject ThreadArgs;
        public JsonObject Goal;
        public int GoalSets;
        public int RepeatedToolCalls = 1;
        public readonly List<JsonObject> ToolResponses = new();
        public readonly List<string> Inputs = new();
        public readonly List<JsonObject> Steers = new();
        public void Drop() => _input.End();
        public CodexConnection Connect() => new(_input, new Writer(Receive), () => { Stopped = true; _input.End(); });
        void Send(JsonObject message) => _input.Add(message.ToJsonString());
        void Reply(JsonNode id, JsonObject result) => Send(new JsonObject { ["id"] = id.DeepClone(), ["result"] = result });
        void Completed(string status) => Send(new JsonObject { ["method"] = "turn/completed", ["params"] = new JsonObject
        { ["threadId"] = "thread", ["turn"] = new JsonObject { ["id"] = "turn" + _turn, ["status"] = status } } });
        void Receive(string line)
        {
            var message = JsonNode.Parse(line).AsObject();
            string method = message["method"]?.ToString();
            if (method == "initialized") return;
            if (method == "initialize") Reply(message["id"], new JsonObject());
            else if (method == "account/read") Reply(message["id"], new JsonObject { ["account"] = LoggedIn ? new JsonObject { ["type"] = "chatgpt" } : null });
            else if (method == "config/read") Reply(message["id"], JsonNode.Parse("{\"config\":{\"mcp_servers\":{\"unrelated\":{}}}}").AsObject());
            else if (method == "thread/start")
            { Threads++; ThreadArgs = message["params"].AsObject(); Reply(message["id"], JsonNode.Parse("{\"thread\":{\"id\":\"thread\"}}").AsObject()); }
            else if (method == "thread/goal/set")
            {
                GoalSets++; Goal ??= new JsonObject();
                foreach (var pair in message["params"].AsObject())
                    if (pair.Key is "objective" or "status" or "tokenBudget") Goal[pair.Key] = pair.Value?.DeepClone();
                Reply(message["id"], new JsonObject { ["goal"] = Goal.DeepClone() });
            }
            else if (method == "thread/goal/get") Reply(message["id"], new JsonObject { ["goal"] = Goal?.DeepClone() });
            else if (method == "turn/start")
            {
                Inputs.Add(message["params"]["input"][0]["text"].ToString());
                _turn++;
                Reply(message["id"], new JsonObject { ["turn"] = new JsonObject { ["id"] = "turn" + _turn } });
                if (FailTurn) { Completed("failed"); return; }
                Send(new JsonObject { ["method"] = "item/tool/call", ["id"] = "tool" + _turn, ["params"] = new JsonObject
                { ["threadId"] = "thread", ["turnId"] = "turn" + _turn, ["callId"] = "call" + _turn, ["tool"] = ToolName, ["arguments"] = new JsonObject() } });
            }
            else if (method == "turn/interrupt") { Interrupts++; Reply(message["id"], new JsonObject()); Completed("interrupted"); }
            else if (method == "turn/steer")
            { Steers.Add(message["params"].AsObject()); Reply(message["id"], new JsonObject { ["turnId"] = "turn" + _turn }); }
            else if (method == null)
            {
                ToolResponses.Add(message["result"].AsObject());
                AfterTool?.Invoke(ToolResponses.Count);
                var usage = new JsonObject { ["method"] = "thread/tokenUsage/updated", ["params"] = new JsonObject
                    { ["threadId"] = "thread", ["turnId"] = "turn" + _turn, ["tokenUsage"] = new JsonObject
                        { ["total"] = new JsonObject { ["totalTokens"] = _turn * 100 } } } };
                Send(usage); Send(usage);
                if (ToolResponses.Count < RepeatedToolCalls)
                {
                    Send(new JsonObject { ["method"] = "item/tool/call", ["id"] = "repeat" + ToolResponses.Count,
                        ["params"] = new JsonObject { ["threadId"] = "thread", ["turnId"] = "turn" + _turn,
                            ["tool"] = ToolName, ["arguments"] = new JsonObject() } });
                    return;
                }
                var completed = new JsonObject { ["method"] = "item/completed", ["params"] = new JsonObject
                { ["threadId"] = "thread", ["turnId"] = "turn" + _turn, ["item"] = new JsonObject
                { ["id"] = "reply" + _turn, ["type"] = "agentMessage", ["text"] = "完成了" } } };
                Send(completed); Send(completed); Completed("completed");
            }
        }
    }
    sealed class Lines : TextReader
    {
        readonly Channel<string> _lines = Channel.CreateUnbounded<string>();
        public void Add(string line) => _lines.Writer.TryWrite(line);
        public void End() => _lines.Writer.TryComplete();
        public override ValueTask<string> ReadLineAsync(CancellationToken token) => _lines.Reader.ReadAsync(token);
    }
    sealed class Writer : TextWriter
    {
        readonly Action<string> _send;
        public Writer(Action<string> send) => _send = send;
        public override Encoding Encoding => Encoding.UTF8;
        public override Task WriteLineAsync(ReadOnlyMemory<char> value, CancellationToken token = default)
        { token.ThrowIfCancellationRequested(); _send(value.ToString()); return Task.CompletedTask; }
    }
