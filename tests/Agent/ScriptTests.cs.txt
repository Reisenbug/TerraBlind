    sealed class DecisionHandler : System.Net.Http.HttpMessageHandler
    {
        public string Reply = "{\"answers\":{\"useful\":{\"type\":\"noul\",\"noul\":0.9},\"route\":{\"type\":\"choice\",\"choice\":\"retreat\"}}}";
        public bool Wait;
        public System.Net.HttpStatusCode Status = System.Net.HttpStatusCode.OK;
        public JsonObject Sent;
        protected override async Task<System.Net.Http.HttpResponseMessage> SendAsync(System.Net.Http.HttpRequestMessage request, CancellationToken token)
        {
            Sent = JsonNode.Parse(await request.Content.ReadAsStringAsync(token)).AsObject();
            if (Wait) await Task.Delay(Timeout.Infinite, token);
            return new System.Net.Http.HttpResponseMessage(Status) { Content = new System.Net.Http.StringContent(Reply) };
        }
    }

    static async Task TestDecisionChannels()
    {
        var channels = new ScriptDecisionChannels();
        var pending = new System.Collections.Concurrent.ConcurrentQueue<(JsonObject Input, CancellationToken Token, TaskCompletionSource<JsonObject> Reply)>();
        Task<JsonObject> Judge(JsonObject input, CancellationToken token)
        {
            var reply = new TaskCompletionSource<JsonObject>(TaskCreationOptions.RunContinuationsAsynchronously);
            pending.Enqueue((input, token, reply));
            return reply.Task.WaitAsync(token);
        }
        async Task WaitFor(int count)
        {
            for (int i = 0; i < 2000 && pending.Count < count; i++) await Task.Delay(1);
            Check(pending.Count == count, "background decision requests did not start independently");
        }
        JsonObject Request(string name)
        {
            var request = JsonNode.Parse("""
            {"channel":"weapon","context":"encounter-1","interval_ms":100,"max_age_ms":100,
             "state":{"x":10},"questions":{"choice":{"type":"choice","instructions":"Choose a concrete action",
             "criteria":{"continue":"Continue along the route","retreat":"Retreat behind cover"}}}}
            """).AsObject();
            request["channel"] = name;
            return request;
        }
        var first = Request("weapon");
        var started = channels.Poll(first, Judge);
        Check(started["status"].ToString() == "pending" && started["in_flight"].GetValue<bool>(),
            "background decision waited for a response or invented an answer");
        channels.Poll(first, Judge);
        channels.Poll(Request("movement"), Judge);
        channels.Poll(Request("opportunity"), Judge);
        await WaitFor(3);
        var calls = pending.ToArray();
        foreach (var call in calls) call.Reply.SetResult(JsonNode.Parse("{\"answers\":{\"choice\":{\"choice\":\"retreat\"}},\"elapsed_ms\":20}").AsObject());
        for (int i = 0; i < 2000 && channels.CompletionVersion < 3; i++) await Task.Delay(1);
        Check(channels.CompletionVersion == 3, "completed decision channels did not notify script scheduling");
        var reactions = new AgentScript.Reactions();
        Check(reactions.TickDue() && !reactions.TickDue(), "ordinary reaction tick lost its throttle");
        Check(reactions.TickDue(channels.CompletionVersion) && !reactions.TickDue(channels.CompletionVersion),
            "a decision answer did not wake the next reaction exactly once");
        var answered = channels.Poll(first, Judge);
        Check(answered["status"].ToString() == "ready" && answered["input_state"]["x"].GetValue<int>() == 10
            && answered["ready_wait_ms"].GetValue<double>() >= 0
            && answered["result"]["answers"]["choice"]["choice"].ToString() == "retreat",
            "background decision lost its sampled scene or typed answer");
        answered["input_state"]["x"] = -100;
        Check(channels.Poll(first, Judge)["input_state"]["x"].GetValue<int>() == 10,
            "script mutated retained decision evidence");
        await Task.Delay(130);
        var expired = channels.Poll(first, Judge);
        Check(expired["status"].ToString() == "pending" && expired["result"] == null,
            "starting a refresh made an expired decision look fresh");
        await WaitFor(4);
        var refresh = pending.ToArray()[3];
        first["context"] = "encounter-2"; first["state"]["x"] = 30;
        var changed = channels.Poll(first, Judge);
        Check(changed["status"].ToString() == "pending" && refresh.Token.IsCancellationRequested,
            "changing the tactical context reused the previous answer or left its request running");
        await WaitFor(5);
        var next = pending.ToArray()[4];
        next.Reply.SetResult(new JsonObject { ["error"] = "jev_http_error", ["status"] = 503 });
        await Task.Delay(10);
        var failed = channels.Poll(first, Judge);
        Check(failed["status"].ToString() == "failed" && failed["result"]["error"].ToString() == "jev_http_error",
            "network failure became a no/continue answer");
        long completedVersion = channels.CompletionVersion;
        channels.Reset();
        Check(refresh.Token.IsCancellationRequested, "stopping decisions retained a previous request");
        await Task.Delay(10);
        Check(channels.CompletionVersion == completedVersion, "retired requests woke a replacement task");
        var retired = new ScriptDecisionChannels();
        var entered = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
        var late = new TaskCompletionSource<JsonObject>(TaskCreationOptions.RunContinuationsAsynchronously);
        retired.Poll(Request("late"), (input, token) => { entered.SetResult(true); return late.Task; });
        await entered.Task;
        retired.Reset();
        late.SetResult(new JsonObject());
        await Task.Delay(10);
        Check(retired.CompletionVersion == 0, "an uncancellable retired response woke a new task");
        using var handler = new DecisionHandler();
        using var http = new System.Net.Http.HttpClient(handler);
        var criteria = new JsonObject { ["retreat"] = "Retreat with the selected weapon" };
        for (int i = 0; i < 49; i++) criteria["movement_weapon_" + i] = "A concrete movement and weapon combination";
        var wide = await ScriptDecision.Judge(http, "test-key", new JsonObject { ["state"] = new JsonObject(),
            ["questions"] = new JsonObject { ["route"] = new JsonObject { ["type"] = "choice",
                ["instructions"] = "Which concrete combination fits this situation?", ["criteria"] = criteria } } }, default);
        Check(wide["answers"]["route"]["choice"].ToString() == "retreat"
            && handler.Sent["questions"]["route"]["criteria"].AsObject().Count == 50,
            "joint movement/weapon choices were truncated before transport");
        Console.WriteLine("PASS: three decision dimensions run without waiting or duplicate requests; context changes, expiry and model errors preserve evidence");
    }

    static async Task TestJointDecision()
    {
        Console.WriteLine("DECISION_TEMPLATE_CHARS " + string.Join(' ', new[] { "decision_reactions", "joint_reactions", "joint_combat" }
            .Select(name => name + "=" + ScriptLibrary.Example(name)["code"].ToString().Length)));
        foreach (string scenario in new[] { "pending", "follow", "move", "invalid", "unsafe", "replayed", "projectile_only" })
        {
            int acted = 0, selected = 0, judged = 0;
            var input = new JsonObject { ["context"] = "journey", ["destination"] = new JsonObject { ["x"] = 50, ["y"] = 20 },
                ["last_request"] = scenario == "replayed" ? 1 : 0 };
            var combat = JsonNode.Parse("""
                {"player":{"hp":100},"targets":[{"who":1,"contact_damage":20,"weapons":[{"type":279,"has_ammo":true,"mana_available":true}]}],
                 "projectiles":[],"forecast":{"status":"ready","id":11,"sample_tick":10,"age_frames":12,"known_route_frames":40,
                 "sample_player":{},"scope":"test engine boundary","movements":[{"id":"follow_route"},{"id":"left"}],
                 "plans":[{"id":"follow_route_weapon_279","movement":"follow_route","weapon_type":279,"movement_available":true},
                          {"id":"left_weapon_279","movement":"left","weapon_type":279,"movement_available":true}]}}
                """).AsObject();
            if (scenario == "pending") combat["forecast"]["status"] = "pending";
            if (scenario == "unsafe") combat["forecast"]["plans"][1]["movement_available"] = false;
            if (scenario == "projectile_only")
            {
                combat["targets"] = new JsonArray();
                combat["projectiles"] = new JsonArray(new JsonObject { ["damage"] = 20 });
            }
            var report = await AgentScript.Run("const input=" + input.ToJsonString() + ";\n" + ScriptLibrary.Example("joint_combat")["code"],
                5, (_, _) => { }, (name, args, _) =>
                {
                    JsonObject result;
                    if (name == "combat_assess") result = combat;
                    else if (name == "jev_judge")
                    {
                        judged++;
                        Check(args["questions"]["response"]["criteria"].AsObject().Count == 2
                            && args["state"]["forecast_id"].GetValue<int>() == 11
                            && (scenario == "projectile_only" ? args["state"]["weapons"].AsArray().Count == 0
                                : args["state"]["weapons"][0]["type"].GetValue<int>() == 279),
                            "joint judgment lost coherent movement/weapon alternatives or source forecast");
                        result = new JsonObject { ["status"] = "ready", ["request_id"] = 1, ["request_age_ms"] = 200,
                            ["input_state"] = args["state"].DeepClone(), ["result"] = new JsonObject { ["elapsed_ms"] = 150,
                                ["answers"] = new JsonObject { ["response"] = new JsonObject {
                                    ["choice"] = scenario == "follow" ? "follow_route_weapon_279" : "left_weapon_279" } } } };
                    }
                    else if (name == "set_combat") { selected++; result = new JsonObject(); }
                    else
                    {
                        Check(name == "act" && args["combat_candidate"]["forecast_id"].GetValue<int>() == 11
                            && args["combat_candidate"]["plan_id"].ToString() == "left_weapon_279",
                            "joint action lost its host-validated candidate reference");
                        acted++;
                        result = scenario == "invalid" ? new JsonObject { ["error"] = "combat_candidate_changed" }
                            : new JsonObject { ["outcome"] = "done" };
                    }
                    return Task.FromResult(result.ToJsonString());
                }, _ => Task.CompletedTask, default);
            Check(report["ok"].GetValue<bool>() && judged == (scenario == "pending" ? 0 : 1)
                && acted == (scenario is "move" or "invalid" or "projectile_only" ? 1 : 0) && selected == (scenario == "follow" ? 1 : 0),
                "joint policy waited, replayed effects or ignored an unavailable candidate: " + scenario + " " + report);
            if (scenario == "invalid") Check(report["value"]["applied"].GetValue<bool>() == false,
                "host rejection was reported as an applied movement");
        }
        Console.WriteLine("PASS: joint combat skill shares movement/weapon candidates, continues while pending, deduplicates and preserves host rejection");
    }

    static async Task TestTacticalReactions()
    {
        foreach (string scenario in new[] { "pending", "continue", "enemy", "stale_enemy", "blocked", "clear", "trap", "failed_clear", "baseline", "pressure", "pressure_denied" })
        {
            int acts = 0, clears = 0, polls = 0;
            JsonObject decision = null;
            var combat = JsonNode.Parse("""
                {"player":{"hp":100,"pos":{"x":160,"y":160},"velocity":{"x":0,"y":0}},"targets":[
                  {"who":1,"attackable":true,"contact_damage":20,"distance_tiles":4,"weapons":[
                    {"type":279,"has_ammo":true,"mana_available":true,"line_or_reach_passed":true}]}]}
                """).AsObject();
            var terrain = JsonNode.Parse("""
                {"player_bounds":{"left":10,"right":11,"bottom":12},"trap_tiles":[],
                 "near_body_tiles":[{"x":10,"y":13,"type":1,"solid":true}]}
                """).AsObject();
            if (scenario == "blocked") combat["targets"][0]["weapons"][0]["line_or_reach_passed"] = false;
            if (scenario is "clear" or "trap" or "failed_clear")
            {
                combat["targets"] = new JsonArray();
                terrain["near_body_tiles"].AsArray().Add(JsonNode.Parse("{\"x\":11,\"y\":12,\"type\":69,\"touch_damage\":10,\"solid\":false,\"falling\":false}"));
                if (scenario == "trap") terrain["trap_tiles"].AsArray().Add(new JsonObject { ["x"] = 11, ["y"] = 12 });
            }
            bool pressure = scenario is "pressure" or "pressure_denied";
            if (pressure)
                terrain["trap_tiles"].AsArray().Add(JsonNode.Parse("{\"x\":11,\"y\":12,\"type\":135,\"frame_x\":0,\"frame_y\":54,\"kind\":\"pressure_plate\"}"));
            var template = ScriptLibrary.Example("tactical_reactions");
            Check(template["code"].ToString().Length <= 16000, "tactical example exceeded saved skill limit");
            string code = "skill('tactical_reactions',0,{instruction:'reach jungle',weapon_order:[279],allow_trigger_removal:"
                + (scenario == "pressure" ? "true" : "false") + ",baseline:"
                + (scenario == "baseline" ? "true" : "false") + "});return tool('nav_to',{x:50,y:20});";
            var result = await AgentScript.Run(code, 5, (_, _) => { }, async (name, args, token) =>
            {
                if (name == "combat_assess") return combat.ToJsonString();
                if (name == "inspect_terrain") return terrain.ToJsonString();
                if (name == "inventory_review") return "{\"inventory\":[{\"slot\":2,\"pick\":35}]}";
                if (name == "opportunity_assess") return "{\"ok\":true,\"target\":{\"has_tile\":false}}";
                if (name == "set_combat") { Check(args["weapon_type"].GetValue<int>() == 279, "tactics lost usable weapon"); return "{}"; }
                if (name == "act")
                {
                    acts++;
                    Check(args["steps"][0]["keys"].AsArray().Count == 0 && args["steps"][0]["until"]["frames"].GetValue<int>() == 16,
                        "tactical threat window stole movement or became unbounded");
                    return "{\"outcome\":\"done\"}";
                }
                if (name == "use_item")
                {
                    clears++;
                    Check(args["slot"].GetValue<int>() == 2 && args["x"].GetValue<int>() == 11 && args["strict"].GetValue<bool>(),
                        "tactics cleared an unobserved coordinate");
                    if (!pressure) Check(args["duration_ticks"].GetValue<int>() == 16, "contact clearance lost its execution bound");
                    if (pressure) terrain["trap_tiles"] = new JsonArray();
                    return scenario == "failed_clear" ? "{\"error\":\"cannot_reach\"}" : "{\"outcome\":\"removed\"}";
                }
                if (name == "jev_judge")
                {
                    polls++;
                    Check(args["state"]["instruction"].ToString() == "reach jungle"
                        && args["questions"]["response"]["criteria"].AsObject().Count == 3, "tactics lost intent or options");
                    if (scenario == "pending") return "{\"status\":\"pending\"}";
                    var answer = new JsonObject { ["status"] = "ready", ["request_id"] = 1, ["input_state"] = args["state"].DeepClone(),
                        ["result"] = new JsonObject { ["answers"] = new JsonObject { ["response"] = new JsonObject {
                            ["choice"] = scenario is "clear" or "failed_clear" ? "contact_terrain" : scenario == "continue" ? "none" : "nearby_enemy" } } } };
                    if (scenario == "stale_enemy") combat["targets"][0]["who"] = 2;
                    return answer.ToJsonString();
                }
                Check(name == "nav_to", "unexpected tactical action: " + name);
                var reactions = AgentScript.CurrentReactions.Value;
                var notice = new JsonObject { ["kind"] = pressure || scenario is "clear" or "failed_clear" ? "action_failed" : "tick",
                    ["tool"] = "nav_to", ["args"] = args.DeepClone(), ["result"] = new JsonObject {
                        ["error"] = pressure ? "navigation_near_trap" : "navigation_contact_escape", ["trap_tiles"] = terrain["trap_tiles"].DeepClone() },
                    ["execution"] = new JsonObject { ["tool"] = "nav_to", ["args"] = args.DeepClone() } };
                decision = await reactions.Dispatch(notice, token);
                if (!pressure && scenario is not ("clear" or "failed_clear" or "baseline")) await reactions.Dispatch(notice, token);
                return "{\"done\":true}";
            }, _ => Task.CompletedTask, default, programs: new JsonArray(template));
            Check(result["ok"].GetValue<bool>() && acts == (scenario is "enemy" or "baseline" ? 3 : 0)
                && clears == (scenario is "clear" or "failed_clear" or "pressure" ? 1 : 0), "tactical action replayed or lost revalidation: " + scenario + " " + result);
            Check(polls == (pressure || scenario is "baseline" or "blocked" or "trap" ? 0 : 2 - (scenario is "clear" or "failed_clear" ? 1 : 0)), "unexpected tactical model polling: " + scenario);
            if (scenario is "clear" or "failed_clear" or "pressure") Check(decision["retry"].GetValue<bool>() == (scenario != "failed_clear"),
                "tactical retry lacked a verified terrain change");
        }
        await ReactiveHostIntegration(emptyWindow: true);
        Console.WriteLine("PASS: tactical skill continues while pending, bounds combat windows, revalidates targets and retries only after tile removal");
    }

    static async Task BoundedCollectUse()
    {
        var agent = new NativeAgent(); agent.Load();
        var handler = HttpServerSystem.Handler;
        var player = Terraria.Main.LocalPlayer;
        var item = player.inventory[49];
        var tick = Terraria.Main.GameUpdateCount;
        int polls = 0, stops = 0, starts = 0;
        try
        {
            AgentTools.NewTask();
            player.inventory[49] = new Terraria.Item { type = 3509, pick = 35, stack = 1 };
            HttpServerSystem.Handler = (path, payload) =>
            {
                Check(!path.StartsWith("/nav"), "bounded item use secretly started navigation");
                if (path == "/item_use") starts++;
                if (path == "/item_use_stop") stops++;
                if (path == "/item_use_status")
                { polls++; Terraria.Main.GameUpdateCount += 4; return (200, "{\"active\":true}"); }
                return (200, "{}");
            };
            var task = AgentTools.Execute("use_item", new JsonObject { ["slot"] = 49, ["x"] = 1000, ["y"] = 1000,
                ["strict"] = true, ["duration_ticks"] = 12 }, default);
            var timer = System.Diagnostics.Stopwatch.StartNew();
            while (!task.IsCompleted)
            {
                Check(timer.Elapsed.TotalSeconds < 5, "bounded item use failed to stop");
                agent.PostUpdateEverything(); await Task.Delay(1);
            }
            var result = JsonNode.Parse(await task);
            Check(result["outcome"].ToString() == "time_elapsed" && polls == 3 && starts == 1 && stops == 1,
                "bounded pick use ignored its frame limit: " + result);
            Console.WriteLine("PASS: explicit-duration pick use stays local and stops at the frame budget");
        }
        finally
        {
            player.inventory[49] = item; Terraria.Main.GameUpdateCount = tick;
            AgentTools.Stop(); HttpServerSystem.Handler = handler; agent.Unload();
        }
    }

    static async Task ConcurrentPolicyContext()
    {
        var agent = new NativeAgent(); agent.Load();
        async Task Pump(Task task)
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();
            while (!task.IsCompleted)
            {
                Check(timer.Elapsed.TotalSeconds < 5, "concurrent policy context deadlocked");
                agent.PostUpdateEverything(); await Task.Delay(1);
            }
            await task;
        }
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset();
            await Pump(AgentTools.Execute("task_policy", new JsonObject { ["operation"] = "set", ["expected_revision"] = 0,
                ["code"] = "on('tick',e=>{tool('say',{text:'concurrent:'+e.execution.tool});return {handled:true};});" }, default));
            Check(AgentTools.ControlSession.Begin("nav_to", null, out _) == null,
                "test could not acquire navigation");
            await Pump(AgentTools.ReactAtBoundary(default, context: new JsonObject { ["tool"] = "fight" }));
            Check(!AgentChat.Messages.Contains("concurrent:fight"), "concurrent fight consumed the primary navigation's policy tick");
            await Pump(AgentTools.ReactAtBoundary(default, context: new JsonObject { ["tool"] = "nav_to",
                ["args"] = new JsonObject { ["x"] = 20, ["y"] = 10 } }));
            Check(AgentChat.Messages.Contains("concurrent:nav_to"), "primary navigation lost its policy observation");
            Console.WriteLine("PASS: concurrent defense leaves task-policy ticks and route handoff with the primary navigation");
        }
        finally
        {
            AgentTools.ControlSession.Reset(); AgentTools.ClearTaskPolicy(true); AgentTools.Stop(); AgentTools.Plan.Reset(); AgentTools.Watches.Reset(); agent.Unload();
        }
    }

    static async Task TestDecisionTravel()
    {
        foreach (string[] dimensions in new[] { Array.Empty<string>(), new[] { "weapon" }, new[] { "movement" },
            new[] { "opportunity" }, new[] { "weapon", "movement", "opportunity" } })
        {
            var enabled = dimensions.ToHashSet();
            int movements = 0, collected = 0, assessments = 0;
            bool opportunityPrefetched = false;
            var selectedWeapons = new List<int>();
            var polls = new Dictionary<string, int>();
            var sceneContexts = new Dictionary<string, string>();
            string firstOpportunityContext = null;
            var config = JsonNode.Parse("""
                {"destination":{"x":50,"y":20},"resources":[{"item_type":20,"count":1,"tile_type":61,"tile_name":"JunglePlants"}],
                 "weapon_order":[11,12],"max_detour_tiles":10,"health_reserve":30}
                """).AsObject();
            config["decisions"] = new JsonArray(dimensions.Select(d => (JsonNode)JsonValue.Create(d)).ToArray());
            var combat = JsonNode.Parse("""
                {"tick":10,"player":{"hp":100,"pos":{"x":160,"y":320},"size_pixels":{"width":20,"height":42},
                  "velocity":{"x":3.2,"y":-4.8},"controls":{"right":true}},"targets":[
                  {"who":0,"type":42,"attackable":true,"contact_damage":20,"distance_tiles":5,"weapons":[
                    {"type":11,"name":"short weapon","has_ammo":true,"mana_available":true,"line_or_reach_passed":true},
                    {"type":12,"name":"ranged weapon","has_ammo":true,"mana_available":true,"line_or_reach_passed":true}]}]}
                """).AsObject();
            var program = ScriptLibrary.Example("decision_travel");
            Check(program["code"].ToString().Length <= 16000, "decision skill exceeded the stored program limit");
            var result = await AgentScript.Run("return skill('decision_travel',0," + config.ToJsonString() + ");",
                5, (_, _) => { }, async (name, args, token) =>
                {
                    if (name == "get_state") return new JsonObject
                    {
                        ["agent"] = new JsonObject { ["auto_attack"] = true, ["combat_weapon_type"] = 0 },
                        ["inventory_counts"] = new JsonObject { ["20"] = collected }
                    }.ToJsonString();
                    if (name == "watch") return "{}";
                    if (name == "combat_assess") return combat.ToJsonString();
                    if (name == "inspect_terrain") return "{\"near_body_tiles\":[],\"player_bounds\":{\"left\":10,\"right\":11}}";
                    if (name == "opportunity_assess") return new JsonObject { ["ok"] = true,
                        ["target"] = new JsonObject { ["has_tile"] = true, ["tile_type"] = 61 },
                        ["approach_direct_distance"] = enabled.Contains("opportunity") && ++assessments <= 2 ? 20 : 2,
                        ["route_relation"] = new JsonObject { ["extra_direct_distance"] = 0 }, ["hazards"] = new JsonArray() }.ToJsonString();
                    if (name == "inventory_review") return "{\"empty_slots\":[10],\"inventory\":[]}";
                    if (name == "set_combat") { selectedWeapons.Add(args["weapon_type"].GetValue<int>()); return "{}"; }
                    if (name == "act") { movements++; return "{\"outcome\":\"done\"}"; }
                    if (name == "gather")
                    {
                        Check(!enabled.Contains("opportunity") || assessments > 2, "speculative opportunity started gathering outside its budget");
                        collected++; return "{\"outcome\":\"enough\"}";
                    }
                    if (name == "jev_judge")
                    {
                        string channel = args["channel"].ToString();
                        if (channel is "movement" or "weapon")
                        {
                            string context = args["context"].ToString();
                            Check(!sceneContexts.TryGetValue(channel, out var previous) || previous == context,
                                "nearby enemy changes cancelled the same journey's decision channel");
                            sceneContexts[channel] = context;
                        }
                        if (channel == "opportunity" && collected == 0)
                        {
                            if (!polls.ContainsKey(channel)) opportunityPrefetched = args["state"]["assessment"]["approach_direct_distance"].GetValue<int>() == 20;
                            firstOpportunityContext ??= args["context"].ToString();
                            Check(args["context"].ToString() == firstOpportunityContext,
                                "a nearer observation cancelled the opportunity judgment before its answer");
                        }
                        var playerScene = args["state"]["combat"]["player"];
                        Check(playerScene["pos"] == null && playerScene["x"].GetValue<double>() == 10.625
                            && playerScene["y"].GetValue<double>() == 21.3125 && playerScene["vx"].GetValue<double>() == 12
                            && playerScene["vy"].GetValue<double>() == -18 && args["state"]["coordinates"] != null,
                            "decision scene mixed pixel and tile coordinates");
                        polls[channel] = polls.GetValueOrDefault(channel) + 1;
                        if (channel == "movement" && polls[channel] == 1)
                        {
                            var newcomer = combat["targets"][0].DeepClone(); newcomer["who"] = 2;
                            combat["targets"].AsArray().Add(newcomer);
                        }
                        if (polls[channel] == 1) return "{\"status\":\"pending\",\"in_flight\":true}";
                        string choice = channel == "weapon" ? "weapon_12" : channel == "movement" ? "jump_right" : "take";
                        return new JsonObject { ["status"] = "ready", ["request_id"] = 1, ["request_age_ms"] = 20,
                            ["input_state"] = args["state"].DeepClone(), ["result"] = new JsonObject { ["elapsed_ms"] = 20,
                                ["answers"] = new JsonObject { ["response"] = new JsonObject { ["choice"] = choice } } } }.ToJsonString();
                    }

    public static async Task Replay(string path)
    {
        var events = JsonNode.Parse(File.ReadAllText(path)).AsArray();
        string code = events.First(e => e?["tool"]?.ToString() == "run_script")["args"]["code"].ToString();
        foreach (bool overshoot in new[] { false, true })
        {
            int stock = 0, attempts = 0;
            bool interrupted = false;
            var said = new List<int>();
            Task<string> Execute(string name, JsonObject args, CancellationToken token)
            {
                if (name == "get_state") return Task.FromResult(new JsonObject { ["inventory_slots"] = new JsonArray(
                    new JsonObject { ["type"] = 29, ["stack"] = stock }) }.ToJsonString());
                if (name == "say") { said.Add(stock); return Task.FromResult("{\"ok\":true}"); }
                Check(name == "gather" && !interrupted, "generated code continued collecting after an unmet condition");
                attempts++;
                if (overshoot) { stock += 2; interrupted = true; }
                else if (attempts == 4) interrupted = true;
                else stock++;
                return Task.FromResult(new JsonObject { ["ok"] = overshoot || !interrupted,
                    ["outcome"] = !overshoot && interrupted ? "source_unavailable" : "enough",
                    ["targets"] = new JsonArray(new JsonObject { ["item_type"] = 29, ["actual"] = stock }) }.ToJsonString());
            }
            var result = await AgentScript.Run(code, 5, (_, _) => { }, Execute, _ => Task.CompletedTask, default);
            Check(result["ok"].GetValue<bool>() && interrupted, "generated code did not return evidence on changed conditions");
            Check(overshoot ? stock == 2 && attempts == 1 && said.Count == 0
                : stock == 3 && attempts == 4 && said.SequenceEqual(new[] { 1, 2, 3 }),
                "generated code falsely announced collection or lost per-item order");
            Console.WriteLine($"PASS: unchanged model code replay overshoot={overshoot} stock={stock} announcements={said.Count} model_calls=0");
        }
        Console.WriteLine("SCRIPT_REPLAY_COMPLETE checks=2 mock_world=true model_calls=0");
    }

                "inventory count projection disagrees with native bag slots");

            AgentTools.Stop(); AgentTools.ClearTaskPolicy(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset(); AgentTools.Watches.Reset();
            HttpServerSystem.Handler = originalHttp; player.inventory[49] = savedItem;
            player.dead = dead; player.suffocating = suffocating; agent.Unload();
        }
    }

    static async Task ReactiveHostIntegration(bool itemUse = false, string candidateMode = null, bool emptyWindow = false)
    {
        var agent = new NativeAgent(); agent.Load();
        var http = HttpServerSystem.Handler;
        var player = Terraria.Main.LocalPlayer;
        var position = player.position;
        int animation = player.itemAnimation, animationPumps = 0;
        bool dead = player.dead, suffocating = player.suffocating;
        int starts = 0, polls = 0, stops = 0;
        bool readingWhileMoving = false, actingAfterStop = false;
        var actedDuringAnimation = new List<bool>();
        string original = null;
        int refreshes = 0;
        bool rejected = candidateMode == "rejected";
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset();
            player.position = new System.Numerics.Vector2(160, 160); player.dead = player.suffocating = false;
            if (candidateMode != null)
            {
                CombatForecast.RefreshHandler = async (args, token) =>
                {
                    if (args["combat_candidate"]["forecast_id"].GetValue<int>() == 2) return args;
                    refreshes++;
                    Check(RecedingNav.Active && stops == 0 && AgentTools.ControlSession.Tool == "nav_to",
                        "candidate recomputation paused navigation before receiving its proof");
                    await Task.Delay(30, token);
                    Check(RecedingNav.Active && stops == 0 && AgentTools.ControlSession.Tool == "nav_to",
                        "candidate recomputation lost the original movement owner");
                    if (rejected) return new JsonObject { ["error"] = "fixture_candidate_rejected" };
                    var refreshed = args.DeepClone().AsObject();
                    refreshed["combat_candidate"]["forecast_id"] = 2;
                    return refreshed;
                };
                CombatForecast.PrepareHandler = args => args;
            }
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/json_roundtrip") return (200, payload);
                if (path == "/nav_recede") { starts++; RecedingNav.Active = true; }
                if (path == "/nav_recede_stop") { stops++; RecedingNav.Active = false; }
                if (path == "/act")
                {
                    actingAfterStop = !RecedingNav.Active && stops == 1;
                    if (emptyWindow) Check(!AgentTools.ControlSession.OwnsDirectMovement, "defense window blocked concurrent evasion");
                    actedDuringAnimation.Add(player.itemAnimation > 0);
                }
                if (path == "/nav_recede_done")
                {
                    polls++;
                    original ??= AgentTools.Actions.ActiveHandle;
                    if (polls <= 2) AgentTools.Watches.Emit("nearby_chest", "case:" + polls,
                        new JsonObject { ["x"] = polls, ["y"] = 10 });
                    else
                    {
                        player.position = new System.Numerics.Vector2(320, 160);
                        return (200, "{\"done\":true}");
                    }
                    if (polls == 2)
                    {
                        readingWhileMoving = RecedingNav.Active && stops == 0;
                        player.itemAnimation = 12;
                    }
                    return (200, "{\"done\":false}");
                }
                return (200, "{}");
            };
            var serialized = AgentTools.Call("/json_roundtrip", new JsonObject {
                ["frames"] = new JsonArray(12, 24), ["metadata"] = JsonValue.Create(new { FrameCount = 24 }) }, default);
            while (!serialized.IsCompleted) { agent.PostUpdateEverything(); await Task.Delay(1); }
            var echoed = await serialized;
            Check(echoed["frames"][1].GetValue<int>() == 24 && echoed["metadata"]["FrameCount"].GetValue<int>() == 24,
                "native JSON values lost their serializer metadata at the HTTP boundary");
            var task = AgentTools.Execute("run_script", new JsonObject { ["workflow"] = "reactive-host-" + Guid.NewGuid().ToString("N"),
                ["code"] = $$$"""
                    on('nearby_chest',event=>{
                      if(event.evidence.x===1) tool('inventory_review');
                      else {
                        const move={steps:[{keys:{{{(emptyWindow ? "[]" : "['right']")}}},until:{frames:1}}]};
                        if({{{(candidateMode != null ? "true" : "false")}}}) move.combat_candidate={context:'fixture',forecast_id:1,plan_id:'right_weapon_279'};
                        tool('act',move);
                        if({{{(itemUse ? "true" : "false")}}}) tool('act',{steps:[{use:true,until:{frames:1}}]});
                      }
                      return {handled:true};
                    });
                    tool('watch',{events:['nearby_chest']});
                    return step('travel',()=>tool('nav_to',{x:20,y:10}));
                    """ }, default);
            var timer = System.Diagnostics.Stopwatch.StartNew();
            while (!task.IsCompleted)
            {
                Check(timer.Elapsed.TotalSeconds < 10, "reactive host navigation deadlocked");
                if (stops == 1 && ++animationPumps >= 20) player.itemAnimation = 0;
                agent.PostUpdateEverything(); await Task.Delay(1);
            }
            var result = JsonNode.Parse(await task);
            Check(result["ok"].GetValue<bool>() && result["value"]["done"].GetValue<bool>()
                && starts == (rejected ? 1 : 2) && readingWhileMoving && actingAfterStop == !rejected,
                "production reaction dispatch failed query/movement handoff: " + result + " starts=" + starts + " stops=" + stops);
            Check(actedDuringAnimation.SequenceEqual(rejected ? Array.Empty<bool>() : itemUse ? new[] { true, false } : new[] { true }),
                "reaction movement waited for attack animation or item use interrupted it: itemUse=" + itemUse);
            Check(refreshes == (candidateMode == null ? 0 : 1), "candidate proof was repeated or skipped");
            var action = AgentTools.Actions.Summary().First(a => a["handle"]?.ToString() == original);
            Check(action["attempt"].GetValue<int>() == 1 && action["status"].ToString() == "completed" && !AgentTools.ControlSession.Active,
                "reaction replaced the original action or leaked its controls");

            var session = new AgentControlSession(new AgentControlPolicy());
            Check(session.Begin("nav_to", null, out long ticket) == null, "could not acquire test navigation");
            var restore = session.SuspendPrimary(); session.Reset(); restore();
            Check(!session.Active, "cancelled reaction restored an obsolete navigation claim");
            Console.WriteLine("PASS: production navigation keeps moving during observation, hands off effects and resumes the same action without leaked controls");
        }
        finally
        {
            AgentTools.Stop(); AgentTools.Actions.Reset(); AgentTools.Watches.Reset(); AgentTools.Plan.Reset();
            player.position = position; player.dead = dead; player.suffocating = suffocating;
            player.itemAnimation = animation;
            HttpServerSystem.Handler = http; agent.Unload();
            CombatForecast.RefreshHandler = null; CombatForecast.PrepareHandler = null;
        }
    }

    static async Task ReactiveFightIntegration()
    {
        var agent = new NativeAgent(); agent.Load();
        var http = HttpServerSystem.Handler;
        bool attack = AgentPermissions.Attack;
        int polls = 0, moves = 0, stops = 0;
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentPermissions.Attack = true;
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/fight" && JsonNode.Parse(payload)?["active"]?.GetValue<bool>() == false) stops++;
                if (path == "/fight_active")
                {
                    polls++;
                    if (polls == 2) AgentTools.Watches.Emit("nearby_chest", "fight-event", new JsonObject());
                    return (200, new JsonObject { ["active"] = polls < 8 }.ToJsonString());
                }
                if (path == "/act")
                {
                    Check(polls < 8 && stops == 0, "fight reaction waited for the attack action to end");
                    moves++;
                }
                return (200, "{}");
            };
            var task = AgentTools.Execute("run_script", new JsonObject {
                ["workflow"] = "reactive-fight-" + Guid.NewGuid().ToString("N"),
                ["code"] = """
                    on('nearby_chest',event=>{
                      if(event.execution.tool!=='fight') throw new Error('missing_fight_context');
                      tool('act',{steps:[{keys:['left'],until:{frames:1}}]});
                      return {handled:true};
                    });
                    tool('watch',{events:['nearby_chest']});
                    return tool('fight',{seconds:10});
                    """ }, default);
            var timer = System.Diagnostics.Stopwatch.StartNew();
            while (!task.IsCompleted)
            {
                Check(timer.Elapsed.TotalSeconds < 10, "fight reaction deadlocked");
                agent.PostUpdateEverything(); await Task.Delay(1);
            }
            var result = JsonNode.Parse(await task);
            Check(result["ok"].GetValue<bool>() && moves == 1 && stops == 1 && !AgentTools.ControlSession.Active,
                "fight reaction lost the attack action or leaked controls: " + result);
            Console.WriteLine("PASS: fight dispatches reactive movement before completion while retaining its attack action");
        }
        finally
        {
            AgentTools.Stop(); AgentTools.Actions.Reset(); AgentTools.Watches.Reset(); AgentTools.Plan.Reset();
            AgentPermissions.Attack = attack; HttpServerSystem.Handler = http; agent.Unload();
        }
    }

    static async Task TaskPolicyIntegration()
    {
        var agent = new NativeAgent(); agent.Load();
        var originalHttp = HttpServerSystem.Handler;
        var player = Terraria.Main.LocalPlayer;
        var position = player.position;
        bool dead = player.dead, suffocating = player.suffocating;
        async Task Pump(Func<bool> done)
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();
            while (!done())
            {
                Check(timer.Elapsed.TotalSeconds < 15, "task policy integration stalled");
                agent.PostUpdateEverything(); await Task.Delay(1);
            }
        }
        async Task<JsonObject> Call(string name, JsonObject args)
        {
            var task = AgentTools.Execute(name, args, default);
            await Pump(() => task.IsCompleted);
            return JsonNode.Parse(await task).AsObject();
        }
        Task<JsonObject> Set(int revision, string code) => Call("task_policy", new JsonObject
            { ["operation"] = "set", ["expected_revision"] = revision, ["code"] = code });
        int phase = 0, polls = 0, goalX = 20;
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset();
            player.position = new System.Numerics.Vector2(160, 160); player.dead = player.suffocating = false;
            int messages = AgentChat.Messages.Count;
            var installed = await Set(0, """
                let count=0;
                tool('watch',{events:['nearby_chest','nearby_drop']});
                on('nearby_chest',e=>{tool('say',{text:'global:'+e.task_node+':'+(++count)});return {handled:true};});
                on('nearby_drop',e=>{tool('say',{text:'global-drop:'+e.task_node+':'+(++count)});return {handled:true};});
                """);
            Check(installed["active"].GetValue<bool>(), "task policy failed to initialize: " + installed);
            await Call("task_plan", JsonNode.Parse("""
                {"goal":"shared reactions across nodes","mode":"dag","steps":[
                 {"id":"first","description":"first leg","depends_on":[],"conditions":[{"kind":"position","x":20,"y":10,"radius":3,"count":1}],
                  "script":"tool('watch',{events:['low_health']});step('travel',()=>tool('nav_to',{x:20,y:10}));return {status:'yield'};"},
                 {"id":"second","description":"second leg","depends_on":["first"],"conditions":[{"kind":"position","x":40,"y":10,"radius":3,"count":1}],
                  "script":"on('nearby_chest',e=>{tool('say',{text:'local:'+e.task_node});return {handled:true};});tool('watch',{events:['low_health']});step('travel',()=>tool('nav_to',{x:40,y:10}));return {status:'yield'};"}]}
                """).AsObject());
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/nav_recede")
                { RecedingNav.Active = true; goalX = JsonNode.Parse(payload)["gx"].GetValue<int>(); phase = goalX == 20 ? 1 : 2; polls = 0; }
                if (path == "/nav_recede_done")
                {
                    polls++;
                    if (polls == 1) AgentTools.Watches.Emit("nearby_chest", "leg:" + phase, new JsonObject { ["x"] = goalX, ["y"] = 10 });
                    if (phase == 2 && polls == 2) AgentTools.Watches.Emit("nearby_drop", "drop:2", new JsonObject { ["x"] = goalX, ["y"] = 10 });
                    if (polls >= 4) { player.position = new System.Numerics.Vector2(goalX * 16, 160); return (200, "{\"done\":true}"); }
                    return (200, "{\"done\":false}");
                }
                return (200, "{}");
            };
            foreach (string node in new[] { "first", "second" })
            {
                var result = await Call("run_script", new JsonObject { ["task_node"] = node });
                Check(result["ok"].GetValue<bool>(), "shared policy failed in node " + node + ": " + result);
                Check(AgentTools.Watches.Subscribed("nearby_drop"), "node watch removed a task policy subscription");
                var verified = await Call("task_step", new JsonObject { ["id"] = node, ["action"] = "verify" });
                Check(verified["verified"].GetValue<bool>(), "node completion was not host verified");
                if (node == "first")
                {
                    var saved = await Call("task_policy", new JsonObject { ["operation"] = "get" });
                    var same = await Set(1, saved["code"].ToString());
                    Check(same["unchanged"]?.GetValue<bool>() == true && same["revision"].GetValue<int>() == 1,
                        "identical policy code discarded live closure state");
                }
            }
            Check(AgentChat.Messages.Skip(messages).Where(m => m.StartsWith("global") || m.StartsWith("local:"))
                    .SequenceEqual(new[] { "global:first:1", "local:second", "global-drop:second:2" }),
                "task policy lost closure state, node ownership or local precedence: " + string.Join(',', AgentChat.Messages.Skip(messages)));
            Check(!AgentTools.TaskPolicyStatus()["active"].GetValue<bool>(), "completed DAG retained its task policy");

            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset();
            bool release = false, started = false, emitted = false, replaced = false, newEmitted = false;
            int navStarts = 0;
            await Set(0, """
                tool('watch',{events:['nearby_chest']});
                on('nearby_chest',e=>{
                  tool('say',{text:'old-start'});
                  tool('act',{steps:[{keys:['right'],until:{frames:1}}]});
                  tool('say',{text:'old-finished'});return {handled:true};
                });
                """);
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/nav_recede") { navStarts++; RecedingNav.Active = true; }
                if (path == "/nav_recede_stop") RecedingNav.Active = false;
                if (path == "/act") started = true;
                if (path == "/act_status") return (200, release ? "{\"active\":false}" : "{\"active\":true}");
                if (path == "/nav_recede_done")
                {
                    if (!emitted) { emitted = true; AgentTools.Watches.Emit("nearby_chest", "hot:old", new JsonObject { ["x"] = 50, ["y"] = 10 }); }
                    if (release && replaced && !newEmitted)
                    { newEmitted = true; AgentTools.Watches.Emit("nearby_chest", "hot:new", new JsonObject { ["x"] = 51, ["y"] = 10 }); }
                    if (AgentChat.Messages.Contains("new-policy"))
                    { player.position = new System.Numerics.Vector2(960, 160); return (200, "{\"done\":true}"); }
                    return (200, "{\"done\":false}");
                }
                return (200, "{}");
            };
            var action = await Call("run_script", new JsonObject { ["code"] = "return tool('action_start',{tool:'nav_to',args:{x:60,y:10}});" });
            string handle = action["value"]["handle"].ToString();
            await Pump(() => started);
            var updated = await Set(1, "tool('watch',{events:['nearby_chest']});on('nearby_chest',e=>{tool('say',{text:'new-policy'});return {handled:true};});");
            replaced = true;
            Check(updated["revision"].GetValue<int>() == 2 && navStarts == 1,
                "updating policy restarted the route or failed to publish a revision");
            var bad = await Set(2, "tool('watch',{events:['nearby_tile'],tile_types:[12]});throw new Error('broken candidate');");
            Check(bad["error"] != null && AgentTools.TaskPolicyStatus()["revision"].GetValue<int>() == 2
                && AgentTools.Watches.Subscribed("nearby_chest") && !AgentTools.Watches.WatchesTile(12),
                "failed policy installation changed the active policy or its subscriptions");
            release = true;
            await Pump(() => AgentTools.Actions.Summary().Any(a => a["handle"]?.ToString() == handle && a["status"]?.ToString() == "completed"));
            var completed = AgentTools.Actions.Summary().First(a => a["handle"]?.ToString() == handle);
            Check(completed["attempt"].GetValue<int>() == 1 && navStarts == 2
                && AgentChat.Messages.Contains("old-finished") && AgentChat.Messages.Contains("new-policy"),
                "hot update cancelled an entered handler or lost the original independent action");
            await Set(2, """
                let calls=0;
                tool('watch',{events:['nearby_chest']});
                on('nearby_chest',e=>{
                  calls++;
                  if(calls===1)tool('act',{steps:[{keys:['right'],until:{frames:1}}]});
                  tool('say',{text:'after-cancel:'+calls});return {handled:true};
                });
                """);
            bool inAction = false, sent = false, resumed = false;
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/nav_recede") RecedingNav.Active = true;
                if (path == "/nav_recede_stop") RecedingNav.Active = false;
                if (path == "/act") inAction = true;
                if (path == "/act_status") return (200, "{\"active\":true}");
                if (path == "/nav_recede_done")
                {
                    if (!sent) { sent = true; AgentTools.Watches.Emit("nearby_chest", "cancel:" + resumed, new JsonObject { ["x"] = 75, ["y"] = 10 }); }
                    if (AgentChat.Messages.Contains("after-cancel:2"))
                    { player.position = new System.Numerics.Vector2(1280, 160); return (200, "{\"done\":true}"); }
                    return (200, "{\"done\":false}");
                }
                return (200, "{}");
            };
            var cancellable = await Call("run_script", new JsonObject { ["code"] = "return tool('action_start',{tool:'nav_to',args:{x:80,y:10}});" });
            string cancelHandle = cancellable["value"]["handle"].ToString();
            await Pump(() => inAction);
            var paused = await Call("action_control", new JsonObject { ["handle"] = cancelHandle, ["command"] = "pause" });
            Check(paused["status"].ToString() == "paused" && !AgentTools.ControlSession.Active,
                "pause returned before the task policy released its active action");
            resumed = true; sent = false;
            await Call("action_control", new JsonObject { ["handle"] = cancelHandle, ["command"] = "resume" });
            await Pump(() => AgentTools.Actions.Summary().Any(a => a["handle"]?.ToString() == cancelHandle && a["status"]?.ToString() == "completed"));
            Check(AgentTools.TaskPolicyStatus()["active"].GetValue<bool>() && AgentChat.Messages.Contains("after-cancel:2"),
                "pausing one actor cancelled the shared policy or lost its state");
            AgentTools.NewTask();
            Check(!AgentTools.TaskPolicyStatus()["active"].GetValue<bool>() && AgentTools.TaskPolicyStatus()["revision"].GetValue<int>() == 0,
                "new task inherited a stale policy");
            Console.WriteLine("PASS: task policy spans DAG nodes, preserves closure/ownership, supports local overrides and updates without restarting active work");
        }
        finally
        {
            AgentTools.ClearTaskPolicy(true); AgentTools.Stop(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset(); AgentTools.Watches.Reset();
            player.position = position; player.dead = dead; player.suffocating = suffocating;
            HttpServerSystem.Handler = originalHttp; agent.Unload();
        }
    }
