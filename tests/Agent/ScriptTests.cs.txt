            return new System.Net.Http.HttpResponseMessage(Status) { Content = new System.Net.Http.StringContent(Reply) };
        }
    }

    static async Task EquipmentAnimationHandoff()
    {
        var agent = new NativeAgent(); agent.Load();
        var player = Terraria.Main.LocalPlayer;
        int animation = player.itemAnimation, itemTime = player.itemTime;
        bool dead = player.dead;
        async Task Pump(Func<bool> done)
        {
            var clock = System.Diagnostics.Stopwatch.StartNew();
            while (!done())
            { Check(clock.Elapsed.TotalSeconds < 5, "equipment animation handoff stalled"); agent.PostUpdateEverything(); await Task.Delay(1); }
        }
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset(); player.dead = false;
            player.itemAnimation = 3; player.itemTime = 5;
            var pending = AgentTools.Execute("equip_item", new JsonObject(), default);
            await Pump(() => AgentTools.ControlSession.OwnsItemUse);
            for (int i = 0; i < 4; i++) { agent.PostUpdateEverything(); await Task.Delay(1); }
            Check(!pending.IsCompleted && player.itemAnimation == 3 && player.itemTime == 5,
                "equipment bypassed or forcibly finished the active animation");
            player.itemAnimation = player.itemTime = 0;
            await Pump(() => pending.IsCompleted);
            var result = JsonNode.Parse(await pending);
            Check(result["error"]?.ToString() == "equipment_requires_engine" && !AgentTools.ControlSession.PrimaryActive,
                "equipment did not reach its adapter after animation completion or leaked its reservation");
            player.itemAnimation = 3;
            using var cancellation = new CancellationTokenSource();
            pending = AgentTools.Execute("equip_item", new JsonObject(), cancellation.Token);
            await Pump(() => AgentTools.ControlSession.OwnsItemUse);
            cancellation.Cancel();
            await Pump(() => pending.IsCompleted);
            try { await pending; throw new Exception("equipment animation wait ignored cancellation"); }
            catch (OperationCanceledException) { }
            Check(!AgentTools.ControlSession.PrimaryActive, "cancelled equipment wait retained input ownership");
            Console.WriteLine("PASS: equipment waits for animation completion with item-use ownership and releases its claim on cancellation (equipment adapter stubbed)");
        }
        finally
        {
            player.itemAnimation = animation; player.itemTime = itemTime; player.dead = dead;
            AgentTools.Stop(); AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset(); agent.Unload();
        }
    }

    static async Task TaskPolicyPatch()
    {
        var agent = new NativeAgent(); agent.Load();
        async Task<JsonObject> Call(JsonObject args)
        {
            var pending = AgentTools.Execute("task_policy", args, default);
            var clock = System.Diagnostics.Stopwatch.StartNew();
            while (!pending.IsCompleted)
            { Check(clock.Elapsed.TotalSeconds < 5, "policy patch stalled"); agent.PostUpdateEverything(); await Task.Delay(1); }
            return JsonNode.Parse(await pending).AsObject();
        }
        Task<JsonObject> Patch(int revision, string oldText, string newText) => Call(new JsonObject {
            ["operation"] = "patch", ["expected_revision"] = revision, ["old_text"] = oldText, ["new_text"] = newText });
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset();
            const string source = "on('tick',()=>{print('before');return {handled:true};});on('action_failed',()=>({retry:false}));";
            await Call(new JsonObject { ["operation"] = "set", ["expected_revision"] = 0, ["code"] = source });
            var result = await Patch(1, "'before'", "'after'");
            Check(result["revision"]?.GetValue<int>() == 2 && result["active"]?.GetValue<bool>() == true
                && result["events"].AsArray().Select(e => e.ToString()).Order().SequenceEqual(new[] { "action_failed", "tick" }),
                "policy patch removed unrelated reactions: " + result);
            string expected = source.Replace("'before'", "'after'");
            var saved = await Call(new JsonObject { ["operation"] = "get" });
            Check(saved["code"]?.ToString() == expected, "policy patch changed more than the selected text");
            foreach (var (revision, oldText, newText) in new[] { (1, "'after'", "'stale'"), (2, "missing", "x"),
                (2, "on(", "bad"), (2, "'after'", "'") })
            {
                var rejected = await Patch(revision, oldText, newText);
                Check(rejected["error"] != null, "invalid policy patch was accepted: " + rejected);
                saved = await Call(new JsonObject { ["operation"] = "get" });
                Check(saved["revision"]?.GetValue<int>() == 2 && saved["code"]?.ToString() == expected
                    && saved["active"]?.GetValue<bool>() == true, "failed patch replaced the working policy");
            }
            Console.WriteLine("PASS: policy patch preserves unrelated handlers and keeps the active version on stale, ambiguous or invalid edits");
        }
        finally { AgentTools.Stop(); AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset(); agent.Unload(); }
    }

        {
            bool boss = scenario == "boss", idle = scenario is "idle" or "idle_after_place" or "fight" or "gather_scan";
            bool feedbackScope=feedbackIdle&&idle&&scenario!="gather_scan";

                + "})({feedback_idle:" + (feedbackIdle ? "true" : "false") + "});return tool('nav_to');", 5, (_, _) => { }, async (name, args, token) =>
                {
                    if (name == "nav_to")
                    {
                        for (int i = 0; i < 3; i++)
                        {
                            phase = i;
                            if (scenario == "idle_after_place" && i == 1) combat["player"]["pos"]["x"] = 320;

                    }
                    if (name == "combat_assess")
                    {
                        if (args["forecast"]?.GetValue<bool>() == true)
                            Check(args["destination"]?["x"]?.GetValue<int>() == (scenario == "idle_after_place" && phase == 1 ? 50
                                : scenario == "idle_after_place" && phase == 2 ? 20 : idle ? 10 : 50),

                        return combat.ToJsonString();
                    }
                    if (name == "get_state") return progress ? """
                        {"codex_goal":{"native":{"status":"active","objective":"finish within 25 minutes"},"host_wall_elapsed_seconds":900},
                         "task_plan":{"status":"active","revision":2,"current_step":"prepare","steps":[{"id":"prepare","conditions":[{"kind":"position","x":10,"y":20}]}]},
                         "inventory_slots":[{"type":9,"stack":100}],"equipment_slots":[]}
                        """ : "{\"codex_goal\":{\"native\":{\"objective\":\"reach the cave\"}}}";
                    if (name == "script_program") return ScriptLibrary.Example(args["name"].ToString()).ToJsonString();
                    if (name == "jev_judge")
                    {
                        decisions++;
                        if (args["questions"]["movement"] != null)
                        {
                            Check(feedbackScope, "feedback motor replaced a navigation-owned controller");
                            return decisions == 1 ? "{\"status\":\"pending\"}" : new JsonObject {
                                ["status"] = "ready", ["request_id"] = 9, ["request_age_ms"] = 100,
                                ["result"] = JsonNode.Parse("""{"elapsed_ms":20,"answers":{"movement":{"choice":"left"},"jump":{"noul":0.1}}}""") }.ToJsonString();

                        if (progress)
                        {
                            Check(args["channel"].ToString() == "task_progress" && args["interval_ms"].GetValue<int>() == 10000
                                && args["state"]["elapsed_seconds"].GetValue<int>() == 900
                                && args["state"]["coordinates"]?.ToString().Contains("world tiles (16 pixels)") == true
                                && args["state"]["current_step"].ToString() == "prepare",

                                    && args["state"]["previous_review_request"]["elapsed_seconds"].GetValue<int>() == 900,

                        if (scenario == "fall_only") Check(args["state"]["fall_forecast"]?["damage"]?.GetValue<int>() == 50,
                            "fall-only evasion lost the projected damage");
                        Check(args["questions"]["attention"]?["criteria"]?["ranged_weapon"] != null, "combat decision could not request a ranged attack");
                        return decisions == 1 ? "{\"status\":\"pending\"}" : new JsonObject {
                            ["status"] = "ready", ["request_id"] = 9, ["input_state"] = args["state"].DeepClone(),
                            ["result"] = JsonNode.Parse("{\"elapsed_ms\":20,\"answers\":{\"response\":{\"choice\":\"avoid\"},\"attention\":{\"choice\":\""

                    }
                    if(name == "set_combat" && args["movement_intent"] != null)
                    { Check(feedbackScope, "movement intent escaped idle/fight scope"); updates++; return "{}"; }
                    Check(name == "act" && args["combat_candidate"]?["plan_id"]?.ToString() == "avoid", "combat policy lost native candidate validation");
                    updates++; return "{}";
                }, _ => Task.CompletedTask, default,
                programs: new ScriptLibrary(Path.Combine(Path.GetTempPath(), "terrablind-empty-skills-" + Guid.NewGuid())).Callable());
            bool inactive = boss || scenario == "quiet";
            Check(result["ok"].GetValue<bool>() && decisions == (inactive ? 0 : 3) && updates == (inactive || progress ? 0 : feedbackScope ? 3 : scenario == "idle_after_place" ? 1 : 2)
                && reviews == (scenario is "equipment_review" or "ranged_review" or "progress_review" ? 1 : 0),
                "default combat policy blocked pending navigation, repeated a response or touched boss movement: " + result);
        }
        Console.WriteLine("PASS: default survival uses shared combat candidates during navigation and idle, deduplicates execution and preserves boss control");
        if(!feedbackIdle) await SurvivalCombatPolicy(feedbackIdle: true);
    }

    static async Task BackpackLoadout()
    {
        var player = Terraria.Main.LocalPlayer;
        var inventory = player.inventory;
        bool dead = player.dead, itemUse = ItemUseCoordinator.IsActive, placing = PlaceAnywhere.IsRunning;
        int animation = player.itemAnimation, itemTime = player.itemTime;
        try
        {
            player.dead = ItemUseCoordinator.IsActive = PlaceAnywhere.IsRunning = false;
            player.itemAnimation = player.itemTime = 0;
            foreach (string scenario in new[] { "swap", "observe", "empty_slot", "keep", "ammo_lost", "source_changed", "destination_changed", "no_slot" })
            {
                player.inventory = Enumerable.Range(0, 58).Select(_ => new Terraria.Item()).ToArray();
                if (scenario != "empty_slot")
                    for (int i = 0; i < 10; i++) player.inventory[i] = new Terraria.Item { type = 9, stack = 1 };
                player.inventory[7] = new Terraria.Item { type = 24, stack = 1, damage = 7 };
                player.inventory[29] = new Terraria.Item { type = 5011, prefix = 41, stack = 1, damage = 18 };
                JsonObject Weapon(int slot) => new() { ["slot"] = slot, ["type"] = player.inventory[slot].type,
                    ["prefix"] = player.inventory[slot].prefix, ["stack"] = player.inventory[slot].stack,
                    ["damage"] = player.inventory[slot].damage, ["has_ammo"] = true, ["mana_available"] = true };
                var combat = new JsonObject { ["preferred_weapon_type"] = 24,
                    ["targets"] = new JsonArray(new JsonObject { ["weapons"] = new JsonArray(Weapon(7), Weapon(29)) }) };
                int swaps = 0, selected = 0;
                if (scenario == "observe") combat["observed_notes"] = new string('x', 9000);
                var code = "const config=" + new JsonObject { ["combat"] = scenario == "observe" ? null : combat.DeepClone(), ["instruction"] = "gather materials while defending" }.ToJsonString()
                    + ";const first=skill('combat_loadout',0,config);skill('combat_loadout',0,{...config,last_request:first.request_id});return first;";
                var result = await AgentScript.Run(code, 5, (_, _) => { }, (name, args, _) =>
                {
                    JsonObject response;
                    if (name == "jev_judge") response = new JsonObject { ["status"] = "ready", ["request_id"] = 7,
                        ["input_state"] = args["state"].DeepClone(), ["result"] = new JsonObject { ["elapsed_ms"] = 200,
                            ["answers"] = new JsonObject { ["response"] = new JsonObject { ["choice"] = scenario == "keep" ? "keep" : "slot_29" } } } };
                    else if (name == "combat_assess")
                    {
                        response = combat.DeepClone().AsObject();
                        if (scenario == "ammo_lost") response["targets"][0]["weapons"][1]["has_ammo"] = false;
                        if (scenario == "no_slot") response["targets"][0]["weapons"].AsArray().RemoveAt(0);
                    }
                    else if (name == "get_state")
                    {
                        if (scenario == "source_changed") player.inventory[29].prefix++;
                        if (scenario == "destination_changed") player.inventory[7].type = 3509;
                        if (scenario == "no_slot") for (int i = 0; i < 10; i++) player.inventory[i] = new Terraria.Item { type = 9, stack = 1 };
                        response = new JsonObject { ["inventory_slots"] = PvpInventory.Snapshot() };
                    }
                    else if (name == "inventory_swap")
                    {
                        Check(args["src"].GetValue<int>() == 29 && args["dst"].GetValue<int>() == (scenario == "empty_slot" ? 0 : 7), "loadout displaced an occupied slot while an empty slot was available");
                        swaps++; response = PvpInventory.Swap(args);
                    }
                    else
                    {
                        Check(name == "set_combat" && player.inventory[scenario == "empty_slot" ? 0 : 7].type == 5011, "loadout selected a weapon before its verified swap");
                        selected++; response = new JsonObject();
                    }
                    return Task.FromResult(response.ToJsonString());
                }, _ => Task.CompletedTask, default,
                    programs: new ScriptLibrary(Path.Combine(Path.GetTempPath(), "terrablind-loadout-" + Guid.NewGuid())).Callable());
                Check(result["ok"].GetValue<bool>() && swaps == (scenario is "swap" or "observe" or "empty_slot" ? 1 : 0) && selected == swaps,
                    "backpack loadout lost inventory validation or repeated a decision: " + scenario + " " + result);
                if (scenario == "swap") Check(player.inventory[29].type == 24 && player.inventory[7].type == 5011,
                    "loadout discarded the displaced weapon");
                if (scenario == "empty_slot") Check(player.inventory[29].type == 0 && player.inventory[7].type == 24,
                    "loadout removed an existing hotbar weapon despite available space");
            }
        }
        finally
        {
            player.inventory = inventory; player.dead = dead; player.itemAnimation = animation; player.itemTime = itemTime;
            ItemUseCoordinator.IsActive = itemUse; PlaceAnywhere.IsRunning = placing;
        }
        Console.WriteLine("PASS: Jev-selected backpack loadout uses verified production swaps, preserves displaced weapons and rejects changed inventory or ammo");
    }

    static async Task ReactiveGatherFightIntegration()
    {
        var agent = new NativeAgent(); agent.Load();
        var previous = Terraria.Main.npc;
        int messages = AgentChat.Messages.Count;
        bool reacted = false;
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.ControlSession.Reset(); AgentPermissions.Reset();
            var player = Terraria.Main.LocalPlayer;
            Terraria.Main.npc = Enumerable.Range(0, Terraria.Main.maxNPCs).Select(_ => new Terraria.NPC()).ToArray();
            Terraria.Main.npc[0] = new Terraria.NPC { active = true, type = 42,
                position = new(player.position.X + 80, player.position.Y) };
            async Task<T> Pump<T>(Task<T> pending)
            {
                var clock = System.Diagnostics.Stopwatch.StartNew();
                while (!pending.IsCompleted)

    static async Task BackgroundRecovery(bool inferOwner = false, bool supplyWorkflow = true)
    {
        var agent = new NativeAgent(); agent.Load();
        var http = HttpServerSystem.Handler;
        bool dead = Terraria.Main.LocalPlayer.dead;
        bool release = false;
        int effects = 0;
        async Task<T> Pump<T>(Task<T> task)
        {
            var clock = System.Diagnostics.Stopwatch.StartNew();
            while (!task.IsCompleted)
            { Check(clock.Elapsed.TotalSeconds < 5, "background recovery blocked the caller"); agent.PostUpdateEverything(); await Task.Delay(1); }
            return await task;
        }
        async Task<JsonObject> Call(string name, JsonObject args) => JsonNode.Parse(
            await Pump(AgentTools.Execute(name, args, default))).AsObject();
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset(); AgentTools.ControlSession.Reset();
            Terraria.Main.LocalPlayer.dead = false;
            AgentTools.Plan.Create(JsonNode.Parse("""
                {"goal":"observe long temporary work","mode":"dag","steps":[{"id":"travel","description":"travel",
                 "depends_on":[],"script":"return {original:true};","conditions":[{"kind":"permanent_health","count":9999}]}]}
                """).AsObject(), AgentTaskEvidence.Measure);
            string original = AgentTools.Plan.Script("travel").ToJsonString();
            HttpServerSystem.Handler = (path, _) =>
            {
                if (path == "/act") effects++;
                return (200, path == "/act_status" ? "{\"active\":" + (!release).ToString().ToLowerInvariant() + "}" : "{}");
            };
            var request = new JsonObject { ["background"] = true,
                ["code"] = "return step('input',()=>tool('act',{steps:[{keys:['right'],until:{frames:60}}]}));" };
            if (!inferOwner) request["task_node"] = "travel";
            if (!inferOwner && supplyWorkflow) request["workflow"] = "temporary";
            var started = await Call("run_script", request);
            Check(started["handle"] != null, "temporary script was rejected or waited for completion: " + started);
            Check(!inferOwner && supplyWorkflow || started["workflow_id"]?.ToString().StartsWith("temporary-") == true,
                "inferred input script omitted its independent checkpoint identity");
            Check(!AgentScript.ContainsInput("tool('task_plan_patch',{});return tool('action_query',{handle:'x'});")
                && AgentScript.ContainsInput("return tool('equip_item',{});"), "preparation was misclassified as physical input");
            string handle = started["handle"].ToString();
            var running = await Pump(AgentTools.Actions.Query(handle, 250, default));
            await Call("get_state", new JsonObject { ["fields"] = new JsonArray("player") });
            Check(running["status"].ToString() == "running" && effects == 1 && !release,
                "long temporary work prevented observation or used the original node script: " + running);
            release = true;
            JsonObject finished;
            do { finished = await Pump(AgentTools.Actions.Query(handle, 250, default)); }
            while (finished["status"].ToString() == "running");
            Check(finished["status"].ToString() == "completed" && effects == 1
                && AgentTools.Plan.Script("travel").ToJsonString() == original,
                "background temporary work replaced the parent script, repeated input or failed: " + finished);
        }
        finally
        {
            HttpServerSystem.Handler = http; Terraria.Main.LocalPlayer.dead = dead;
            AgentTools.Stop(); AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset(); agent.Unload();
        }
        Console.WriteLine("PASS: long temporary node scripts return a live handle, permit observation and retain original node code");
    }

    static async Task GatherReactionConsumption()
    {
        var agent = new NativeAgent(); agent.Load();
        var handler = HttpServerSystem.Handler;
        var player = Terraria.Main.LocalPlayer;
        var previousItem = player.inventory[49];
        var previousPotion = player.inventory[48];

        uint tick = Terraria.Main.GameUpdateCount;
        int uses = 0, heals = 0, navStarts = 0, navStops = 0;
        async Task<string> Pump(Task<string> task)
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();
            while (!task.IsCompleted)
            {
                Check(timer.Elapsed.TotalSeconds < 8, "gather reaction consumption deadlocked");
                Terraria.Main.GameUpdateCount++;
                agent.PostUpdateEverything(); await Task.Delay(1);
            }
            return await task;
        }
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset();

            player.inventory[48] = new Terraria.Item { type = 28, stack = 1, consumable = true };
            player.inventory[47] = new Terraria.Item { type = 3509, stack = 1, pick = 35 };
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/item_use")
                {
                    int slot = JsonNode.Parse(payload)["slot"].GetValue<int>();
                    if (slot == 49) uses++; else if (slot == 48) heals++;
                    player.inventory[slot] = new Terraria.Item { type = 0, stack = 0 };
                }
                if (path == "/item_use_status") return (200, "{\"active\":false,\"outcome\":\"consumed\"}");
                if (path == "/nav_recede")
                {
                    navStarts++; RecedingNav.Active = true;
                    player.inventory[49] = new Terraria.Item { type = 29, stack = 1, consumable = true };
                }
                if (path == "/nav_recede_stop") { navStops++; RecedingNav.Active = false; }
                if (path == "/nav_recede_done") return (200, "{\"done\":false}");
                return (200, "{}");
            };
            var output = JsonNode.Parse(await Pump(AgentTools.Execute("run_script", new JsonObject
            {
                ["workflow"] = "gather-reaction-consumption", ["code"] = """
                    let healed=false;
                    on('tick',()=>{
                      const gathered=tool('gather',{targets:[{item_type:29,count:1}]});
                      const used=tool('use_item',{slot:49,x:10,y:10,duration_ticks:1});
                      if(!healed){tool('use_item',{slot:48,x:10,y:10,duration_ticks:1});healed=true;}

                      return {handled:true};
                    });

                    """
            }, default)));

                && output["logs"].ToJsonString().Contains("deferred_consumption")
                && player.inventory[49].stack == 1 && uses == 0 && heals == 1,
                "reaction consumed the parent's collection before inventory verification: " + output);

                { ["slot"] = 49, ["x"] = 10, ["y"] = 10, ["duration_ticks"] = 1 }, default)));
            Check(usedAfter["outcome"].ToString() == "consumed" && uses == 1,
                "completed gather kept blocking later item consumption");
            Terraria.Main.tile[400, 400] = new Terraria.Tile { HasUnactuatedTile = true, TileType = 12 };
            Reach.MineTest = (_, _, _) => false;
            var world = new GameGather(new[] { new GatherTarget(29, 1, Array.Empty<string>(), Array.Empty<int>()) });
            async Task<string> Approach() => (await world.Act(new JsonObject { ["item_type"] = 29,
                ["kind"] = "tile", ["id"] = "old-crystal", ["x"] = 400, ["y"] = 400, ["tile_type"] = 12 }, default)).ToJsonString();

    static async Task IdlePolicyHandoff()
    {
        var agent = new NativeAgent(); agent.Load();

        int actions = 0;
        async Task Pump(Func<bool> done)
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();

            {
                Check(timer.Elapsed.TotalSeconds < 5, "idle action handoff stalled");
                agent.PostUpdateEverything(); await Task.Delay(1);

            {
                if (path == "/act") actions++;

                    ? (release ? "{\"active\":false,\"outcome\":\"done\"}" : "{\"active\":true}") : "{}");
            };

                | System.Reflection.BindingFlags.NonPublic).SetValue(agent, idle);

                tool('say',{text:'foreground-ready'});
                return tool('act',{steps:[{keys:['left'],until:{frames:1}}]});
                """ }, default);
            await Pump(() => AgentChat.Messages.Skip(messages).Contains("foreground-ready"));
            for (int i = 0; i < 30; i++) { agent.PostUpdateEverything(); await Task.Delay(1); }
            Check(!main.IsCompleted && actions == 1, "foreground action failed instead of waiting for the in-flight idle action");
            release = true;
            await Pump(() => main.IsCompleted && idle.IsCompleted);
            var result = JsonNode.Parse(await main);
            Check(result["ok"]?.GetValue<bool>() == true && result["value"]?["outcome"]?.ToString() == "done"
                && actions == 2 && !AgentTools.ControlSession.PrimaryActive,
                "foreground handoff replayed idle work or leaked control: " + result);

        finally

            AgentTools.Stop(); AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset(); agent.Unload();

    static async Task ConcurrentPolicyContext()
    {
        var agent = new NativeAgent(); agent.Load();
        long completion = CombatForecast.CompletionVersion;
        async Task Pump(Task task)
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();
            while (!task.IsCompleted)
            {
                Check(timer.Elapsed.TotalSeconds < 5, "concurrent policy context deadlocked");
                agent.PostUpdateEverything(); await Task.Delay(1);
            }
            await task;
        }
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset();
            await Pump(AgentTools.Execute("task_policy", new JsonObject { ["operation"] = "set", ["expected_revision"] = 0,
                ["code"] = "on('tick',e=>{tool('say',{text:'concurrent:'+e.execution.tool});return {handled:true};});" }, default));
            Check(AgentTools.ControlSession.Begin("nav_to", null, out _) == null,
                "test could not acquire navigation");
            await Pump(AgentTools.ReactAtBoundary(default, context: new JsonObject { ["tool"] = "fight" }));
            Check(!AgentChat.Messages.Contains("concurrent:fight"), "concurrent fight consumed the primary navigation's policy tick");
            await Pump(AgentTools.ReactAtBoundary(default, context: new JsonObject { ["tool"] = "nav_to",
                ["args"] = new JsonObject { ["x"] = 20, ["y"] = 10 } }));
            Check(AgentChat.Messages.Contains("concurrent:nav_to"), "primary navigation lost its policy observation");
            int observations = AgentChat.Messages.Count(m => m == "concurrent:nav_to");
            CombatForecast.CompletionVersion++;
            await Pump(AgentTools.ReactAtBoundary(default, context: new JsonObject { ["tool"] = "nav_to",
                ["args"] = new JsonObject { ["x"] = 20, ["y"] = 10 } }));
            Check(AgentChat.Messages.Count(m => m == "concurrent:nav_to") == observations + 1,
                "native forecast completion remained behind the ordinary tick throttle");
            Console.WriteLine("PASS: concurrent defense leaves task-policy ticks and route handoff with the primary navigation");
        }
        finally
        {
            CombatForecast.CompletionVersion = completion;
            AgentTools.ControlSession.Reset(); AgentTools.ClearTaskPolicy(true); AgentTools.Stop(); AgentTools.Plan.Reset(); AgentTools.Watches.Reset(); agent.Unload();
        }
    }

    public static async Task Replay(string path)
    {
        var events = JsonNode.Parse(File.ReadAllText(path)).AsArray();
        string code = events.First(e => e?["tool"]?.ToString() == "run_script")["args"]["code"].ToString();
        foreach (bool overshoot in new[] { false, true })
        {
            int stock = 0, attempts = 0;
            bool interrupted = false;
            var said = new List<int>();
            Task<string> Execute(string name, JsonObject args, CancellationToken token)
            {
                if (name == "get_state") return Task.FromResult(new JsonObject { ["inventory_slots"] = new JsonArray(
                    new JsonObject { ["type"] = 29, ["stack"] = stock }) }.ToJsonString());
                if (name == "say") { said.Add(stock); return Task.FromResult("{\"ok\":true}"); }
                Check(name == "gather" && !interrupted, "generated code continued collecting after an unmet condition");
                attempts++;
                if (overshoot) { stock += 2; interrupted = true; }
                else if (attempts == 4) interrupted = true;
                else stock++;
                return Task.FromResult(new JsonObject { ["ok"] = overshoot || !interrupted,
                    ["outcome"] = !overshoot && interrupted ? "source_unavailable" : "enough",
                    ["targets"] = new JsonArray(new JsonObject { ["item_type"] = 29, ["actual"] = stock }) }.ToJsonString());
            }
            var result = await AgentScript.Run(code, 5, (_, _) => { }, Execute, _ => Task.CompletedTask, default);
            Check(result["ok"].GetValue<bool>() && interrupted, "generated code did not return evidence on changed conditions");
            Check(overshoot ? stock == 2 && attempts == 1 && said.Count == 0
                : stock == 3 && attempts == 4 && said.SequenceEqual(new[] { 1, 2, 3 }),
                "generated code falsely announced collection or lost per-item order");
            Console.WriteLine($"PASS: unchanged model code replay overshoot={overshoot} stock={stock} announcements={said.Count} model_calls=0");
        }
        Console.WriteLine("SCRIPT_REPLAY_COMPLETE checks=2 mock_world=true model_calls=0");
    }

                "inventory count projection disagrees with native bag slots");

    static async Task PendingCombatNavigation(bool rejected = false, bool ownerEnded = false)
    {
        var agent = new NativeAgent(); agent.Load();
        var http = HttpServerSystem.Handler;
        var refresh = CombatForecast.RefreshHandler;
        var prepare = CombatForecast.PrepareHandler;
        bool manual = Manual.On;
        using var lifetime = new CancellationTokenSource();
        var release = new TaskCompletionSource<JsonObject>(TaskCreationOptions.RunContinuationsAsynchronously);
        int starts = 0, stops = 0, moves = 0, computations = 0, pendingPolls = 0;
        string handle = null;
        bool notified = false, finished = false, proofCancelled = false;
        int messages = AgentChat.Messages.Count;
        var clock = System.Diagnostics.Stopwatch.StartNew();
        long pendingAt = 0;
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset(); Manual.On = false;
            CombatForecast.RefreshHandler = async (args, token) =>
            {
                if (args["combat_candidate"]["forecast_id"].GetValue<int>() == 2) return args;
                computations++;
                pendingAt = clock.ElapsedMilliseconds;
                Check(RecedingNav.Active && stops == 0 && AgentTools.ControlSession.Tool == "nav_to",
                    "background preparation took navigation input before its proof");
                handle = AgentTools.Actions.ActiveHandle;
                try { return await release.Task.WaitAsync(token); }
                catch (OperationCanceledException) { proofCancelled = true; throw; }
            };
            CombatForecast.PrepareHandler = args => args;
            HttpServerSystem.Handler = (path, _) =>
            {
                if (path == "/nav_recede") { starts++; RecedingNav.Active = true; }
                if (path == "/nav_recede_stop") { stops++; RecedingNav.Active = false; }
                if (path == "/act")
                {
                    Check(!RecedingNav.Active && stops == 1, "ready maneuver skipped navigation handoff");
                    moves++; finished = true;
                }
                if (path == "/nav_recede_done")
                {
                    if (computations > 0 && !release.Task.IsCompleted)
                    {
                        pendingPolls++;
                        if (!notified)
                        { notified = true; AgentTools.Watches.Emit("nearby_chest", "pending-proof", new JsonObject()); }
                    }
                    if (AgentChat.Messages.Skip(messages).Contains("proof rejected")) finished = true;
                    if (ownerEnded && pendingPolls > 5 && AgentChat.Messages.Skip(messages).Contains("observed during proof")) finished = true;
                    return (200, new JsonObject { ["done"] = finished && (ownerEnded || rejected || starts > 1) }.ToJsonString());
                }
                return (200, "{\"active\":false,\"outcome\":\"done\"}");
            };
            var task = AgentTools.Execute("run_script", new JsonObject { ["code"] = """
                let done=false;
                on('nearby_chest',()=>{tool('say',{text:'observed during proof'});return {handled:true};});
                on('tick',()=>{
                  if(done||tool('get_state',{fields:['control_execution']}).control_execution.tool!=='nav_to')return {handled:true};
                  const r=tool('act',{steps:[{keys:['right'],until:{frames:1}}],
                    combat_candidate:{context:'pending-test',forecast_id:1,plan_id:'right_weapon_279'}});
                  if(r.status==='combat_preparing')return {handled:true};
                  done=true;
                  if(r.error)tool('say',{text:'proof rejected'});
                  return {handled:true};
                });
                tool('watch',{events:['nearby_chest']});
                return tool('nav_to',{x:20,y:10});
                """ }, lifetime.Token);
            long releasedAt = 0;
            while (!task.IsCompleted)
            {
                Check(clock.Elapsed.TotalSeconds < 10, "pending combat/navigation integration stalled");
                agent.PostUpdateEverything(); await Task.Delay(1);
                if (!ownerEnded && !release.Task.IsCompleted && pendingPolls > 5 && clock.ElapsedMilliseconds - pendingAt >= 2300
                    && AgentChat.Messages.Skip(messages).Contains("observed during proof"))
                {
                    Check(RecedingNav.Active && stops == 0 && moves == 0 && computations == 1,
                        "pending physics stopped navigation, duplicated preparation or executed without proof");
                    releasedAt = clock.ElapsedMilliseconds;
                    release.SetResult(rejected ? new JsonObject { ["error"] = "fixture_candidate_changed" }
                        : JsonNode.Parse("""{"combat_candidate":{"context":"pending-test","forecast_id":2,"plan_id":"right_weapon_279"}}""").AsObject());
                }
            }
            var result = JsonNode.Parse(await task);
            if (ownerEnded)
            {
                var cleanup = System.Diagnostics.Stopwatch.StartNew();
                while (!proofCancelled && cleanup.ElapsedMilliseconds < 250)
                { agent.PostUpdateEverything(); await Task.Delay(1); }
                Check(proofCancelled && moves == 0 && starts == 1 && stops == 0,
                    "finished navigation left its background physics running or applied a late maneuver");
                Console.WriteLine("COMBAT_PENDING_NAV owner_ended=true proof_cancelled=true model_calls=0");
                return;
            }
            Check(result["ok"]?.GetValue<bool>() == true && result["value"]?["done"]?.GetValue<bool>() == true
                && starts == (rejected ? 1 : 2) && stops == (rejected ? 0 : 1) && moves == (rejected ? 0 : 1),
                "ready or rejected combat proof failed navigation continuation: " + result);
            Check(AgentTools.Actions.Summary().Any(a => a["handle"]?.ToString() == handle
                && a["attempt"]?.GetValue<int>() == 1 && a["status"]?.ToString() == "completed"),
                "combat proof replaced the original navigation handle");
            Console.WriteLine($"COMBAT_PENDING_NAV rejected={rejected} pending_ms={releasedAt - pendingAt} pending_polls={pendingPolls} completion_after_proof_ms={clock.ElapsedMilliseconds - releasedAt} model_calls=0");
        }
        finally
        {
            lifetime.Cancel(); AgentTools.Stop(); AgentTools.ClearTaskPolicy(); AgentTools.Actions.Reset(); AgentTools.Plan.Reset();
            HttpServerSystem.Handler = http; CombatForecast.RefreshHandler = refresh; CombatForecast.PrepareHandler = prepare;
            Manual.On = manual; agent.Unload();
        }
    }

            var session = new AgentControlSession(new AgentControlPolicy());
            Check(session.Begin("nav_to", null, out long ticket) == null, "could not acquire test navigation");
            var restore = session.SuspendPrimary(); session.Reset(); restore();
            Check(!session.Active, "cancelled reaction restored an obsolete navigation claim");
            Console.WriteLine("PASS: production navigation keeps moving during observation, hands off effects and resumes the same action without leaked controls");
        }
        finally
        {
            AgentTools.Stop(); AgentTools.Actions.Reset(); AgentTools.Watches.Reset(); AgentTools.Plan.Reset();
            player.position = position; player.dead = dead; player.suffocating = suffocating;
            player.itemAnimation = animation;
            HttpServerSystem.Handler = http; agent.Unload();
            CombatForecast.RefreshHandler = null; CombatForecast.PrepareHandler = null;
        }
    }

    static async Task ReactiveFightIntegration()
    {
        var agent = new NativeAgent(); agent.Load();
        var http = HttpServerSystem.Handler;
        bool attack = AgentPermissions.Attack;
        int polls = 0, moves = 0, stops = 0;
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentPermissions.Attack = true;
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/fight" && JsonNode.Parse(payload)?["active"]?.GetValue<bool>() == false) stops++;
                if (path == "/fight_active")
                {
                    polls++;
                    if (polls == 2) AgentTools.Watches.Emit("nearby_chest", "fight-event", new JsonObject());
                    return (200, new JsonObject { ["active"] = polls < 8 }.ToJsonString());
                }
                if (path == "/act")
                {
                    Check(polls < 8 && stops == 0, "fight reaction waited for the attack action to end");
                    moves++;
                }
                return (200, "{}");
            };
            var task = AgentTools.Execute("run_script", new JsonObject {
                ["workflow"] = "reactive-fight-" + Guid.NewGuid().ToString("N"),
                ["code"] = """
                    on('nearby_chest',event=>{
                      if(event.execution.tool!=='fight') throw new Error('missing_fight_context');
                      tool('act',{steps:[{keys:['left'],until:{frames:1}}]});
                      return {handled:true};
                    });
                    tool('watch',{events:['nearby_chest']});
                    return tool('fight',{seconds:10});
                    """ }, default);
            var timer = System.Diagnostics.Stopwatch.StartNew();
            while (!task.IsCompleted)
            {
                Check(timer.Elapsed.TotalSeconds < 10, "fight reaction deadlocked");
                agent.PostUpdateEverything(); await Task.Delay(1);
            }
            var result = JsonNode.Parse(await task);
            Check(result["ok"].GetValue<bool>() && moves == 1 && stops == 1 && !AgentTools.ControlSession.Active,
                "fight reaction lost the attack action or leaked controls: " + result);
            Console.WriteLine("PASS: fight dispatches reactive movement before completion while retaining its attack action");
        }
        finally
        {
            AgentTools.Stop(); AgentTools.Actions.Reset(); AgentTools.Watches.Reset(); AgentTools.Plan.Reset();
            AgentPermissions.Attack = attack; HttpServerSystem.Handler = http; agent.Unload();
        }
    }

    static async Task TaskPolicyIntegration()
    {
        var agent = new NativeAgent(); agent.Load();
        var originalHttp = HttpServerSystem.Handler;
        var player = Terraria.Main.LocalPlayer;
        var position = player.position;
        bool dead = player.dead, suffocating = player.suffocating;
        async Task Pump(Func<bool> done)
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();
            while (!done())
            {
                Check(timer.Elapsed.TotalSeconds < 15, "task policy integration stalled");
                agent.PostUpdateEverything(); await Task.Delay(1);
            }
        }
        async Task<JsonObject> Call(string name, JsonObject args)
        {
            var task = AgentTools.Execute(name, args, default);
            await Pump(() => task.IsCompleted);
            return JsonNode.Parse(await task).AsObject();
        }
        Task<JsonObject> Set(int revision, string code) => Call("task_policy", new JsonObject
            { ["operation"] = "set", ["expected_revision"] = revision, ["code"] = code });
        int phase = 0, polls = 0, goalX = 20;
        try
        {
            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset();
            player.position = new System.Numerics.Vector2(160, 160); player.dead = player.suffocating = false;
            int messages = AgentChat.Messages.Count;
            var installed = await Set(0, """
                let count=0;
                tool('watch',{events:['nearby_chest','nearby_drop']});
                on('nearby_chest',e=>{tool('say',{text:'global:'+e.task_node+':'+(++count)});return {handled:true};});
                on('nearby_drop',e=>{tool('say',{text:'global-drop:'+e.task_node+':'+(++count)});return {handled:true};});
                """);
            Check(installed["active"].GetValue<bool>(), "task policy failed to initialize: " + installed);
            await Call("task_plan", JsonNode.Parse("""
                {"goal":"shared reactions across nodes","mode":"dag","steps":[
                 {"id":"first","description":"first leg","depends_on":[],"conditions":[{"kind":"position","x":20,"y":10,"radius":3,"count":1}],
                  "script":"tool('watch',{events:['low_health']});step('travel',()=>tool('nav_to',{x:20,y:10}));return {status:'yield'};"},
                 {"id":"second","description":"second leg","depends_on":["first"],"conditions":[{"kind":"position","x":40,"y":10,"radius":3,"count":1}],
                  "script":"on('nearby_chest',e=>{tool('say',{text:'local:'+e.task_node});return {handled:true};});tool('watch',{events:['low_health']});step('travel',()=>tool('nav_to',{x:40,y:10}));return {status:'yield'};"}]}
                """).AsObject());
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/nav_recede")
                { RecedingNav.Active = true; goalX = JsonNode.Parse(payload)["gx"].GetValue<int>(); phase = goalX == 20 ? 1 : 2; polls = 0; }
                if (path == "/nav_recede_done")
                {
                    polls++;
                    if (polls == 1) AgentTools.Watches.Emit("nearby_chest", "leg:" + phase, new JsonObject { ["x"] = goalX, ["y"] = 10 });
                    if (phase == 2 && polls == 2) AgentTools.Watches.Emit("nearby_drop", "drop:2", new JsonObject { ["x"] = goalX, ["y"] = 10 });
                    if (polls >= 4) { player.position = new System.Numerics.Vector2(goalX * 16, 160); return (200, "{\"done\":true}"); }
                    return (200, "{\"done\":false}");
                }
                return (200, "{}");
            };
            foreach (string node in new[] { "first", "second" })
            {
                var result = await Call("run_script", new JsonObject { ["task_node"] = node });
                Check(result["ok"].GetValue<bool>(), "shared policy failed in node " + node + ": " + result);
                Check(AgentTools.Watches.Subscribed("nearby_drop"), "node watch removed a task policy subscription");
                var verified = await Call("task_step", new JsonObject { ["id"] = node, ["action"] = "verify" });
                Check(verified["verified"].GetValue<bool>(), "node completion was not host verified: " + verified);
                if (node == "first")
                {
                    var saved = await Call("task_policy", new JsonObject { ["operation"] = "get" });
                    var same = await Set(1, saved["code"].ToString());
                    Check(same["unchanged"]?.GetValue<bool>() == true && same["revision"].GetValue<int>() == 1,
                        "identical policy code discarded live closure state");
                }
            }
            Check(AgentChat.Messages.Skip(messages).Where(m => m.StartsWith("global") || m.StartsWith("local:"))
                    .SequenceEqual(new[] { "global:first:1", "local:second", "global-drop:second:2" }),
                "task policy lost closure state, node ownership or local precedence: " + string.Join(',', AgentChat.Messages.Skip(messages)));
            Check(!AgentTools.TaskPolicyStatus()["active"].GetValue<bool>(), "completed DAG retained its task policy");

            AgentTools.NewTask(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset();
            bool release = false, started = false, emitted = false, replaced = false, newEmitted = false;
            int navStarts = 0;
            await Set(0, """
                tool('watch',{events:['nearby_chest']});
                on('nearby_chest',e=>{
                  tool('say',{text:'old-start'});
                  tool('act',{steps:[{keys:['right'],until:{frames:1}}]});
                  tool('say',{text:'old-finished'});return {handled:true};
                });
                """);
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/nav_recede") { navStarts++; RecedingNav.Active = true; }
                if (path == "/nav_recede_stop") RecedingNav.Active = false;
                if (path == "/act") started = true;
                if (path == "/act_status") return (200, release ? "{\"active\":false}" : "{\"active\":true}");
                if (path == "/nav_recede_done")
                {
                    if (!emitted) { emitted = true; AgentTools.Watches.Emit("nearby_chest", "hot:old", new JsonObject { ["x"] = 50, ["y"] = 10 }); }
                    if (release && replaced && !newEmitted)
                    { newEmitted = true; AgentTools.Watches.Emit("nearby_chest", "hot:new", new JsonObject { ["x"] = 51, ["y"] = 10 }); }
                    if (AgentChat.Messages.Contains("new-policy"))
                    { player.position = new System.Numerics.Vector2(960, 160); return (200, "{\"done\":true}"); }
                    return (200, "{\"done\":false}");
                }
                return (200, "{}");
            };
            var action = await Call("run_script", new JsonObject { ["code"] = "return tool('action_start',{tool:'nav_to',args:{x:60,y:10}});" });
            string handle = action["value"]["handle"].ToString();
            await Pump(() => started);
            var updated = await Set(1, "tool('watch',{events:['nearby_chest']});on('nearby_chest',e=>{tool('say',{text:'new-policy'});return {handled:true};});");
            replaced = true;
            Check(updated["revision"].GetValue<int>() == 2 && navStarts == 1,
                "updating policy restarted the route or failed to publish a revision");
            var bad = await Set(2, "tool('watch',{events:['nearby_tile'],tile_types:[12]});throw new Error('broken candidate');");
            Check(bad["error"] != null && AgentTools.TaskPolicyStatus()["revision"].GetValue<int>() == 2
                && AgentTools.Watches.Subscribed("nearby_chest") && !AgentTools.Watches.WatchesTile(12),
                "failed policy installation changed the active policy or its subscriptions");
            release = true;
            await Pump(() => AgentTools.Actions.Summary().Any(a => a["handle"]?.ToString() == handle && a["status"]?.ToString() == "completed"));
            var completed = AgentTools.Actions.Summary().First(a => a["handle"]?.ToString() == handle);
            Check(completed["attempt"].GetValue<int>() == 1 && navStarts == 2
                && AgentChat.Messages.Contains("old-finished") && AgentChat.Messages.Contains("new-policy"),
                "hot update cancelled an entered handler or lost the original independent action");
            await Set(2, """
                let calls=0;
                tool('watch',{events:['nearby_chest']});
                on('nearby_chest',e=>{
                  calls++;
                  if(calls===1)tool('act',{steps:[{keys:['right'],until:{frames:1}}]});
                  tool('say',{text:'after-cancel:'+calls});return {handled:true};
                });
                """);
            bool inAction = false, sent = false, resumed = false;
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/nav_recede") RecedingNav.Active = true;
                if (path == "/nav_recede_stop") RecedingNav.Active = false;
                if (path == "/act") inAction = true;
                if (path == "/act_status") return (200, "{\"active\":true}");
                if (path == "/nav_recede_done")
                {
                    if (!sent) { sent = true; AgentTools.Watches.Emit("nearby_chest", "cancel:" + resumed, new JsonObject { ["x"] = 75, ["y"] = 10 }); }
                    if (AgentChat.Messages.Contains("after-cancel:2"))
                    { player.position = new System.Numerics.Vector2(1280, 160); return (200, "{\"done\":true}"); }
                    return (200, "{\"done\":false}");
                }
                return (200, "{}");
            };
            var cancellable = await Call("run_script", new JsonObject { ["code"] = "return tool('action_start',{tool:'nav_to',args:{x:80,y:10}});" });
            string cancelHandle = cancellable["value"]["handle"].ToString();
            await Pump(() => inAction);
            var paused = await Call("action_control", new JsonObject { ["handle"] = cancelHandle, ["command"] = "pause" });
            Check(paused["status"].ToString() == "paused" && !AgentTools.ControlSession.Active,
                "pause returned before the task policy released its active action");
            resumed = true; sent = false;
            await Call("action_control", new JsonObject { ["handle"] = cancelHandle, ["command"] = "resume" });
            await Pump(() => AgentTools.Actions.Summary().Any(a => a["handle"]?.ToString() == cancelHandle && a["status"]?.ToString() == "completed"));
            Check(AgentTools.TaskPolicyStatus()["active"].GetValue<bool>() && AgentChat.Messages.Contains("after-cancel:2"),
                "pausing one actor cancelled the shared policy or lost its state");
            AgentTools.NewTask();
            Check(!AgentTools.TaskPolicyStatus()["active"].GetValue<bool>() && AgentTools.TaskPolicyStatus()["revision"].GetValue<int>() == 0,
                "new task inherited a stale policy");
            Console.WriteLine("PASS: task policy spans DAG nodes, preserves closure/ownership, supports local overrides and updates without restarting active work");
        }
        finally
        {
            AgentTools.ClearTaskPolicy(true); AgentTools.Stop(); AgentTools.Plan.Reset(); AgentTools.Actions.Reset(); AgentTools.Watches.Reset();
            player.position = position; player.dead = dead; player.suffocating = suffocating;
            HttpServerSystem.Handler = originalHttp; agent.Unload();
        }
    }
