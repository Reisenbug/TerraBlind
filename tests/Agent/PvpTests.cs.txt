    static bool JevariaCombatExtraJump() => TerraBlind.JevariaCombat.GameMovement.ExtraJumpObserved;

    sealed class DecisionHandler : HttpMessageHandler
    {
        public JsonObject Request;
        public string Choice = "retreat";
        public string Mode = "ranged";
        public Func<JsonObject, string> ModeForState;
        protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken token)
        {
            Check(request.Headers.Authorization?.Parameter == "fake-key", "Jev authentication missing");
            Request = JsonNode.Parse(await request.Content.ReadAsStringAsync(token)).AsObject();
            string mode = ModeForState?.Invoke(Request["state"].AsObject()) ?? Mode;
            string selected = Request["state"]["plans"].AsObject().FirstOrDefault(pair => pair.Value["move"].ToString() == Choice
                && pair.Value["weapon"].ToString() == mode).Key ?? "invalid";
            return new HttpResponseMessage(HttpStatusCode.OK)
            { Content = new StringContent(new JsonObject { ["answers"] = new JsonObject { ["plan"] = new JsonObject { ["choice"] = selected } } }.ToJsonString()) };
        }
    }

            Check(PvpDuel.TakeLoadoutReview() && !PvpDuel.TakeLoadoutReview(), "ammo loss did not wake exactly once");
            report = PvpDuel.Report();
            Check(!report["loadout"]["advice_ready"].GetValue<bool>()
                && report["directive"]["instructions"]["general"].GetValue<string>() == "Keep distance.",
                "loadout refresh cleared other instructions or retained old equipment advice");
            advice["expected_revision"] = revision;
            Check((await Publish(advice))["error"]?.ToString() == "revision_conflict", "late advice was accepted for the wrong loadout");
            Console.WriteLine("PASS: Agent equipment advice reaches the single live Jev request path, changes weapon selection and expires on ammo loss");
        }
        finally
        {
            PvpDuel.Stop();
            Game.netMode = Terraria.ID.NetmodeID.SinglePlayer;
            Config.I.AiClient = false;
            self.HasAmmunition = true;
            self.armor[3] = armor;
            self.miscEquips[4] = hook;
            Terraria.ID.ContentSamples.ProjectilesByType.Remove(Terraria.ID.ProjectileID.DeathSickle);
        }
    }

    static void CheckCoverFight()
    {
        var self = Game.LocalPlayer;
        var target = Game.player[1];
        const int floor = 50;
        Game.netMode = Terraria.ID.NetmodeID.MultiplayerClient;
        Config.I.AiClient = true;
        self.active = target.active = self.hostile = target.hostile = true;
        self.dead = target.dead = false;
        self.position = new(200, floor * 16 - self.height);
        target.position = new(360, self.position.Y);
        self.velocity = target.velocity = Microsoft.Xna.Framework.Vector2.Zero;
        self.itemAnimation = self.itemTime = target.selectedItem = 0;
        self.selectedItem = 0;
        foreach (var item in self.inventory) item.stack = 0;
        self.inventory[0] = new Terraria.Item { type = Terraria.ID.ItemID.DeathSickle, stack = 1, damage = 25,
            useStyle = 1, shoot = Terraria.ID.ProjectileID.DeathSickle, shootSpeed = 10 };
        self.inventory[1] = new Terraria.Item { type = 3, stack = 10, createTile = 1, damage = 0, useTime = 12 };
        target.inventory[0] = new Terraria.Item { type = 96, stack = 1, shoot = Terraria.ID.ProjectileID.Bullet };
        Terraria.ID.ContentSamples.ProjectilesByType[Terraria.ID.ProjectileID.DeathSickle] = new()
            { tileCollide = false, aiStyle = 18, width = 42, height = 42 };
        Terraria.ID.ContentSamples.ProjectilesByType[Terraria.ID.ProjectileID.Bullet] = new() { tileCollide = true };
        for (int x = 4; x < 26; x++) Game.tile[x, floor] = new Terraria.Tile { HasUnactuatedTile = true, TileType = 1 };
        int row = (int)(self.Center.Y / 16);
        Game.tile[16, row - 1] = new Terraria.Tile { HasUnactuatedTile = true, TileType = 1 };
        var solid = Terraria.Collision.SolidTest;
        Terraria.Collision.SolidTest = (position, width, height) =>
        {
            for (int x = (int)(position.X / 16); x <= (int)((position.X + width - .01f) / 16); x++)
            for (int y = (int)(position.Y / 16); y <= (int)((position.Y + height - .01f) / 16); y++)
                if (Game.tile[x, y].HasUnactuatedTile) return true;
            return false;
        };
        JsonObject last = null;
        bool placeCover = true;
        try
        {
            Check(PvpDuel.Start(1, 30, 0, 120, "use cover", out _, (scene, _) =>
            {
                last = scene;
                return Task.FromResult(placeCover && scene["cover_option"] != null
                    ? new PvpTactics.Decision("m0", "", "body", false, Cover: true) : new("m0", "ranged"));
            }), "cover fight failed to start");
            PvpDuel.Tick(self);
            Check(last["cover_option"] == null, "terrain mutations offered without the player's permission");
            PvpDuel.UpdateDirective(new JsonObject { ["session"] = PvpDuel.Session,
                ["constraints"] = new JsonObject { ["allow_placing"] = true } }, fromPlayer: true);
            PvpDuel.Tick(self);
            Check(last["cover_option"]?["x"]?.GetValue<int>() == 16, "safe cover cell was not offered");
            self.itemAnimation = 4;
            PvpDuel.Tick(self);
            Check(!PvpDuel.ExternalAction && !self.controlUseItem, "cover interrupted an active attack animation");
            self.itemAnimation = 0;
            PvpDuel.Tick(self);
            Check(PvpDuel.ExternalAction && ItemUseCoordinator.LastRequest.TargetWx == 16,
                "cover was not placed after the animation finished");
            Game.tile[16, row] = new Terraria.Tile { HasUnactuatedTile = true, TileType = 1 };
            placeCover = false;
            PvpDuel.Tick(self); PvpDuel.Tick(self);
            Check(!PvpDuel.ExternalAction && self.controlUseItem, "through-wall projectile did not attack from verified cover");
            Check(last["weapon_modes"]["ranged"]["aims"]["velocity"]["through_tiles"].GetValue<bool>(),
                "through-wall capability not exposed to Jev");
            Console.WriteLine("PASS: PvP guarded placement preserves attack animation and uses a through-wall weapon from cover");
        }
        finally
        {
            PvpDuel.Stop();
            Terraria.Collision.SolidTest = solid;
            Game.netMode = Terraria.ID.NetmodeID.SinglePlayer; Config.I.AiClient = false;
            Game.tile[16, row] = Game.tile[16, row - 1] = default;
            for (int x = 4; x < 26; x++) Game.tile[x, floor] = default;
            Terraria.ID.ContentSamples.ProjectilesByType.Remove(Terraria.ID.ProjectileID.DeathSickle);
            Terraria.ID.ContentSamples.ProjectilesByType.Remove(Terraria.ID.ProjectileID.Bullet);
            target.inventory[0] = new Terraria.Item();
        }
    }

    public static async Task Smoke()
    {
        string configPath = System.IO.Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.UserProfile),
            "Library/Application Support/Terraria/tModLoader/ModConfigs/TerraBlind_Config.json");
        var config = JsonNode.Parse(await System.IO.File.ReadAllTextAsync(configPath));
        string key = config["TypeSafeKey"]?.GetValue<string>();
        Check(!string.IsNullOrWhiteSpace(key), "No configured TypeSafe key for live smoke");
        using var http = new HttpClient { Timeout = TimeSpan.FromSeconds(5) };
        for (int i = 0; i < 3; i++)
        {
            var scene = new JsonObject
            {
                ["strategy"] = "持续追击目标，近距离用 Volcano 接触近战；利用跳跃、额外跳追上并躲避攻击。",
                ["movement_options"] = new JsonObject { ["stay"] = "保持", ["left"] = "向左小幅移动", ["right"] = "向右小幅移动",
                    ["up"] = "向上跳跃或使用额外跳", ["up_right"] = "向右跳跃，必要时使用额外跳或飞行" },
                ["weapon_modes"] = new JsonObject {
                    ["melee"] = new JsonObject { ["name"] = "Volcano", ["damage"] = 40, ["contact_reach_pixels"] = 60,
                        ["can_attack_now"] = i == 1, ["description"] = "接触近战，需要够到目标" },
                    ["ranged"] = new JsonObject { ["name"] = "Molten Fury", ["damage"] = 31, ["shoot_speed"] = 8,
                        ["can_attack_now"] = true, ["description"] = "弓箭，需要射界" } },
                ["current_mode"] = "melee",
                ["dx"] = i == 1 ? 30 : 400, ["dy"] = 0,
                ["self_hp"] = i == 1 ? 50 : 100, ["target_hp"] = 100,
                ["aim_clear"] = i != 2, ["target_vx"] = -3, ["target_vy"] = 0,
                ["previous_choice"] = "hold", ["moved_pixels"] = 0,
                ["nearby_projectiles"] = new JsonArray()
            };
            var clock = System.Diagnostics.Stopwatch.StartNew();
            var choice = await PvpTactics.Choose(http, key, scene, default);
            Check(scene["movement_options"].AsObject().ContainsKey(choice.Move), "invalid live decision");
            if (i != 1) Check(choice.WeaponMode == "ranged", "Live Jev kept out-of-reach melee despite an attackable ranged weapon");
            Console.WriteLine($"PVP_SMOKE scene={i} choice={choice.Move} mode={choice.WeaponMode} ms={clock.ElapsedMilliseconds} mock_world=true");
        }
        Console.WriteLine("PVP_SMOKE_COMPLETE checks=3 mock_world=true");
    }
