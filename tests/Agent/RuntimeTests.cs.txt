    static async Task PumpUntil(Func<bool> done)
    {
        var deadline = DateTime.UtcNow.AddSeconds(5);
        while (!done())
        {
            Check(DateTime.UtcNow < deadline, "main-thread test timed out");
            Agent.PostUpdateEverything();
            for (int frame = 0; frame < 10; frame++) PlayerCombat.Tick(Game.LocalPlayer);
            await Task.Delay(5);
        }
    }

    static async Task<T> Execute<T>(Task<T> task)
    {
        await PumpUntil(() => task.IsCompleted);
        return await task;
    }

    static void Receive(int type, int sender, Action<BinaryWriter> payload)
    {
        using var data = new MemoryStream();
        using var writer = new BinaryWriter(data, Encoding.UTF8, true);
        writer.Write((byte)type); payload(writer); writer.Flush(); data.Position = 0;
        using var reader = new BinaryReader(data);
        AgentNetwork.Receive(reader, sender);
    }

    static void Request(int sender, string command) => Receive(1, sender, writer => { writer.Write(1); writer.Write(command); });

        using var reader = new BinaryReader(new MemoryStream(packet));
        AgentNetwork.Receive(reader, 256);
    }

            {
                string code = emptyYield ? "return {status:'yield'};" : call == 0
                    ? "step('first',()=>tool('say',{text:'workflow-first'})); return {status:'yield',completed:1};"
                    : "step('first',()=>tool('say',{text:'workflow-first'})); step('second',()=>tool('say',{text:'workflow-second'})); return {status:'done'};";
                message["tool_calls"] = new JsonArray(new JsonObject { ["id"] = "workflow-" + call, ["type"] = "function",
                    ["function"] = new JsonObject { ["name"] = "run_script", ["arguments"] = new JsonObject
                        { ["workflow"] = "test-flow", ["code"] = code }.ToJsonString() } });
            }
            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK) { Content = new StringContent(
                new JsonObject { ["choices"] = new JsonArray(new JsonObject { ["message"] = message }) }.ToJsonString()) });
        }
    }

    sealed class NavHandler : HttpMessageHandler
    {
        int calls;
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken token)
        {
            string message = calls++ == 0
                ? new JsonObject { ["tool_calls"] = new JsonArray(new JsonObject { ["id"] = "nav", ["type"] = "function",
                    ["function"] = new JsonObject { ["name"] = "run_script", ["arguments"] = new JsonObject
                        { ["code"] = "return tool('nav_to',{x:100,y:100});" }.ToJsonString() } }) }.ToJsonString()
                : "{\"content\":\"新任务完成\"}";
            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK)
            { Content = new StringContent("{\"choices\":[{\"message\":" + message + "}]}", Encoding.UTF8, "application/json") });
        }
    }
