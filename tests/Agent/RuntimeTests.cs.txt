static class RuntimeTests
{
    static readonly NativeAgent Agent = new();
    static void Check(bool condition, string message)
    { if (!condition) throw new Exception(message); }

    static async Task PumpUntil(Func<bool> done)
    {
        var deadline = DateTime.UtcNow.AddSeconds(5);
        while (!done())
        {
            Check(DateTime.UtcNow < deadline, "main-thread test timed out");
            Agent.PostUpdateEverything();
            for (int frame = 0; frame < 10; frame++) PlayerCombat.Tick(Game.LocalPlayer);
            await Task.Delay(5);
        }
    }

    static async Task<T> Execute<T>(Task<T> task)
    {
        await PumpUntil(() => task.IsCompleted);
        return await task;
    }

    public static async Task Run()
    {
        Agent.Load();
        try
        {
            using var cancel = new CancellationTokenSource();
            bool executed = false;
            var queued = NativeAgent.OnMain(() => executed = true, cancel.Token);
            cancel.Cancel();
            Agent.PostUpdateEverything();
            Check(queued.IsCanceled && !executed, "cancelled main-thread callback executed");
            Console.WriteLine("PASS: cancelled queued game operation never executes");

            string previousKey = Config.I.AgentApiKey;
            Config.I.AgentApiKey = "";
            NativeAgent.Submit("检查配置失败");
            Agent.PostUpdateEverything();
            Check(NativeAgent.Status == "failed", "invalid configuration left task queued");
            Config.I.AgentApiKey = previousKey;
            NativeAgent.Stop();
            Console.WriteLine("PASS: invalid configuration terminates queued task as failed");

            HttpServerSystem.Handler = (path, payload) =>
            {
                Check(path == "/fight" && JsonNode.Parse(payload)["active"]?.GetValue<bool>() == false,
                    "death cleanup restarted fight instead of stopping it");
                return (200, "{\"active\":false}");
            };
            Game.LocalPlayer.dead = true;
            try
            {
                var interrupted = await Execute(AgentTools.Wait("/fight_active", "active", "/fight", 1, default));
                Check(interrupted["outcome"].ToString() == "interrupted_by_death", "death interruption was not reported");
            }
            finally { Game.LocalPlayer.dead = false; HttpServerSystem.Handler = null; }
            Console.WriteLine("PASS: death during fight sends explicit stop and reports interruption");

            int combatStops = Combat.Stops;
            FightCoordinator.Start(20);
            await Execute(AgentTools.Execute("set_combat", new JsonObject { ["auto_dodge"] = false, ["auto_attack"] = false }, default));
            Check(!AgentPermissions.Dodge && !AgentPermissions.Attack, "combat permission was not applied");
            Check(Combat.Stops == combatStops + 1 && !FightCoordinator.IsActive, "active attack was not stopped");
            int calls = HttpServerSystem.Calls.Count;
            string denied = await Execute(AgentTools.Execute("fight", new JsonObject(), default));
            Check(denied.Contains("error") && HttpServerSystem.Calls.Count == calls, "forbidden fight reached game dispatcher");
            await Execute(AgentTools.Execute("set_combat", new JsonObject { ["auto_dodge"] = true }, default));
            Check(AgentPermissions.Dodge && !AgentPermissions.Attack, "partial permission update changed unrelated permission");
            Console.WriteLine("PASS: combat denial stops attack, rejects fight and preserves partial updates");

            Game.LocalPlayer.inventory[0].Name = "木材";
            Game.LocalPlayer.inventory[0].stack = 20;
            HttpServerSystem.Handler = (path, payload) =>
            {
                Check(path == "/craft", "unexpected craft endpoint");
                var args = JsonNode.Parse(payload);
                Check(args["item_name"].ToString() == "工作台" && args["amount"].GetValue<int>() == 1, "craft arguments changed");
                Game.LocalPlayer.inventory[0].stack = 10;
                Game.LocalPlayer.inventory[1].Name = "工作台";
                Game.LocalPlayer.inventory[1].stack = 1;
                return (200, "{\"ok\":true}");
            };
            var crafted = JsonNode.Parse(await Execute(AgentTools.Execute("craft", new JsonObject { ["name"] = "工作台" }, default)));
            Check(crafted["got"]["工作台"].GetValue<int>() == 1 && crafted["lost"]["木材"].GetValue<int>() == 10, "tool result did not measure inventory delta");
            Check(crafted["state"]["pos"]["x"].GetValue<int>() == 12 && crafted["state"]["pos"]["y"].GetValue<int>() == 34, "tool position is not taken from player origin");
            Console.WriteLine("PASS: craft adapter reports measured inventory delta and player origin");

            Game.tile[100, 100] = new Terraria.Tile { HasUnactuatedTile = true, TileType = 0 };
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/nav_recede")
                {
                    var request = JsonNode.Parse(payload);
                    bool solid = Game.tile[100, 100].TileType == 1;
                    Check(request["exact"].GetValue<bool>() == solid && request["reach"].GetValue<bool>() != solid,
                        "navigation mode does not match live target solidity");
                }
                return (200, path == "/nav_recede_done" ? "{\"done\":true}" : "{\"accepted\":true}");
            };
            var adapted = await Execute(AgentTools.Navigate(new JsonObject { ["x"] = 100, ["y"] = 100, ["exact"] = true }, default));
            Check(adapted["mode_adjustment"] != null && adapted["done"].GetValue<bool>(), "non-solid target did not use interaction reach");
            Game.tile[100, 100] = new Terraria.Tile { HasUnactuatedTile = true, TileType = 1 };
            var solidResult = await Execute(AgentTools.Navigate(new JsonObject { ["x"] = 100, ["y"] = 100, ["exact"] = true }, default));
            Check(solidResult["mode_adjustment"] == null && !solidResult["done"].GetValue<bool>(), "solid mining target changed modes or ignored arrival check");
            Game.tile[100, 100] = default;
            Console.WriteLine("PASS: live non-solid targets switch exact mining to interaction reach; solid targets retain mining and arrival checks");

            Game.tile[100, 100] = new Terraria.Tile { HasUnactuatedTile = true, TileType = 0 };
            var noTool = JsonNode.Parse(await Execute(AgentTools.Execute("mine", new JsonObject { ["dir"] = "left", ["target_x"] = 100, ["target_y"] = 100 }, default)));
            Check(noTool["error"]?.ToString() == "no_collect_tool", "non-solid mining succeeded without a tool");
            Game.LocalPlayer.inventory[6] = new Terraria.Item { stack = 1, pick = 50 };
            HttpServerSystem.Handler = (path, payload) =>
            {
                Check(path is "/item_use" or "/item_use_status", "non-solid collection used tunnel mining");
                if (path == "/item_use")
                {
                    var request = JsonNode.Parse(payload);
                    Check(request["slot"].GetValue<int>() == 6 && request["strict"].GetValue<bool>(), "collection failed to select real tool and strict coordinates");
                }
                return (200, path == "/item_use_status" ? "{\"active\":false,\"outcome\":\"removed\"}" : "{\"ok\":true}");
            };
            var collected = JsonNode.Parse(await Execute(AgentTools.Execute("mine", new JsonObject { ["dir"] = "left", ["target_x"] = 100, ["target_y"] = 100 }, default)));
            Check(collected["outcome"].ToString() == "removed" && collected["mode_adjustment"] != null, "non-solid collection invented tunnel completion");
            Game.LocalPlayer.inventory[6] = new Terraria.Item(); Game.tile[100, 100] = default;
            Console.WriteLine("PASS: non-solid mining selects an actual collect tool, rejects missing tools and preserves measured removal outcome");

            using var http = new HttpClient(new NavHandler());
            ((HttpClient)typeof(NativeAgent).GetField("_http", BindingFlags.Instance | BindingFlags.NonPublic).GetValue(Agent)).Dispose();
            typeof(NativeAgent).GetField("_http", BindingFlags.Instance | BindingFlags.NonPublic).SetValue(Agent, http);
            typeof(NativeAgent).GetField("_conversation", BindingFlags.Instance | BindingFlags.NonPublic).SetValue(Agent, new AgentConversation(http));
            HttpServerSystem.Handler = (path, payload) =>
            {
                if (path == "/nav_recede") RecedingNav.Active = true;
                return (200, path == "/nav_recede_done" ? "{\"done\":false}" : "{\"accepted\":true}");
            };
            int navCallsBefore = HttpServerSystem.Calls.Count(p => p == "/nav_recede");
            NativeAgent.Submit("移动");
            await PumpUntil(() => RecedingNav.Active);
            int stops = Halt.Stops;
            NativeAgent.Submit("换个任务");
            Check(!RecedingNav.Active && Halt.Stops == stops + 1, "replacement task did not stop old navigation");
            await PumpUntil(() => AgentChat.Messages.Contains("新任务完成"));
            Check(HttpServerSystem.Calls.Count(p => p == "/nav_recede") == navCallsBefore + 1, "old navigation restarted after cancellation");
            Console.WriteLine("PASS: replacement instruction cancels active navigation and completes next task");

            await PumpUntil(() => NativeAgent.Status == "idle");
            var pauseHandler = new ContinuationLimitHandler();
            using var pauseHttp = new HttpClient(pauseHandler);
            var conversationField = typeof(NativeAgent).GetField("_conversation", BindingFlags.Instance | BindingFlags.NonPublic);
            var originalConversation = conversationField.GetValue(Agent);
            conversationField.SetValue(Agent, new AgentConversation(pauseHttp));
            await Execute(AgentTools.Execute("task_plan", JsonNode.Parse("""
                {"goal":"unfinished goal","steps":[{"id":"health","description":"300 permanent health","conditions":[{"kind":"permanent_health","count":300}]}]}
                """).AsObject(), default));
            NativeAgent.Submit("继续未完成目标");
            await PumpUntil(() => NativeAgent.Status == "paused");
            var paused = JsonNode.Parse(await Execute(AgentTools.State(default)));
            Check(paused["task_plan"]["current_step"].ToString() == "health" && AgentTools.Plan.Incomplete,
                "model final reply discarded unfinished task or marked it completed");
            Check(pauseHandler.Calls == 2, "no-progress continuation did not stop after one extra turn");
            conversationField.SetValue(Agent, originalConversation);
            AgentTools.NewTask();
            Check(AgentTools.Plan.Incomplete, "continuation cleared task cursor");
            AgentTools.Plan.Reset();
            NativeAgent.Stop();
            Console.WriteLine("PASS: model ending does not complete a registered goal; follow-up retains task cursor");

            using (var continuationHttp = new HttpClient(new ContinuationHandler()))
            {
                var field = typeof(NativeAgent).GetField("_conversation", BindingFlags.Instance | BindingFlags.NonPublic);
                var previousConversation = field.GetValue(Agent);
                field.SetValue(Agent, new AgentConversation(continuationHttp));
                try
                {
                    Game.LocalPlayer.statLifeMax = 300;
                    await Execute(AgentTools.Execute("task_plan", JsonNode.Parse("""
                        {"goal":"two stages","steps":[
                          {"id":"first","description":"first milestone","conditions":[{"kind":"permanent_health","count":300}]},
                          {"id":"last","description":"final requirements","conditions":[{"kind":"permanent_health","count":300}]}]}
                        """).AsObject(), default));
                    NativeAgent.Submit("执行两个阶段");
                    await PumpUntil(() => NativeAgent.Status == "idle");
                    Check(AgentTools.Plan.Snapshot()["status"].ToString() == "completed", "host did not continue after verified progress");
                }
                finally
                {
                    field.SetValue(Agent, previousConversation);
                    Game.LocalPlayer.statLifeMax = 100;
                    AgentTools.Plan.Reset();
                }
            }
            Console.WriteLine("PASS: verified progress resumes the next model turn and finishes both steps without another player message");
            await TestPartialProgress();
            await TestToolFailureEvidence();

            await TestNetwork();
            TestRoleSwitch();
            TestDefaultOwner();
            AgentPermissions.Dodge = AgentPermissions.Attack = false;
            Agent.OnWorldUnload();
            Check(AgentPermissions.Dodge && AgentPermissions.Attack, "world unload did not reset permissions");
            Console.WriteLine("PASS: world unload resets combat permissions");
            await TestPvp();
        }
        finally { Agent.Unload(); }
    }

    static async Task TestToolFailureEvidence()
    {
        await Execute(AgentTools.Execute("task_plan", JsonNode.Parse("""
            {"goal":"failure evidence","steps":[{"id":"health","description":"300 health","conditions":[{"kind":"permanent_health","count":300}]}]}
            """).AsObject(), default));
        try
        {
            HttpServerSystem.Handler = (_, _) => throw new IOException("fixture dispatch failed");
            var result = JsonNode.Parse(await Execute(AgentTools.Execute("craft", new JsonObject { ["name"] = "fixture" }, default)));
            Check(result["error"]?.ToString() == "tool_exception" && result["detail"].ToString().Contains("fixture dispatch failed"),
                "tool exception was not returned with actionable detail");
            Check(result["task_plan"]["steps"][0]["blocking_evidence"]["tool"].ToString() == "craft", "tool exception evidence not retained in plan");
            await Execute(AgentTools.Execute("task_step", new JsonObject { ["id"] = "health", ["action"] = "block", ["reason"] = "fixture dispatch failed" }, default));
            int stops = 0;
            HttpServerSystem.Handler = (path, _) =>
            {
                if (path == "/item_use") ItemUseCoordinator.IsActive = true;
                if (path == "/item_use_status") throw new IOException("fixture status failed");
                if (path == "/item_use_stop") { stops++; ItemUseCoordinator.IsActive = false; }
                return (200, "{\"ok\":true}");
            };
            var interrupted = JsonNode.Parse(await Execute(AgentTools.Execute("use_item", new JsonObject { ["slot"] = 0, ["x"] = 0, ["y"] = 0 }, default)));
            Check(stops == 1 && !ItemUseCoordinator.IsActive && interrupted["detail"].ToString().Contains("fixture status failed"),
                "failed polling left its action active or lost original error");
            HttpServerSystem.Handler = (path, _) =>
            {
                if (path == "/item_use") ItemUseCoordinator.IsActive = true;
                if (path == "/item_use_status") return (500, "{\"error\":\"fixture status rejected\"}");
                if (path == "/item_use_stop") { stops++; ItemUseCoordinator.IsActive = false; }
                return (200, "{\"ok\":true}");
            };
            var rejectedStatus = JsonNode.Parse(await Execute(AgentTools.Execute("use_item", new JsonObject { ["slot"] = 0, ["x"] = 0, ["y"] = 0 }, default)));
            Check(stops == 2 && !ItemUseCoordinator.IsActive && rejectedStatus["error"]?.ToString() == "tool_exception"
                && rejectedStatus["detail"].ToString().Contains("fixture status rejected"), "status rejection appeared as normal completion or left action running");
            HttpServerSystem.Handler = (path, _) => throw new IOException(path == "/item_use_stop" ? "fixture cleanup failed" : "fixture status failed");
            var cleanupFailure = await Execute(AgentTools.Wait("/item_use_status", "active", "/item_use_stop", 1, default).ContinueWith(task => task.Exception, TaskScheduler.Default));
            Check(cleanupFailure != null && cleanupFailure.ToString().Contains("fixture cleanup failed") && cleanupFailure.ToString().Contains("fixture status failed"),
                "cleanup failure hid original polling error");
            HttpServerSystem.Handler = (path, _) => (500, path == "/item_use_stop"
                ? "{\"error\":\"fixture cleanup rejected\"}" : "{\"error\":\"fixture status rejected\"}");
            var rejectedCleanup = await Execute(AgentTools.Wait("/item_use_status", "active", "/item_use_stop", 1, default).ContinueWith(task => task.Exception, TaskScheduler.Default));
            Check(rejectedCleanup != null && rejectedCleanup.ToString().Contains("fixture cleanup rejected") && rejectedCleanup.ToString().Contains("fixture status rejected"),
                "rejected cleanup was reported as successful");
            string before = AgentTools.Plan.Snapshot().ToJsonString();
            using var cancel = new CancellationTokenSource(); cancel.Cancel();
            try { await Execute(AgentTools.Execute("craft", new JsonObject { ["name"] = "stale action" }, cancel.Token)); throw new Exception("cancelled tool ran"); }
            catch (OperationCanceledException) { }
            Check(AgentTools.Plan.Snapshot().ToJsonString() == before, "cancelled tool contaminated task evidence");
        }
        finally { HttpServerSystem.Handler = null; AgentTools.Plan.Reset(); }
        Console.WriteLine("PASS: tool exceptions retain evidence and stop failed polling actions; cleanup errors remain visible and cancellation stays cancellation");
    }

    static async Task TestPartialProgress()
    {
        var handler = new PartialProgressHandler();
        using var http = new HttpClient(handler);
        var field = typeof(NativeAgent).GetField("_conversation", BindingFlags.Instance | BindingFlags.NonPublic);
        var previous = field.GetValue(Agent);
        field.SetValue(Agent, new AgentConversation(http));
        try
        {
            Game.LocalPlayer.statLifeMax = 100;
            await Execute(AgentTools.Execute("task_plan", JsonNode.Parse("""
                {"goal":"partial progress","steps":[{"id":"health","description":"160 permanent health","conditions":[{"kind":"permanent_health","count":160}]}]}
                """).AsObject(), default));
            HttpServerSystem.Handler = (path, payload) =>
            {
                Check(path is "/item_use" or "/item_use_status", "unexpected partial-progress endpoint");
                if (path == "/item_use") Game.LocalPlayer.statLifeMax += 20;
                return (200, path == "/item_use_status" ? "{\"active\":false,\"outcome\":\"n/a\"}" : "{\"ok\":true}");
            };
            NativeAgent.Submit("逐步提高永久血量到160");
            await PumpUntil(() => handler.Calls >= 2 && NativeAgent.Status is "paused" or "idle");
            Check(Game.LocalPlayer.statLifeMax == 160 && AgentTools.Plan.Snapshot()["status"].ToString() == "completed",
                $"partial progress stopped early: hp={Game.LocalPlayer.statLifeMax} calls={handler.Calls} status={NativeAgent.Status}");
            Check(handler.Calls == 6, "partial progress caused extra model turns");
        }
        finally
        {
            NativeAgent.Stop(); field.SetValue(Agent, previous);
            Game.LocalPlayer.statLifeMax = 100; AgentTools.Plan.Reset(); HttpServerSystem.Handler = null;
        }
        Console.WriteLine("PASS: measured progress within one step resumes three turns without another player instruction");
    }

    static void Receive(int type, int sender, Action<BinaryWriter> payload)
    {
        using var data = new MemoryStream();
        using var writer = new BinaryWriter(data, Encoding.UTF8, true);
        writer.Write((byte)type); payload(writer); writer.Flush(); data.Position = 0;
        using var reader = new BinaryReader(data);
        AgentNetwork.Receive(reader, sender);
    }

    static void Request(int sender, string command) => Receive(1, sender, writer => { writer.Write(1); writer.Write(command); });

    static void DeliverToAi(byte[] packet)
    {
        Game.netMode = NetmodeID.MultiplayerClient; Game.dedServ = false; Config.I.AiClient = true;
        using var reader = new BinaryReader(new MemoryStream(packet));
        AgentNetwork.Receive(reader, 256);
    }

    static async Task TestNetwork()
    {
        var network = new AgentNetwork();
        network.OnWorldLoad();
        Game.netMode = NetmodeID.Server; Game.dedServ = true;
        ModPacket.Sent.Clear();
        Receive(0, 1, _ => { });
        Request(0, "bind");
        var bound = ModPacket.Sent.Last();
        using (var reader = new BinaryReader(new MemoryStream(bound.Data)))
            Check(bound.Recipient == 1 && reader.ReadByte() == 2 && reader.ReadInt32() == 0 && reader.ReadString() == "status", "binding was not sent to AI client with real owner");
        DeliverToAi(bound.Data);
        Game.netMode = NetmodeID.Server; Game.dedServ = true;
        Request(2, "task 捣乱");
        Check(ModPacket.Sent.Last().Recipient == 2 && ModPacket.Sent.Last().Data[0] == 4, "unbound sender was not rejected");
        string goal = "task 先别合成，检查一下背包";
        Request(0, goal);
        var forwarded = ModPacket.Sent.Last();
        using (var reader = new BinaryReader(new MemoryStream(forwarded.Data)))
            Check(forwarded.Recipient == 1 && reader.ReadByte() == 2 && reader.ReadInt32() == 0 && reader.ReadString() == goal, "owner task text was changed or misrouted");
        int messages = AgentChat.Messages.Count;
        DeliverToAi(forwarded.Data);
        await PumpUntil(() => AgentChat.Messages.Count > messages && AgentChat.Messages.Last() == "新任务完成");
        ModPacket.Sent.Clear();
        Check(AgentNetwork.RelayAgentReply(new string('测', 2000)), "AI reply was not relayed");
        Check(ModPacket.Sent.Count == 3, "long reply was not split within packet limit");
        string joined = "";
        foreach (var packet in ModPacket.Sent)
        {
            using var reader = new BinaryReader(new MemoryStream(packet.Data));
            Check(reader.ReadByte() == 3, "AI reply used wrong packet type");
            string chunk = reader.ReadString();
            Check(chunk.Length <= 900, "reply chunk exceeds server limit");
            joined += chunk;
        }
        Check(joined == new string('测', 2000), "reply splitting lost text");
        Game.netMode = NetmodeID.Server; Game.dedServ = true;
        Receive(3, 1, writer => writer.Write("任务完成"));
        var reply = ModPacket.Sent.Last();
        using (var reader = new BinaryReader(new MemoryStream(reply.Data)))
            Check(reply.Recipient == 0 && reader.ReadByte() == 4 && reader.ReadString() == "AI[1] 任务完成", "AI reply did not return to owner");
        AgentNetwork.Disconnect(0);
        var stopped = ModPacket.Sent.Last();
        using (var reader = new BinaryReader(new MemoryStream(stopped.Data)))
            Check(stopped.Recipient == 1 && reader.ReadByte() == 2 && reader.ReadInt32() == -1 && reader.ReadString() == "stop", "owner disconnect did not stop AI");
        DeliverToAi(stopped.Data);
        Check(Manual.On && !Combat.Enabled && !Dodge.Enabled, "AI client did not apply disconnect stop");
        Check(!AgentNetwork.RelayAgentReply("不应转发"), "unbound AI client continued relaying");
        Game.netMode = NetmodeID.Server; Game.dedServ = true;
        int packets = ModPacket.Sent.Count;
        Receive(3, 1, writer => writer.Write("不应转发"));
        Check(ModPacket.Sent.Count == packets, "reply was relayed after owner disconnected");
        Game.netMode = NetmodeID.MultiplayerClient; Game.dedServ = false; Config.I.AiClient = false;
        Game.LocalPlayer.active = true;
        new AiCommand().Action(new CommandCaller(), "/ai list", new[] { "list" });
        Check(Game.Chat.Last().Contains("查询"), "list command had no immediate acknowledgement");
        var listPacket = ModPacket.Sent.Last();
        Game.netMode = NetmodeID.Server; Game.dedServ = true;
        using (var reader = new BinaryReader(new MemoryStream(listPacket.Data))) AgentNetwork.Receive(reader, 0);
        var listReply = ModPacket.Sent.Last();
        Check(listReply.Recipient == 0 && listReply.Data[0] == 4, "server did not reply to list request");
        Game.netMode = NetmodeID.MultiplayerClient; Game.dedServ = false;
        using (var reader = new BinaryReader(new MemoryStream(listReply.Data))) AgentNetwork.Receive(reader, 256);
        int chatCount = Game.Chat.Count;
        Game.GameUpdateCount += 600; network.PostUpdateEverything();
        Check(Game.Chat.Count == chatCount, "completed list request reported timeout");
        AgentNetwork.Request(-1, "list"); Game.GameUpdateCount += 600; network.PostUpdateEverything();
        Check(Game.Chat.Last().Contains("10秒"), "missing server reply stayed silent");
        Game.netMode = NetmodeID.SinglePlayer; Game.dedServ = false; Config.I.AiClient = false;
        Console.WriteLine("PASS: production packet handlers enforce binding, AI execution, reply splitting/routing and disconnect stop");
    }

    static void TestRoleSwitch()
    {
        var network = new AgentNetwork();
        network.OnWorldLoad();
        Game.netMode = NetmodeID.MultiplayerClient; Config.I.AiClient = true;
        ModPacket.Sent.Clear();
        network.PostUpdateEverything();
        Check(ModPacket.Sent.Count == 1 && ModPacket.Sent[0].Data[0] == 0, "enabling AI did not register immediately");
        Game.netMode = NetmodeID.Server; Game.dedServ = true;
        Receive(0, 1, _ => { });
        Request(0, "bind");
        DeliverToAi(ModPacket.Sent.Last().Data);
        Combat.Enabled = Dodge.Enabled = RecedingNav.Active = true;
        Config.I.AiClient = false;
        network.PostUpdateEverything();
        Check(!Combat.Enabled && !Dodge.Enabled && !RecedingNav.Active, "disabling AI left active operations");
        var unregister = ModPacket.Sent.Last();
        Check(unregister.Data[0] == 5, "disabling AI did not unregister");
        Game.netMode = NetmodeID.Server; Game.dedServ = true;
        using (var reader = new BinaryReader(new MemoryStream(unregister.Data))) AgentNetwork.Receive(reader, 1);
        Request(0, "status");
        using (var reader = new BinaryReader(new MemoryStream(ModPacket.Sent.Last().Data)))
            Check(reader.ReadByte() == 4 && reader.ReadString().Contains("不在线"), "disabled AI still registered on server");
        Game.netMode = NetmodeID.MultiplayerClient; Game.dedServ = false; Config.I.AiClient = true;
        network.PostUpdateEverything();
        Check(ModPacket.Sent.Last().Data[0] == 0 && !AgentNetwork.RelayAgentReply("旧主人"), "reenabling retained old binding or failed to register");
        Game.netMode = NetmodeID.Server; Game.dedServ = true;
        Receive(0, 1, _ => { });
        Request(0, "task 不该执行");
        using (var reader = new BinaryReader(new MemoryStream(ModPacket.Sent.Last().Data)))
            Check(reader.ReadByte() == 4 && reader.ReadString().Contains("没有绑定"), "reenabled AI retained server owner");
        Game.netMode = NetmodeID.SinglePlayer; Game.dedServ = false; Config.I.AiClient = false;
        Console.WriteLine("PASS: live AI toggle registers, stops, unregisters and requires fresh binding");
    }

    static void TestDefaultOwner()
    {
        var network = new AgentNetwork();
        string previousName = Game.player[0].name;
        Game.netMode = NetmodeID.Server; Game.dedServ = true;
        network.OnWorldLoad();
        try
        {
            Game.player[0].name = "not_the_owner";
            Game.player[0].active = Game.player[1].active = true;
            ModPacket.Sent.Clear();
            Receive(0, 1, _ => { });
            Check(ModPacket.Sent.Count == 0, "bound to an arbitrary player before Reisenberg joined");
            Game.player[0].name = "Reisenberg";
            network.PostUpdateEverything();
            using (var reader = new BinaryReader(new MemoryStream(ModPacket.Sent.Last().Data)))
                Check(reader.ReadByte() == 2 && reader.ReadInt32() == 0 && reader.ReadString() == "status", "late default owner did not auto-bind");
            Request(0, "unbind");
            int sent = ModPacket.Sent.Count;
            network.PostUpdateEverything();
            Check(ModPacket.Sent.Count == sent, "manual unbind was immediately overridden");
            Receive(5, 1, _ => { }); Receive(0, 1, _ => { });
            using (var reader = new BinaryReader(new MemoryStream(ModPacket.Sent.Last().Data)))
                Check(reader.ReadByte() == 2 && reader.ReadInt32() == 0, "new AI registration did not default to Reisenberg");
            Game.player[0].active = false;
            AgentNetwork.Disconnect(0);
            using (var reader = new BinaryReader(new MemoryStream(ModPacket.Sent.Last().Data)))
                Check(reader.ReadByte() == 2 && reader.ReadInt32() == -1 && reader.ReadString() == "stop", "owner disconnect did not stop AI");
            Game.player[0].active = true; network.PostUpdateEverything();
            using (var reader = new BinaryReader(new MemoryStream(ModPacket.Sent.Last().Data)))
                Check(reader.ReadByte() == 2 && reader.ReadInt32() == 0 && reader.ReadString() == "status", "owner rejoin did not restore binding");
        }
        finally
        {
            network.OnWorldLoad(); Game.player[0].name = previousName;
            Game.netMode = NetmodeID.SinglePlayer; Game.dedServ = false;
        }
        Console.WriteLine("PASS: default Reisenberg owner binds on registration/join, honors unbind and reconnects safely");
    }

    static async Task TestPvp()
    {
        Game.netMode = NetmodeID.MultiplayerClient; Config.I.AiClient = true;
        var self = Game.LocalPlayer;
        var target = Game.player[1];
        foreach (var other in Game.player) other.active = false;
        self.active = target.active = true;

        Check(!PlayerCombat.Start(1, 1, 80, out var reason) && reason == "target_pvp_disabled" && !self.hostile, "disabled target PvP was ignored");
        target.hostile = true; self.team = target.team = 2;
        Check(!PlayerCombat.Start(1, 1, 80, out reason) && reason == "same_team", "same-team PvP was allowed");
        self.team = 0;
        Check(!PlayerCombat.Start(0, 1, 80, out reason) && reason == "self_target", "self attack was allowed");
        Check(!PlayerCombat.Start(Game.maxPlayers, 1, 80, out reason) && reason == "invalid_player", "invalid player ID was allowed");
        Check(!PlayerCombat.Start(1, 1, 80, out reason) && reason == "no_weapon_in_hotbar", "attack without weapon was allowed");
        self.inventory[2] = new Terraria.Item { stack = 1, damage = 20, useStyle = 1, pick = 50 };
        self.inventory[3] = new Terraria.Item { stack = 1, damage = 20, useStyle = 1 };
        self.selectedItem = 2;
        AgentPermissions.Attack = false;
        Check(!PlayerCombat.Start(1, 1, 80, out reason) && reason == "attack_forbidden", "PvP bypassed attack permission");
        AgentPermissions.Attack = true;
        Check(PlayerCombat.Start(1, 1, 80, out reason), "valid PvP start failed: " + reason);
        Check(self.hostile && Terraria.NetMessage.Sent.Last() == (30, self.whoAmI), "self PvP toggle was not synced");
        PlayerCombat.Tick(self);
        Check(self.controlUseItem && self.selectedItem == 3 && Cursor.X == 110 && Cursor.Y == 40, "PvP did not aim at player center with a weapon");
        target.hostile = false; PlayerCombat.Tick(self);
        Check(!PlayerCombat.Active && !self.controlUseItem && PlayerCombat.Outcome == "target_pvp_disabled", "target disabling PvP did not release attack");
        target.hostile = true;
        PlayerCombat.Start(1, 1, 80, out _);
        target.dead = true; PlayerCombat.Tick(self);
        Check(!PlayerCombat.Active && PlayerCombat.Outcome == "target_dead", "dead target did not stop attack");
        target.dead = false;
        PlayerCombat.Start(1, 1, 80, out _);
        target.position = new System.Numerics.Vector2(5000, 0); PlayerCombat.Tick(self);
        Check(!PlayerCombat.Active && PlayerCombat.Outcome == "out_of_range", "out-of-range target did not stop attack");
        target.position = new System.Numerics.Vector2(100, 20);
        var state = JsonNode.Parse(await Execute(AgentTools.State(default)));
        Check(state["other_players"].AsArray().Count == 1 && state["other_players"][0]["id"].GetValue<int>() == 1
            && state["other_players"][0]["pvp"].GetValue<bool>() && state["agent"]["player_id"].GetValue<int>() == 0,
            "agent state does not expose real player identity and PvP status");
        var result = JsonNode.Parse(await Execute(AgentTools.Execute("fight_player", JsonNode.Parse("{\"player_id\":1,\"seconds\":1}").AsObject(), default)));
        Check(result["outcome"].ToString() == "duration_complete" && result["target_hp_before"].GetValue<int>() == target.statLife
            && result["target_hp_after"].GetValue<int>() == target.statLife && !PlayerCombat.Active && !self.controlUseItem,
            "PvP tool did not finish or invented damage: " + result.ToJsonString());
        PlayerCombat.Start(1, 1, 80, out _); PlayerCombat.Tick(self); AgentTools.Stop();
        Check(!PlayerCombat.Active && !self.controlUseItem, "agent stop left PvP firing");
        Game.netMode = NetmodeID.SinglePlayer; Config.I.AiClient = false;
        Console.WriteLine("PASS: production PvP validates players/teams/permissions, syncs toggle, aims, stops and reports actual HP");
    }

    sealed class ContinuationLimitHandler : HttpMessageHandler
    {
        public int Calls;
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken token)
        {
            Check(++Calls <= 2, "unbounded continuation without progress");
            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK)
            { Content = new StringContent("{\"choices\":[{\"message\":{\"content\":\"准备执行\"}}]}") });
        }
    }

    sealed class ContinuationHandler : HttpMessageHandler
    {
        int calls;
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken token)
        {
            int call = calls++;
            Check(call < 4, "host retried without verified progress");
            string id = call < 2 ? "first" : "last";
            var message = new JsonObject { ["content"] = "阶段结束" };
            if (call % 2 == 0)
                message["tool_calls"] = new JsonArray(new JsonObject
                {
                    ["id"] = "step-" + call, ["type"] = "function",
                    ["function"] = new JsonObject { ["name"] = "task_step",
                        ["arguments"] = new JsonObject { ["id"] = id, ["action"] = "verify" }.ToJsonString() }
                });
            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK)
            { Content = new StringContent(new JsonObject { ["choices"] = new JsonArray(new JsonObject
                { ["message"] = message }) }.ToJsonString(), Encoding.UTF8, "application/json") });
        }
    }

    sealed class PartialProgressHandler : HttpMessageHandler
    {
        public int Calls;
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken token)
        {
            int call = Calls++;
            Check(call < 6, "partial-progress loop exceeded expected budget");
            var message = new JsonObject { ["content"] = "本轮结束" };
            if (call % 2 == 0)
            {
                var calls = new JsonArray(new JsonObject { ["id"] = "use-" + call, ["type"] = "function",
                    ["function"] = new JsonObject { ["name"] = "use_item", ["arguments"] = "{\"x\":0,\"y\":0,\"slot\":0}" } });
                if (call == 4) calls.Add(new JsonObject { ["id"] = "verify-final", ["type"] = "function",
                    ["function"] = new JsonObject { ["name"] = "task_step", ["arguments"] = "{\"id\":\"health\",\"action\":\"verify\"}" } });
                message["tool_calls"] = calls;
            }
            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK)
            { Content = new StringContent(new JsonObject { ["choices"] = new JsonArray(new JsonObject { ["message"] = message }) }.ToJsonString()) });
        }
    }

    }
