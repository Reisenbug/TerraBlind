using System;
using System.Text.Json.Nodes;
using TerraBlind;
using Terraria;
using Terraria.ID;

static class TaskPlanTests
{
    static void Check(bool ok, string message) { if (!ok) throw new Exception(message); }
    static JsonObject Parse(string text) => JsonNode.Parse(text).AsObject();

    public static void Run()
    {
        var plan = new AgentTaskPlan();
        var hook = new AgentTaskEvidence();
        Main.netMode = NetmodeID.SinglePlayer;
        AgentTaskEvidence.Reset();
        hook.OnKill(new NPC { type = 222 });
        var request = Parse("""
        {"goal":"test long task","steps":[
          {"id":"health","description":"increase permanent health","conditions":[{"kind":"permanent_health","count":300}]},
          {"id":"weapon","description":"obtain weapon","conditions":[{"kind":"item","type":191,"count":1}]},
          {"id":"finish","description":"all final requirements","conditions":[{"kind":"permanent_health","count":300},{"kind":"item","type":191,"count":1},{"kind":"npc_kills","type":222,"count":1}]}
        ]}
        """);
        plan.Create(request, AgentTaskEvidence.Measure);
        Main.LocalPlayer.statLifeMax = 100;
        Main.LocalPlayer.statLifeMax2 = 400;
        Check(plan.Advance("health", AgentTaskEvidence.Measure)["verified"].GetValue<bool>() == false,
            "temporary buff health completed permanent-health goal");
        int progress = plan.Progress;
        Main.LocalPlayer.statLifeMax = 120; plan.Observe(AgentTaskEvidence.Measure);
        Check(plan.Progress == progress + 1 && plan.Cursor == 0, "partial health progress was not measured or completed step prematurely");
        Main.LocalPlayer.statLifeMax = 100; plan.Observe(AgentTaskEvidence.Measure);
        Main.LocalPlayer.statLifeMax = 120; plan.Observe(AgentTaskEvidence.Measure);
        Check(plan.Progress == progress + 1, "repeated gain/loss inflated progress");
        Main.LocalPlayer.statLifeMax = 300; plan.Observe(AgentTaskEvidence.Measure);
        progress = plan.Progress;
        Main.LocalPlayer.statLifeMax = 400; plan.Observe(AgentTaskEvidence.Measure);
        Check(plan.Progress == progress && plan.Incomplete, "surplus resources inflated progress or observation completed the plan");
        try { plan.Advance("finish", AgentTaskEvidence.Measure); throw new Exception("skipped steps"); }
        catch (InvalidOperationException) { }
        Main.LocalPlayer.statLifeMax = 300;
        plan.Advance("health", AgentTaskEvidence.Measure);
        Main.LocalPlayer.inventory[5] = new Item { type = 999, Name = "same name", stack = 1 };
        Check(!plan.Advance("weapon", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "wrong type counted");
        Main.LocalPlayer.inventory[5].type = 191;
        plan.Advance("weapon", AgentTaskEvidence.Measure);
        Check(!plan.Advance("finish", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "old kill counted");
        plan.Record("find_tiles", "{\"tiles\":[{\"x\":10,\"y\":20}]}");
        try { plan.Block("finish", "not sure about these candidates"); throw new Exception("uncertainty counted as blocker"); }
        catch (InvalidOperationException) { }
        plan.Record("find_tiles", "{\"error\":\"unknown_tile_name\"}");
        try { plan.Block("finish", "spelled the query wrong"); throw new Exception("query typo counted as blocker"); }
        catch (InvalidOperationException) { }
        plan.Record("nav_to", "{\"status\":\"failed\",\"reason\":\"no_path\"}", true);
        plan.Record("nav_to", "{\"status\":\"failed\",\"reason\":\"no_path\",\"task_plan\":{\"steps\":[\"old plan\"]}}", true);
        Check(!plan.Snapshot()["steps"][2]["last_action"]["result"].ToString().Contains("task_plan"), "action recursively retained its own task plan");
        plan.Block("finish", "no_path");
        plan.Record("mine", "{\"outcome\":\"removed\",\"got\":{\"resource\":1}}", true);
        Check(plan.Snapshot()["status"].ToString() == "active", "successful recovery left current step blocked");
        Check(plan.Snapshot()["steps"][2]["previous_failure"]["reason"].ToString() == "no_path"
            && plan.Snapshot()["steps"][2]["blocking_evidence"] == null, "recovery lost failure history or retained stale blocker");
        plan.Record("nav_to", "{\"status\":\"failed\",\"reason\":\"no_path\"}", true);
        plan.Block("finish", "no_path");
        try { plan.Create(request, AgentTaskEvidence.Measure); throw new Exception("lost active plan"); }
        catch (InvalidOperationException) { }
        var snapshot = plan.Snapshot();
        snapshot["steps"][2]["failure"] = "modified externally";
        Check(plan.Snapshot()["steps"][2]["failure"].ToString() == "no_path", "snapshot aliases mutable plan");
        Check(plan.Snapshot()["steps"][2]["last_action"]["result"].ToString().Contains("no_path"), "failure evidence lost");
        hook.OnKill(new NPC { type = 222 });
        Main.LocalPlayer.inventory[5].stack = 0;
        Check(!plan.Advance("finish", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "spent final weapon ignored");
        Main.LocalPlayer.inventory[5].stack = 1;
        plan.Advance("finish", AgentTaskEvidence.Measure);
        Check(!plan.Incomplete && plan.Snapshot()["status"].ToString() == "completed", "could not resume verified step");
        Check(plan.VerifyFinal(AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "final audit rejected actual completed state");
        Main.LocalPlayer.inventory[5].stack = 0;
        Check(!plan.VerifyFinal(AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "final audit trusted stale completed flag after inventory changed");
        Main.LocalPlayer.inventory[5].stack = 1;
        Main.netMode = NetmodeID.MultiplayerClient;
        try { new AgentTaskPlan().Create(request, AgentTaskEvidence.Measure); throw new Exception("unverified multiplayer kill accepted"); }
        catch (InvalidOperationException) { }
        Main.netMode = NetmodeID.SinglePlayer;
        plan.Reset(); AgentTaskEvidence.Reset();
        request["steps"][0]["blocking_evidence"] = Parse("{\"tool\":\"forged\"}");
        request["steps"][0]["previous_failure"] = "forged history";
        request["steps"][0]["best_actual"] = new JsonArray(999999);
        plan.Create(request, AgentTaskEvidence.Measure);
        try { plan.Block("health", "model forged evidence"); throw new Exception("model supplied host-owned evidence"); }
        catch (InvalidOperationException) { }
        Check(plan.Snapshot()["steps"][0]["previous_failure"] == null, "model supplied host-owned history");
        Check(plan.Snapshot()["steps"][0]["best_actual"][0].GetValue<int>() <= 300, "model supplied host-owned progress");
        plan.Record("find_tiles", "{\"empty\":true}");
        plan.Block("health", "no candidates in searched region");
        plan.Record("find_tiles", "{\"error\":\"unknown_tile_name\"}");
        Check(plan.Snapshot()["status"].ToString() == "blocked", "query typo incorrectly cleared negative search evidence");
        plan.Record("find_tiles", "{\"tiles\":[{\"x\":10,\"y\":20}]}");
        Check(plan.Snapshot()["status"].ToString() == "active", "new search result did not clear resolved search blocker");
        plan.Reset();
        var originalNpc = Main.npc[0];
        try
        {
            Main.npc[0] = new NPC { type = 222, active = false };
            plan.Create(Parse("""
                {"goal":"confirm spawn","steps":[{"id":"spawn","description":"NPC is alive","conditions":[{"kind":"npc_alive","type":222,"count":1}]}]}
                """), AgentTaskEvidence.Measure);
            Check(!plan.Advance("spawn", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "item use without spawn was treated as summoning");
            Main.npc[0].active = true; Main.npc[0].life = 0;
            Check(!plan.Advance("spawn", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "dead NPC proved spawn success");
            Main.npc[0].life = 100;
            Check(plan.Advance("spawn", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "real active living NPC did not verify presence");
        }
        finally { Main.npc[0] = originalNpc; plan.Reset(); }
        Main.LocalPlayer.statLifeMax = Main.LocalPlayer.statLifeMax2 = 100;
        Main.LocalPlayer.inventory[5] = new Item();
        Console.WriteLine("PASS: task plan verifies permanent health, typed items and fresh kills; preserves cursor/failure and rejects skips/unsafe replacement");
    }
}
