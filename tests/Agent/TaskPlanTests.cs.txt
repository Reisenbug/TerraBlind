using System;
using System.Text.Json.Nodes;
using TerraBlind;
using Terraria;
using Terraria.ID;

static class TaskPlanTests
{
    static void Check(bool ok, string message) { if (!ok) throw new Exception(message); }
    static JsonObject Parse(string text) => JsonNode.Parse(text).AsObject();

    public static void Run()
    {
        VerifyDag();
        VerifyDagInsertion();
        VerifyChestEvidence();
        VerifyPosition();
        VerifyBiome();
        VerifyTiles();
        var plan = new AgentTaskPlan();
        var missingPlan = plan.VerifyFinal(AgentTaskEvidence.Measure);
        Check(missingPlan["verified"]?.GetValue<bool>() == false && missingPlan["reason"]?.ToString() == "plan_required",
            "empty goal verifier did not identify the missing executable verification plan");
        var hook = new AgentTaskEvidence();
        Main.netMode = NetmodeID.SinglePlayer;
        AgentTaskEvidence.Reset();
        hook.OnKill(new NPC { type = 222 });
        var request = Parse("""
        {"goal":"test long task","steps":[
          {"id":"health","description":"increase permanent health","conditions":[{"kind":"permanent_health","count":300}]},
          {"id":"weapon","description":"obtain weapon","conditions":[{"kind":"item","type":191,"count":1}]},
          {"id":"finish","description":"all final requirements","conditions":[{"kind":"permanent_health","count":300},{"kind":"item","type":191,"count":1},{"kind":"npc_kills","type":222,"count":1}]}
        ]}
        """);
        plan.Create(request, AgentTaskEvidence.Measure);
        Main.LocalPlayer.statLifeMax = 100;
        Main.LocalPlayer.statLifeMax2 = 400;
        Check(plan.Advance("health", AgentTaskEvidence.Measure)["verified"].GetValue<bool>() == false,
            "temporary buff health completed permanent-health goal");
        int progress = plan.Progress;
        Main.LocalPlayer.statLifeMax = 120; plan.Observe(AgentTaskEvidence.Measure);
        Check(plan.Progress == progress + 1 && plan.Cursor == 0, "partial health progress was not measured or completed step prematurely");
        Main.LocalPlayer.statLifeMax = 100; plan.Observe(AgentTaskEvidence.Measure);
        Main.LocalPlayer.statLifeMax = 120; plan.Observe(AgentTaskEvidence.Measure);
        Check(plan.Progress == progress + 1, "repeated gain/loss inflated progress");
        Main.LocalPlayer.statLifeMax = 300; plan.Observe(AgentTaskEvidence.Measure);
        progress = plan.Progress;
        Main.LocalPlayer.statLifeMax = 400; plan.Observe(AgentTaskEvidence.Measure);
        Check(plan.Progress == progress && plan.Incomplete, "surplus resources inflated progress or observation completed the plan");
        try { plan.Advance("finish", AgentTaskEvidence.Measure); throw new Exception("skipped steps"); }
        catch (InvalidOperationException) { }
        Main.LocalPlayer.statLifeMax = 300;
        plan.Advance("health", AgentTaskEvidence.Measure);
        Main.LocalPlayer.inventory[5] = new Item { type = 999, Name = "same name", stack = 1 };
        Check(!plan.Advance("weapon", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "wrong type counted");
        Main.LocalPlayer.inventory[5].type = 191;
        plan.Advance("weapon", AgentTaskEvidence.Measure);
        Check(!plan.Advance("finish", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "old kill counted");
        plan.Record("find_tiles", "{\"tiles\":[{\"x\":10,\"y\":20}]}");
        try { plan.Block("finish", "not sure about these candidates"); throw new Exception("uncertainty counted as blocker"); }
        catch (InvalidOperationException) { }
        plan.Record("find_tiles", "{\"error\":\"unknown_tile_name\"}");
        try { plan.Block("finish", "spelled the query wrong"); throw new Exception("query typo counted as blocker"); }
        catch (InvalidOperationException) { }
        plan.Record("nav_to", "{\"status\":\"failed\",\"reason\":\"no_path\"}", true);
        plan.Record("nav_to", "{\"status\":\"failed\",\"reason\":\"no_path\",\"task_plan\":{\"steps\":[\"old plan\"]}}", true);
        Check(!plan.Snapshot()["steps"][2]["last_action"]["result"].ToString().Contains("task_plan"), "action recursively retained its own task plan");
        plan.Block("finish", "no_path");
        plan.Record("mine", "{\"outcome\":\"removed\",\"got\":{\"resource\":1}}", true);
        Check(plan.Snapshot()["status"].ToString() == "active", "successful recovery left current step blocked");
        Check(plan.Snapshot()["steps"][2]["previous_failure"]["reason"].ToString() == "no_path"
            && plan.Snapshot()["steps"][2]["blocking_evidence"] == null, "recovery lost failure history or retained stale blocker");
        plan.Record("nav_to", "{\"status\":\"failed\",\"reason\":\"no_path\"}", true);
        plan.Block("finish", "no_path");
        try { plan.Create(request, AgentTaskEvidence.Measure); throw new Exception("lost active plan"); }
        catch (InvalidOperationException) { }
        var snapshot = plan.Snapshot();
        snapshot["steps"][2]["failure"] = "modified externally";
        Check(plan.Snapshot()["steps"][2]["failure"].ToString() == "no_path", "snapshot aliases mutable plan");
        Check(plan.Snapshot()["steps"][2]["last_action"]["result"].ToString().Contains("no_path"), "failure evidence lost");
        hook.OnKill(new NPC { type = 222 });
        Main.LocalPlayer.inventory[5].stack = 0;
        Check(!plan.Advance("finish", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "spent final weapon ignored");
        Main.LocalPlayer.inventory[5].stack = 1;
        plan.Advance("finish", AgentTaskEvidence.Measure);
        Check(!plan.Incomplete && plan.Snapshot()["status"].ToString() == "completed", "could not resume verified step");
        Check(plan.VerifyFinal(AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "final audit rejected actual completed state");
        Main.LocalPlayer.inventory[5].stack = 0;
        Check(!plan.VerifyFinal(AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "final audit trusted stale completed flag after inventory changed");
        Main.LocalPlayer.inventory[5].stack = 1;
        Main.netMode = NetmodeID.MultiplayerClient;
        try { new AgentTaskPlan().Create(request, AgentTaskEvidence.Measure); throw new Exception("unverified multiplayer kill accepted"); }
        catch (InvalidOperationException) { }
        Main.netMode = NetmodeID.SinglePlayer;
        plan.Reset(); AgentTaskEvidence.Reset();
        request["steps"][0]["blocking_evidence"] = Parse("{\"tool\":\"forged\"}");
        request["steps"][0]["previous_failure"] = "forged history";
        request["steps"][0]["best_actual"] = new JsonArray(999999);
        plan.Create(request, AgentTaskEvidence.Measure);
        try { plan.Block("health", "model forged evidence"); throw new Exception("model supplied host-owned evidence"); }
        catch (InvalidOperationException) { }
        Check(plan.Snapshot()["steps"][0]["previous_failure"] == null, "model supplied host-owned history");
        Check(plan.Snapshot()["steps"][0]["best_actual"][0].GetValue<int>() <= 300, "model supplied host-owned progress");
        plan.Record("find_tiles", "{\"empty\":true}");
        plan.Block("health", "no candidates in searched region");
        plan.Record("find_tiles", "{\"error\":\"unknown_tile_name\"}");
        Check(plan.Snapshot()["status"].ToString() == "blocked", "query typo incorrectly cleared negative search evidence");
        plan.Record("find_tiles", "{\"tiles\":[{\"x\":10,\"y\":20}]}");
        Check(plan.Snapshot()["status"].ToString() == "active", "new search result did not clear resolved search blocker");
        plan.Reset();
        var originalNpc = Main.npc[0];
        try
        {
            Main.npc[0] = new NPC { type = 222, active = false };
            plan.Create(Parse("""
                {"goal":"confirm spawn","steps":[{"id":"spawn","description":"NPC is alive","conditions":[{"kind":"npc_alive","type":222,"count":1}]}]}
                """), AgentTaskEvidence.Measure);
            Check(!plan.Advance("spawn", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "item use without spawn was treated as summoning");
            Main.npc[0].active = true; Main.npc[0].life = 0;
            Check(!plan.Advance("spawn", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "dead NPC proved spawn success");
            Main.npc[0].life = 100;
            Check(plan.Advance("spawn", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "real active living NPC did not verify presence");
        }
        finally { Main.npc[0] = originalNpc; plan.Reset(); }
        Main.LocalPlayer.statLifeMax = Main.LocalPlayer.statLifeMax2 = 100;
        Main.LocalPlayer.inventory[5] = new Item();
        Console.WriteLine("PASS: task plan verifies permanent health, typed items and fresh kills; preserves cursor/failure and rejects skips/unsafe replacement");
        PatchChecks();
    }

    static void VerifyChestEvidence()
    {
        AgentTaskEvidence.Reset(); Main.netMode = NetmodeID.SinglePlayer;
        var chest = new Chest { x = 20, y = 30 };
        Main.chest[0] = chest;
        chest.item[0] = new Item { type = 191, stack = 1 };
        Main.LocalPlayer.inventory[0] = new Item { type = 191, stack = 0 };
        var plan = new AgentTaskPlan();
        plan.Create(Parse("""
            {"goal":"actual chest loot","steps":[{"id":"loot","description":"loot one chest",
              "conditions":[{"kind":"chests_looted","count":1}]}]}
            """), AgentTaskEvidence.Measure);
        var before = AgentTaskEvidence.ObserveChest(21, 31);
        Check(!AgentTaskEvidence.VerifyChest(before), "opening unchanged chest counted as loot");
        chest.item[0].stack = 0;
        Check(!AgentTaskEvidence.VerifyChest(before), "chest removal without inventory gain counted as loot");
        Main.LocalPlayer.inventory[0].stack = 1;
        Check(AgentTaskEvidence.VerifyChest(before) && !AgentTaskEvidence.VerifyChest(before), "verified chest transfer missing or counted twice");
        Check(plan.Advance("loot", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "host chest evidence did not verify plan");
        plan.Create(Parse("""
            {"goal":"another chest","steps":[{"id":"loot","description":"loot new chest",
              "conditions":[{"kind":"chests_looted","count":1}]}]}
            """), AgentTaskEvidence.Measure);
        Check(!plan.Advance("loot", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "new plan reused historical chest loot");
        Main.chest[0] = null; Main.LocalPlayer.inventory[0] = new Item(); AgentTaskEvidence.Reset();
        Console.WriteLine("PASS: chest conditions verify fresh unique container-to-inventory transfers without guessing contents");
    }

    static void VerifyDagInsertion()
    {
        var plan = new AgentTaskPlan();
        int Measure(JsonObject _) => 0;
        plan.Create(Parse("""
            {"goal":"insert opportunity","mode":"dag","steps":[
              {"id":"approach","description":"approach","depends_on":[],"script":"return 0;","conditions":[{"kind":"item","type":1,"count":1}]},
              {"id":"original","description":"original","depends_on":["approach"],"script":"return 1;","conditions":[{"kind":"item","type":2,"count":1}]}]}
            """), Measure);
        plan.Advance("approach", _ => 1);
        string workflow = plan.Script("original")["workflow"].ToString();
        var edit = Parse("""
            {"reason":"observed nearby opportunity","nodes":[{"id":"original","depends_on":["approach","opportunity"]}],
             "add_nodes":[{"id":"opportunity","description":"opportunity","depends_on":["approach"],"script":"return 2;",
               "status":"completed","workflow":"forged","conditions":[{"kind":"item","type":3,"count":1}]}]}
            """);
        edit["expected_revision"] = plan.Snapshot()["revision"].DeepClone();
        var inserted = plan.Revise(edit, Measure);
        var brief = plan.Summary()["steps"][1];
        Check(brief["revision_count"]?.GetValue<int>() == 1 && brief["revisions"] == null
            && plan.Snapshot()["steps"][1]["revisions"].AsArray().Count == 1
            && brief["script"]?.ToString() == "return 1;",
            "plan summary lost current script or retained full revision history");
        Check(inserted["steps"][0]["status"].ToString() == "completed"
            && inserted["current_step"].ToString() == "opportunity"
            && inserted["steps"][1]["workflow"].ToString() == workflow,
            "insertion lost shared prerequisite, original continuation or new dependency scheduling");
        Check(inserted["steps"][2]["status"].ToString() != "completed" && inserted["steps"][2]["workflow"].ToString() != "forged",
            "insertion trusted model-owned runtime progress");
        string before = plan.Snapshot().ToJsonString();
        edit["expected_revision"] = inserted["revision"].DeepClone();
        edit["add_nodes"][0]["id"] = "cycle";
        edit["add_nodes"][0]["depends_on"] = new JsonArray("original");
        edit["nodes"][0]["depends_on"] = new JsonArray("cycle");
        try { plan.Revise(edit, Measure); throw new Exception("insertion introduced dependency cycle"); }
        catch (InvalidOperationException) { }
        Check(before == plan.Snapshot().ToJsonString(), "invalid insertion partially changed existing graph");
        Check(!plan.Advance("opportunity", Measure)["verified"].GetValue<bool>(), "inserted opportunity fabricated collection");
        plan.Advance("opportunity", _ => 1);
        Check(plan.Script("original")["workflow"].ToString() == workflow, "opportunity completion discarded original workflow");
        plan.Advance("original", _ => 1);
        Check(!plan.Incomplete, "could not finish after dynamic opportunity");
        Console.WriteLine("PASS: dynamic DAG insertion shares prerequisites and preserves original progress; cyclic edits are atomic");
    }

    static void VerifyDag()
    {
        var plan = new AgentTaskPlan();
        var request = Parse("""
        {"goal":"shared prerequisite","mode":"dag","steps":[
          {"id":"a","description":"shared approach","depends_on":[],"script":"return tool('get_state',{});","conditions":[{"kind":"item","type":1,"count":1}]},
          {"id":"b","description":"first branch","depends_on":["a"],"script":"return tool('get_state',{});","conditions":[{"kind":"item","type":2,"count":1}]},
          {"id":"c","description":"second branch","depends_on":["a"],"script":"return tool('get_state',{});","conditions":[{"kind":"item","type":3,"count":1}]},
          {"id":"d","description":"joined result","depends_on":["b","c"],"script":"return tool('get_state',{});","conditions":[{"kind":"item","type":4,"count":1}]},
          {"id":"e","description":"independent final requirement","depends_on":[],"script":"return tool('get_state',{});","conditions":[{"kind":"item","type":5,"count":1}]}
        ]}
        """);
        int Measure(JsonObject c) => c["type"].GetValue<int>() == 4 ? 0 : 1;
        plan.Create(request, Measure);
        Check(plan.Snapshot()["ready_steps"].ToJsonString() == "[\"a\",\"e\"]", "DAG did not expose independent roots");
        try { plan.Advance("b", Measure); throw new Exception("DAG bypassed prerequisite"); }
        catch (InvalidOperationException) { }
        plan.Advance("a", Measure);
        string workflow = plan.Script("b")["workflow"].ToString();
        int revision = plan.Snapshot()["revision"].GetValue<int>();
        var update = Parse("""
            {"reason":"new observed route","nodes":[{"id":"b","script":"return 7;","depends_on":["a","c"]}]}
            """);
        update["expected_revision"] = revision;
        plan.Revise(update, Measure);
        Check(plan.Snapshot()["steps"][0]["status"].ToString() == "completed", "graph edit lost shared prerequisite evidence");
        try { plan.Revise(update, Measure); throw new Exception("stale graph edit accepted"); }
        catch (InvalidOperationException) { }
        plan.Select("c"); plan.Advance("c", Measure);
        plan.Select("e");
        plan.Record("nav_to", "{\"error\":\"blocked\"}", true, "b");
        Check(plan.Snapshot()["steps"][1]["blocking_evidence"] != null
            && plan.Snapshot()["steps"][4]["last_action"] == null, "late action evidence was attributed to selected node");
        plan.Record("fight", "{\"ok\":true}", true);
        Check(plan.Snapshot()["steps"][4]["last_action"] == null, "global survival action polluted task evidence");
        Check(plan.Script("b")["workflow"].ToString() == workflow && plan.Script("b")["code"].ToString() == "return 7;",
            "script revision discarded stable continuation identity");
        Check(plan.Snapshot()["steps"][1]["status"].ToString() != "completed", "DAG completed unrelated branch");
        try { plan.Advance("d", Measure); throw new Exception("join ignored incomplete branch"); }
        catch (InvalidOperationException) { }
        plan.Advance("b", Measure);
        Check(!plan.Advance("d", Measure)["verified"].GetValue<bool>(), "DAG trusted execution instead of state");
        plan.Advance("e", Measure); plan.Advance("d", _ => 1);
        Check(!plan.Incomplete && plan.VerifyFinal(_ => 1)["verified"].GetValue<bool>(), "DAG could not finish out of order");
        Check(!plan.VerifyFinal(Measure)["verified"].GetValue<bool>(), "DAG audited only last array element");
        var before = plan.Snapshot().ToJsonString();
        request["steps"][0]["depends_on"] = new JsonArray("d");
        try { plan.Create(request, Measure); throw new Exception("cyclic DAG accepted"); }
        catch (InvalidOperationException) { }
        Check(plan.Snapshot().ToJsonString() == before, "invalid graph destroyed verified progress");
        request["steps"][0]["depends_on"] = new JsonArray("missing");
        try { plan.Create(request, Measure); throw new Exception("missing dependency accepted"); }
        catch (InvalidOperationException) { }
        Console.WriteLine("PASS: DAG shared prerequisites, independent branches, joins, atomic validation and all-sink verification");
    }

    static void PatchChecks()
    {
        var plan = new AgentTaskPlan();
        int kills = 7;
        int Measure(JsonObject c) => c["kind"].ToString() switch
        { "permanent_health" => 120, "npc_kills" => kills, _ => c["type"].GetValue<int>() == 191 ? 1 : 0 };
        plan.Create(Parse("""
            {"goal":"preserve original objective","steps":[
              {"id":"health","description":"already earned health","conditions":[{"kind":"permanent_health","count":120}]},
              {"id":"weapon","description":"wrong item identity","conditions":[{"kind":"item","type":190,"count":1}]},
              {"id":"final","description":"weapon and fresh kill","conditions":[{"kind":"item","type":190,"count":1},{"kind":"npc_kills","type":222,"count":1}]}]}
            """), Measure);
        plan.Advance("health", Measure); kills++;
        string health = plan.Snapshot()["steps"][0].ToJsonString();
        int progress = plan.Progress, revision = plan.Snapshot()["revision"].GetValue<int>();
        var request = Parse($$"""
            {"expected_revision":{{revision}},"reason":"catalog confirmed identity","changes":[
              {"step":"weapon","condition":0,"expected_type":190,"type":191},
              {"step":"final","condition":0,"expected_type":190,"type":191}]}
            """);
        var invalid = (JsonObject)request.DeepClone();
        invalid["changes"][1]["expected_type"] = 999;
        string before = plan.Snapshot().ToJsonString();
        try { plan.Patch(invalid, Measure); throw new Exception("partial invalid patch accepted"); }
        catch (InvalidOperationException) { }
        Check(plan.Snapshot().ToJsonString() == before, "failed patch partially changed conditions or revision");
        var patched = plan.Patch(request, Measure);
        Check(plan.Cursor == 1 && plan.Progress == progress && patched["verified"].GetValue<bool>() == false
            && patched["goal"].ToString() == "preserve original objective" && patched["steps"][0].ToJsonString() == health,
            "identity repair reset earned steps, goal or counted fake game progress");
        Check(patched["steps"][2]["conditions"][1]["baseline"].GetValue<int>() == 7
            && patched["steps"][1]["conditions"][0]["count"].GetValue<int>() == 1, "patch altered kill baseline or quantity");
        try { plan.Patch(request, Measure); throw new Exception("stale revision accepted"); }
        catch (InvalidOperationException) { }
        foreach (var change in new[]
        {
            Parse("{\"step\":\"health\",\"condition\":0,\"expected_type\":190,\"type\":191}"),
            Parse("{\"step\":\"final\",\"condition\":1,\"expected_type\":222,\"type\":223}")
        })
        {
            var denied = new JsonObject { ["expected_revision"] = plan.Snapshot()["revision"].DeepClone(),
                ["reason"] = "invalid correction", ["changes"] = new JsonArray(change) };
            string unchanged = plan.Snapshot().ToJsonString();
            try { plan.Patch(denied, Measure); throw new Exception("completed step or kill rewrite accepted"); }
            catch (InvalidOperationException) { }
            Check(plan.Snapshot().ToJsonString() == unchanged, "rejected edit changed original progress");
        }
        Check(!plan.VerifyFinal(Measure)["verified"].GetValue<bool>(), "patch itself completed the task");
        plan.Advance("weapon", Measure); plan.Advance("final", Measure);
        Check(plan.VerifyFinal(Measure)["verified"].GetValue<bool>(), "corrected identity did not verify earned inventory and preserved fresh kill");
        Console.WriteLine("PASS: atomic plan identity correction preserves earned steps and kill baseline, rejects stale/completed edits and never invents progress");
    }

    static void VerifyTiles()
    {
        var original = new[] { Main.tile[910, 910], Main.tile[911, 910], Main.tile[912, 910] };
        try
        {
            var request = Parse("""{"goal":"three wooden tiles","steps":[{"id":"floor","description":"wooden floor","conditions":[{"kind":"tiles","type":30,"x":910,"y":910,"width":3,"height":1,"count":3}]}]}""");
            for (int i = 0; i < 3; i++) Main.tile[910 + i, 910] = default;
            var plan = new AgentTaskPlan(); plan.Create(request, AgentTaskEvidence.Measure);
            for (int i = 0; i < 2; i++) Main.tile[910 + i, 910] = new Tile { HasUnactuatedTile = true, TileType = 30 };
            Check(!plan.Advance("floor", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "partial floor passed structure verification");
            Main.tile[912, 910] = new Tile { HasUnactuatedTile = true, TileType = 30 };
            Check(plan.Advance("floor", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "complete real floor failed verification");
            Main.tile[911, 910] = new Tile { HasUnactuatedTile = true, TileType = 1 };
            Check(!plan.VerifyFinal(AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "changed structure retained stale completion");
            request["steps"][0]["conditions"][0]["count"] = 1;
            bool rejected = false;
            try { new AgentTaskPlan().Create(request, AgentTaskEvidence.Measure); } catch (InvalidOperationException) { rejected = true; }
            Check(rejected, "rectangle accepted count smaller than complete structure");
            var empty = Parse("""{"kind":"tiles","type":-1,"x":910,"y":910,"width":3,"height":1,"count":3}""");
            Check(AgentTaskEvidence.Measure(empty) == 0, "solid tiles counted as cleared region");
            for (int i = 0; i < 3; i++) Main.tile[910 + i, 910] = default;
            Check(AgentTaskEvidence.Measure(empty) == 3, "empty real cells failed clear verification");
        }
        finally { for (int i = 0; i < 3; i++) Main.tile[910 + i, 910] = original[i]; }
        Console.WriteLine("PASS: structure verification reads actual tile type and every cell, then rechecks the final world");
    }

    static void VerifyBiome()
    {
        var p = Main.LocalPlayer;
        var saved = (p.active, p.dead, p.statLife, p.ZoneJungle, p.ZoneDirtLayerHeight, p.ZoneRockLayerHeight, p.position);
        try
        {
            p.active = true; p.dead = false; p.statLife = 100;
            p.ZoneJungle = true; p.ZoneDirtLayerHeight = p.ZoneRockLayerHeight = false;
            var request = Parse("""{"goal":"finish in underground jungle","steps":[{"id":"finish","description":"live location","conditions":[{"kind":"biome","name":"jungle","depth":"below_surface","count":1}]}]}""");
            var plan = new AgentTaskPlan();
            plan.Create(request, AgentTaskEvidence.Measure);
            Check(!plan.Advance("finish", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "surface jungle met underground condition");
            p.ZoneRockLayerHeight = true; p.ZoneJungle = false;
            Check(!plan.Advance("finish", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "other cavern met jungle condition");
            p.ZoneJungle = true; p.dead = true;
            Check(!plan.Advance("finish", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "death completed live biome condition");
            p.dead = false; p.position = new(8000, 12000);
            Check(plan.Advance("finish", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "real jungle cavern required arbitrary return coordinates");
            p.ZoneRockLayerHeight = false;
            Check(!plan.VerifyFinal(AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "final audit accepted stale biome evidence");
            foreach (string invalid in new[] { "{\"kind\":\"biome\",\"name\":\"jungle\",\"count\":2}", "{\"kind\":\"biome\",\"name\":\"unknown\",\"count\":1}", "{\"kind\":\"biome\",\"name\":\"jungle\",\"depth\":\"unknown\",\"count\":1}" })
            {
                request["steps"][0]["conditions"][0] = Parse(invalid);
                bool rejected = false;
                try { new AgentTaskPlan().Create(request, AgentTaskEvidence.Measure); }
                catch (InvalidOperationException) { rejected = true; }
                Check(rejected, "invalid biome condition accepted");
            }
        }
        finally { (p.active, p.dead, p.statLife, p.ZoneJungle, p.ZoneDirtLayerHeight, p.ZoneRockLayerHeight, p.position) = saved; }
        Console.WriteLine("PASS: biome completion uses live native flags, depth and life, with fresh final verification");
    }

    static void VerifyPosition()
    {
        var player = Main.LocalPlayer;
        var savedPosition = player.position;
        bool savedActive = player.active, savedDead = player.dead;
        int savedLife = player.statLife;
        try
        {
            player.active = true; player.dead = false; player.statLife = 100;
            var request = Parse("""
            {"goal":"arrive then collect","steps":[
              {"id":"arrival","description":"reach target","conditions":[{"kind":"position","x":10,"y":20,"count":1}]},
              {"id":"health","description":"recover","conditions":[{"kind":"permanent_health","count":500}]}
            ]}
            """);
            var plan = new AgentTaskPlan();
            player.position = new(168 - player.width / 2f + 32, 328 - player.height / 2f + 32);
            var snapshot = plan.Create(request, AgentTaskEvidence.Measure);
            var condition = snapshot["steps"][0]["conditions"][0].AsObject();
            Check(condition["x"].GetValue<int>() == 10 && condition["y"].GetValue<int>() == 20
                && condition["radius"].GetValue<int>() == 2, "position target or default radius lost");
            Check(!plan.Advance("arrival", AgentTaskEvidence.Measure)["verified"].GetValue<bool>() && plan.Cursor == 0,
                "diagonal outside circular radius completed arrival");
            player.position = new(168 - player.width / 2f + 32, 328 - player.height / 2f);
            Check(AgentTaskEvidence.Measure(condition) == 1, "exact radius boundary rejected");
            player.dead = true;
            Check(!plan.Advance("arrival", AgentTaskEvidence.Measure)["verified"].GetValue<bool>(), "dead player completed arrival");
            player.dead = false; player.active = false;
            Check(AgentTaskEvidence.Measure(condition) == 0, "inactive player completed arrival");
            player.active = true; player.statLife = 0;
            Check(AgentTaskEvidence.Measure(condition) == 0, "zero life completed arrival");
            player.statLife = 100;
            Check(plan.Advance("arrival", AgentTaskEvidence.Measure)["verified"].GetValue<bool>() && plan.Cursor == 1 && plan.Incomplete,
                "verified arrival did not preserve following task");
            foreach (var invalid in new[]
