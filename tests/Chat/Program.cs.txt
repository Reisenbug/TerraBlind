extern alias engine;
using System;
using System.Collections.Generic;
using System.Linq;
using Microsoft.Xna.Framework;
using ReLogic.Graphics;
using TerraBlind;

class Program
{
    static void Main()
    {
        var font = Terraria.GameContent.FontAssets.MouseText.Value;
        foreach (char c in "AI[1] second_R 所有者=-1正在查询服务器。中文长消息尾行 alpha beta\n".Distinct())
            font.SpriteCharacters[c] = new DynamicSpriteFont.SpriteCharacterData(null, new Rectangle(0, 0, 10, 16), default, new Vector3(0, 10, 0));
        engine::Terraria.Utils.WordwrapString("AI[1]", font, 700, 10, out int index);
        if (index != 0) throw new Exception("Unexpected installed engine wrapping contract");
        foreach (string text in new[] { "AI[1] second_R 所有者=-1", "正在查询服务器。", "alpha\nbeta", string.Concat(Enumerable.Repeat("中文长消息", 40)) + "尾行" })
        {
            Terraria.Main.Chat.Clear();
            AgentChat.Display(text);
            string actual = string.Concat(Terraria.Main.Chat).Replace(" ", "").Replace("\n", "").Replace("-", "");
            string expected = text.Replace(" ", "").Replace("\n", "").Replace("-", "");
            if (Terraria.Main.Chat.Count == 0 || actual != expected) throw new Exception("Chat text lost: " + text);
            Console.WriteLine($"CHAT_CHECK chars={text.Length} displayed_lines={Terraria.Main.Chat.Count}");
        }
        Console.WriteLine("CHAT_CHECKS_COMPLETE checks=4 real_engine_wrap=true renderer_stub=true");
    }
}
