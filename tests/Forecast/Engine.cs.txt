using System;
using System.IO;
using System.Linq;
using System.Net;
using System.Net.Sockets;
using System.Text.Json.Nodes;
using Microsoft.Xna.Framework;
using Terraria;
using Terraria.ID;
using Terraria.ModLoader;
using TerraBlind;

namespace Forecast;

public sealed class Forecast : Mod { }

public sealed class Engine : ModSystem
{
    const int X = 500, Y = 400;
    readonly Rectangle area = new(X - 30, Y - 65, 120, 100);
    JsonObject checkpoint;
    Player source;
    public Player Source => source;
    public Rectangle Area => area;
    public override void Load() => Main.OnEnginePreload += Run;
    public override void Unload() => Main.OnEnginePreload -= Run;

    void Run()
    {
        if (!Main.dedServ || !Terraria.Program.LaunchParameters.TryGetValue("-vanillaforecastport", out string value)) return;
        VanillaFrame.RequireWorker();
        var listener = new TcpListener(IPAddress.Loopback, int.Parse(value));
        listener.Start();
        Console.WriteLine("VANILLA_FORECAST_READY");
        try
        {
            using var client = listener.AcceptTcpClient(); client.NoDelay = true;
            using var reader = new StreamReader(client.GetStream());
            using var writer = new StreamWriter(client.GetStream()) { AutoFlush = true };
            string line;
            while ((line = reader.ReadLine()) != null)
            {
                var request = JsonNode.Parse(line).AsObject();
                if (request["command"]?.GetValue<string>() == "close") { writer.WriteLine("{\"closed\":true}"); break; }
                try { writer.WriteLine(Handle(request).ToJsonString()); }
                catch (Exception error) { writer.WriteLine(new JsonObject { ["error"] = error.ToString() }.ToJsonString()); }
            }
            Environment.Exit(0);
        }
        catch (Exception error) { Console.WriteLine(error); Environment.Exit(1); }
        finally { listener.Stop(); }
    }

    JsonObject Handle(JsonObject request)
    {
        string command = request["command"].GetValue<string>();
        if (command == "replay")
            return new VanillaPrediction(request["snapshot"].AsObject()).Predict(
                request["controls"].AsArray().Select(key => key.GetValue<int>()).ToArray());
        if (command == "transit_contract")
        {
            Setup(new JsonObject { ["equipment"] = new JsonArray((int)ItemID.SandstorminaBottle, (int)ItemID.HermesBoots, (int)ItemID.CobaltShield) });
            var snapshot = VanillaFrame.Capture(source, area, true);
            var guidance = new JsonArray();
            for (int x = area.Left; x < area.Right; x++)
            for (int y = area.Top; y < area.Bottom; y++) guidance.Add(new JsonArray(x, y, area.Right - x));
            var goal = source.position + new Vector2(22 * 16, 0);
            var walk = new VanillaLocalPlanner().Plan(snapshot, goal, guidance: guidance);
            var continued = new VanillaLocalPlanner().Plan(snapshot, goal, Enumerable.Repeat(2, 12).ToArray(), guidance);
            foreach (var plan in new[] { walk, continued })
                if (plan["status"]?.GetValue<string>() != "planned"
                    || plan["controls"].AsArray().Any(key => (key.GetValue<int>() & 3) != 2))
                    throw new InvalidOperationException("transit_inserted_a_stop");
            if (!continued["candidates"].AsArray().Any(candidate => candidate["name"].GetValue<string>() == "continue"))
                throw new InvalidOperationException("transit_discarded_valid_route");
            var baseline = new VanillaPrediction(snapshot).Predict(Enumerable.Repeat(2, VanillaLocalPlanner.Horizon).ToArray(), false);
            float walked = baseline["frames"].AsArray()[^1]["x"].GetValue<float>() - source.position.X;
            float advanced = walk["forecast"]["frames"].AsArray()[^1]["x"].GetValue<float>() - source.position.X;
            if (advanced < walked) throw new InvalidOperationException("mobility_search_lost_to_walking");
            return new JsonObject { ["passed"] = true, ["walk_ms"] = walk["elapsed_ms"].DeepClone(),
                ["continue_ms"] = continued["elapsed_ms"].DeepClone(), ["selected"] = walk["selected"].DeepClone(),
                ["walking_distance"] = walked, ["chosen_distance"] = advanced };
        }
        if (command == "async_cleanup")
        {
            Setup(new JsonObject());
            var snapshot = VanillaFrame.Capture(source, area);
            var fake = new JsonObject { ["status"] = "planned", ["anchor"] = VanillaPrediction.Observe(source),
                ["forecast"] = new JsonObject { ["frames"] = new JsonArray(VanillaPrediction.Observe(source)) },
                ["controls"] = new JsonArray(Enumerable.Repeat(2, 90).Select(key => JsonValue.Create(key)).ToArray()) };
            var controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0);
            controller.Poll(VanillaAsyncMovement.LeadFrames, source, 0);
            if (controller.Accepted != 1 || controller.InputAt(31) != 2) throw new InvalidOperationException("contract_plan_not_installed");
            controller.Request(snapshot, 30, source.position, 0);
            source.velocity.X += 1;
            controller.Poll(60, source, 0);
            if (controller.Status != "state_changed:vx" || controller.InputAt(61) != 0)
                throw new InvalidOperationException("rejected_plan_kept_old_inputs");
            source.velocity.X = 0;
            controller.Request(snapshot, 60, source.position, 0);
            Main.projectile[0].SetDefaults(ProjectileID.WoodenArrowFriendly);
            Main.projectile[0].active = true; Main.projectile[0].Center = source.Center;
            controller.Poll(90, source, 0);
            Main.projectile[0].active = false;
            if (controller.Status != "context_changed" || controller.InputAt(91) != 0)
                throw new InvalidOperationException("new_projectile_did_not_invalidate_plan");
            snapshot = VanillaFrame.Capture(source, area, true);
            fake["anchor"] = VanillaPrediction.Observe(source);
            controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0);
            source.statLife--;
            controller.Poll(30, source, 0);
            if (controller.Accepted != 1) throw new InvalidOperationException("navigation_rejected_health_only_change");
            controller.Request(snapshot, 30, source.position, 0);
            source.runAcceleration += .01f;
            controller.Poll(60, source, 0);
            if (controller.Status != "physics_changed:run_accel") throw new InvalidOperationException("navigation_ignored_movement_change");
            Setup(new JsonObject());
            snapshot = VanillaFrame.Capture(source, area);
            fake["anchor"] = VanillaPrediction.Observe(source);
            var delayed = new System.Threading.Tasks.TaskCompletionSource<JsonObject>();
            int calls = 0;
            controller = new VanillaAsyncMovement(_ => ++calls == 1
                ? System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()) : delayed.Task);
            controller.Request(snapshot, 0, source.position, 0); controller.Poll(30, source, 0);
            controller.Request(snapshot, 30, source.position, 0);
            delayed.SetResult((JsonObject)fake.DeepClone()); controller.Poll(61, source, 0);
            if (controller.Status != "late_result" || controller.InputAt(62) != 2 || !controller.HasInputAt(62))
                throw new InvalidOperationException("late_result_erased_valid_queue");
            controller.Poll(62, source, 1);
            if (controller.HasInputAt(63)) throw new InvalidOperationException("revision_kept_old_queue");
            Setup(new JsonObject());
            Main.npc[0].SetDefaults(NPCID.BlueSlime); Main.npc[0].active = true; Main.npc[0].Center = source.Center;
            snapshot = VanillaFrame.Capture(source, area, true);
            fake["anchor"] = VanillaPrediction.Observe(source);
            controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0);
            Main.npc[0].active = false;
            Main.item[0].SetDefaults(ItemID.Gel); Main.item[0].active = true; Main.item[0].position = source.position;
            controller.Poll(30, source, 0);
            if (controller.Accepted != 1) throw new InvalidOperationException("kill_and_loot_invalidated_navigation");
            controller.Request(snapshot, 30, source.position, 0);
            var obstruction = Main.tile[(int)(source.Center.X / 16), (int)(source.Center.Y / 16)];
            obstruction.HasTile = true; obstruction.TileType = TileID.Stone;
            controller.Poll(60, source, 0);
            if (controller.Status != "context_changed") throw new InvalidOperationException("navigation_ignored_terrain_change");
            Main.item[0].active = false;
            Setup(new JsonObject());
            snapshot = VanillaFrame.Capture(source, area, true);
            fake["anchor"] = VanillaPrediction.Observe(source);
            controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0); controller.Poll(1, source, 0);
            if (controller.Accepted != 1 || controller.InputAt(2) != 2)
                throw new InvalidOperationException("matching_state_waited_for_handoff");
            Setup(new JsonObject());
            var plant = Main.tile[(int)(source.Center.X / 16), (int)(source.Center.Y / 16)];
            plant.HasTile = true; plant.TileType = TileID.Plants;
            snapshot = VanillaFrame.Capture(source, area, true); fake["anchor"] = VanillaPrediction.Observe(source);
            controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0); plant.ClearEverything(); controller.Poll(1, source, 0);
            if (controller.Accepted != 1) throw new InvalidOperationException("cut_plant_invalidated_navigation");
            controller.Request(snapshot, 1, source.position, 0);
            plant.HasTile = true; plant.TileType = TileID.Cobweb;
            controller.Poll(1 + controller.PlanningLead, source, 0);
            if (controller.Status != "context_changed") throw new InvalidOperationException("web_change_was_ignored");
            Setup(new JsonObject()); source.controlRight = true;
            snapshot = VanillaFrame.Capture(source, area, true);
            fake["anchor"] = VanillaPrediction.Observe(source);
            fake["forecast"]["frames"] = new JsonArray(Enumerable.Range(0, 90).Select(_ => (JsonNode)VanillaPrediction.Observe(source)).ToArray());
            delayed = new System.Threading.Tasks.TaskCompletionSource<JsonObject>(); calls = 0;
            controller = new VanillaAsyncMovement(_ => ++calls == 1
                ? System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()) : delayed.Task);
            controller.Request(snapshot, 0, source.position, 0); controller.Poll(1, source, 0);
            controller.Request(snapshot, 1, source.position, 0);
            int arrival = 1 + controller.PlanningLead + 3;
            for (int frame = 2; frame < arrival; frame++) controller.Poll(frame, source, 0);
            delayed.SetResult((JsonObject)fake.DeepClone()); controller.Poll(arrival, source, 0);
            if (controller.Accepted != 2 || controller.InputAt(arrival + 1) != 2)
                throw new InvalidOperationException("matching_late_route_was_discarded");
            delayed = new System.Threading.Tasks.TaskCompletionSource<JsonObject>(); calls = 0;
            controller = new VanillaAsyncMovement(_ => ++calls == 1
                ? System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()) : delayed.Task);
            controller.Request(snapshot, 0, source.position, 0); controller.Poll(1, source, 0);
            controller.Request(snapshot, 1, source.position, 0); arrival = 1 + controller.PlanningLead + 3;
            source.controlRight = false; source.controlLeft = true;
            for (int frame = 2; frame < arrival; frame++) controller.Poll(frame, source, 0);
            delayed.SetResult((JsonObject)fake.DeepClone()); controller.Poll(arrival, source, 0);
            if (controller.Accepted != 1 || controller.LastRejection != "late_result")
                throw new InvalidOperationException("different_input_history_accepted_late_route");
            return new JsonObject { ["passed"] = true, ["scope"] = "coordinator contract with a synthetic transport, not movement validation" };
        }
        if (command == "plan")
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();
            var state = request["snapshot"].AsObject();
            try
            {
                var prefix = request["prefix"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                var advanced = new VanillaPrediction(state).Predict(prefix, false, true);
                if (advanced["end_snapshot"] is not JsonObject future)
                    return new JsonObject { ["status"] = "unsafe_handoff" };
                var anchor = VanillaPrediction.Observe(VanillaFrame.RestoreInWorker(future));
                var result = new VanillaLocalPlanner().Plan(future,
                    new Vector2(request["x"].GetValue<float>(), request["y"].GetValue<float>()),
                    continuation: request["continuation"]?.AsArray().Select(key => key.GetValue<int>()).ToArray(),
                    guidance: request["guidance"]?.AsArray(), terminal: request["terminal"]?.GetValue<bool>() == true);
                result["anchor"] = anchor;
                result["request_compute_ms"] = timer.Elapsed.TotalMilliseconds;
                return result;
            }
            finally { VanillaFrame.RestoreInWorker(state); }
        }
        if (command == "reset")
        {
            Setup(request);
            checkpoint = VanillaFrame.Capture(source, area);
            return new JsonObject { ["initial"] = VanillaPrediction.Observe(source), ["snapshot"] = checkpoint.DeepClone() };
        }
        if (command == "load")
        {
            var incoming = request["snapshot"].AsObject();
            source = VanillaFrame.RestoreInWorker(incoming);
            checkpoint = (JsonObject)incoming.DeepClone();
            return new JsonObject { ["initial"] = VanillaPrediction.Observe(source) };
        }
        if (checkpoint == null) throw new InvalidOperationException("reset_required");
        if (command == "navigate")
        {
            var trace = new JsonArray();
            var plans = new JsonArray();
            int[] continuation = null;
            var goal = new Vector2(request["x"].GetValue<float>(), request["y"].GetValue<float>());
            int limit = Math.Clamp(request["limit"]?.GetValue<int>() ?? 240, 8, 600);
            bool reached = false;
            float maximumError = 0;
            for (int frame = 0; frame < limit;)
            {
                var snapshotTimer = System.Diagnostics.Stopwatch.StartNew();
                checkpoint = VanillaFrame.Capture(source, VanillaLocalPlanner.SnapshotArea(source, goal));
                double snapshotMs = snapshotTimer.Elapsed.TotalMilliseconds;
                var before = VanillaFrame.CaptureFields(source);
                var plan = new VanillaLocalPlanner().Plan(checkpoint, goal, continuation);
                plan["snapshot_ms"] = snapshotMs;
                plan["snapshot_tiles"] = checkpoint["tiles"].AsArray().Count;
                if (!JsonNode.DeepEquals(before, VanillaFrame.CaptureFields(source))) throw new InvalidOperationException("planner_mutated_source");
                plans.Add(plan);
                if (plan["status"].GetValue<string>() != "planned") break;
                var keys = plan["controls"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                Main.player[source.whoAmI] = source;
                int count = Math.Min(VanillaLocalPlanner.ExecuteFrames, limit - frame);
                for (int i = 0; i < count; i++)
                {
                    VanillaPrediction.Advance(source, (VanillaKeys)keys[i]);
                    var expected = plan["forecast"]["frames"][i];
                    float error = Vector2.Distance(source.position, new Vector2(expected["x"].GetValue<float>(), expected["y"].GetValue<float>()));
                    maximumError = Math.Max(error, maximumError);
                    if (error > .001f) throw new InvalidOperationException("execution_diverged_from_prediction");
                    trace.Add(VanillaPrediction.Observe(source)); frame++;
                    if (source.dead) throw new InvalidOperationException("navigation_died");
                    if (Vector2.Distance(source.position, goal) <= 20 && Math.Abs(source.velocity.Y) < .01f) { reached = true; break; }
                }
                continuation = keys.Skip(count).ToArray();
                if (reached) break;
            }
            checkpoint = VanillaFrame.Capture(source, area);
            return new JsonObject { ["reached"] = reached, ["trace"] = trace, ["plans"] = plans, ["max_error"] = maximumError };
        }
        int[] controls = request["controls"]?.AsArray().Select(node => node.GetValue<int>()).ToArray() ?? Array.Empty<int>();
        if (command == "predict")
        {
            var original = VanillaFrame.CaptureFields(source);
            var result = new VanillaPrediction(checkpoint).Predict(controls);
            result["source_unchanged"] = JsonNode.DeepEquals(original, VanillaFrame.CaptureFields(source));
            result["checkpoint_restored"] = JsonNode.DeepEquals(checkpoint,
                VanillaFrame.Capture(Main.player[source.whoAmI], area));
            return result;
        }
        if (command == "actual")
        {
            var trace = new JsonArray();
            Main.player[source.whoAmI] = source;
            foreach (int control in controls) { VanillaPrediction.Advance(source, (VanillaKeys)control); trace.Add(VanillaPrediction.Observe(source)); }
            return new JsonObject { ["frames"] = trace };
        }
        throw new ArgumentException("unknown_command");
    }

    public void Setup(JsonObject request, bool worker = true)
    {
        Main.gameMenu = false;
        if (worker) Main.myPlayer = 255;
        Main.rand = new Terraria.Utilities.UnifiedRandom(73);
        VanillaFrame.RestoreFields(WorldGen.genRand, VanillaFrame.CaptureFields(new Terraria.Utilities.UnifiedRandom(81)));
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        foreach (var item in Main.item) item.active = false;
        for (int x = area.Left; x < area.Right; x++)
        for (int y = area.Top; y < area.Bottom; y++)
        {
            var tile = Main.tile[x, y]; tile.ClearEverything();
            if (y >= Y) { tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        source = new Player { whoAmI = 0, active = true, name = "Forecast subject", statLife = 400, statLifeMax = 400, ConsumedLifeCrystals = 15 };
        if (!worker)
        {
            VanillaFrame.RestoreFields(Main.LocalPlayer, VanillaFrame.CaptureFields(source));
            source = Main.LocalPlayer;
            foreach (var item in source.armor.Concat(source.inventory).Concat(source.miscEquips)) item.SetDefaults(ItemID.None);
        }
        source.position = new Vector2(X * 16, Y * 16 - source.height);
        source.fallStart = (int)(source.position.Y / 16);
        Main.player[0] = source;
        int slot = 3;
        if (request["equipment"] is JsonArray items)
            foreach (var item in items) source.armor[slot++].SetDefaults(item.GetValue<int>());
        if (!worker) Main.ActivePlayerFileData.Player = source;
        if (worker) for (int i = 0; i < 3; i++) VanillaPrediction.Advance(source, VanillaKeys.None);
        if (request["buffs"] is JsonArray buffs)
            foreach (var buff in buffs) source.AddBuff(buff[0].GetValue<int>(), buff[1].GetValue<int>());
        string terrain = request["terrain"]?.GetValue<string>() ?? "flat";
        if (terrain == "ceiling")
            for (int x = X - 3; x < X + 18; x++) { var tile = Main.tile[x, Y - 6]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
        if (terrain is "half" or "slope")
            for (int x = X + 2; x < X + 12; x++)
            {
                var tile = Main.tile[x, Y - 1]; tile.HasTile = true; tile.TileType = TileID.Dirt;
                if (terrain == "half") tile.IsHalfBlock = true;
                else tile.Slope = SlopeType.SlopeDownRight;
            }
        if (terrain == "honey_block")
            for (int x = X - 2; x < X + 20; x++) { var tile = Main.tile[x, Y]; tile.TileType = TileID.HoneyBlock; }
        if (terrain == "honey")
            for (int x = X - 2; x < X + 20; x++)
            for (int y = Y - 4; y < Y; y++) { var tile = Main.tile[x, y]; tile.LiquidType = LiquidID.Honey; tile.LiquidAmount = 255; }
        if (terrain == "embedded")
        {
            var tile = Main.tile[X, Y - 2]; tile.HasTile = true; tile.TileType = TileID.Dirt;
        }
        if (terrain is "ledge" or "wall")
        {
            int height = request["height"]?.GetValue<int>() ?? 12;
            for (int x = X + 9; x < X + (terrain == "ledge" ? 38 : 12); x++)
            for (int y = Y - height; y < Y; y++) { var tile = Main.tile[x, y]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        if (terrain == "lava_gap")
            for (int x = X + 7; x < X + 20; x++)
            for (int y = Y; y < Y + 12; y++)
            {
                var tile = Main.tile[x, y]; tile.ClearEverything();
                if (y >= Y + 10) { tile.LiquidType = LiquidID.Lava; tile.LiquidAmount = 255; }
            }
        if (worker && terrain is ("honey" or "honey_block"))
            for (int i = 0; i < 3; i++) VanillaPrediction.Advance(source, VanillaKeys.None);
        if (request["airborne"]?.GetValue<bool>() == true)
        {
            source.position.Y -= request["airborne_height"]?.GetValue<int>() ?? 320;
            source.velocity.Y = 3; source.fallStart = (int)(source.position.Y / 16);
        }
        if (worker && request["pre_controls"] is JsonArray prior)
            foreach (var key in prior) VanillaPrediction.Advance(source, (VanillaKeys)key.GetValue<int>());
    }
}
