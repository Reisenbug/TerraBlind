    static readonly System.Reflection.FieldInfo RandomIndex = typeof(Terraria.Utilities.UnifiedRandom).GetField("inext", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic);
    static int WorldRandomIndex() => (int)RandomIndex.GetValue(WorldGen.genRand);
    static string traceRandom;
    static void TraceKillTile(On_WorldGen.orig_KillTile original, int x, int y, bool fail, bool effectOnly, bool noItem)
    {
        if (traceRandom == null) { original(x, y, fail, effectOnly, noItem); return; }
        int before = WorldRandomIndex();
        original(x, y, fail, effectOnly, noItem);
        if (traceRandom != null)
            Console.WriteLine($"RANDOM_KILL_TILE {traceRandom} {x},{y} fail={fail} effect={effectOnly} {before}->{WorldRandomIndex()}");
    }
    static void TraceTileFrame(On_WorldGen.orig_TileFrame original, int x, int y, bool reset, bool noBreak)
    {
        if (traceRandom == null) { original(x, y, reset, noBreak); return; }
        int before = WorldRandomIndex();
        original(x, y, reset, noBreak);
        if (traceRandom != null && before != WorldRandomIndex())
            Console.WriteLine($"RANDOM_TILE_FRAME {traceRandom} {x},{y} reset={reset} {before}->{WorldRandomIndex()}");
    }

    sealed record RegionNode(VanillaFrame.WorkerState State, int[] Controls, int[] Uses, Point[] Targets, int[] RandomIndices, float Distance, Vector2 Position, Vector2 Velocity, bool ReleaseJump, int MiningDamage, int ClearedTiles,
        (int horizontal, int vertical, bool jumping, int available, float gravity, bool placed, int terrainType) Mode);

    static JsonObject RegionSurfaces(JsonObject request)
    {
        var initial = request["snapshot"].AsObject();
        var bounds = request["region"].AsArray();
        var region = new Rectangle(bounds[0].GetValue<int>(), bounds[1].GetValue<int>(),
            bounds[2].GetValue<int>(), bounds[3].GetValue<int>());
        var timer = System.Diagnostics.Stopwatch.StartNew();
        var points = new System.Collections.Generic.HashSet<(int x, int y)>();
        int probes = 0;
        try
        {
            var player = VanillaFrame.RestoreInWorker(initial);
            if (player.gravDir != 1) throw new NotSupportedException("surface_extraction_requires_normal_gravity");
            var a = initial["area"].AsArray();
            var observed = new Rectangle(a[0].GetValue<int>() * 16, a[1].GetValue<int>() * 16,
                a[2].GetValue<int>() * 16, a[3].GetValue<int>() * 16);
            if (region.Width <= 0 || region.Height <= 0 || !observed.Contains(region))
                throw new ArgumentException("surface_region_outside_snapshot");
            int padding = (player.width + 1) / 2;
            int left = Math.Max(observed.Left / 16, (region.Left - padding) / 16);
            int right = Math.Min(observed.Right / 16 - 1, (region.Right + padding) / 16);
            for (int y = region.Top / 16; y <= Math.Min(observed.Bottom / 16 - 1, region.Bottom / 16); y++)
            for (int x = left; x <= right; x++)
            {
                var tile = Main.tile[x, y];
                if (!tile.HasTile || tile.IsActuated || !(Main.tileSolid[tile.TileType] || Main.tileSolidTop[tile.TileType])) continue;
                int top = y * 16 + (tile.IsHalfBlock ? 8 : 0);
                int bottom = tile.Slope == 0 ? top : y * 16 + 16;
                for (int footY = Math.Max(region.Top, top); footY <= Math.Min(region.Bottom, bottom); footY++)
                for (int footX = Math.Max(region.Left, x * 16 - padding + 1);
                    footX <= Math.Min(region.Right, (x + 1) * 16 + padding - 1); footX++)
                {
                    if (points.Contains((footX, footY))) continue;
                    player.position = new Vector2(footX - player.width / 2f, footY - player.height);
                    player.velocity = Vector2.Zero;
                    if (!observed.Contains(player.Hitbox)) continue;
                    probes++;
                    if (VanillaContactRegion.SafeBody(player.position, player.width, player.height)
                        && VanillaTerrainPlan.HasSupport(player)) points.Add((footX, footY));
                }
            }
            var regions = new JsonArray();
            foreach (var row in points.GroupBy(point => point.y).OrderBy(row => row.Key))
            {
                int start = -1, end = -1;
                void Add() { if (start >= 0) regions.Add(new JsonArray(start, row.Key, end - start, 0)); }
                foreach (int x in row.Select(point => point.x).OrderBy(x => x))
                {
                    if (start < 0) start = x;
                    else if (x != end + 1) { Add(); start = x; }
                    end = x;
                }
                Add();
            }
            var placements = new JsonArray();
            for (int y = Math.Max(1, (region.Top + 15) / 16); y <= Math.Min(Main.maxTilesY - 2, region.Bottom / 16); y++)
            for (int x = Math.Max(1, region.Left / 16); x <= Math.Min(Main.maxTilesX - 2, region.Right / 16); x++)
            {
                int footX = x * 16 + 8, footY = y * 16;
                if (footX < region.Left || footX > region.Right || Main.tile[x, y].HasTile
                    || LavaSafety.IsLava(x, y) || !VanillaTerrainPlan.HasPlatformAnchor(x, y)) continue;
                var position = new Vector2(footX - player.width / 2f, footY - player.height);
                if (!observed.Contains(new Rectangle((int)position.X, (int)position.Y, player.width, player.height))
                    || !VanillaContactRegion.SafeBody(position, player.width, player.height)) continue;
                int leftFoot = footX, rightFoot = footX;
                for (int offset = 1; offset < 8 + player.width / 2; offset++)
                {
                    if (!VanillaContactRegion.SafeBody(position - new Vector2(offset, 0), player.width, player.height)) break;
                    leftFoot = footX - offset;
                }
                for (int offset = 1; offset < 8 + player.width / 2; offset++)
                {
                    if (!VanillaContactRegion.SafeBody(position + new Vector2(offset, 0), player.width, player.height)) break;
                    rightFoot = footX + offset;
                }
                placements.Add(new JsonObject { ["tile"] = new JsonArray(x, y),
                    ["region"] = new JsonArray(leftFoot, footY, rightFoot - leftFoot, 0), ["execution_verified"] = false });
            }
            int validated = 0;
            var invalid = new JsonArray();
            if (request["validate_samples"]?.GetValue<bool>() == true)
            {
                var baseline = new VanillaFrame.WorkerState(initial);
                foreach (var surface in regions)
                {
                    int x = surface[0].GetValue<int>(), y = surface[1].GetValue<int>(), width = surface[2].GetValue<int>();
                    foreach (int footX in new[] { x, x + width / 2, x + width }.Distinct())
                    {
                        player = baseline.Restore();
                        var position = new Vector2(footX - player.width / 2f, y - player.height);
                        player.position = position; player.velocity = Vector2.Zero; player.jump = 0;
                        player.fallStart = (int)(position.Y / 16);
                        int hp = player.statLife;
                        for (int frame = 0; frame < 12; frame++) VanillaPrediction.Advance(player, VanillaKeys.None);
                        validated++;
                        if (Vector2.Distance(player.position, position) > .01f || player.dead || player.statLife < hp)
                            invalid.Add(new JsonObject { ["x"] = footX, ["y"] = y,
                                ["end_x"] = player.Bottom.X, ["end_y"] = player.Bottom.Y, ["hp_loss"] = hp - player.statLife });
                    }
                }
            }
            return new JsonObject { ["surfaces"] = regions, ["platform_candidates"] = placements, ["supported_samples"] = points.Count,
                ["collision_probes"] = probes, ["validated_samples"] = validated, ["invalid_samples"] = invalid,
                ["elapsed_ms"] = timer.Elapsed.TotalMilliseconds };
        }
        finally { VanillaFrame.RestoreInWorker(initial); }
    }

    static JsonObject CorridorRegions(JsonObject request)
    {
        var initial = request["snapshot"].AsObject();
        var field = VanillaLocalPlanner.ReadGuidance(request["guidance"]);
        var player = VanillaFrame.RestoreInWorker(initial);
        var cell = VanillaLocalPlanner.FootCell(player.position, player.width, player.height);
        if (field == null || !field.TryGetValue(cell, out int startCost))
            throw new ArgumentException("corridor_start_missing_from_guidance");
        int threshold = startCost;
        var path = new JsonArray(new JsonArray(cell.Item1, cell.Item2));
        for (int step = 0; step < 12; step++)
        {
            var next = Enumerable.Range(-1, 3).SelectMany(dx => Enumerable.Range(-1, 3)
                .Select(dy => (cell.Item1 + dx, cell.Item2 + dy)))
                .Where(point => field.TryGetValue(point, out int cost) && cost < threshold)
                .OrderBy(point => field[point]).Take(1).ToArray();
            if (next.Length == 0) break;
            cell = next[0]; threshold = field[cell];
            path.Add(new JsonArray(cell.Item1, cell.Item2));
        }
        var extracted = RegionSurfaces(request);
        var surfaces = extracted["surfaces"].AsArray();
        var platformCandidates = new JsonArray();
        if (player.inventory.Any(item => !item.IsAir && item.stack > 0 && item.createTile == TileID.Platforms))
        foreach (var candidate in extracted["platform_candidates"].AsArray())
        {
            var tile = candidate["tile"].AsArray();
            if (!field.TryGetValue((tile[0].GetValue<int>(), tile[1].GetValue<int>() - 1), out int cost) || cost >= startCost) continue;
            var copy = candidate.DeepClone().AsObject();
            copy["field_cost"] = cost;
            platformCandidates.Add(copy);
        }
        var preferred = path[Math.Min(4, path.Count - 1)].AsArray();
        int preferredCost = field[(preferred[0].GetValue<int>(), preferred[1].GetValue<int>())];
        platformCandidates = new JsonArray(platformCandidates.OrderBy(candidate =>
            Math.Abs(candidate["field_cost"].GetValue<int>() - preferredCost)).Select(candidate => candidate.DeepClone()).ToArray());
        var regions = new JsonArray();
        foreach (var surface in surfaces)
        {
            int x = surface[0].GetValue<int>(), y = surface[1].GetValue<int>(), width = surface[2].GetValue<int>();
            int run = -1;
            for (int footX = x; footX <= x + width + 1; footX++)
            {
                bool eligible = footX <= x + width && field.TryGetValue((footX / 16, (y - 1) / 16), out int cost)
                    && cost <= threshold && cost < startCost;
                if (eligible && run < 0) run = footX;
                if (!eligible && run >= 0) { regions.Add(new JsonArray(run, y, footX - run - 1, 0)); run = -1; }
            }
        }
        var excavations = new JsonArray();
        var passageTiles = new System.Collections.Generic.HashSet<Point>();
        int pathStep = 0;
        var snapshotArea = initial["area"].AsArray();
        var observed = new Rectangle(snapshotArea[0].GetValue<int>() * 16, snapshotArea[1].GetValue<int>() * 16,
            snapshotArea[2].GetValue<int>() * 16, snapshotArea[3].GetValue<int>() * 16);
        foreach (var point in path.Skip(1))
        {
            pathStep++;
            int cx = point[0].GetValue<int>(), cy = point[1].GetValue<int>();
            int footX = cx * 16 + 8, footY = cy * 16 + 1;
            var body = new Rectangle(footX - 4 - player.width / 2, footY - player.height,
                player.width + 8, player.height + 15);
            if (!observed.Contains(body)) continue;
            var blockers = new JsonArray();
            bool safe = true;
            for (int tx = body.Left / 16; tx <= (body.Right - 1) / 16; tx++)
            for (int ty = body.Top / 16; ty <= (body.Bottom - 1) / 16; ty++)
            {
                var tile = Main.tile[tx, ty];
                if (!tile.HasTile || tile.IsActuated || !Main.tileSolid[tile.TileType] || Main.tileSolidTop[tile.TileType]) continue;
                var overlap = Rectangle.Intersect(body, new Rectangle(tx * 16, ty * 16, 16, 16));
                if (!Collision.SolidCollision(new Vector2(overlap.X, overlap.Y), overlap.Width, overlap.Height)) continue;
                safe &= LavaSafety.CanMine(tx, ty) && WorldGen.CanKillTile(tx, ty);
                blockers.Add(new JsonArray(tx, ty));
            }
            if (!safe) break;
            foreach (var blocker in blockers)
                passageTiles.Add(new Point(blocker[0].GetValue<int>(), blocker[1].GetValue<int>()));
            if (passageTiles.Count == 0) continue;
            JsonArray movementBlockedBy = null;
            foreach (var blocker in blockers)
            {
                int tx = blocker[0].GetValue<int>(), ty = blocker[1].GetValue<int>();
                var tile = Main.tile[tx, ty];
                bool stable = tile.TileType is TileID.Dirt or TileID.Stone or TileID.Mud or TileID.Ash;
                if (!stable || tile.IsHalfBlock || tile.Slope != 0) continue;
                if (player.Hitbox.Intersects(new Rectangle(tx * 16, ty * 16, 16, 16))) continue;
                if (footX - 4 + player.width / 2f > tx * 16 && footX + 4 - player.width / 2f < (tx + 1) * 16
                    && footY > ty * 16 && footY + 15 - player.height < (ty + 1) * 16)
                {
                    movementBlockedBy = new JsonArray(tx, ty);
                    break;
                }
            }
            excavations.Add(new JsonObject { ["goal_region"] = new JsonArray(footX - 4, footY, 8, 15),
                ["mining_tiles"] = new JsonArray(passageTiles.Select(tile => (JsonNode)new JsonArray(tile.X, tile.Y)).ToArray()),
                ["path_step"] = pathStep, ["field_cost"] = field[(cx, cy)],
                ["movement_blocked_by"] = movementBlockedBy,
                ["body_sweep"] = new JsonArray(body.X, body.Y, body.Width, body.Height),
                ["reason"] = "solid collision inside player body sweep at descending field cell" });
        }
        VanillaFrame.RestoreInWorker(initial);
        return new JsonObject { ["goal_regions"] = regions, ["start_cost"] = startCost,
            ["threshold_cost"] = threshold, ["gradient_path"] = path, ["excavation_candidates"] = excavations,
            ["platform_candidates"] = platformCandidates };
    }

    static JsonObject RegionPlan(JsonObject request)
    {
        var timer = System.Diagnostics.Stopwatch.StartNew();
        var corridor = CorridorRegions(request);
        var attempts = new JsonArray();
        var search = request.DeepClone().AsObject();
        JsonObject result = null;
        bool support = true;
        bool direct = true;
        bool Terminal() => result?["status"]?.GetValue<string>() is "reached" or "snapshot_required";
        JsonObject Search(string mode, JsonNode region, bool grounded, JsonNode goals = null,
            JsonNode platform = null, JsonNode mining = null)
        {
            support = grounded;
            search["region"] = region.DeepClone();
            search["require_support"] = grounded;
            search["goal_regions"] = goals?.DeepClone();
            search["placement_tile"] = platform?.DeepClone();
            search["mining_tiles"] = mining?.DeepClone();
            search["direct"] = direct;
            result = RegionSearch(search);
            attempts.Add(new JsonObject { ["mode"] = mode, ["region"] = region.DeepClone(),
                ["direct"] = direct,
                ["status"] = result["status"]?.DeepClone(), ["elapsed_ms"] = result["elapsed_ms"]?.DeepClone(),
                ["expanded"] = result["expanded"]?.DeepClone(), ["frames"] = result["controls"].AsArray().Count });
            return result;
        }
        void Explore()
        {
            if (corridor["goal_regions"].AsArray().Count > 0)
                Search("movement", request["region"], true, corridor["goal_regions"]);
            if (!Terminal())
            foreach (var candidate in corridor["platform_candidates"].AsArray().Take(3))
            {
                Search("platform", candidate["region"], true, platform: candidate["tile"]);
                if (result["status"]?.GetValue<string>() == "reached") corridor["selected_platform"] = candidate.DeepClone();
                if (Terminal()) break;
            }
            if (!Terminal())
            {
                var candidates = corridor["excavation_candidates"].AsArray();
                var priorities = candidates.OrderBy(candidate => Math.Abs(candidate["path_step"].GetValue<int>() - 4)).Take(1).ToList();
                if (!direct)
                    priorities.AddRange(candidates.Reverse().Where(candidate => candidate["path_step"].GetValue<int>() <= 2
                        && !priorities.Contains(candidate)));
                foreach (var candidate in priorities)
                {
                    if (candidate["movement_blocked_by"] == null)
                    {
                        Search("local_movement", candidate["goal_region"], false);
                        if (result["status"]?.GetValue<string>() == "reached")
                            corridor["selected_local_transit"] = candidate["goal_region"].DeepClone();
                        if (Terminal()) break;
                    }
                    Search("mining", candidate["goal_region"], false, mining: candidate["mining_tiles"]);
                    if (result["status"]?.GetValue<string>() == "reached") corridor["selected_excavation"] = candidate.DeepClone();
                    if (Terminal()) break;
                }
            }
        }
        Explore();
        if (!Terminal()) { direct = false; Explore(); }
        result ??= new JsonObject { ["status"] = "no_local_route", ["controls"] = new JsonArray() };
        if (result["status"]?.GetValue<string>() == "snapshot_required")
        {
            var boundary = search.DeepClone().AsObject();
            boundary.Remove("snapshot"); boundary.Remove("guidance"); boundary.Remove("command");
            result["boundary_request"] = boundary;
        }
        result["corridor"] = corridor;
        result["attempts"] = attempts;
        result["require_support"] = support;
        if (result["status"]?.GetValue<string>() == "reached")
        {
            var snapshot = request["snapshot"].AsObject();
            var traceTimer = System.Diagnostics.Stopwatch.StartNew();
            try
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                result["anchor"] = VanillaPrediction.Observe(player, false);
                var frames = new JsonArray();
                foreach (var input in VanillaPrediction.RegionInputs(result))
                {
                    VanillaPrediction.Advance(player, input.Keys, input.UseSlot, input.Target);
                    frames.Add(VanillaPrediction.Observe(player, false));
                }
                var bounds = snapshot["area"].AsArray();
                var actual = VanillaFrame.Capture(player, new Rectangle(bounds[0].GetValue<int>(),
                    bounds[1].GetValue<int>(), bounds[2].GetValue<int>(), bounds[3].GetValue<int>()), true);
                if (!JsonNode.DeepEquals(actual, result["end_snapshot"]))
                    throw new InvalidOperationException("region_execution_trace_mismatch");
                result["frames"] = frames;
            }
            finally { VanillaFrame.RestoreInWorker(snapshot); }
            result["execution_trace_ms"] = traceTimer.Elapsed.TotalMilliseconds;
        }
        result["planning_wall_ms"] = timer.Elapsed.TotalMilliseconds;
        return result;
    }

    static JsonObject RegionSearch(JsonObject request)
    {
        var initial = request["snapshot"].AsObject();
        var bounds = request["region"].AsArray();
        var region = new Rectangle(bounds[0].GetValue<int>(), bounds[1].GetValue<int>(),
            bounds[2].GetValue<int>(), bounds[3].GetValue<int>());
        var goalRegions = request["goal_regions"] is JsonArray goals
            ? goals.Select(goal => new Rectangle(goal[0].GetValue<int>(), goal[1].GetValue<int>(),
                goal[2].GetValue<int>(), goal[3].GetValue<int>())).ToArray() : new[] { region };
        if (goalRegions.Length == 0 || goalRegions.Any(goal => goal.Width < 0 || goal.Height < 0))
            throw new ArgumentException("invalid_goal_regions");
        int horizon = request["horizon"]?.GetValue<int>() ?? 90;
        int beam = request["beam"]?.GetValue<int>() ?? 6;
        int quantum = request["quantum"]?.GetValue<int>() ?? 6;
        bool direct = request["direct"]?.GetValue<bool>() == true;
        bool requireSupport = request["require_support"]?.GetValue<bool>() ?? true;
        Point? placement = request["placement_tile"] is JsonArray place
            ? new Point(place[0].GetValue<int>(), place[1].GetValue<int>()) : null;
        Point? mining = request["mining_tile"] is JsonArray mine
            ? new Point(mine[0].GetValue<int>(), mine[1].GetValue<int>()) : null;
        var miningTiles = request["mining_tiles"] is JsonArray mineList
            ? mineList.Select(tile => new Point(tile[0].GetValue<int>(), tile[1].GetValue<int>())).Distinct().ToArray()
            : mining is Point single ? new[] { single } : Array.Empty<Point>();
        if (placement != null && miningTiles.Length > 0) throw new ArgumentException("mixed_terrain_edits_not_supported");
        int useSlot = -1, initialStock = 0, playerIndex = 0;
        bool Placed() => placement is Point tile && Main.tile[tile.X, tile.Y].HasTile
            && Main.tile[tile.X, tile.Y].TileType == TileID.Platforms;
        bool Edited() => miningTiles.Length > 0 ? miningTiles.All(tile => !Main.tile[tile.X, tile.Y].HasTile) : Placed();
        int MiningDamage(Player player) => player.hitTile.data
            .Where(hit => hit.type == 1 && miningTiles.Contains(new Point(hit.X, hit.Y))).Sum(hit => hit.damage);
        Point? Target(Player player)
        {
            if (placement != null) return placement;
            var foot = player.Bottom;
            var destination = goalRegions.Select(goal => new Vector2(Math.Clamp(foot.X, goal.Left, goal.Right),
                Math.Clamp(foot.Y, goal.Top, goal.Bottom))).OrderBy(point => Vector2.DistanceSquared(foot, point)).First();
            var delta = destination - foot;
            if (delta.LengthSquared() > 16 * 16) delta = Vector2.Normalize(delta) * 16;
            var next = player.Hitbox;
            next.Offset((int)Math.Round(delta.X), (int)Math.Round(delta.Y));
            var sweep = Rectangle.Union(player.Hitbox, next);
            return miningTiles.Where(tile => Main.tile[tile.X, tile.Y].HasTile && LavaSafety.CanMine(tile.X, tile.Y)
                    && sweep.Intersects(new Rectangle(tile.X * 16, tile.Y * 16, 16, 16)))
                .OrderBy(tile => Vector2.DistanceSquared(player.Center, tile.ToWorldCoordinates()))
                .Select(tile => (Point?)tile).FirstOrDefault();
        }
        if (region.Width < 0 || region.Height < 0 || horizon < 1 || horizon > 240
            || beam < 1 || beam > 16 || quantum < 1 || quantum > 12)
            throw new ArgumentException("invalid_region_search_budget");
        var timer = System.Diagnostics.Stopwatch.StartNew();
        int expanded = 0, simulated = 0, shared = 0;
        long tileChanges = 0, tileWrites = 0;
        var layers = new JsonArray();
        double restoreMs = 0, updateMs = 0, captureMs = 0;
        double restoreFields = 0, restoreItems = 0, restoreTiles = 0;
        double playerConstructors = 0;
        double captureFields = 0, captureItems = 0, captureTiles = 0;
        float Distance(Player player)
        {
            var point = player.Bottom;
            return goalRegions.Min(goal =>
            {
                float dx = Math.Max(goal.Left - point.X, Math.Max(0, point.X - goal.Right));
                float dy = Math.Max(goal.Top - point.Y, Math.Max(0, point.Y - goal.Bottom));
                return MathF.Sqrt(dx * dx + dy * dy);
            });
        }
        bool Reached(Player player) => requireSupport
            ? Distance(player) <= .5f && player.velocity.Y == 0
                && VanillaContactRegion.SafeBody(player.position, player.width, player.height)
                && VanillaTerrainPlan.HasSupport(player)
            : Distance(player) == 0;
        (int, int, bool, int, float, bool, int) Mode(Player player)
        {
            int available = 0;
            foreach (var jump in player.ExtraJumps) if (jump.Available) available++;
            return (Math.Sign(player.velocity.X), Math.Sign(player.velocity.Y), player.jump > 0, available, player.gravDir, Edited(),
                miningTiles.Aggregate(-1, (hash, tile) => unchecked(hash * 31
                    + (Main.tile[tile.X, tile.Y].HasTile ? Main.tile[tile.X, tile.Y].TileType + 1 : 0))));
        }
        var a = initial["area"].AsArray();
        var tiles = new Rectangle(a[0].GetValue<int>(), a[1].GetValue<int>(), a[2].GetValue<int>(), a[3].GetValue<int>());
        JsonObject Finish(RegionNode node, bool reached) => new()
        {
            ["status"] = reached ? "reached" : "budget_exhausted",
            ["quantum"] = quantum,
            ["distance"] = node.Distance,
            ["controls"] = new JsonArray(node.Controls.Select(key => JsonValue.Create(key)).ToArray()),
            ["world_random_indices"] = new JsonArray(node.RandomIndices.Select(value => JsonValue.Create(value)).ToArray()),
            ["use_slots"] = new JsonArray(node.Uses.Select(slot => JsonValue.Create(slot)).ToArray()),
            ["use_targets"] = new JsonArray(node.Targets.Select(tile => (JsonNode)new JsonArray(tile.X, tile.Y)).ToArray()),
            ["placement_tile"] = placement is Point tile ? new JsonArray(tile.X, tile.Y) : null,
            ["mining_tile"] = mining is Point mineTile ? new JsonArray(mineTile.X, mineTile.Y) : null,
            ["mining_tiles"] = new JsonArray(miningTiles.Select(tile => (JsonNode)new JsonArray(tile.X, tile.Y)).ToArray()),
            ["end_snapshot"] = VanillaFrame.Capture(node.State.Restore(), tiles, true),
            ["placed"] = Placed(),
            ["cleared"] = miningTiles.Length > 0 && Edited(),
            ["cleared_count"] = miningTiles.Count(tile => !Main.tile[tile.X, tile.Y].HasTile),
            ["consumed"] = useSlot < 0 ? 0 : initialStock - Main.player[playerIndex].inventory[useSlot].stack,
            ["expanded"] = expanded, ["simulated_frames"] = simulated, ["reused_prefix_frames"] = shared,
            ["captured_tile_changes"] = tileChanges, ["restored_tile_writes"] = tileWrites,
            ["layers"] = layers,
            ["elapsed_ms"] = timer.Elapsed.TotalMilliseconds,
            ["timing"] = new JsonObject { ["restore_ms"] = restoreMs, ["update_ms"] = updateMs, ["capture_ms"] = captureMs,
                ["restore_fields_ms"] = restoreFields, ["restore_items_ms"] = restoreItems, ["restore_tiles_ms"] = restoreTiles,
                ["player_constructor_ms"] = playerConstructors,
                ["capture_fields_ms"] = captureFields, ["capture_items_ms"] = captureItems, ["capture_tiles_ms"] = captureTiles }
        };
        try
        {
            var start = VanillaFrame.RestoreInWorker(initial);
            playerIndex = start.whoAmI;
            if (placement is Point target)
            {
                if (!tiles.Contains(target) || Main.tile[target.X, target.Y].HasTile)
                    throw new ArgumentException("placement_target_outside_snapshot_or_occupied");
                useSlot = Array.FindIndex(start.inventory, item => !item.IsAir && item.createTile == TileID.Platforms);
                if (useSlot < 0) throw new InvalidOperationException("missing_platform_item");
                initialStock = start.inventory[useSlot].stack;
            }
            if (miningTiles.Length > 0)
            {
                foreach (var mineTarget in miningTiles)
                if (!tiles.Contains(mineTarget) || !Main.tile[mineTarget.X, mineTarget.Y].HasTile
                    || !LavaSafety.CanMine(mineTarget.X, mineTarget.Y) || !WorldGen.CanKillTile(mineTarget.X, mineTarget.Y))
                    throw new ArgumentException("unsafe_or_missing_mining_target");
                useSlot = Enumerable.Range(0, start.inventory.Length).Where(slot => start.inventory[slot].pick > 0)
                    .OrderByDescending(slot => start.inventory[slot].pick).DefaultIfEmpty(-1).First();
                if (useSlot < 0) throw new InvalidOperationException("missing_pickaxe");
                initialStock = start.inventory[useSlot].stack;
            }
            int hp = start.statLife;
            var interior = new Rectangle((tiles.X + 2) * 16, (tiles.Y + 2) * 16, (tiles.Width - 4) * 16, (tiles.Height - 4) * 16);
            var best = new RegionNode(new VanillaFrame.WorkerState(initial), Array.Empty<int>(), Array.Empty<int>(), Array.Empty<Point>(), Array.Empty<int>(), Distance(start), start.Bottom, start.velocity, start.releaseJump, MiningDamage(start), 0, Mode(start));
            if (Reached(start))
                return Finish(best, true);
            var frontier = new System.Collections.Generic.List<RegionNode> { best };
            VanillaFrame.WorkerState knownWorld = null;
            bool gravity = start.buffType.Contains(BuffID.Gravitation);
            int[] actions = gravity ? new[] { 0, 1, 2, 4, 5, 6, 16, 17, 18, 8, 9, 10 }
                : new[] { 0, 1, 2, 4, 5, 6, 16, 17, 18 };
            if (direct) actions = actions.Where(keys => (keys & 1) == 0).ToArray();
            int segment = direct ? horizon : quantum;
            for (int depth = 0; depth < horizon && frontier.Count > 0; depth += segment)
            {
                var next = new System.Collections.Generic.List<RegionNode>();
                foreach (var parent in frontier)
                {
                    var checkpoint = parent.State;
                    bool continueMining = miningTiles.Length > 0 && parent.Uses.Length > 0 && parent.Uses[^1] >= 0;
                    var choices = actions.Select(keys => (keys, use: continueMining));
                    if (useSlot >= 0 && !parent.Mode.placed && !continueMining)
                        choices = choices.Concat(actions.Select(keys => (keys, use: true)));
                    foreach (var action in choices)
                    {
                        VanillaPrediction.CheckCancellation();
                        long at = System.Diagnostics.Stopwatch.GetTimestamp();
                        var player = checkpoint.Restore(knownWorld);
                        knownWorld = null;
                        restoreFields += checkpoint.FieldsMs; restoreItems += checkpoint.ItemsMs; restoreTiles += checkpoint.TilesMs;
                        playerConstructors += checkpoint.PlayerConstructorMs;
                        tileWrites += checkpoint.RestoredTileWrites;
                        restoreMs += System.Diagnostics.Stopwatch.GetElapsedTime(at).TotalMilliseconds;
                        expanded++; shared += parent.Controls.Length;
                        int count = 0;
                        var uses = new System.Collections.Generic.List<int>();
                        var targets = new System.Collections.Generic.List<Point>();
                        var randomIndices = new System.Collections.Generic.List<int>();
                        var controls = new System.Collections.Generic.List<int>();
                        bool safe = true, reached = false;
                        for (; count < Math.Min(segment, horizon - depth);)
                        {
                            at = System.Diagnostics.Stopwatch.GetTimestamp();
                            Point? aim = action.use && !Edited() ? Target(player) : null;
                            int slot = aim != null ? useSlot : -1;
                            randomIndices.Add(WorldRandomIndex());
                            traceRandom = parent.Controls.Length == 0 && action.keys == 0 && slot >= 0 && count == 0 ? "search" : null;
                            int keys = action.keys;
                            if (direct && (keys & 2) != 0)
                            {
                                var foot = player.Bottom;
                                var destination = goalRegions.OrderBy(goal => Vector2.DistanceSquared(foot,
                                    new Vector2(Math.Clamp(foot.X, goal.Left, goal.Right), Math.Clamp(foot.Y, goal.Top, goal.Bottom)))).First();
                                keys = (keys & ~3) | (foot.X < destination.Left ? 2 : foot.X > destination.Right ? 1 : 0);
                            }
                            VanillaPrediction.Advance(player, (VanillaKeys)keys, slot, aim);
                            traceRandom = null;
                            randomIndices.Add(WorldRandomIndex());
                            uses.Add(slot);
                            controls.Add(keys);
                            targets.Add(aim ?? Point.Zero);
                            updateMs += System.Diagnostics.Stopwatch.GetElapsedTime(at).TotalMilliseconds;
                            count++; simulated++;
                            safe = !player.dead && player.statLife >= hp && interior.Contains(player.Hitbox)
                                && !Main.projectile.Any(projectile => projectile.active)
                                && !Main.npc.Any(npc => npc.active);
                            if (!safe) break;
                            reached = Reached(player);
                            if (reached) break;
                        }
                        if (!safe) continue;
                        at = System.Diagnostics.Stopwatch.GetTimestamp();
                        var state = checkpoint.CaptureCurrent(player);
                        knownWorld = state;
                        captureFields += state.FieldsMs; captureItems += state.ItemsMs; captureTiles += state.TilesMs;
                        tileChanges += state.CapturedTileChanges;
                        captureMs += System.Diagnostics.Stopwatch.GetElapsedTime(at).TotalMilliseconds;
                        var node = new RegionNode(state, parent.Controls.Concat(controls).ToArray(),
                            parent.Uses.Concat(uses).ToArray(), parent.Targets.Concat(targets).ToArray(), parent.RandomIndices.Concat(randomIndices).ToArray(), Distance(player), player.Bottom, player.velocity, player.releaseJump, MiningDamage(player),
                            miningTiles.Count(tile => !Main.tile[tile.X, tile.Y].HasTile), Mode(player));
                        if (node.Distance < best.Distance) best = node;
                        if (reached) return Finish(node, true);
                        next.Add(node);
                    }
                }
                var ordered = next.OrderBy(node => node.Distance).ToArray();
                frontier = new System.Collections.Generic.List<RegionNode>();
                if (ordered.Length > 0) frontier.Add(ordered[0]);
                float Separation(RegionNode left, RegionNode right) =>
                    Vector2.DistanceSquared(left.Position, right.Position)
                    + Vector2.DistanceSquared(left.Velocity * quantum, right.Velocity * quantum)
                    + (left.Mode.available != right.Mode.available ? 256 : 0)
                    + (left.Mode.placed != right.Mode.placed ? 256 : 0)
                    + (left.Mode.terrainType != right.Mode.terrainType ? 256 : 0)
                    + (left.ReleaseJump != right.ReleaseJump ? 64 : 0)
                    + MathF.Pow((left.MiningDamage - right.MiningDamage) * .32f, 2);
                while (frontier.Count < beam && frontier.Count < ordered.Length)
                {
                    var candidate = ordered.Except(frontier)
                        .OrderBy(node => node.Distance - MathF.Sqrt(frontier.Min(selected => Separation(node, selected)))).First();
                    if (frontier.Min(selected => Separation(candidate, selected)) == 0) break;
                    frontier.Add(candidate);
                }
                layers.Add(new JsonObject { ["frame"] = depth + segment, ["generated"] = next.Count,
                    ["candidates"] = new JsonArray(next.Select(node => (JsonNode)new JsonObject {
                        ["position"] = new JsonArray(node.Position.X, node.Position.Y),
                        ["velocity"] = new JsonArray(node.Velocity.X, node.Velocity.Y),
                        ["release_jump"] = node.ReleaseJump,
                        ["mining_damage"] = node.MiningDamage,
                        ["cleared_tiles"] = node.ClearedTiles,
                        ["distance"] = node.Distance, ["mode"] = node.Mode.ToString(),
                        ["retained"] = frontier.Contains(node),
                        ["controls"] = new JsonArray(node.Controls.Select(key => JsonValue.Create(key)).ToArray()) }).ToArray()),
                    ["retained"] = new JsonArray(frontier.Select(node => (JsonNode)new JsonObject {
                        ["distance"] = node.Distance,
                        ["mode"] = node.Mode.ToString(),
                        ["controls"] = new JsonArray(node.Controls.Select(key => JsonValue.Create(key)).ToArray()) }).ToArray()) });
            }
            return Finish(best, false);
        }
        catch (VanillaFrame.SnapshotBoundaryException error)
        {
            return new JsonObject { ["status"] = "snapshot_required", ["controls"] = new JsonArray(),
                ["missing_tile"] = new JsonArray(error.Tile.X, error.Tile.Y),
                ["elapsed_ms"] = timer.Elapsed.TotalMilliseconds };
        }
        finally { VanillaFrame.RestoreInWorker(initial); }
    }

    public override void Load() { Main.OnEnginePreload += Run; On_WorldGen.TileFrame += TraceTileFrame; On_WorldGen.KillTile += TraceKillTile; }
    public override void Unload() { Main.OnEnginePreload -= Run; On_WorldGen.TileFrame -= TraceTileFrame; On_WorldGen.KillTile -= TraceKillTile; }

    sealed class Work
    {
        public JsonObject Request;
        public long Id;
        public int Cancelled;
    }

    void Run()
    {
        if (!Main.dedServ || !Terraria.Program.LaunchParameters.TryGetValue("-vanillaforecastport", out string value)) return;
        VanillaFrame.RequireWorker();
        var listener = new TcpListener(IPAddress.Loopback, int.Parse(value));
        listener.Start();
        Console.WriteLine("VANILLA_FORECAST_READY");
        try
        {
            using var client = listener.AcceptTcpClient(); client.NoDelay = true;
            using var reader = new StreamReader(client.GetStream());
            using var writer = new StreamWriter(client.GetStream()) { AutoFlush = true };
            using var incoming = new BlockingCollection<Work>();
            var requests = new ConcurrentDictionary<long, Work>();
            long anonymousId = 0;
            _ = Task.Run(() =>
            {
                try
                {
                    string line;
                    while ((line = reader.ReadLine()) != null)
                    {
                        var request = JsonNode.Parse(line).AsObject();
                        long id = request["request_id"]?.GetValue<long>() ?? Interlocked.Decrement(ref anonymousId);
                        if (request["command"]?.GetValue<string>() == "cancel")
                        {
                            if (requests.TryGetValue(id, out var cancelled)) Interlocked.Exchange(ref cancelled.Cancelled, 1);
                            continue;
                        }
                        var work = new Work { Request = request, Id = id };
                        requests[id] = work; incoming.Add(work);
                    }
                }
                catch (Exception error) { Console.WriteLine(error); }
                finally { incoming.CompleteAdding(); }
            });
            foreach (var work in incoming.GetConsumingEnumerable())
            {
                var request = work.Request;
                if (request["command"]?.GetValue<string>() == "close") { writer.WriteLine("{\"closed\":true}"); break; }
                VanillaPrediction.CancellationRequested = () => Volatile.Read(ref work.Cancelled) != 0;
                try
                {
                    if (Terraria.Program.LaunchParameters.ContainsKey("-forecasttrace")) Console.WriteLine("FORECAST_REQUEST_STARTED " + work.Id);
                    VanillaPrediction.CheckCancellation();
                    var result = Handle(request); result["request_id"] = work.Id;
                    writer.WriteLine(result.ToJsonString());
                }
                catch (OperationCanceledException)
                { writer.WriteLine(new JsonObject { ["status"] = "cancelled", ["request_id"] = work.Id }.ToJsonString()); }
                catch (Exception error) { writer.WriteLine(new JsonObject { ["error"] = error.ToString() }.ToJsonString()); }
                finally { VanillaPrediction.CancellationRequested = null; requests.TryRemove(work.Id, out _); }
            }
            Environment.Exit(0);
        }
        catch (Exception error) { Console.WriteLine(error); Environment.Exit(1); }
        finally { listener.Stop(); }
    }

    JsonObject Handle(JsonObject request)
    {
        string command = request["command"].GetValue<string>();
        if (command == "platform_anchor")
        {
            var player = VanillaFrame.RestoreInWorker(request["snapshot"].AsObject());
            int x = request["tile_x"].GetValue<int>(), y = request["tile_y"].GetValue<int>();
            player.inventory[player.selectedItem].SetDefaults(ItemID.WoodPlatform);
            int previousX = Player.tileTargetX, previousY = Player.tileTargetY;
            try
            {
                Player.tileTargetX = x; Player.tileTargetY = y;
                var method = typeof(Player).GetMethod("PlaceThing_Tiles_BlockPlacementForAssortedThings",
                    System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic);
                bool native = (bool)method.Invoke(player, new object[] { false });
                return new JsonObject { ["native"] = native, ["planner"] = VanillaTerrainPlan.HasPlatformAnchor(x, y) };
            }
            finally { Player.tileTargetX = previousX; Player.tileTargetY = previousY; }
        }
        if (command == "terrain_plan") return VanillaTerrainPlan.Build(request);
        if (command == "settled_probe")
        {
            var player = VanillaFrame.RestoreInWorker(request["snapshot"].AsObject());
            foreach (var key in request["prefix"].AsArray()) VanillaPrediction.Advance(player, (VanillaKeys)key.GetValue<int>());
            var state = VanillaPrediction.Observe(player);
            bool supported = VanillaTerrainPlan.HasSupport(player);
            bool safeBody = VanillaContactRegion.SafeBody(player.position, player.width, player.height);
            bool solidCollision = Collision.SolidCollision(player.position, player.width, player.height);
            var contacts = new JsonArray();
            for (int x = (int)Math.Floor(player.position.X / 16); x <= (int)Math.Floor(player.Right.X / 16); x++)
            for (int y = (int)Math.Floor(player.position.Y / 16); y <= (int)Math.Floor(player.Bottom.Y / 16); y++)
            {
                var tile = Main.tile[x, y];
                if (tile.HasTile || VanillaContactRegion.Hazard(x, y))
                    contacts.Add(new JsonObject { ["x"] = x, ["y"] = y, ["type"] = tile.TileType,
                        ["slope"] = (int)tile.Slope, ["hazard"] = VanillaContactRegion.Hazard(x, y) });
            }
            bool contactSupported = VanillaContactRegion.Supported(player.position, player.width, player.height);
            var released = new JsonArray();
            for (int i = 0; i < 60; i++)
            {
                VanillaPrediction.Advance(player, VanillaKeys.None);
                released.Add(VanillaPrediction.Observe(player, false));
            }
            return new JsonObject { ["state"] = state, ["supported"] = supported,
                ["contact_supported"] = contactSupported, ["safe_body"] = safeBody,
                ["solid_collision"] = solidCollision, ["contacts"] = contacts, ["released"] = released };
        }
        if (command == "jump_place_context")
        {
            var snapshot = request["snapshot"].AsObject();
            var plan = VanillaTerrainPlan.Build(request);
            if (plan["status"]?.GetValue<string>() != "planned") throw new InvalidOperationException("context_probe_requires_plan");
            var trajectory = plan["validation_frames"].AsArray();
            try
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                if (!VanillaFrame.ContextMatches(snapshot, player, out _, trajectory))
                    throw new InvalidOperationException("unchanged_placement_context_rejected");
                int farX = snapshot["area"][0].GetValue<int>() + 2, farY = snapshot["area"][1].GetValue<int>() + 2;
                if (trajectory.Any(frame => Math.Abs(frame["x"].GetValue<float>() / 16 - farX) < 8
                    && Math.Abs(frame["y"].GetValue<float>() / 16 - farY) < 8))
                    throw new InvalidOperationException("probe_tile_not_outside_trajectory");
                var far = Main.tile[farX, farY]; far.LiquidType = LiquidID.Lava; far.LiquidAmount = 255;
                if (VanillaFrame.ContextMatches(snapshot, player) || !VanillaFrame.ContextMatches(snapshot, player, out _, trajectory))
                    throw new InvalidOperationException("unrelated_lava_did_not_stay_outside_placement_context");
                player = VanillaFrame.RestoreInWorker(snapshot);
                var target = Main.tile[request["tile_x"].GetValue<int>(), request["tile_y"].GetValue<int>()];
                target.LiquidType = LiquidID.Lava; target.LiquidAmount = 255;
                if (VanillaFrame.ContextMatches(snapshot, player, out _, trajectory))
                    throw new InvalidOperationException("placement_context_ignored_target_lava");
                return new JsonObject { ["passed"] = true, ["cases"] = 3 };
            }
            finally { VanillaFrame.RestoreInWorker(snapshot); }
        }
        if (command == "placement_sequence")
        {
            var snapshot = request["snapshot"].AsObject();
            int x = request["tile_x"].GetValue<int>(), y = request["tile_y"].GetValue<int>();
            int placeAt = request["place_after"].GetValue<int>();
            var sequence = request["controls"].AsArray();
            if (placeAt < 0 || placeAt >= sequence.Count || sequence.Count > 240)
                throw new ArgumentException("invalid_placement_sequence");
            var timer = System.Diagnostics.Stopwatch.StartNew();
            try
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                int hp = player.statLife;
                var trace = new JsonArray();
                bool placed = false;
                bool nativeUse = request["native_use"]?.GetValue<bool>() == true;
                int slot = Array.FindIndex(player.inventory, item => !item.IsAir && item.createTile == TileID.Platforms);
                if (nativeUse && slot < 0) throw new InvalidOperationException("missing_platform_item");
                int stock = slot >= 0 ? player.inventory[slot].stack : 0;
                string status = "complete";
                for (int frame = 0; frame < sequence.Count; frame++)
                {
                    VanillaPrediction.Advance(player, (VanillaKeys)sequence[frame].GetValue<int>(),
                        nativeUse && frame >= placeAt && !placed ? slot : -1, new Point(x, y));
                    if (nativeUse) placed |= Main.tile[x, y].HasTile && Main.tile[x, y].TileType == TileID.Platforms;
                    if (!nativeUse && frame == placeAt)
                    {
                        if (player.Hitbox.Intersects(new Rectangle(x * 16, y * 16, 16, 16)))
                        { status = "body_blocks_placement"; break; }
                        if (LavaSafety.IsLava(x, y) || Main.tile[x, y].HasTile)
                        { status = "occupied_or_lava"; break; }
                        WorldGen.PlaceTile(x, y, TileID.Platforms, mute: true, plr: player.whoAmI);
                        placed = Main.tile[x, y].HasTile && Main.tile[x, y].TileType == TileID.Platforms;
                        if (!placed) { status = "placement_rejected"; break; }
                    }
                    trace.Add(VanillaPrediction.Observe(player, false));
                    if (player.dead || player.statLife < hp) { status = "damaged"; break; }
                }
                return new JsonObject { ["status"] = status, ["placed"] = placed, ["frames"] = trace,
                    ["consumed"] = slot >= 0 ? stock - player.inventory[slot].stack : 0,
                    ["supported"] = VanillaContactRegion.Supported(player.position, player.width, player.height),
                    ["elapsed_ms"] = timer.Elapsed.TotalMilliseconds,
                    ["scope"] = nativeUse ? "native Player.Update item use; isolated engine"
                        : "geometry probe using WorldGen placement; item reach and use timing are not validated" };
            }
            finally { VanillaFrame.RestoreInWorker(snapshot); }
        }
        if (command == "contact_descent_region")
        {
            Setup(new JsonObject());
            var landing = source.position;
            var snapshot = VanillaFrame.Capture(source, area, true);
            int passed = 0;
            foreach (int offset in new[] { -24, 24 })
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                player.position += new Vector2(offset, -64);
                player.velocity = new Vector2(0, 2);
                player.fallStart = (int)(player.position.Y / 16);
                if (!VanillaContactRegion.SafeDescent(player, landing))
                    throw new InvalidOperationException("safe_shifted_descent_rejected");
                var anchor = VanillaPrediction.Observe(player, false);
                var endpoint = anchor.DeepClone();
                endpoint["x"] = landing.X; endpoint["y"] = landing.Y;
                endpoint["vx"] = 0f; endpoint["vy"] = 0f;
                var plan = new JsonObject { ["selected"] = "contact_region", ["wait_frames"] = 0,
                    ["context"] = VanillaFrame.Capture(player, area, true), ["anchor"] = anchor,
                    ["forecast"] = new JsonObject { ["frames"] = new JsonArray(endpoint) },
                    ["controls"] = new JsonArray(0), ["route_progress"] = 1 };
                var controller = new VanillaAsyncMovement(_ => throw new InvalidOperationException("unexpected_request"));
                if (!controller.InstallTerrain(plan, player, 0, landing))
                    throw new InvalidOperationException("contact_region_install_failed");
                player.position.Y = landing.Y; player.velocity = Vector2.Zero;
                if (!VanillaContactRegion.SafeDescent(player, landing))
                    throw new InvalidOperationException("safe_shifted_landing_rejected");
                if (!controller.CurrentMatches(0, player) || !controller.NeedsPlan(0, player, false)
                    || !controller.ContactDescentLanded)
                    throw new InvalidOperationException("early_shifted_landing_did_not_handoff");
                passed += 3;
            }
            foreach (string hazard in new[] { "lava", "spike", "wall", "gap", "fall_damage", "horizontal_speed" })
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                player.position += new Vector2(48, -64);
                player.velocity = new Vector2(0, 2);
                player.fallStart = (int)(player.position.Y / 16);
                int x = (int)(player.position.X / 16), y = (int)((landing.Y + player.height) / 16);
                if (hazard == "lava") { var tile = Main.tile[x, y - 1]; tile.LiquidType = LiquidID.Lava; tile.LiquidAmount = 255; }
                if (hazard == "spike") { var tile = Main.tile[x, y]; tile.HasTile = true; tile.TileType = TileID.Spikes; }
                if (hazard == "wall") { var tile = Main.tile[x, y - 3]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
                if (hazard == "gap")
                    for (int tx = (int)(landing.X / 16) + 1; tx <= x + 2; tx++) Main.tile[tx, y].ClearEverything();
                if (hazard == "fall_damage") player.fallStart = (int)(landing.Y / 16) - 100;
                if (hazard == "horizontal_speed") player.velocity.X = 1;
                if (VanillaContactRegion.SafeDescent(player, landing))
                    throw new InvalidOperationException("unsafe_shifted_descent_accepted:" + hazard);
                passed++;
            }
            VanillaFrame.RestoreInWorker(snapshot);
            return new JsonObject { ["passed"] = true, ["cases"] = passed };
        }
        if (command == "contact_descent_replay")
        {
            var evidence = request["evidence"].AsObject();
            var snapshot = evidence["snapshot"].AsObject();
            var player = VanillaFrame.RestoreInWorker(snapshot);
            var remaining = evidence["remaining_controls"].AsArray();
            var frames = evidence["active_plan"]["forecast"]["frames"].AsArray();
            var tail = new JsonArray(frames.Skip(frames.Count - remaining.Count).Select(frame => frame.DeepClone()).ToArray());
            var plan = new JsonObject { ["status"] = "planned", ["selected"] = "recorded_descent", ["wait_frames"] = 0,
                ["context"] = snapshot.DeepClone(), ["anchor"] = VanillaPrediction.Observe(player, false),
                ["route_progress"] = 100, ["forecast"] = new JsonObject { ["frames"] = tail },
                ["controls"] = remaining.DeepClone() };
            var controller = new VanillaAsyncMovement(_ => throw new InvalidOperationException("unexpected_request"));
            if (!controller.InstallTerrain(plan, player, 0, new Vector2(evidence["x"].GetValue<float>(), evidence["y"].GetValue<float>())))
                throw new InvalidOperationException("descent_install_failed");
            float maxError = 0;
            int landedAt = 0, hp = player.statLife;
            for (int frame = 1; frame < remaining.Count; frame++)
            {
                VanillaPrediction.Advance(player, (VanillaKeys)controller.InputAt(frame));
                controller.Poll(frame, player, 0);
                if (!controller.CurrentMatches(frame, player) || player.statLife < hp || player.dead)
                    throw new InvalidOperationException("unsafe_contact_descent:" + frame + ":" + controller.LastMismatch
                        + ":actual=" + VanillaPrediction.Observe(player, false) + ":landing=" + tail[^1]);
                maxError = Math.Max(maxError, Math.Abs(player.position.Y - tail[frame - 1]["y"].GetValue<float>()));
                bool needsPlan = controller.NeedsPlan(frame, player, false);
                if (controller.ContactDescentLanded) { landedAt = frame; if (!needsPlan) throw new InvalidOperationException("landed_without_handoff"); break; }
                if (needsPlan) throw new InvalidOperationException("descent_replanned_in_air");
            }
            if (landedAt == 0 || maxError < 1 || controller.ContactDescents != 1)
                throw new InvalidOperationException("descent_replay_did_not_exercise_tolerance");
            var landing = player.position;
            player = VanillaFrame.RestoreInWorker(snapshot);
            var foot = Main.tile[(int)(landing.X / 16), (int)((landing.Y + player.height - 4) / 16)];
            foot.LiquidType = LiquidID.Lava; foot.LiquidAmount = 255;
            if (VanillaContactRegion.SafeDescent(player, landing)) throw new InvalidOperationException("descent_ignored_lava");
            player = VanillaFrame.RestoreInWorker(snapshot);
            player.fallStart = (int)(landing.Y / 16) - 100;
            if (VanillaContactRegion.SafeDescent(player, landing)) throw new InvalidOperationException("descent_ignored_fall_damage");
            VanillaFrame.RestoreInWorker(snapshot);
            return new JsonObject { ["landed_frame"] = landedAt, ["max_y_deviation"] = maxError, ["hp"] = hp,
                ["contact_descents"] = controller.ContactDescents, ["hazards_rejected"] = true };
        }
        if (command == "contact_cycle_replay")
        {
            var player = VanillaFrame.RestoreInWorker(request["snapshot"].AsObject());
            var progress = new VanillaContactRegion.Progress();
            var field = VanillaLocalPlanner.ReadGuidance(request["guidance"]);
            var events = new JsonArray();

                else controller.Poll(1, source, 1);
                if (!received.IsCancellationRequested || controller.Busy)
                    throw new InvalidOperationException("obsolete_request_was_not_cancelled");
                delayed.SetCanceled();
            }
            return new JsonObject { ["passed"] = true, ["scope"] = "coordinator contract with a synthetic transport, not movement validation" };
        }
        if (command == "terrain_probe")
        {
            var state = request["snapshot"].AsObject();
            try
            {
                var prefix = request["prefix"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                var future = new VanillaPrediction(state).Predict(prefix, false, true)["end_snapshot"].AsObject();
                var player = VanillaFrame.RestoreInWorker(future);
                int left = (int)Math.Floor(player.Left.X / 16), right = (int)Math.Floor((player.Right.X - 1) / 16);
                int bottom = (int)Math.Floor(player.Bottom.Y / 16);
                var clearing = new JsonArray();
                for (int yy = bottom; yy <= bottom + 2; yy++)
                for (int xx = left - 1; xx <= right + 1; xx++)
                {
                    var tile = Main.tile[xx, yy];
                    if (tile.HasTile && Main.tileCut[tile.TileType] && !Main.tileSolid[tile.TileType])
                        clearing.Add(new JsonArray(xx, yy));
                }
                var results = new JsonArray();
                for (int y = bottom; y <= bottom + 1; y++)
                for (int x = left; x <= right; x++)
                foreach (int wait in new[] { 1, 24 })
                foreach (bool clear in new[] { false, true })
                {
                    var targets = clear ? (JsonArray)clearing.DeepClone() : new JsonArray();
                    if (!targets.Any(tile => tile[0].GetValue<int>() == x && tile[1].GetValue<int>() == y))
                        targets.Add(new JsonArray(x, y));
                    var probe = new JsonObject { ["command"] = "terrain_plan", ["snapshot"] = future.DeepClone(),
                        ["tile_x"] = x, ["tile_y"] = y, ["tile_type"] = -1, ["wait_frames"] = wait,
                        ["tiles"] = targets,
                        ["x"] = request["x"].DeepClone(), ["y"] = request["y"].DeepClone(),
                        ["route_direction"] = request["route_direction"].DeepClone(), ["guidance"] = request["guidance"].DeepClone() };
                    var result = VanillaTerrainPlan.Build(probe);
                    if (result["status"].GetValue<string>() == "planned")
                    {
                        var actual = VanillaFrame.RestoreInWorker(future);
                        foreach (var target in targets)
                        {
                            int tx = target[0].GetValue<int>(), ty = target[1].GetValue<int>();
                            int before = Main.tile[tx, ty].TileType;
                            WorldGen.KillTile(tx, ty, noItem: true);
                            if (Main.tile[tx, ty].HasTile && Main.tile[tx, ty].TileType != before)
                                WorldGen.KillTile(tx, ty, noItem: true);
                        }
                        float error = 0;
                        int initialHp = actual.statLife;
                        var replayKeys = result["controls"].AsArray();
                        for (int i = 0; i < replayKeys.Count; i++)
                        {
                            VanillaPrediction.Advance(actual, (VanillaKeys)replayKeys[i].GetValue<int>());
                            var expected = result["forecast"]["frames"][i];
                            error = Math.Max(error, Vector2.Distance(actual.position,
                                new Vector2(expected["x"].GetValue<float>(), expected["y"].GetValue<float>())));
                            if (error > .001f || actual.statLife < initialHp || actual.dead)
                                throw new InvalidOperationException("terrain_probe_replay_diverged");
                        }
                        result["replay_max_error"] = error;
                    }
                    results.Add(new JsonObject { ["x"] = x, ["y"] = y, ["wait"] = wait, ["clear"] = clear, ["tiles"] = targets.DeepClone(), ["result"] = result });
                }
                return new JsonObject { ["results"] = results };
            }
            finally { VanillaFrame.RestoreInWorker(state); }
        }
        if (command == "native_mining_probe")
        {
            Setup(new JsonObject());
            source.inventory[0].SetDefaults(ItemID.CopperPickaxe);
            var target = new Point(X + 2, Y - 1);
            var targetTile = Main.tile[target.X, target.Y];
            targetTile.HasTile = true;
            targetTile.TileType = TileID.Stone;
            int Damage(Player player) => player.hitTile.data.Where(hit => hit.type == 1
                && hit.X == target.X && hit.Y == target.Y).Select(hit => hit.damage).DefaultIfEmpty().Max();
            var initial = VanillaFrame.Capture(source, area, true);
            var result = new JsonArray();
            foreach (int checkpointMode in new[] { 0, 1, 2 })
            {
                var parent = new VanillaFrame.WorkerState(initial);
                var player = parent.Restore();
                int firstDamage = 0, restoredDamage = 0, frames = 0;
                for (; frames < 120 && Main.tile[target.X, target.Y].HasTile; frames++)
                {
                    VanillaPrediction.Advance(player, VanillaKeys.None, 0, target);
                    if (firstDamage == 0) firstDamage = Damage(player);
                    if (checkpointMode != 0 && (frames + 1) % 6 == 0)
                    {
                        if (checkpointMode == 1)
                        {
                            parent = parent.CaptureCurrent(player);
                            player = parent.Restore();
                        }
                        else player = VanillaFrame.RestoreInWorker(VanillaFrame.Capture(player, area, true));
                        if (restoredDamage == 0) restoredDamage = Damage(player);
                    }
                }
                result.Add(new JsonObject { ["checkpoint_mode"] = checkpointMode,
                    ["broken"] = !Main.tile[target.X, target.Y].HasTile, ["frames"] = frames,
                    ["first_damage"] = firstDamage, ["restored_damage"] = restoredDamage });
            }
            return new JsonObject { ["results"] = result };
        }
        if (command == "worker_state_contract")
        {
            Setup(new JsonObject { ["terrain"] = "flat" });
            var initial = VanillaFrame.Capture(source, area, true);
            var parent = new VanillaFrame.WorkerState(initial);
            var player = parent.Restore();
            var tile = Main.tile[X + 5, Y - 5];
            tile.HasTile = true; tile.TileType = TileID.Platforms; tile.LiquidAmount = 71;
            tile.WallType = WallID.DirtUnsafe; tile.TileFrameX = 18; tile.TileFrameY = 36;
            tile.Slope = SlopeType.SlopeDownRight; tile.IsActuated = true; tile.LiquidType = LiquidID.Honey;
            tile.TileFrameNumber = 2;
            var edge = Main.tile[area.Right - 1, area.Bottom - 1];
            edge.HasTile = true; edge.TileType = TileID.Stone; edge.IsHalfBlock = true;
            player.position.X += 7; player.buffTime[0] += 17; player.inventory[0].stack += 3;
            var expected = VanillaFrame.Capture(player, area, true);
            var child = parent.CaptureCurrent(player);
            if (child.CapturedTileChanges != 2) throw new InvalidOperationException("branch_tile_delta_mismatch");
            if (child.CaptureCurrent(player).CapturedTileChanges != 0)
                throw new InvalidOperationException("unchanged_branch_reported_tile_changes");
            player = parent.Restore(child);
            if (!JsonNode.DeepEquals(initial, VanillaFrame.Capture(player, area, true)))
                throw new InvalidOperationException("child_polluted_parent");
            parent.Restore(parent);
            if (parent.RestoredTileWrites != 0) throw new InvalidOperationException("unchanged_world_rewritten");
            player = child.Restore(parent);
            if (Main.tile[X + 5, Y - 5].TileFrameNumber != 2)
                throw new InvalidOperationException($"branch_frame_number_lost:{Main.tile[X + 5, Y - 5].TileFrameNumber}");
            if (!JsonNode.DeepEquals(expected, VanillaFrame.Capture(player, area, true)))
                throw new InvalidOperationException("branch_restore_changed_state");
            player.buffTime[0]++; player.inventory[0].stack++;
            player = child.Restore();
            if (!JsonNode.DeepEquals(expected, VanillaFrame.Capture(player, area, true)))
                throw new InvalidOperationException("branch_arrays_or_items_aliased");
            parent.Restore();
            return new JsonObject { ["passed"] = true, ["changed_tiles"] = child.CapturedTileChanges };
        }
        if (command == "region_replay")
        {
            var snapshot = request["snapshot"].AsObject();
            var inputs = VanillaPrediction.RegionInputs(request);
            try
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                int hp = player.statLife;
                var playback = request["frames"] is JsonArray ? new VanillaPrediction.RegionPlayback(request, 0) : null;
                if (playback != null && !playback.Matches(player, 0, out _))
                    throw new InvalidOperationException("region_playback_anchor_mismatch");
                int checkpointFrames = request["checkpoint_frames"]?.GetValue<int>() ?? 0;
                var randomIndices = new JsonArray();
                var checkpoint = checkpointFrames > 0 ? new VanillaFrame.WorkerState(snapshot) : null;
                for (int i = 0; i < inputs.Length; i++)
                {
                    randomIndices.Add(WorldRandomIndex());
                    traceRandom = i == 0 ? "replay" : null;
                    var input = playback?.At(i + 1) ?? inputs[i];
                    if (input != inputs[i] || playback != null && playback.At(i + 1) != input)
                        throw new InvalidOperationException("region_playback_input_mismatch");
                    VanillaPrediction.Advance(player, input.Keys, input.UseSlot, input.Target);
                    if (playback != null && !playback.Matches(player, i + 1, out string mismatch))
                        throw new InvalidOperationException("region_playback_state_mismatch:" + mismatch);
                    traceRandom = null;
                    randomIndices.Add(WorldRandomIndex());
                    if (player.dead || player.statLife < hp) return new JsonObject { ["outcome"] = "damaged" };
                    if (checkpointFrames > 0 && (i + 1) % checkpointFrames == 0)
                    {
                        checkpoint = checkpoint.CaptureCurrent(player);
                        player = checkpoint.Restore();
                    }
                }
                var a = snapshot["area"].AsArray();
                if (playback != null && (!playback.Complete(inputs.Length) || playback.HasInput(inputs.Length + 1)))
                    throw new InvalidOperationException("region_playback_end_mismatch");
                return new JsonObject { ["outcome"] = "complete", ["playback_verified"] = playback != null,
                    ["world_random_indices"] = randomIndices, ["end_snapshot"] = VanillaFrame.Capture(player,
                    new Rectangle(a[0].GetValue<int>(), a[1].GetValue<int>(), a[2].GetValue<int>(), a[3].GetValue<int>()), true) };
            }
            finally { VanillaFrame.RestoreInWorker(snapshot); }
        }
        if (command == "slope_body_contract")
        {
            Setup(new JsonObject { ["terrain"] = "flat" });
            var snapshot = VanillaFrame.Capture(source, area, true);
            int checkedBodies = 0;
            try
            {
                var tile = Main.tile[X + 5, Y - 5];
                tile.ClearEverything(); tile.HasTile = true; tile.TileType = TileID.Stone;
                foreach (var slope in new[] { SlopeType.SlopeDownRight, SlopeType.SlopeDownLeft,
                    SlopeType.SlopeUpRight, SlopeType.SlopeUpLeft })
                foreach (int dx in new[] { 2, 10 })
                foreach (int dy in new[] { 3, 11 })
                {
                    tile.Slope = slope;
                    var position = new Vector2((X + 5) * 16 + dx, (Y - 5) * 16 + dy);
                    var native = Collision.SlopeCollision(position, Vector2.Zero, 2, 2);
                    bool corrected = native.X != position.X || native.Y != position.Y;
                    if (VanillaContactRegion.SafeBody(position, 2, 2) == corrected)
                        throw new InvalidOperationException($"slope_body_mismatch:{slope}:{dx},{dy}");
                    checkedBodies++;
                }
                return new JsonObject { ["passed"] = true, ["cases"] = checkedBodies };
            }
            finally { VanillaFrame.RestoreInWorker(snapshot); }
        }
        if (command == "snapshot_expansion_contract")
        {
            Setup(new JsonObject { ["terrain"] = "flat" });
            int x = area.Left - 1, y = Y - 5;
            var tile = Main.tile[x, y];
            tile.ClearEverything(); tile.HasTile = true; tile.TileType = TileID.Stone;
            var snapshot = VanillaFrame.Capture(source, area, true);
            var expanded = Handle(new JsonObject { ["command"] = "expand_snapshot", ["snapshot"] = snapshot,
                ["missing_tile"] = new JsonArray(x, y) });
            if (expanded["status"]?.GetValue<string>() != "expanded")
                throw new InvalidOperationException("snapshot_expansion_failed");
            tile.ClearEverything();
            VanillaFrame.RestoreInWorker(expanded["snapshot"].AsObject());
            if (!tile.HasTile || tile.TileType != TileID.Stone)
                throw new InvalidOperationException("expanded_snapshot_missing_source_tile");
            if (!VanillaFrame.WorkerBounds.Contains(area) || !VanillaFrame.WorkerBounds.Contains(x, y))
                throw new InvalidOperationException("expanded_snapshot_missing_previous_area");
            return new JsonObject { ["passed"] = true };
        }
        if (command == "expand_snapshot")
        {
            var snapshot = request["snapshot"].AsObject();
            var bounds = snapshot["area"].AsArray();
            var missing = request["missing_tile"].AsArray();
            int x = missing[0].GetValue<int>(), y = missing[1].GetValue<int>();
            var current = new Rectangle(bounds[0].GetValue<int>(), bounds[1].GetValue<int>(),
                bounds[2].GetValue<int>(), bounds[3].GetValue<int>());
            if (current.Contains(x, y)) throw new ArgumentException("missing_tile_already_observed");
            var expanded = Rectangle.Union(current, Rectangle.Intersect(new Rectangle(x - 16, y - 16, 33, 33),
                new Rectangle(0, 0, Main.maxTilesX, Main.maxTilesY)));
            if (expanded.Width > 256 || expanded.Height > 256)
                return new JsonObject { ["status"] = "snapshot_size_limit" };
            return new JsonObject { ["status"] = "expanded",
                ["snapshot"] = VanillaFrame.Capture(VanillaFrame.RestoreInWorker(snapshot), expanded, true) };
        }
        if (command == "region_boundary_contract")
        {
            var snapshot = request["snapshot"].AsObject();
            var bounds = snapshot["area"].AsArray();
            var missing = request["missing_tile"].AsArray();
            var outer = Rectangle.Union(new Rectangle(bounds[0].GetValue<int>(), bounds[1].GetValue<int>(),
                bounds[2].GetValue<int>(), bounds[3].GetValue<int>()),
                new Rectangle(missing[0].GetValue<int>() - 1, missing[1].GetValue<int>() - 1, 3, 3));
            var before = VanillaFrame.Capture(VanillaFrame.RestoreInWorker(snapshot), outer, true);
            for (int attempt = 0; attempt < 2; attempt++)
            {
                var result = RegionSearch(request);
                if (result["status"]?.GetValue<string>() != "snapshot_required"
                    || !JsonNode.DeepEquals(result["missing_tile"], missing))
                    throw new InvalidOperationException("boundary_failure_not_repeatable");
                var after = VanillaFrame.Capture(VanillaFrame.RestoreInWorker(snapshot), outer, true);
                if (!JsonNode.DeepEquals(before, after)) throw new InvalidOperationException("boundary_probe_changed_world");
            }
            return new JsonObject { ["passed"] = true, ["attempts"] = 2, ["missing_tile"] = missing.DeepClone() };
        }
        if (command == "region_surfaces") return RegionSurfaces(request);
        if (command == "region_corridor") return CorridorRegions(request);
        if (command == "region_plan") return RegionPlan(request);
        if (command == "region_search") return RegionSearch(request);
        if (command == "plan")
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();
            var state = request["snapshot"].AsObject();
            try
            {
                var prefix = request["prefix"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                var advanced = new VanillaPrediction(state).Predict(prefix, false, true);
                if (advanced["end_snapshot"] is not JsonObject future)
                    return new JsonObject { ["status"] = "unsafe_handoff" };
                var anchor = VanillaPrediction.Observe(VanillaFrame.RestoreInWorker(future));
                var result = new VanillaLocalPlanner().Plan(future,
                    new Vector2(request["x"].GetValue<float>(), request["y"].GetValue<float>()),
                    continuation: request["continuation"]?.AsArray().Select(key => key.GetValue<int>()).ToArray(),
                    guidance: request["guidance"], terminal: request["terminal"]?.GetValue<bool>() == true,
                    routeDirection: request["route_direction"]?.GetValue<int>() ?? 0,
                    preferContinuation: request["recover"]?.GetValue<bool>() == true,
                    continuationOnly: request["continuation_only"]?.GetValue<bool>() == true);
                result["anchor"] = anchor;
                result["request_compute_ms"] = timer.Elapsed.TotalMilliseconds;
                return result;
            }
            finally { VanillaFrame.RestoreInWorker(state); }
        }
        if (command == "reset")
        {
            Setup(request);
            checkpoint = VanillaFrame.Capture(source, area);
            return new JsonObject { ["initial"] = VanillaPrediction.Observe(source), ["snapshot"] = checkpoint.DeepClone() };
        }
        if (command == "load")
        {
            var incoming = request["snapshot"].AsObject();
            source = VanillaFrame.RestoreInWorker(incoming);
            checkpoint = (JsonObject)incoming.DeepClone();
            return new JsonObject { ["initial"] = VanillaPrediction.Observe(source) };
        }
        if (checkpoint == null) throw new InvalidOperationException("reset_required");
        if (command == "navigate")
        {
            var trace = new JsonArray();
            var plans = new JsonArray();
            int[] continuation = null;
            var goal = new Vector2(request["x"].GetValue<float>(), request["y"].GetValue<float>());
            int limit = Math.Clamp(request["limit"]?.GetValue<int>() ?? 240, 8, 600);
            bool reached = false;
            float maximumError = 0;
            for (int frame = 0; frame < limit;)
            {
                var snapshotTimer = System.Diagnostics.Stopwatch.StartNew();
                checkpoint = VanillaFrame.Capture(source, VanillaLocalPlanner.SnapshotArea(source, goal));
                double snapshotMs = snapshotTimer.Elapsed.TotalMilliseconds;
                var before = VanillaFrame.CaptureFields(source);
                var plan = new VanillaLocalPlanner().Plan(checkpoint, goal, continuation);
                plan["snapshot_ms"] = snapshotMs;
                plan["snapshot_tiles"] = checkpoint["area"][2].GetValue<int>() * checkpoint["area"][3].GetValue<int>();
                if (!JsonNode.DeepEquals(before, VanillaFrame.CaptureFields(source))) throw new InvalidOperationException("planner_mutated_source");
                plans.Add(plan);
                if (plan["status"].GetValue<string>() != "planned") break;
                var keys = plan["controls"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                Main.player[source.whoAmI] = source;
                int count = Math.Min(VanillaLocalPlanner.ExecuteFrames, limit - frame);
                for (int i = 0; i < count; i++)
                {
                    VanillaPrediction.Advance(source, (VanillaKeys)keys[i]);
                    var expected = plan["forecast"]["frames"][i];
                    float error = Vector2.Distance(source.position, new Vector2(expected["x"].GetValue<float>(), expected["y"].GetValue<float>()));
                    maximumError = Math.Max(error, maximumError);
                    if (error > .001f) throw new InvalidOperationException("execution_diverged_from_prediction");
                    trace.Add(VanillaPrediction.Observe(source)); frame++;
                    if (source.dead) throw new InvalidOperationException("navigation_died");
                    if (Vector2.Distance(source.position, goal) <= 20 && Math.Abs(source.velocity.Y) < .01f) { reached = true; break; }
                }
                continuation = keys.Skip(count).ToArray();
                if (reached) break;
            }
            checkpoint = VanillaFrame.Capture(source, area);
            return new JsonObject { ["reached"] = reached, ["trace"] = trace, ["plans"] = plans, ["max_error"] = maximumError };
        }
        int[] controls = request["controls"]?.AsArray().Select(node => node.GetValue<int>()).ToArray() ?? Array.Empty<int>();
        if (command == "predict")
        {
            var original = VanillaFrame.CaptureFields(source);
            var result = new VanillaPrediction(checkpoint).Predict(controls);
            result["source_unchanged"] = JsonNode.DeepEquals(original, VanillaFrame.CaptureFields(source));
            result["checkpoint_restored"] = JsonNode.DeepEquals(checkpoint,
                VanillaFrame.Capture(Main.player[source.whoAmI], area));
            return result;
        }
        if (command == "actual")
        {
            var trace = new JsonArray();
            Main.player[source.whoAmI] = source;
            foreach (int control in controls) { VanillaPrediction.Advance(source, (VanillaKeys)control); trace.Add(VanillaPrediction.Observe(source)); }
            return new JsonObject { ["frames"] = trace };
        }
        throw new ArgumentException("unknown_command");
    }

    public void Setup(JsonObject request, bool worker = true)
    {
        Main.gameMenu = false;
        if (worker) Main.myPlayer = 255;
        Main.rand = new Terraria.Utilities.UnifiedRandom(73);
        VanillaFrame.RestoreFields(WorldGen.genRand, VanillaFrame.CaptureFields(new Terraria.Utilities.UnifiedRandom(81)));
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        foreach (var item in Main.item) item.active = false;
        for (int x = area.Left; x < area.Right; x++)
        for (int y = area.Top; y < area.Bottom; y++)
        {
            var tile = Main.tile[x, y]; tile.ClearEverything();
            if (y >= Y) { tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        source = new Player { whoAmI = 0, active = true, name = "Forecast subject", statLife = 400, statLifeMax = 400, ConsumedLifeCrystals = 15 };
        if (!worker)
        {
            VanillaFrame.RestoreFields(Main.LocalPlayer, VanillaFrame.CaptureFields(source));
            source = Main.LocalPlayer;
            foreach (var item in source.armor.Concat(source.inventory).Concat(source.miscEquips)) item.SetDefaults(ItemID.None);
        }
        source.position = new Vector2(X * 16, Y * 16 - source.height);
        source.fallStart = (int)(source.position.Y / 16);
        Main.player[0] = source;
        int slot = 3;
        if (request["equipment"] is JsonArray items)
            foreach (var item in items) source.armor[slot++].SetDefaults(item.GetValue<int>());
        if (!worker) Main.ActivePlayerFileData.Player = source;
        if (worker) for (int i = 0; i < 3; i++) VanillaPrediction.Advance(source, VanillaKeys.None);
        if (request["buffs"] is JsonArray buffs)
            foreach (var buff in buffs) source.AddBuff(buff[0].GetValue<int>(), buff[1].GetValue<int>());
        string terrain = request["terrain"]?.GetValue<string>() ?? "flat";
        if (terrain == "ceiling")
            for (int x = X - 3; x < X + 18; x++) { var tile = Main.tile[x, Y - 6]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
        if (terrain is "half" or "slope")
            for (int x = X + 2; x < X + 12; x++)
            {
                var tile = Main.tile[x, Y - 1]; tile.HasTile = true; tile.TileType = TileID.Dirt;
                if (terrain == "half") tile.IsHalfBlock = true;
                else tile.Slope = SlopeType.SlopeDownRight;
            }
        if (terrain == "honey_block")
            for (int x = X - 2; x < X + 20; x++) { var tile = Main.tile[x, Y]; tile.TileType = TileID.HoneyBlock; }
        if (terrain == "honey")
            for (int x = X - 2; x < X + 20; x++)
            for (int y = Y - 4; y < Y; y++) { var tile = Main.tile[x, y]; tile.LiquidType = LiquidID.Honey; tile.LiquidAmount = 255; }
        if (terrain == "embedded")
        {
            var tile = Main.tile[X, Y - 2]; tile.HasTile = true; tile.TileType = TileID.Dirt;
        }
        if (terrain is "ledge" or "wall")
        {
            int height = request["height"]?.GetValue<int>() ?? 12;
            for (int x = X + 9; x < X + (terrain == "ledge" ? 38 : 12); x++)
            for (int y = Y - height; y < Y; y++) { var tile = Main.tile[x, y]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        if (terrain == "lava_gap")
            for (int x = X + 7; x < X + 20; x++)
            for (int y = Y; y < Y + 12; y++)
            {
                var tile = Main.tile[x, y]; tile.ClearEverything();
                if (y >= Y + 10) { tile.LiquidType = LiquidID.Lava; tile.LiquidAmount = 255; }
            }
        if (worker && terrain is ("honey" or "honey_block"))
            for (int i = 0; i < 3; i++) VanillaPrediction.Advance(source, VanillaKeys.None);
        if (request["airborne"]?.GetValue<bool>() == true)
        {
            source.position.Y -= request["airborne_height"]?.GetValue<int>() ?? 320;
            source.velocity.Y = 3; source.fallStart = (int)(source.position.Y / 16);
        }
        if (worker && request["pre_controls"] is JsonArray prior)
            foreach (var key in prior) VanillaPrediction.Advance(source, (VanillaKeys)key.GetValue<int>());
    }
