    sealed record RegionNode(VanillaFrame.WorkerState State, int[] Controls, float Distance);

    static JsonObject RegionSearch(JsonObject request)
    {
        var initial = request["snapshot"].AsObject();
        var bounds = request["region"].AsArray();
        var region = new Rectangle(bounds[0].GetValue<int>(), bounds[1].GetValue<int>(),
            bounds[2].GetValue<int>(), bounds[3].GetValue<int>());
        int horizon = request["horizon"]?.GetValue<int>() ?? 90;
        int beam = request["beam"]?.GetValue<int>() ?? 6;
        int quantum = request["quantum"]?.GetValue<int>() ?? 6;
        bool requireSupport = request["require_support"]?.GetValue<bool>() ?? true;
        if (region.Width <= 0 || region.Height <= 0 || horizon < 1 || horizon > 240
            || beam < 1 || beam > 16 || quantum < 1 || quantum > 12)
            throw new ArgumentException("invalid_region_search_budget");
        var timer = System.Diagnostics.Stopwatch.StartNew();
        int expanded = 0, simulated = 0, shared = 0;
        long tileChanges = 0, tileWrites = 0;
        double restoreMs = 0, updateMs = 0, captureMs = 0;
        float Distance(Player player)
        {
            var point = player.Bottom;
            float dx = Math.Max(region.Left - point.X, Math.Max(0, point.X - region.Right));
            float dy = Math.Max(region.Top - point.Y, Math.Max(0, point.Y - region.Bottom));
            return MathF.Sqrt(dx * dx + dy * dy);
        }
        var a = initial["area"].AsArray();
        var tiles = new Rectangle(a[0].GetValue<int>(), a[1].GetValue<int>(), a[2].GetValue<int>(), a[3].GetValue<int>());
        JsonObject Finish(RegionNode node, bool reached) => new()
        {
            ["status"] = reached ? "reached" : "budget_exhausted",
            ["distance"] = node.Distance,
            ["controls"] = new JsonArray(node.Controls.Select(key => JsonValue.Create(key)).ToArray()),
            ["end_snapshot"] = VanillaFrame.Capture(node.State.Restore(), tiles, true),
            ["expanded"] = expanded, ["simulated_frames"] = simulated, ["reused_prefix_frames"] = shared,
            ["captured_tile_changes"] = tileChanges, ["restored_tile_writes"] = tileWrites,
            ["elapsed_ms"] = timer.Elapsed.TotalMilliseconds,
            ["timing"] = new JsonObject { ["restore_ms"] = restoreMs, ["update_ms"] = updateMs, ["capture_ms"] = captureMs }
        };
        try
        {
            var start = VanillaFrame.RestoreInWorker(initial);
            int hp = start.statLife;
            var interior = new Rectangle((tiles.X + 2) * 16, (tiles.Y + 2) * 16, (tiles.Width - 4) * 16, (tiles.Height - 4) * 16);
            var best = new RegionNode(new VanillaFrame.WorkerState(initial), Array.Empty<int>(), Distance(start));
            if (best.Distance == 0 && (!requireSupport || start.velocity.Y == 0 && VanillaTerrainPlan.HasSupport(start)))
                return Finish(best, true);
            var frontier = new System.Collections.Generic.List<RegionNode> { best };
            bool gravity = start.buffType.Contains(BuffID.Gravitation);
            int[] actions = gravity ? new[] { 0, 1, 2, 4, 5, 6, 16, 17, 18, 8, 9, 10 }
                : new[] { 0, 1, 2, 4, 5, 6, 16, 17, 18 };
            for (int depth = 0; depth < horizon && frontier.Count > 0; depth += quantum)
            {
                var next = new System.Collections.Generic.List<RegionNode>();
                foreach (var parent in frontier)
                {
                    var checkpoint = parent.State;
                    foreach (int keys in actions)
                    {
                        VanillaPrediction.CheckCancellation();
                        long at = System.Diagnostics.Stopwatch.GetTimestamp();
                        var player = checkpoint.Restore();
                        tileWrites += checkpoint.RestoredTileWrites;
                        restoreMs += System.Diagnostics.Stopwatch.GetElapsedTime(at).TotalMilliseconds;
                        expanded++; shared += parent.Controls.Length;
                        int count = 0;
                        bool safe = true, reached = false;
                        for (; count < Math.Min(quantum, horizon - depth);)
                        {
                            at = System.Diagnostics.Stopwatch.GetTimestamp();
                            VanillaPrediction.Advance(player, (VanillaKeys)keys);
                            updateMs += System.Diagnostics.Stopwatch.GetElapsedTime(at).TotalMilliseconds;
                            count++; simulated++;
                            safe = !player.dead && player.statLife >= hp && interior.Contains(player.Hitbox)
                                && !Main.projectile.Any(projectile => projectile.active)
                                && !Main.npc.Any(npc => npc.active);
                            if (!safe) break;
                            reached = Distance(player) == 0
                                && (!requireSupport || player.velocity.Y == 0 && VanillaTerrainPlan.HasSupport(player));
                            if (reached) break;
                        }
                        if (!safe) continue;
                        at = System.Diagnostics.Stopwatch.GetTimestamp();
                        var state = checkpoint.CaptureCurrent(player);
                        tileChanges += state.CapturedTileChanges;
                        captureMs += System.Diagnostics.Stopwatch.GetElapsedTime(at).TotalMilliseconds;
                        var node = new RegionNode(state, parent.Controls.Concat(Enumerable.Repeat(keys, count)).ToArray(), Distance(player));
                        if (node.Distance < best.Distance) best = node;
                        if (reached) return Finish(node, true);
                        next.Add(node);
                    }
                }
                frontier = next.OrderBy(node => node.Distance).Take(beam).ToList();
            }
            return Finish(best, false);
        }
        finally { VanillaFrame.RestoreInWorker(initial); }
    }

    public override void Load() => Main.OnEnginePreload += Run;
    public override void Unload() => Main.OnEnginePreload -= Run;

    sealed class Work
    {
        public JsonObject Request;
        public long Id;
        public int Cancelled;
    }

    void Run()
    {
        if (!Main.dedServ || !Terraria.Program.LaunchParameters.TryGetValue("-vanillaforecastport", out string value)) return;
        VanillaFrame.RequireWorker();
        var listener = new TcpListener(IPAddress.Loopback, int.Parse(value));
        listener.Start();
        Console.WriteLine("VANILLA_FORECAST_READY");
        try
        {
            using var client = listener.AcceptTcpClient(); client.NoDelay = true;
            using var reader = new StreamReader(client.GetStream());
            using var writer = new StreamWriter(client.GetStream()) { AutoFlush = true };
            using var incoming = new BlockingCollection<Work>();
            var requests = new ConcurrentDictionary<long, Work>();
            long anonymousId = 0;
            _ = Task.Run(() =>
            {
                try
                {
                    string line;
                    while ((line = reader.ReadLine()) != null)
                    {
                        var request = JsonNode.Parse(line).AsObject();
                        long id = request["request_id"]?.GetValue<long>() ?? Interlocked.Decrement(ref anonymousId);
                        if (request["command"]?.GetValue<string>() == "cancel")
                        {
                            if (requests.TryGetValue(id, out var cancelled)) Interlocked.Exchange(ref cancelled.Cancelled, 1);
                            continue;
                        }
                        var work = new Work { Request = request, Id = id };
                        requests[id] = work; incoming.Add(work);
                    }
                }
                catch (Exception error) { Console.WriteLine(error); }
                finally { incoming.CompleteAdding(); }
            });
            foreach (var work in incoming.GetConsumingEnumerable())
            {
                var request = work.Request;
                if (request["command"]?.GetValue<string>() == "close") { writer.WriteLine("{\"closed\":true}"); break; }
                VanillaPrediction.CancellationRequested = () => Volatile.Read(ref work.Cancelled) != 0;
                try
                {
                    if (Terraria.Program.LaunchParameters.ContainsKey("-forecasttrace")) Console.WriteLine("FORECAST_REQUEST_STARTED " + work.Id);
                    VanillaPrediction.CheckCancellation();
                    var result = Handle(request); result["request_id"] = work.Id;
                    writer.WriteLine(result.ToJsonString());
                }
                catch (OperationCanceledException)
                { writer.WriteLine(new JsonObject { ["status"] = "cancelled", ["request_id"] = work.Id }.ToJsonString()); }
                catch (Exception error) { writer.WriteLine(new JsonObject { ["error"] = error.ToString() }.ToJsonString()); }
                finally { VanillaPrediction.CancellationRequested = null; requests.TryRemove(work.Id, out _); }
            }
            Environment.Exit(0);
        }
        catch (Exception error) { Console.WriteLine(error); Environment.Exit(1); }
        finally { listener.Stop(); }
    }

    JsonObject Handle(JsonObject request)
    {
        string command = request["command"].GetValue<string>();
        if (command == "platform_anchor")
        {
            var player = VanillaFrame.RestoreInWorker(request["snapshot"].AsObject());
            int x = request["tile_x"].GetValue<int>(), y = request["tile_y"].GetValue<int>();
            player.inventory[player.selectedItem].SetDefaults(ItemID.WoodPlatform);
            int previousX = Player.tileTargetX, previousY = Player.tileTargetY;
            try
            {
                Player.tileTargetX = x; Player.tileTargetY = y;
                var method = typeof(Player).GetMethod("PlaceThing_Tiles_BlockPlacementForAssortedThings",
                    System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic);
                bool native = (bool)method.Invoke(player, new object[] { false });
                return new JsonObject { ["native"] = native, ["planner"] = VanillaTerrainPlan.HasPlatformAnchor(x, y) };
            }
            finally { Player.tileTargetX = previousX; Player.tileTargetY = previousY; }
        }
        if (command == "terrain_plan") return VanillaTerrainPlan.Build(request);
        if (command == "settled_probe")
        {
            var player = VanillaFrame.RestoreInWorker(request["snapshot"].AsObject());
            foreach (var key in request["prefix"].AsArray()) VanillaPrediction.Advance(player, (VanillaKeys)key.GetValue<int>());
            var state = VanillaPrediction.Observe(player);
            bool supported = VanillaTerrainPlan.HasSupport(player);
            bool contactSupported = VanillaContactRegion.Supported(player.position, player.width, player.height);
            var released = new JsonArray();
            for (int i = 0; i < 60; i++)
            {
                VanillaPrediction.Advance(player, VanillaKeys.None);
                released.Add(VanillaPrediction.Observe(player, false));
            }
            return new JsonObject { ["state"] = state, ["supported"] = supported,
                ["contact_supported"] = contactSupported, ["released"] = released };
        }
        if (command == "jump_place_context")
        {
            var snapshot = request["snapshot"].AsObject();
            var plan = VanillaTerrainPlan.Build(request);
            if (plan["status"]?.GetValue<string>() != "planned") throw new InvalidOperationException("context_probe_requires_plan");
            var trajectory = plan["validation_frames"].AsArray();
            try
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                if (!VanillaFrame.ContextMatches(snapshot, player, out _, trajectory))
                    throw new InvalidOperationException("unchanged_placement_context_rejected");
                int farX = snapshot["area"][0].GetValue<int>() + 2, farY = snapshot["area"][1].GetValue<int>() + 2;
                if (trajectory.Any(frame => Math.Abs(frame["x"].GetValue<float>() / 16 - farX) < 8
                    && Math.Abs(frame["y"].GetValue<float>() / 16 - farY) < 8))
                    throw new InvalidOperationException("probe_tile_not_outside_trajectory");
                var far = Main.tile[farX, farY]; far.LiquidType = LiquidID.Lava; far.LiquidAmount = 255;
                if (VanillaFrame.ContextMatches(snapshot, player) || !VanillaFrame.ContextMatches(snapshot, player, out _, trajectory))
                    throw new InvalidOperationException("unrelated_lava_did_not_stay_outside_placement_context");
                player = VanillaFrame.RestoreInWorker(snapshot);
                var target = Main.tile[request["tile_x"].GetValue<int>(), request["tile_y"].GetValue<int>()];
                target.LiquidType = LiquidID.Lava; target.LiquidAmount = 255;
                if (VanillaFrame.ContextMatches(snapshot, player, out _, trajectory))
                    throw new InvalidOperationException("placement_context_ignored_target_lava");
                return new JsonObject { ["passed"] = true, ["cases"] = 3 };
            }
            finally { VanillaFrame.RestoreInWorker(snapshot); }
        }
        if (command == "placement_sequence")
        {
            var snapshot = request["snapshot"].AsObject();
            int x = request["tile_x"].GetValue<int>(), y = request["tile_y"].GetValue<int>();
            int placeAt = request["place_after"].GetValue<int>();
            var sequence = request["controls"].AsArray();
            if (placeAt < 0 || placeAt >= sequence.Count || sequence.Count > 240)
                throw new ArgumentException("invalid_placement_sequence");
            var timer = System.Diagnostics.Stopwatch.StartNew();
            try
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                int hp = player.statLife;
                var trace = new JsonArray();
                bool placed = false;
                string status = "complete";
                for (int frame = 0; frame < sequence.Count; frame++)
                {
                    VanillaPrediction.Advance(player, (VanillaKeys)sequence[frame].GetValue<int>());
                    if (frame == placeAt)
                    {
                        if (player.Hitbox.Intersects(new Rectangle(x * 16, y * 16, 16, 16)))
                        { status = "body_blocks_placement"; break; }
                        if (LavaSafety.IsLava(x, y) || Main.tile[x, y].HasTile)
                        { status = "occupied_or_lava"; break; }
                        WorldGen.PlaceTile(x, y, TileID.Platforms, mute: true, plr: player.whoAmI);
                        placed = Main.tile[x, y].HasTile && Main.tile[x, y].TileType == TileID.Platforms;
                        if (!placed) { status = "placement_rejected"; break; }
                    }
                    trace.Add(VanillaPrediction.Observe(player, false));
                    if (player.dead || player.statLife < hp) { status = "damaged"; break; }
                }
                return new JsonObject { ["status"] = status, ["placed"] = placed, ["frames"] = trace,
                    ["supported"] = VanillaContactRegion.Supported(player.position, player.width, player.height),
                    ["elapsed_ms"] = timer.Elapsed.TotalMilliseconds,
                    ["scope"] = "geometry probe using WorldGen placement; item reach and use timing are not validated" };
            }
            finally { VanillaFrame.RestoreInWorker(snapshot); }
        }
        if (command == "contact_descent_region")
        {
            Setup(new JsonObject());
            var landing = source.position;
            var snapshot = VanillaFrame.Capture(source, area, true);
            int passed = 0;
            foreach (int offset in new[] { -24, 24 })
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                player.position += new Vector2(offset, -64);
                player.velocity = new Vector2(0, 2);
                player.fallStart = (int)(player.position.Y / 16);
                if (!VanillaContactRegion.SafeDescent(player, landing))
                    throw new InvalidOperationException("safe_shifted_descent_rejected");
                var anchor = VanillaPrediction.Observe(player, false);
                var endpoint = anchor.DeepClone();
                endpoint["x"] = landing.X; endpoint["y"] = landing.Y;
                endpoint["vx"] = 0f; endpoint["vy"] = 0f;
                var plan = new JsonObject { ["selected"] = "contact_region", ["wait_frames"] = 0,
                    ["context"] = VanillaFrame.Capture(player, area, true), ["anchor"] = anchor,
                    ["forecast"] = new JsonObject { ["frames"] = new JsonArray(endpoint) },
                    ["controls"] = new JsonArray(0), ["route_progress"] = 1 };
                var controller = new VanillaAsyncMovement(_ => throw new InvalidOperationException("unexpected_request"));
                if (!controller.InstallTerrain(plan, player, 0, landing))
                    throw new InvalidOperationException("contact_region_install_failed");
                player.position.Y = landing.Y; player.velocity = Vector2.Zero;
                if (!VanillaContactRegion.SafeDescent(player, landing))
                    throw new InvalidOperationException("safe_shifted_landing_rejected");
                if (!controller.CurrentMatches(0, player) || !controller.NeedsPlan(0, player, false)
                    || !controller.ContactDescentLanded)
                    throw new InvalidOperationException("early_shifted_landing_did_not_handoff");
                passed += 3;
            }
            foreach (string hazard in new[] { "lava", "spike", "wall", "gap", "fall_damage", "horizontal_speed" })
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                player.position += new Vector2(48, -64);
                player.velocity = new Vector2(0, 2);
                player.fallStart = (int)(player.position.Y / 16);
                int x = (int)(player.position.X / 16), y = (int)((landing.Y + player.height) / 16);
                if (hazard == "lava") { var tile = Main.tile[x, y - 1]; tile.LiquidType = LiquidID.Lava; tile.LiquidAmount = 255; }
                if (hazard == "spike") { var tile = Main.tile[x, y]; tile.HasTile = true; tile.TileType = TileID.Spikes; }
                if (hazard == "wall") { var tile = Main.tile[x, y - 3]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
                if (hazard == "gap")
                    for (int tx = (int)(landing.X / 16) + 1; tx <= x + 2; tx++) Main.tile[tx, y].ClearEverything();
                if (hazard == "fall_damage") player.fallStart = (int)(landing.Y / 16) - 100;
                if (hazard == "horizontal_speed") player.velocity.X = 1;
                if (VanillaContactRegion.SafeDescent(player, landing))
                    throw new InvalidOperationException("unsafe_shifted_descent_accepted:" + hazard);
                passed++;
            }
            VanillaFrame.RestoreInWorker(snapshot);
            return new JsonObject { ["passed"] = true, ["cases"] = passed };
        }
        if (command == "contact_descent_replay")
        {
            var evidence = request["evidence"].AsObject();
            var snapshot = evidence["snapshot"].AsObject();
            var player = VanillaFrame.RestoreInWorker(snapshot);
            var remaining = evidence["remaining_controls"].AsArray();
            var frames = evidence["active_plan"]["forecast"]["frames"].AsArray();
            var tail = new JsonArray(frames.Skip(frames.Count - remaining.Count).Select(frame => frame.DeepClone()).ToArray());
            var plan = new JsonObject { ["status"] = "planned", ["selected"] = "recorded_descent", ["wait_frames"] = 0,
                ["context"] = snapshot.DeepClone(), ["anchor"] = VanillaPrediction.Observe(player, false),
                ["route_progress"] = 100, ["forecast"] = new JsonObject { ["frames"] = tail },
                ["controls"] = remaining.DeepClone() };
            var controller = new VanillaAsyncMovement(_ => throw new InvalidOperationException("unexpected_request"));
            if (!controller.InstallTerrain(plan, player, 0, new Vector2(evidence["x"].GetValue<float>(), evidence["y"].GetValue<float>())))
                throw new InvalidOperationException("descent_install_failed");
            float maxError = 0;
            int landedAt = 0, hp = player.statLife;
            for (int frame = 1; frame < remaining.Count; frame++)
            {
                VanillaPrediction.Advance(player, (VanillaKeys)controller.InputAt(frame));
                controller.Poll(frame, player, 0);
                if (!controller.CurrentMatches(frame, player) || player.statLife < hp || player.dead)
                    throw new InvalidOperationException("unsafe_contact_descent:" + frame + ":" + controller.LastMismatch
                        + ":actual=" + VanillaPrediction.Observe(player, false) + ":landing=" + tail[^1]);
                maxError = Math.Max(maxError, Math.Abs(player.position.Y - tail[frame - 1]["y"].GetValue<float>()));
                bool needsPlan = controller.NeedsPlan(frame, player, false);
                if (controller.ContactDescentLanded) { landedAt = frame; if (!needsPlan) throw new InvalidOperationException("landed_without_handoff"); break; }
                if (needsPlan) throw new InvalidOperationException("descent_replanned_in_air");
            }
            if (landedAt == 0 || maxError < 1 || controller.ContactDescents != 1)
                throw new InvalidOperationException("descent_replay_did_not_exercise_tolerance");
            var landing = player.position;
            player = VanillaFrame.RestoreInWorker(snapshot);
            var foot = Main.tile[(int)(landing.X / 16), (int)((landing.Y + player.height - 4) / 16)];
            foot.LiquidType = LiquidID.Lava; foot.LiquidAmount = 255;
            if (VanillaContactRegion.SafeDescent(player, landing)) throw new InvalidOperationException("descent_ignored_lava");
            player = VanillaFrame.RestoreInWorker(snapshot);
            player.fallStart = (int)(landing.Y / 16) - 100;
            if (VanillaContactRegion.SafeDescent(player, landing)) throw new InvalidOperationException("descent_ignored_fall_damage");
            VanillaFrame.RestoreInWorker(snapshot);
            return new JsonObject { ["landed_frame"] = landedAt, ["max_y_deviation"] = maxError, ["hp"] = hp,
                ["contact_descents"] = controller.ContactDescents, ["hazards_rejected"] = true };
        }
        if (command == "contact_cycle_replay")
        {
            var player = VanillaFrame.RestoreInWorker(request["snapshot"].AsObject());
            var progress = new VanillaContactRegion.Progress();
            var field = VanillaLocalPlanner.ReadGuidance(request["guidance"]);
            var events = new JsonArray();
            foreach (var sample in request["samples"].AsArray())
            {
                player.position = new Vector2(sample["x"].GetValue<float>(), sample["y"].GetValue<float>());
                player.velocity = new Vector2(sample["vx"].GetValue<float>(), sample["vy"].GetValue<float>());
                var cell = VanillaLocalPlanner.FootCell(player.position, player.width, player.height);
                if (field.TryGetValue(cell, out int cost) && progress.ReturnedWithoutProgress(player, cost))
                    events.Add(sample.DeepClone());
            }
            return new JsonObject { ["events"] = events };
        }
        if (command == "contact_replay")
        {
            var snapshot = request["snapshot"].AsObject();
            var results = new JsonArray();
            foreach (int direction in new[] { 1, 2 })
            foreach (int offset in new[] { -8, 8 })
            {
                var player = VanillaFrame.RestoreInWorker(snapshot);
                var keys = Enumerable.Repeat(direction, 90).ToArray();
                var forecast = new VanillaPrediction(snapshot).Predict(keys, false);
                if (forecast["outcome"].GetValue<string>() != "complete") continue;
                var plan = new JsonObject { ["status"] = "planned", ["selected"] = "contact_replay", ["wait_frames"] = 0,
                    ["context"] = snapshot.DeepClone(), ["anchor"] = VanillaPrediction.Observe(player, false),
                    ["forecast"] = forecast, ["route_progress"] = 100d,
                    ["controls"] = new JsonArray(keys.Select(key => JsonValue.Create(key)).ToArray()) };
                var controller = new VanillaAsyncMovement(_ => throw new InvalidOperationException("unexpected_request"));
                if (!controller.InstallTerrain(plan, player, 0, player.position + new Vector2(300, 0)))
                    throw new InvalidOperationException("contact_install_failed");
                VanillaPrediction.Advance(player, (VanillaKeys)controller.InputAt(1));
                player.position.X += offset;
                controller.Poll(1, player, 0);
                controller.NeedsPlan(1, player, false);
                int retained = controller.ContactRetainedFrames;
                int hp = player.statLife;
                for (int frame = 2; frame <= retained + 1; frame++)
                {
                    VanillaPrediction.Advance(player, (VanillaKeys)controller.InputAt(frame));
                    if (player.dead || player.statLife < hp || !controller.CurrentMatches(frame, player))
                        throw new InvalidOperationException("contact_continuation_diverged:" + frame + " direction=" + direction
                            + " offset=" + offset + " actual=" + VanillaPrediction.Observe(player, false)
                            + " expected=" + controller.ActiveResult["forecast"]["frames"][frame - 1]);
                }
                results.Add(new JsonObject { ["direction"] = direction, ["offset"] = offset,
                    ["retained_frames"] = retained, ["accepted"] = controller.ContactContinuations,
                    ["hp"] = player.statLife, ["x"] = player.position.X, ["y"] = player.position.Y });
            }
            VanillaFrame.RestoreInWorker(snapshot);
            return new JsonObject { ["cases"] = results };
        }
        if (command == "water_fall_contract")
        {
            Setup(new JsonObject { ["equipment"] = new JsonArray((int)ItemID.SandstorminaBottle) });
            for (int x = X + 3; x < area.Right; x++)
            for (int y = Y; y < Y + 30; y++)
            {
                var tile = Main.tile[x, y]; tile.ClearEverything();
                if (y >= Y + 28) { tile.LiquidType = LiquidID.Water; tile.LiquidAmount = 255; }
            }
            var wet = VanillaFrame.Capture(source, area, true);
            var keys = Enumerable.Repeat(2, 180).ToArray();
            var wetRun = new VanillaPrediction(wet).Predict(keys, false);
            var wetFrames = wetRun["frames"].AsArray();
            if (wetFrames.Any(frame => frame["hp"].GetValue<int>() < 400))
                throw new InvalidOperationException("wet_fixture_not_safe");
            var plan = new VanillaLocalPlanner().Plan(wet,
                new Vector2(wetFrames[^1]["x"].GetValue<float>(), wetFrames[^1]["y"].GetValue<float>()),
                keys, preferContinuation: true);
            var candidate = plan["candidates"][0];
            if (candidate["safe"].GetValue<bool>() || candidate["dry_hp_loss"].GetValue<int>() <= 0)
                throw new InvalidOperationException("planner_relied_on_water_for_fall_survival");
            if (plan["status"].GetValue<string>() != "planned")
                throw new InvalidOperationException("no_safe_alternative_to_water_landing");
            return new JsonObject { ["passed"] = true, ["candidate"] = candidate.DeepClone(),
                ["selected"] = plan["selected"].DeepClone() };
        }
        if (command == "landing_probe")
        {
            var recorded = request["plan"]?.AsObject();
            if (recorded == null)
            {
                Setup(new JsonObject { ["equipment"] = new JsonArray((int)ItemID.SandstorminaBottle) });
                for (int x = area.Left; x < area.Right; x++)
                for (int y = Y; y < Y + 30; y++)
                    if (x > X + 2 || y > Y) Main.tile[x, y].ClearEverything();
                var initial = VanillaFrame.Capture(source, area, true);

            if (!JsonNode.DeepEquals(expected, VanillaFrame.Capture(player, area, true)))
                throw new InvalidOperationException("branch_arrays_or_items_aliased");
            parent.Restore();
            return new JsonObject { ["passed"] = true, ["changed_tiles"] = child.CapturedTileChanges };
        }
        if (command == "region_search") return RegionSearch(request);
        if (command == "plan")
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();
            var state = request["snapshot"].AsObject();
            try
            {
                var prefix = request["prefix"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                var advanced = new VanillaPrediction(state).Predict(prefix, false, true);
                if (advanced["end_snapshot"] is not JsonObject future)
                    return new JsonObject { ["status"] = "unsafe_handoff" };
                var anchor = VanillaPrediction.Observe(VanillaFrame.RestoreInWorker(future));
                var result = new VanillaLocalPlanner().Plan(future,
                    new Vector2(request["x"].GetValue<float>(), request["y"].GetValue<float>()),
                    continuation: request["continuation"]?.AsArray().Select(key => key.GetValue<int>()).ToArray(),
                    guidance: request["guidance"], terminal: request["terminal"]?.GetValue<bool>() == true,
                    routeDirection: request["route_direction"]?.GetValue<int>() ?? 0,
                    preferContinuation: request["recover"]?.GetValue<bool>() == true,
                    continuationOnly: request["continuation_only"]?.GetValue<bool>() == true);
                result["anchor"] = anchor;
                result["request_compute_ms"] = timer.Elapsed.TotalMilliseconds;
                return result;
            }
            finally { VanillaFrame.RestoreInWorker(state); }
        }
        if (command == "reset")
        {
            Setup(request);
            checkpoint = VanillaFrame.Capture(source, area);
            return new JsonObject { ["initial"] = VanillaPrediction.Observe(source), ["snapshot"] = checkpoint.DeepClone() };
        }
        if (command == "load")
        {
            var incoming = request["snapshot"].AsObject();
            source = VanillaFrame.RestoreInWorker(incoming);
            checkpoint = (JsonObject)incoming.DeepClone();
            return new JsonObject { ["initial"] = VanillaPrediction.Observe(source) };
        }
        if (checkpoint == null) throw new InvalidOperationException("reset_required");
        if (command == "navigate")
        {
            var trace = new JsonArray();
            var plans = new JsonArray();
            int[] continuation = null;
            var goal = new Vector2(request["x"].GetValue<float>(), request["y"].GetValue<float>());
            int limit = Math.Clamp(request["limit"]?.GetValue<int>() ?? 240, 8, 600);
            bool reached = false;
            float maximumError = 0;
            for (int frame = 0; frame < limit;)
            {
                var snapshotTimer = System.Diagnostics.Stopwatch.StartNew();
                checkpoint = VanillaFrame.Capture(source, VanillaLocalPlanner.SnapshotArea(source, goal));
                double snapshotMs = snapshotTimer.Elapsed.TotalMilliseconds;
                var before = VanillaFrame.CaptureFields(source);
                var plan = new VanillaLocalPlanner().Plan(checkpoint, goal, continuation);
                plan["snapshot_ms"] = snapshotMs;
                plan["snapshot_tiles"] = checkpoint["area"][2].GetValue<int>() * checkpoint["area"][3].GetValue<int>();
                if (!JsonNode.DeepEquals(before, VanillaFrame.CaptureFields(source))) throw new InvalidOperationException("planner_mutated_source");
                plans.Add(plan);
                if (plan["status"].GetValue<string>() != "planned") break;
                var keys = plan["controls"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                Main.player[source.whoAmI] = source;
                int count = Math.Min(VanillaLocalPlanner.ExecuteFrames, limit - frame);
                for (int i = 0; i < count; i++)
                {
                    VanillaPrediction.Advance(source, (VanillaKeys)keys[i]);
                    var expected = plan["forecast"]["frames"][i];
                    float error = Vector2.Distance(source.position, new Vector2(expected["x"].GetValue<float>(), expected["y"].GetValue<float>()));
                    maximumError = Math.Max(error, maximumError);
                    if (error > .001f) throw new InvalidOperationException("execution_diverged_from_prediction");
                    trace.Add(VanillaPrediction.Observe(source)); frame++;
                    if (source.dead) throw new InvalidOperationException("navigation_died");
                    if (Vector2.Distance(source.position, goal) <= 20 && Math.Abs(source.velocity.Y) < .01f) { reached = true; break; }
                }
                continuation = keys.Skip(count).ToArray();
                if (reached) break;
            }
            checkpoint = VanillaFrame.Capture(source, area);
            return new JsonObject { ["reached"] = reached, ["trace"] = trace, ["plans"] = plans, ["max_error"] = maximumError };
        }
        int[] controls = request["controls"]?.AsArray().Select(node => node.GetValue<int>()).ToArray() ?? Array.Empty<int>();
        if (command == "predict")
        {
            var original = VanillaFrame.CaptureFields(source);
            var result = new VanillaPrediction(checkpoint).Predict(controls);
            result["source_unchanged"] = JsonNode.DeepEquals(original, VanillaFrame.CaptureFields(source));
            result["checkpoint_restored"] = JsonNode.DeepEquals(checkpoint,
                VanillaFrame.Capture(Main.player[source.whoAmI], area));
            return result;
        }
        if (command == "actual")
        {
            var trace = new JsonArray();
            Main.player[source.whoAmI] = source;
            foreach (int control in controls) { VanillaPrediction.Advance(source, (VanillaKeys)control); trace.Add(VanillaPrediction.Observe(source)); }
            return new JsonObject { ["frames"] = trace };
        }
        throw new ArgumentException("unknown_command");
    }

    public void Setup(JsonObject request, bool worker = true)
    {
        Main.gameMenu = false;
        if (worker) Main.myPlayer = 255;
        Main.rand = new Terraria.Utilities.UnifiedRandom(73);
        VanillaFrame.RestoreFields(WorldGen.genRand, VanillaFrame.CaptureFields(new Terraria.Utilities.UnifiedRandom(81)));
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        foreach (var item in Main.item) item.active = false;
        for (int x = area.Left; x < area.Right; x++)
        for (int y = area.Top; y < area.Bottom; y++)
        {
            var tile = Main.tile[x, y]; tile.ClearEverything();
            if (y >= Y) { tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        source = new Player { whoAmI = 0, active = true, name = "Forecast subject", statLife = 400, statLifeMax = 400, ConsumedLifeCrystals = 15 };
        if (!worker)
        {
            VanillaFrame.RestoreFields(Main.LocalPlayer, VanillaFrame.CaptureFields(source));
            source = Main.LocalPlayer;
            foreach (var item in source.armor.Concat(source.inventory).Concat(source.miscEquips)) item.SetDefaults(ItemID.None);
        }
        source.position = new Vector2(X * 16, Y * 16 - source.height);
        source.fallStart = (int)(source.position.Y / 16);
        Main.player[0] = source;
        int slot = 3;
        if (request["equipment"] is JsonArray items)
            foreach (var item in items) source.armor[slot++].SetDefaults(item.GetValue<int>());
        if (!worker) Main.ActivePlayerFileData.Player = source;
        if (worker) for (int i = 0; i < 3; i++) VanillaPrediction.Advance(source, VanillaKeys.None);
        if (request["buffs"] is JsonArray buffs)
            foreach (var buff in buffs) source.AddBuff(buff[0].GetValue<int>(), buff[1].GetValue<int>());
        string terrain = request["terrain"]?.GetValue<string>() ?? "flat";
        if (terrain == "ceiling")
            for (int x = X - 3; x < X + 18; x++) { var tile = Main.tile[x, Y - 6]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
        if (terrain is "half" or "slope")
            for (int x = X + 2; x < X + 12; x++)
            {
                var tile = Main.tile[x, Y - 1]; tile.HasTile = true; tile.TileType = TileID.Dirt;
                if (terrain == "half") tile.IsHalfBlock = true;
                else tile.Slope = SlopeType.SlopeDownRight;
            }
        if (terrain == "honey_block")
            for (int x = X - 2; x < X + 20; x++) { var tile = Main.tile[x, Y]; tile.TileType = TileID.HoneyBlock; }
        if (terrain == "honey")
            for (int x = X - 2; x < X + 20; x++)
            for (int y = Y - 4; y < Y; y++) { var tile = Main.tile[x, y]; tile.LiquidType = LiquidID.Honey; tile.LiquidAmount = 255; }
        if (terrain == "embedded")
        {
            var tile = Main.tile[X, Y - 2]; tile.HasTile = true; tile.TileType = TileID.Dirt;
        }
        if (terrain is "ledge" or "wall")
        {
            int height = request["height"]?.GetValue<int>() ?? 12;
            for (int x = X + 9; x < X + (terrain == "ledge" ? 38 : 12); x++)
            for (int y = Y - height; y < Y; y++) { var tile = Main.tile[x, y]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        if (terrain == "lava_gap")
            for (int x = X + 7; x < X + 20; x++)
            for (int y = Y; y < Y + 12; y++)
            {
                var tile = Main.tile[x, y]; tile.ClearEverything();
                if (y >= Y + 10) { tile.LiquidType = LiquidID.Lava; tile.LiquidAmount = 255; }
            }
        if (worker && terrain is ("honey" or "honey_block"))
            for (int i = 0; i < 3; i++) VanillaPrediction.Advance(source, VanillaKeys.None);
        if (request["airborne"]?.GetValue<bool>() == true)
        {
            source.position.Y -= request["airborne_height"]?.GetValue<int>() ?? 320;
            source.velocity.Y = 3; source.fallStart = (int)(source.position.Y / 16);
        }
        if (worker && request["pre_controls"] is JsonArray prior)
            foreach (var key in prior) VanillaPrediction.Advance(source, (VanillaKeys)key.GetValue<int>());
    }
