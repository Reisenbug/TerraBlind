public sealed class Forecast : Mod { }

public sealed class Engine : ModSystem
{
    const int X = 500, Y = 400;
    readonly Rectangle area = new(X - 30, Y - 65, 120, 100);
    JsonObject checkpoint;
    Player source;
    public Player Source => source;
    public Rectangle Area => area;
    public override void Load() => Main.OnEnginePreload += Run;
    public override void Unload() => Main.OnEnginePreload -= Run;

    sealed class Work
    {
        public JsonObject Request;
        public long Id;
        public int Cancelled;
    }

    void Run()
    {
        if (!Main.dedServ || !Terraria.Program.LaunchParameters.TryGetValue("-vanillaforecastport", out string value)) return;
        VanillaFrame.RequireWorker();
        var listener = new TcpListener(IPAddress.Loopback, int.Parse(value));
        listener.Start();
        Console.WriteLine("VANILLA_FORECAST_READY");
        try
        {
            using var client = listener.AcceptTcpClient(); client.NoDelay = true;
            using var reader = new StreamReader(client.GetStream());
            using var writer = new StreamWriter(client.GetStream()) { AutoFlush = true };
            using var incoming = new BlockingCollection<Work>();
            var requests = new ConcurrentDictionary<long, Work>();
            long anonymousId = 0;
            _ = Task.Run(() =>
            {
                try
                {
                    string line;
                    while ((line = reader.ReadLine()) != null)
                    {
                        var request = JsonNode.Parse(line).AsObject();
                        long id = request["request_id"]?.GetValue<long>() ?? Interlocked.Decrement(ref anonymousId);
                        if (request["command"]?.GetValue<string>() == "cancel")
                        {
                            if (requests.TryGetValue(id, out var cancelled)) Interlocked.Exchange(ref cancelled.Cancelled, 1);
                            continue;
                        }
                        var work = new Work { Request = request, Id = id };
                        requests[id] = work; incoming.Add(work);
                    }
                }
                catch (Exception error) { Console.WriteLine(error); }
                finally { incoming.CompleteAdding(); }
            });
            foreach (var work in incoming.GetConsumingEnumerable())
            {
                var request = work.Request;
                if (request["command"]?.GetValue<string>() == "close") { writer.WriteLine("{\"closed\":true}"); break; }
                VanillaPrediction.CancellationRequested = () => Volatile.Read(ref work.Cancelled) != 0;
                try
                {
                    if (Terraria.Program.LaunchParameters.ContainsKey("-forecasttrace")) Console.WriteLine("FORECAST_REQUEST_STARTED " + work.Id);
                    VanillaPrediction.CheckCancellation();
                    var result = Handle(request); result["request_id"] = work.Id;
                    writer.WriteLine(result.ToJsonString());
                }
                catch (OperationCanceledException)
                { writer.WriteLine(new JsonObject { ["status"] = "cancelled", ["request_id"] = work.Id }.ToJsonString()); }
                catch (Exception error) { writer.WriteLine(new JsonObject { ["error"] = error.ToString() }.ToJsonString()); }
                finally { VanillaPrediction.CancellationRequested = null; requests.TryRemove(work.Id, out _); }
            }
            Environment.Exit(0);
        }
        catch (Exception error) { Console.WriteLine(error); Environment.Exit(1); }
        finally { listener.Stop(); }
    }

    JsonObject Handle(JsonObject request)
    {
        string command = request["command"].GetValue<string>();
        if (command == "terrain_plan") return VanillaTerrainPlan.Build(request);
        if (command == "route_reuse_contract")
        {
            Setup(new JsonObject());
            var snapshot = VanillaFrame.Capture(source, area, true);
            var keys = Enumerable.Repeat(2, 180).ToArray();
            var forecast = new VanillaPrediction(snapshot).Predict(keys, false);
            var plan = new JsonObject { ["status"] = "planned", ["selected"] = "reuse_walk", ["wait_frames"] = 0,
                ["context"] = snapshot.DeepClone(), ["anchor"] = VanillaPrediction.Observe(source, false),
                ["forecast"] = forecast, ["route_progress"] = 100d,
                ["controls"] = new JsonArray(keys.Select(key => JsonValue.Create(key)).ToArray()) };
            var controller = new VanillaAsyncMovement(_ => throw new InvalidOperationException("unexpected_replanning"));
            if (!controller.InstallTerrain(plan, source, 0, source.position + new Vector2(600, 0)))
                throw new InvalidOperationException("reuse_install_failed");
            double checksMs = 0;
            int firstRequest = -1;
            for (int frame = 1; frame <= 180; frame++)
            {
                VanillaPrediction.Advance(source, (VanillaKeys)controller.InputAt(frame));
                controller.Poll(frame, source, 0);
                var expected = forecast["frames"][frame - 1];
                if (Vector2.Distance(source.position, new Vector2(expected["x"].GetValue<float>(), expected["y"].GetValue<float>())) > .001f)
                    throw new InvalidOperationException("reused_route_diverged");
                long started = System.Diagnostics.Stopwatch.GetTimestamp();
                bool needsPlan = controller.NeedsPlan(frame, source, false);
                checksMs += System.Diagnostics.Stopwatch.GetElapsedTime(started).TotalMilliseconds;
                if (needsPlan) { firstRequest = frame; break; }
            }
            if (firstRequest != 138) throw new InvalidOperationException("reuse_did_not_preserve_planning_runway:" + firstRequest);
            foreach (string change in new[] { "velocity", "tile", "equipment", "distant_tile", "water", "lava", "wet" })
            {
                source = VanillaFrame.RestoreInWorker(snapshot);
                controller = new VanillaAsyncMovement(_ => throw new InvalidOperationException("unexpected_replanning"));
                controller.InstallTerrain(plan, source, 0, source.position + new Vector2(600, 0));
                if (change == "velocity") source.velocity.X += 1;
                if (change == "equipment") source.armor[3].SetDefaults(ItemID.SandstorminaBottle);
                if (change == "wet") source.wet = !source.wet;
                if (change is "water" or "lava")
                {
                    var tile = Main.tile[X + 1, Y - 1];
                    tile.LiquidType = change == "water" ? LiquidID.Water : LiquidID.Lava;
                    tile.LiquidAmount = 128;
                }
                if (change is "tile" or "distant_tile")
                {
                    var tile = Main.tile[X + (change == "tile" ? 1 : 80), Y - 1];
                    tile.HasTile = true; tile.TileType = TileID.Stone;
                }
                bool needsPlan = controller.NeedsPlan(0, source, false);
                if (needsPlan != (change is not "distant_tile" and not "water")) throw new InvalidOperationException("reuse_invalidation:" + change);
            }
            return new JsonObject { ["passed"] = true, ["first_request_frame"] = firstRequest,
                ["validation_total_ms"] = checksMs, ["validated_frames"] = firstRequest, ["max_error"] = 0 };
        }
        if (command == "damage_cutoff_contract")
        {
            Setup(new JsonObject());
            var controlsToTest = Enumerable.Repeat(2, 180).ToArray();
            var safeState = VanillaFrame.Capture(source, area, true);
            var safeFull = new VanillaPrediction(safeState).Predict(controlsToTest, false);
            var safeCutoff = new VanillaPrediction(safeState).Predict(controlsToTest, false, stopOnDamage: true);
            if (!JsonNode.DeepEquals(safeFull["frames"], safeCutoff["frames"]))
                throw new InvalidOperationException("damage_cutoff_changed_safe_route");
            var spike = Main.tile[X + 3, Y - 1]; spike.HasTile = true; spike.TileType = TileID.Spikes;
            var dangerState = VanillaFrame.Capture(source, area, true);
            var full = new VanillaPrediction(dangerState).Predict(controlsToTest, false);
            var cutoff = new VanillaPrediction(dangerState).Predict(controlsToTest, false, stopOnDamage: true);
            var all = full["frames"].AsArray(); var shortened = cutoff["frames"].AsArray();
            int firstHurt = all.ToList().FindIndex(frame => frame["hp"].GetValue<int>() < 400);
            if (firstHurt < 0 || cutoff["outcome"].GetValue<string>() != "damaged" || shortened.Count != firstHurt + 1)
                throw new InvalidOperationException("damage_cutoff_did_not_stop_at_first_hit");
            for (int i = 0; i < shortened.Count; i++)
                if (!JsonNode.DeepEquals(all[i], shortened[i])) throw new InvalidOperationException("damage_cutoff_changed_prefix");
            if (!JsonNode.DeepEquals(dangerState, VanillaFrame.Capture(Main.player[source.whoAmI], area, true)))
                throw new InvalidOperationException("damage_cutoff_failed_to_restore_state");
            return new JsonObject { ["passed"] = true, ["full_frames"] = all.Count, ["cutoff_frames"] = shortened.Count };
        }
        if (command == "falling_shelf_contract")
        {
            Setup(new JsonObject { ["equipment"] = new JsonArray((int)ItemID.SandstorminaBottle, (int)ItemID.HermesBoots) });
            for (int x = area.Left; x < area.Right; x++)
            for (int y = Y; y < area.Bottom; y++)
            {
                var tile = Main.tile[x, y]; tile.ClearEverything();
                if (y >= Y + 25) { tile.LiquidType = LiquidID.Lava; tile.LiquidAmount = 255; }
                if (x >= X + 14 && y == Y + 4) { tile.HasTile = true; tile.TileType = TileID.Dirt; }
            }
            source.velocity.Y = 6;
            var snapshot = VanillaFrame.Capture(source, area, true);
            var walk = new VanillaPrediction(snapshot).Predict(Enumerable.Repeat(2, VanillaLocalPlanner.Horizon).ToArray(), false);
            var plan = new VanillaLocalPlanner().Plan(snapshot, new Vector2((X + 22) * 16, (Y + 4) * 16 - source.height), terminal: true);
            if (plan["status"].GetValue<string>() != "planned") throw new InvalidOperationException("falling_shelf_no_plan");
            var frames = plan["forecast"]["frames"].AsArray();
            bool OnShelf(JsonNode frame) => frame["grounded"].GetValue<bool>()
                && frame["x"].GetValue<float>() >= (X + 14) * 16 - source.width
                && Math.Abs(frame["y"].GetValue<float>() - ((Y + 4) * 16 - source.height)) < .01f;
            if (walk["frames"].AsArray().Any(OnShelf) || !frames.Any(OnShelf)
                || !frames.Any(frame => frame["available_jumps"].GetValue<int>() == 0)
                || frames.Any(frame => frame["hp"].GetValue<int>() < 400))
                throw new InvalidOperationException("falling_shelf_jump_not_used_for_traversal");
            source = VanillaFrame.RestoreInWorker(snapshot);
            float error = 0;
            var keys = plan["controls"].AsArray();
            for (int i = 0; i < keys.Count; i++)
            {
                VanillaPrediction.Advance(source, (VanillaKeys)keys[i].GetValue<int>());
                error = Math.Max(error, Vector2.Distance(source.position,
                    new Vector2(frames[i]["x"].GetValue<float>(), frames[i]["y"].GetValue<float>())));
            }
            if (error > .001f) throw new InvalidOperationException("falling_shelf_replay_diverged");
            return new JsonObject { ["passed"] = true, ["selected"] = plan["selected"].DeepClone(),
                ["max_error"] = error, ["compute_ms"] = plan["elapsed_ms"].DeepClone() };
        }
        if (command == "terrain_contract")
        {
            int passed = 0;
            float error = 0;
            var miningBatch = VanillaTerrainPlan.MiningBatch((10, 20),
                new[] { (11, 20), (12, 19), (10, 21), (15, 19) }, 160, 190, 20);
            if (!miningBatch.SequenceEqual(new[] { (12, 19), (10, 20), (11, 20) }))

            for (int i = 0; i < 30; i++) VanillaPrediction.Advance(source, VanillaKeys.Right);

            if (halfCell.y != Y - 1 || Math.Abs(source.Bottom.Y - (Y * 16 - 8)) > .01f)
                throw new InvalidOperationException("halfbrick_foot_cell_wrong");
            passed++;
            foreach (int mode in new[] { 0, 1, 2, 3, 4, 5, 6, 7 })
            {
                bool place = mode == 1;
                Setup(new JsonObject { ["equipment"] = new JsonArray((int)ItemID.SandstorminaBottle, (int)ItemID.HermesBoots) });
                source.inventory[0].SetDefaults(place ? ItemID.WoodPlatform : ItemID.MoltenPickaxe);
                source.inventory[0].stack = 999;
                int x = X + 3, y = Y - 1;

                var targets = new JsonArray();
                for (int xx = x; xx <= x + (mode == 2 ? 2 : 0); xx++)
                for (int yy = mode is 2 or 6 or 7 ? Y - 3 : y; yy <= y; yy++)
                {
                    targets.Add(new JsonArray(xx, yy));
                    if (!place) { var tile = Main.tile[xx, yy]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
                }
                if (mode == 7)
                    for (int yy = Y - 25; yy < Y - 3; yy++)
                    { var tile = Main.tile[x, yy]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
                if (mode == 3) { var tile = Main.tile[x, y]; tile.TileType = TileID.JungleGrass; }
                if (mode is 4 or 5)
                {
                    x = X; y = Y;
                    targets = new JsonArray(new JsonArray(X, Y), new JsonArray(X + 1, Y));
                }
                var snapshot = VanillaFrame.Capture(source, area, true);
                var guidance = new JsonArray();
                for (int xx = area.Left; xx < area.Right; xx++)
                for (int yy = area.Top; yy < area.Bottom; yy++) guidance.Add(new JsonArray(xx, yy, mode == 5 ? area.Bottom - yy : area.Right - xx));
                var edit = new JsonObject { ["snapshot"] = snapshot, ["tile_x"] = x, ["tile_y"] = y,
                    ["tile_type"] = place ? (int)TileID.Platforms : -1, ["wait_frames"] = 24,
                    ["tiles"] = targets,
                    ["x"] = source.position.X + (mode == 5 ? 0 : 22 * 16), ["y"] = source.position.Y + (mode == 5 ? 16 : 0),
                    ["guidance"] = guidance, ["route_direction"] = 1 };
                int[] arrival = null;
                if (mode is 6 or 7)
                {
                    var walk = Enumerable.Repeat(2, 180).ToArray();
                    var approach = new VanillaPrediction(snapshot).Predict(walk, false);
                    var arriving = new JsonObject { ["selected"] = "arrival", ["wait_frames"] = 0,
                        ["context"] = snapshot.DeepClone(), ["anchor"] = VanillaPrediction.Observe(source, false),
                        ["forecast"] = approach, ["controls"] = new JsonArray(walk.Select(key => JsonValue.Create(key)).ToArray()) };
                    var approaching = new VanillaAsyncMovement(_ => throw new InvalidOperationException("unexpected_transport"));
                    if (!approaching.InstallTerrain(arriving, source, 0, source.position + new Vector2(300, 0))
                        || !approaching.TryArrival(0, out arrival, out var expectedArrival))
                        throw new InvalidOperationException("terrain_arrival_not_extracted");
                    edit["arrival"] = new JsonArray(arrival.Select(key => JsonValue.Create(key)).ToArray());
                    Console.WriteLine("TERRAIN_PREFETCH_LEAD frames=" + arrival.Length);
                }
                var plan = VanillaTerrainPlan.Build(edit);
                if (plan["status"].GetValue<string>() != "planned") throw new InvalidOperationException("terrain_contract:" + plan.ToJsonString());
                bool skipEdit = plan["terrain_required"]?.GetValue<bool>() == false;
                if (mode is 6 or 7 && skipEdit != (mode == 6))
                    throw new InvalidOperationException("terrain_did_not_check_unedited_route:" + mode);
                if (arrival != null && !JsonNode.DeepEquals(snapshot, VanillaFrame.Capture(Main.player[source.whoAmI], area, true)))
                    throw new InvalidOperationException("terrain_arrival_did_not_restore_snapshot");
                if (mode == 5 && (plan["edit_progress"].GetValue<double>() != 16
                    || plan["movement_progress"].GetValue<double>() != 0 || plan["route_progress"].GetValue<double>() != 16))
                    throw new InvalidOperationException("support_removal_progress_not_counted");
                if (!Main.tile[x, y].HasTile && !place || Main.tile[x, y].HasTile && place)
                    throw new InvalidOperationException("terrain_probe_mutated_world");
                source = VanillaFrame.RestoreInWorker(snapshot);
                if (arrival != null)
                {
                    foreach (int key in arrival) VanillaPrediction.Advance(source, (VanillaKeys)key);
                    if (!VanillaAsyncMovement.StateMatches(source, plan["anchor"].AsObject(), true, out _, out _))
                        throw new InvalidOperationException("terrain_arrival_anchor_differs");
                    source.velocity.X += 1;
                    if (VanillaAsyncMovement.StateMatches(source, plan["anchor"].AsObject(), true, out _, out _))
                        throw new InvalidOperationException("terrain_changed_arrival_accepted");
                    source.velocity.X -= 1;
                }
                if (place) WorldGen.PlaceTile(x, y, TileID.Platforms, mute: true, plr: source.whoAmI);
                else if (!skipEdit) foreach (var target in targets)
                {
                    int tx = target[0].GetValue<int>(), ty = target[1].GetValue<int>();
                    WorldGen.KillTile(tx, ty, noItem: true);
                    if (mode == 3) WorldGen.KillTile(tx, ty, noItem: true);
                }
                var controller = new VanillaAsyncMovement(_ => throw new InvalidOperationException("unexpected_transport"));
                if (!controller.InstallTerrain(plan, source, 0, source.position + new Vector2(22 * 16, 0)))
                    throw new InvalidOperationException("terrain_handoff_rejected_matching_state");
                source.itemAnimation = 22; source.itemTime = 10;
                var editControls = plan["controls"].AsArray();
                int installedOffset = controller.PlanFrameAt(1);
                if (mode == 2 && installedOffset != 24)
                    throw new InvalidOperationException("stationary_terrain_wait_was_not_skipped:" + installedOffset);
                for (int i = 0; controller.HasInputAt(i + 1); i++)
                {
                    int key = controller.InputAt(i + 1);
                    VanillaPrediction.Advance(source, (VanillaKeys)key);
                    var expected = plan["forecast"]["frames"][controller.PlanFrameAt(i + 1)];
                    error = Math.Max(error, Vector2.Distance(source.position,
                        new Vector2(expected["x"].GetValue<float>(), expected["y"].GetValue<float>())));
                    if (error > .001f || source.dead) throw new InvalidOperationException("terrain_execution_diverged");
                }
                if (mode == 2 && source.position.X < (x + 3) * 16)
                    throw new InvalidOperationException("mined_passage_not_traversed");
                passed++;
            }
            Setup(new JsonObject());
            for (int x = X - 4; x <= X + 20; x++)
            for (int y = Y; y <= Y + 20; y++)
            {
                var tile = Main.tile[x, y]; tile.ClearEverything();
                if (y >= Y + 8) { tile.LiquidType = LiquidID.Lava; tile.LiquidAmount = 255; }
            }
            var seal = Main.tile[X, Y]; seal.HasTile = true; seal.TileType = TileID.Dirt;
            var unsafeState = VanillaFrame.Capture(source, area, true);
            var unsafeEdit = new JsonObject { ["snapshot"] = unsafeState, ["tile_x"] = X, ["tile_y"] = Y,
                ["tile_type"] = -1, ["wait_frames"] = 60, ["x"] = source.position.X, ["y"] = source.position.Y + 100,
                ["route_direction"] = 0 };
            if (VanillaTerrainPlan.Build(unsafeEdit)["status"].GetValue<string>() != "unsafe_edit_wait")
                throw new InvalidOperationException("unsafe_terrain_wait_was_accepted");
            passed++;
            Setup(new JsonObject());
            source.inventory[0].SetDefaults(ItemID.MoltenPickaxe);
            source.itemAnimation = 10; source.itemTime = 5;
            var activePick = VanillaFrame.Capture(source, area, true);
            var dirt = Main.tile[X + 3, Y];
            dirt.TileFrameY += 18;
            if (!VanillaFrame.ContextMatches(activePick, source))
                throw new InvalidOperationException("cosmetic_dirt_frame_invalidated_navigation");
            dirt.HasTile = false;
            if (VanillaFrame.ContextMatches(activePick, source))
                throw new InvalidOperationException("removed_support_did_not_invalidate_navigation");
            passed++;
            return new JsonObject { ["passed"] = true, ["cases"] = passed, ["max_error"] = error };
        }
        if (command == "replay")
            return new VanillaPrediction(request["snapshot"].AsObject()).Predict(
                request["controls"].AsArray().Select(key => key.GetValue<int>()).ToArray());
        if (command == "transit_contract")
        {
            Setup(new JsonObject { ["equipment"] = new JsonArray((int)ItemID.SandstorminaBottle, (int)ItemID.HermesBoots, (int)ItemID.CobaltShield) });
            var snapshot = VanillaFrame.Capture(source, area, true);
            var guidance = new JsonArray();
            for (int x = area.Left; x < area.Right; x++)
            for (int y = area.Top; y < area.Bottom; y++) guidance.Add(new JsonArray(x, y, area.Right - x));
            var goal = source.position + new Vector2(22 * 16, 0);
            var walk = new VanillaLocalPlanner().Plan(snapshot, goal, guidance: guidance);
            var packedGuidance = guidance.SelectMany(row => row.AsArray().Select(value => value.GetValue<int>())).ToArray();
            var packedPlan = new VanillaLocalPlanner().Plan(snapshot, goal, guidance: JsonValue.Create(VanillaFrame.PackInts(packedGuidance)));
            if (!JsonNode.DeepEquals(walk["controls"], packedPlan["controls"]))
                throw new InvalidOperationException("packed_guidance_changed_decision");
            var staleGoal = new VanillaLocalPlanner().Plan(snapshot, source.position - new Vector2(32, 0),
                guidance: guidance, routeDirection: 1);
            if (staleGoal["controls"].AsArray().Any(key => (key.GetValue<int>() & 3) != 2))
                throw new InvalidOperationException("stale_waypoint_reversed_travel");
            var queuedSnapshot = VanillaFrame.Capture(source, VanillaLocalPlanner.SnapshotArea(source, goal, 90), true);
            var handoff = new VanillaPrediction(queuedSnapshot).Predict(Enumerable.Repeat(2, 90).ToArray(), false, true);
            if (handoff["end_snapshot"] is not JsonObject future)
                throw new InvalidOperationException("queued_motion_left_snapshot");
            var afterHandoff = new VanillaLocalPlanner().Plan(future, goal, guidance: guidance, routeDirection: 1);
            if (afterHandoff["candidates"].AsArray().Any(candidate => candidate["outcome"].GetValue<string>() == "outside_observed_world"))
                throw new InvalidOperationException("snapshot_omitted_handoff_distance");
            var continued = new VanillaLocalPlanner().Plan(snapshot, goal, Enumerable.Repeat(2, 12).ToArray(), guidance);
            foreach (var plan in new[] { walk, continued })
                if (plan["status"]?.GetValue<string>() != "planned"
                    || plan["controls"].AsArray().Any(key => (key.GetValue<int>() & 3) != 2))
                    throw new InvalidOperationException("transit_inserted_a_stop");
            if (!continued["candidates"].AsArray().Any(candidate => candidate["name"].GetValue<string>() == "continue"))
                throw new InvalidOperationException("transit_discarded_valid_route");
            var baseline = new VanillaPrediction(snapshot).Predict(Enumerable.Repeat(2, VanillaLocalPlanner.Horizon).ToArray(), false);
            float walked = baseline["frames"].AsArray()[^1]["x"].GetValue<float>() - source.position.X;
            float advanced = walk["forecast"]["frames"].AsArray()[^1]["x"].GetValue<float>() - source.position.X;
            if (advanced < walked) throw new InvalidOperationException("mobility_search_lost_to_walking");
            return new JsonObject { ["passed"] = true, ["walk_ms"] = walk["elapsed_ms"].DeepClone(),
                ["continue_ms"] = continued["elapsed_ms"].DeepClone(), ["selected"] = walk["selected"].DeepClone(),
                ["walking_distance"] = walked, ["chosen_distance"] = advanced };
        }
        if (command == "async_cleanup")
        {
            Setup(new JsonObject());
            var snapshot = VanillaFrame.Capture(source, area);
            var fake = new JsonObject { ["status"] = "planned", ["anchor"] = VanillaPrediction.Observe(source),
                ["forecast"] = new JsonObject { ["frames"] = new JsonArray(VanillaPrediction.Observe(source)) },
                ["controls"] = new JsonArray(Enumerable.Repeat(2, 90).Select(key => JsonValue.Create(key)).ToArray()) };
            var controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0);
            controller.Poll(VanillaAsyncMovement.LeadFrames, source, 0);
            if (controller.Accepted != 1 || controller.InputAt(31) != 2) throw new InvalidOperationException("contract_plan_not_installed");

            source.velocity.X += 1;
            controller.Poll(60, source, 0);
            if (controller.Status != "state_changed:vx" || controller.InputAt(61) != 0 || controller.ActiveResult != null)
                throw new InvalidOperationException("rejected_plan_kept_old_inputs");

            controller.Poll(90, source, 0);
            Main.projectile[0].active = false;
            if (controller.Status != "context_changed" || controller.InputAt(91) != 0)
                throw new InvalidOperationException("new_projectile_did_not_invalidate_plan");

            fake["anchor"] = VanillaPrediction.Observe(source);

            source.statLife--;
            controller.Poll(30, source, 0);
            if (controller.Accepted != 1) throw new InvalidOperationException("navigation_rejected_health_only_change");

            controller.Poll(60, source, 0);

            var delayed = new System.Threading.Tasks.TaskCompletionSource<JsonObject>();
            int calls = 0;
            controller = new VanillaAsyncMovement(_ => ++calls == 1
                ? System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()) : delayed.Task);
            controller.Request(snapshot, 0, source.position, 0); controller.Poll(30, source, 0);
            controller.Request(snapshot, 30, source.position, 0);
            delayed.SetResult((JsonObject)fake.DeepClone()); controller.Poll(61, source, 0);
            if (controller.Status != "late_result" || controller.InputAt(62) != 2 || !controller.HasInputAt(62)
                || controller.ActiveResult == null)
                throw new InvalidOperationException("late_result_erased_valid_queue");
            controller.Poll(62, source, 1);
            if (controller.HasInputAt(63)) throw new InvalidOperationException("revision_kept_old_queue");
            if (controller.ActiveResult != null) throw new InvalidOperationException("revision_kept_active_result");
            Setup(new JsonObject());
            snapshot = VanillaFrame.Capture(source, area, true);
            fake["anchor"] = VanillaPrediction.Observe(source);
            fake["route_progress"] = 10;
            controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0); controller.Poll(1, source, 0);
            fake["route_progress"] = 0;
            controller.Request(snapshot, 1, source.position, 0); controller.Poll(2, source, 0);
            if (controller.LastResult["route_progress"].GetValue<int>() != 0
                || controller.ActiveResult["route_progress"].GetValue<int>() != 10 || controller.Accepted != 1)
                throw new InvalidOperationException("early_pending_result_replaced_active_result");
            controller.Poll(1 + controller.PlanningLead, source, 0);
            if (controller.ActiveResult["route_progress"].GetValue<int>() != 0 || controller.Accepted != 2)
                throw new InvalidOperationException("handoff_did_not_replace_active_result");
            Setup(new JsonObject());
            Main.npc[0].SetDefaults(NPCID.BlueSlime); Main.npc[0].active = true; Main.npc[0].Center = source.Center;
            snapshot = VanillaFrame.Capture(source, area, true);
            fake["anchor"] = VanillaPrediction.Observe(source);
            controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0);
            Main.npc[0].active = false;
            Main.item[0].SetDefaults(ItemID.Gel); Main.item[0].active = true; Main.item[0].position = source.position;
            controller.Poll(30, source, 0);
            if (controller.Accepted != 1) throw new InvalidOperationException("kill_and_loot_invalidated_navigation");
            controller.Request(snapshot, 30, source.position, 0);
            var obstruction = Main.tile[(int)(source.Center.X / 16), (int)(source.Center.Y / 16)];
            obstruction.HasTile = true; obstruction.TileType = TileID.Stone;
            controller.Poll(60, source, 0);
            if (controller.Status != "context_changed") throw new InvalidOperationException("navigation_ignored_terrain_change");
            Main.item[0].active = false;
            Setup(new JsonObject());
            snapshot = VanillaFrame.Capture(source, area, true);
            fake["anchor"] = VanillaPrediction.Observe(source);
            controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0); controller.Poll(1, source, 0);
            if (controller.Accepted != 1 || controller.InputAt(2) != 2)
                throw new InvalidOperationException("matching_state_waited_for_handoff");
            Setup(new JsonObject());
            var plant = Main.tile[(int)(source.Center.X / 16), (int)(source.Center.Y / 16)];
            plant.HasTile = true; plant.TileType = TileID.Plants;
            snapshot = VanillaFrame.Capture(source, area, true); fake["anchor"] = VanillaPrediction.Observe(source);
            controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0); plant.ClearEverything(); controller.Poll(1, source, 0);
            if (controller.Accepted != 1) throw new InvalidOperationException("cut_plant_invalidated_navigation");
            controller.Request(snapshot, 1, source.position, 0);
            plant.HasTile = true; plant.TileType = TileID.Cobweb;
            controller.Poll(1 + controller.PlanningLead, source, 0);
            if (controller.Status != "context_changed") throw new InvalidOperationException("web_change_was_ignored");
            Setup(new JsonObject()); source.controlRight = true;
            snapshot = VanillaFrame.Capture(source, area, true);
            fake["anchor"] = VanillaPrediction.Observe(source);
            fake["forecast"]["frames"] = new JsonArray(Enumerable.Range(0, 90).Select(_ => (JsonNode)VanillaPrediction.Observe(source)).ToArray());
            delayed = new System.Threading.Tasks.TaskCompletionSource<JsonObject>(); calls = 0;
            controller = new VanillaAsyncMovement(_ => ++calls == 1
                ? System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()) : delayed.Task);
            controller.Request(snapshot, 0, source.position, 0); controller.Poll(1, source, 0);
            controller.Request(snapshot, 1, source.position, 0);
            int arrival = 1 + controller.PlanningLead + 3;
            for (int frame = 2; frame < arrival; frame++) controller.Poll(frame, source, 0);
            delayed.SetResult((JsonObject)fake.DeepClone()); controller.Poll(arrival, source, 0);
            if (controller.Accepted != 2 || controller.InputAt(arrival + 1) != 2)
                throw new InvalidOperationException("matching_late_route_was_discarded");
            delayed = new System.Threading.Tasks.TaskCompletionSource<JsonObject>(); calls = 0;
            controller = new VanillaAsyncMovement(_ => ++calls == 1
                ? System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()) : delayed.Task);
            controller.Request(snapshot, 0, source.position, 0); controller.Poll(1, source, 0);
            controller.Request(snapshot, 1, source.position, 0); arrival = 1 + controller.PlanningLead + 3;
            source.controlRight = false; source.controlLeft = true;
            for (int frame = 2; frame < arrival; frame++) controller.Poll(frame, source, 0);
            delayed.SetResult((JsonObject)fake.DeepClone()); controller.Poll(arrival, source, 0);
            if (controller.Accepted != 1 || controller.LastRejection != "late_result")
                throw new InvalidOperationException("different_input_history_accepted_late_route");
            Setup(new JsonObject()); snapshot = VanillaFrame.Capture(source, area, true);
            fake["anchor"] = VanillaPrediction.Observe(source);
            controller = new VanillaAsyncMovement(_ => System.Threading.Tasks.Task.FromResult((JsonObject)fake.DeepClone()));
            controller.Request(snapshot, 0, source.position, 0); source.direction = -source.direction;
            controller.Poll(1, source, 0);
            if (controller.Accepted != 1) throw new InvalidOperationException("aim_facing_invalidated_movement");
            foreach (bool terrainChange in new[] { false, true })
            {
                CancellationToken received = default;
                delayed = new TaskCompletionSource<JsonObject>();
                controller = new VanillaAsyncMovement((_, token) => { received = token; return delayed.Task; });
                controller.Request(snapshot, 0, source.position, 0);
                if (terrainChange) controller.TerrainChanged(X, Y, source, 1);
                else controller.Poll(1, source, 1);
                if (!received.IsCancellationRequested || controller.Busy)
                    throw new InvalidOperationException("obsolete_request_was_not_cancelled");
                delayed.SetCanceled();
            }
            return new JsonObject { ["passed"] = true, ["scope"] = "coordinator contract with a synthetic transport, not movement validation" };
        }
        if (command == "terrain_probe")
        {
            var state = request["snapshot"].AsObject();
            try
            {
                var prefix = request["prefix"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                var future = new VanillaPrediction(state).Predict(prefix, false, true)["end_snapshot"].AsObject();
                var player = VanillaFrame.RestoreInWorker(future);
                int left = (int)Math.Floor(player.Left.X / 16), right = (int)Math.Floor((player.Right.X - 1) / 16);
                int bottom = (int)Math.Floor(player.Bottom.Y / 16);
                var clearing = new JsonArray();
                for (int yy = bottom; yy <= bottom + 2; yy++)
                for (int xx = left - 1; xx <= right + 1; xx++)
                {
                    var tile = Main.tile[xx, yy];
                    if (tile.HasTile && Main.tileCut[tile.TileType] && !Main.tileSolid[tile.TileType])
                        clearing.Add(new JsonArray(xx, yy));
                }
                var results = new JsonArray();
                for (int y = bottom; y <= bottom + 1; y++)
                for (int x = left; x <= right; x++)
                foreach (int wait in new[] { 1, 24 })
                foreach (bool clear in new[] { false, true })
                {
                    var targets = clear ? (JsonArray)clearing.DeepClone() : new JsonArray();
                    if (!targets.Any(tile => tile[0].GetValue<int>() == x && tile[1].GetValue<int>() == y))
                        targets.Add(new JsonArray(x, y));
                    var probe = new JsonObject { ["command"] = "terrain_plan", ["snapshot"] = future.DeepClone(),
                        ["tile_x"] = x, ["tile_y"] = y, ["tile_type"] = -1, ["wait_frames"] = wait,
                        ["tiles"] = targets,
                        ["x"] = request["x"].DeepClone(), ["y"] = request["y"].DeepClone(),
                        ["route_direction"] = request["route_direction"].DeepClone(), ["guidance"] = request["guidance"].DeepClone() };
                    var result = VanillaTerrainPlan.Build(probe);
                    if (result["status"].GetValue<string>() == "planned")
                    {
                        var actual = VanillaFrame.RestoreInWorker(future);
                        foreach (var target in targets)
                        {
                            int tx = target[0].GetValue<int>(), ty = target[1].GetValue<int>();
                            int before = Main.tile[tx, ty].TileType;
                            WorldGen.KillTile(tx, ty, noItem: true);
                            if (Main.tile[tx, ty].HasTile && Main.tile[tx, ty].TileType != before)
                                WorldGen.KillTile(tx, ty, noItem: true);
                        }
                        float error = 0;
                        int initialHp = actual.statLife;
                        var replayKeys = result["controls"].AsArray();
                        for (int i = 0; i < replayKeys.Count; i++)
                        {
                            VanillaPrediction.Advance(actual, (VanillaKeys)replayKeys[i].GetValue<int>());
                            var expected = result["forecast"]["frames"][i];
                            error = Math.Max(error, Vector2.Distance(actual.position,
                                new Vector2(expected["x"].GetValue<float>(), expected["y"].GetValue<float>())));
                            if (error > .001f || actual.statLife < initialHp || actual.dead)
                                throw new InvalidOperationException("terrain_probe_replay_diverged");
                        }
                        result["replay_max_error"] = error;
                    }
                    results.Add(new JsonObject { ["x"] = x, ["y"] = y, ["wait"] = wait, ["clear"] = clear, ["tiles"] = targets.DeepClone(), ["result"] = result });
                }
                return new JsonObject { ["results"] = results };
            }
            finally { VanillaFrame.RestoreInWorker(state); }
        }
        if (command == "plan")
        {
            var timer = System.Diagnostics.Stopwatch.StartNew();
            var state = request["snapshot"].AsObject();
            try
            {
                var prefix = request["prefix"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                var advanced = new VanillaPrediction(state).Predict(prefix, false, true);
                if (advanced["end_snapshot"] is not JsonObject future)
                    return new JsonObject { ["status"] = "unsafe_handoff" };
                var anchor = VanillaPrediction.Observe(VanillaFrame.RestoreInWorker(future));
                var result = new VanillaLocalPlanner().Plan(future,
                    new Vector2(request["x"].GetValue<float>(), request["y"].GetValue<float>()),
                    continuation: request["continuation"]?.AsArray().Select(key => key.GetValue<int>()).ToArray(),
                    guidance: request["guidance"], terminal: request["terminal"]?.GetValue<bool>() == true,
                    routeDirection: request["route_direction"]?.GetValue<int>() ?? 0);
                result["anchor"] = anchor;
                result["request_compute_ms"] = timer.Elapsed.TotalMilliseconds;
                return result;
            }
            finally { VanillaFrame.RestoreInWorker(state); }
        }
        if (command == "reset")
        {
            Setup(request);
            checkpoint = VanillaFrame.Capture(source, area);
            return new JsonObject { ["initial"] = VanillaPrediction.Observe(source), ["snapshot"] = checkpoint.DeepClone() };
        }
        if (command == "load")
        {
            var incoming = request["snapshot"].AsObject();
            source = VanillaFrame.RestoreInWorker(incoming);
            checkpoint = (JsonObject)incoming.DeepClone();
            return new JsonObject { ["initial"] = VanillaPrediction.Observe(source) };
        }
        if (checkpoint == null) throw new InvalidOperationException("reset_required");
        if (command == "navigate")
        {
            var trace = new JsonArray();
            var plans = new JsonArray();
            int[] continuation = null;
            var goal = new Vector2(request["x"].GetValue<float>(), request["y"].GetValue<float>());
            int limit = Math.Clamp(request["limit"]?.GetValue<int>() ?? 240, 8, 600);
            bool reached = false;
            float maximumError = 0;
            for (int frame = 0; frame < limit;)
            {
                var snapshotTimer = System.Diagnostics.Stopwatch.StartNew();
                checkpoint = VanillaFrame.Capture(source, VanillaLocalPlanner.SnapshotArea(source, goal));
                double snapshotMs = snapshotTimer.Elapsed.TotalMilliseconds;
                var before = VanillaFrame.CaptureFields(source);
                var plan = new VanillaLocalPlanner().Plan(checkpoint, goal, continuation);
                plan["snapshot_ms"] = snapshotMs;
                plan["snapshot_tiles"] = checkpoint["area"][2].GetValue<int>() * checkpoint["area"][3].GetValue<int>();
                if (!JsonNode.DeepEquals(before, VanillaFrame.CaptureFields(source))) throw new InvalidOperationException("planner_mutated_source");
                plans.Add(plan);
                if (plan["status"].GetValue<string>() != "planned") break;
                var keys = plan["controls"].AsArray().Select(key => key.GetValue<int>()).ToArray();
                Main.player[source.whoAmI] = source;
                int count = Math.Min(VanillaLocalPlanner.ExecuteFrames, limit - frame);
                for (int i = 0; i < count; i++)
                {
                    VanillaPrediction.Advance(source, (VanillaKeys)keys[i]);
                    var expected = plan["forecast"]["frames"][i];
                    float error = Vector2.Distance(source.position, new Vector2(expected["x"].GetValue<float>(), expected["y"].GetValue<float>()));
                    maximumError = Math.Max(error, maximumError);
                    if (error > .001f) throw new InvalidOperationException("execution_diverged_from_prediction");
                    trace.Add(VanillaPrediction.Observe(source)); frame++;
                    if (source.dead) throw new InvalidOperationException("navigation_died");
                    if (Vector2.Distance(source.position, goal) <= 20 && Math.Abs(source.velocity.Y) < .01f) { reached = true; break; }
                }
                continuation = keys.Skip(count).ToArray();
                if (reached) break;
            }
            checkpoint = VanillaFrame.Capture(source, area);
            return new JsonObject { ["reached"] = reached, ["trace"] = trace, ["plans"] = plans, ["max_error"] = maximumError };
        }
        int[] controls = request["controls"]?.AsArray().Select(node => node.GetValue<int>()).ToArray() ?? Array.Empty<int>();
        if (command == "predict")
        {
            var original = VanillaFrame.CaptureFields(source);
            var result = new VanillaPrediction(checkpoint).Predict(controls);
            result["source_unchanged"] = JsonNode.DeepEquals(original, VanillaFrame.CaptureFields(source));
            result["checkpoint_restored"] = JsonNode.DeepEquals(checkpoint,
                VanillaFrame.Capture(Main.player[source.whoAmI], area));
            return result;
        }
        if (command == "actual")
        {
            var trace = new JsonArray();
            Main.player[source.whoAmI] = source;
            foreach (int control in controls) { VanillaPrediction.Advance(source, (VanillaKeys)control); trace.Add(VanillaPrediction.Observe(source)); }
            return new JsonObject { ["frames"] = trace };
        }
        throw new ArgumentException("unknown_command");
    }

    public void Setup(JsonObject request, bool worker = true)
    {
        Main.gameMenu = false;
        if (worker) Main.myPlayer = 255;
        Main.rand = new Terraria.Utilities.UnifiedRandom(73);
        VanillaFrame.RestoreFields(WorldGen.genRand, VanillaFrame.CaptureFields(new Terraria.Utilities.UnifiedRandom(81)));
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        foreach (var item in Main.item) item.active = false;
        for (int x = area.Left; x < area.Right; x++)
        for (int y = area.Top; y < area.Bottom; y++)
        {
            var tile = Main.tile[x, y]; tile.ClearEverything();
            if (y >= Y) { tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        source = new Player { whoAmI = 0, active = true, name = "Forecast subject", statLife = 400, statLifeMax = 400, ConsumedLifeCrystals = 15 };
        if (!worker)
        {
            VanillaFrame.RestoreFields(Main.LocalPlayer, VanillaFrame.CaptureFields(source));
            source = Main.LocalPlayer;
            foreach (var item in source.armor.Concat(source.inventory).Concat(source.miscEquips)) item.SetDefaults(ItemID.None);
        }
        source.position = new Vector2(X * 16, Y * 16 - source.height);
        source.fallStart = (int)(source.position.Y / 16);
        Main.player[0] = source;
        int slot = 3;
        if (request["equipment"] is JsonArray items)
            foreach (var item in items) source.armor[slot++].SetDefaults(item.GetValue<int>());
        if (!worker) Main.ActivePlayerFileData.Player = source;
        if (worker) for (int i = 0; i < 3; i++) VanillaPrediction.Advance(source, VanillaKeys.None);
        if (request["buffs"] is JsonArray buffs)
            foreach (var buff in buffs) source.AddBuff(buff[0].GetValue<int>(), buff[1].GetValue<int>());
        string terrain = request["terrain"]?.GetValue<string>() ?? "flat";
        if (terrain == "ceiling")
            for (int x = X - 3; x < X + 18; x++) { var tile = Main.tile[x, Y - 6]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
        if (terrain is "half" or "slope")
            for (int x = X + 2; x < X + 12; x++)
            {
                var tile = Main.tile[x, Y - 1]; tile.HasTile = true; tile.TileType = TileID.Dirt;
                if (terrain == "half") tile.IsHalfBlock = true;
                else tile.Slope = SlopeType.SlopeDownRight;
            }
        if (terrain == "honey_block")
            for (int x = X - 2; x < X + 20; x++) { var tile = Main.tile[x, Y]; tile.TileType = TileID.HoneyBlock; }
        if (terrain == "honey")
            for (int x = X - 2; x < X + 20; x++)
            for (int y = Y - 4; y < Y; y++) { var tile = Main.tile[x, y]; tile.LiquidType = LiquidID.Honey; tile.LiquidAmount = 255; }
        if (terrain == "embedded")
        {
            var tile = Main.tile[X, Y - 2]; tile.HasTile = true; tile.TileType = TileID.Dirt;
        }
        if (terrain is "ledge" or "wall")
        {
            int height = request["height"]?.GetValue<int>() ?? 12;
            for (int x = X + 9; x < X + (terrain == "ledge" ? 38 : 12); x++)
            for (int y = Y - height; y < Y; y++) { var tile = Main.tile[x, y]; tile.HasTile = true; tile.TileType = TileID.Dirt; }
        }
        if (terrain == "lava_gap")
            for (int x = X + 7; x < X + 20; x++)
            for (int y = Y; y < Y + 12; y++)
            {
                var tile = Main.tile[x, y]; tile.ClearEverything();
                if (y >= Y + 10) { tile.LiquidType = LiquidID.Lava; tile.LiquidAmount = 255; }
            }
        if (worker && terrain is ("honey" or "honey_block"))
            for (int i = 0; i < 3; i++) VanillaPrediction.Advance(source, VanillaKeys.None);
        if (request["airborne"]?.GetValue<bool>() == true)
        {
            source.position.Y -= request["airborne_height"]?.GetValue<int>() ?? 320;
            source.velocity.Y = 3; source.fallStart = (int)(source.position.Y / 16);
        }
        if (worker && request["pre_controls"] is JsonArray prior)
            foreach (var key in prior) VanillaPrediction.Advance(source, (VanillaKeys)key.GetValue<int>());
    }
