namespace Demo;

public sealed class Demo : Mod { }

public sealed class DamageTrace : ModPlayer
{
    public override void PostUpdate()
    {
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavdescentcheck") && Player.FindBuffIndex(BuffID.Shine) < 0)
            Player.AddBuff(BuffID.Shine, 216000);
    }

    public override void OnHurt(Player.HurtInfo info)
    {
        if (!Terraria.Program.LaunchParameters.ContainsKey("-vnavdescentcheck")) return;
        var source = info.DamageSource;
        Console.WriteLine(FormattableString.Invariant($"VNAV_HURT tick={Main.GameUpdateCount} damage={info.Damage} hp_before={Player.statLife} x={Player.Center.X / 16} y={Player.Bottom.Y / 16} fall_start={Player.fallStart} other={source.SourceOtherIndex} npc={source.SourceNPCIndex} projectile={source.SourceProjectileType}"));
    }
}

public sealed class Caller : CommandCaller
{
    public CommandType CommandType => CommandType.Chat;
    public Player Player => Main.LocalPlayer;
    public string Last;
    public void Reply(string text, Color color = default) { Last = text; Main.NewText(text, color == default ? Color.White : color); }
}

public sealed class NoSpawns : GlobalNPC
{
    public override void EditSpawnRate(Player player, ref int spawnRate, ref int maxSpawns)
    {
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavdemocheck")
            || Terraria.Program.LaunchParameters.ContainsKey("-vnavdescentcheck")) maxSpawns = 0;
    }
}

public sealed class Driver : ModSystem
{
    bool enabled, opened, prepared, started, shot;
    int trial, ticks, warmup;
    int menuTicks;
    readonly Caller caller = new();
    readonly JsonArray results = new();
    JsonArray trace = new();
    bool routed;
    bool descent;
    bool continuity;
    int continuityLeg;
    int[] offsets = { 180 };
    int barrierTiles;
    StreamWriter controlTrace;
    JsonObject recordedPlan;
    System.Threading.Tasks.Task<JsonObject> descentGoals;

    public override void Load()
    {
        descent = Terraria.Program.LaunchParameters.ContainsKey("-vnavdescentcheck");
        continuity = Terraria.Program.LaunchParameters.ContainsKey("-vnavcontinuitycheck");
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavcircuitcheck")) offsets = new[] { 180, 360, -180 };
        enabled = !Main.dedServ && (Terraria.Program.LaunchParameters.ContainsKey("-vnavdemocheck") || descent);
        if (!enabled) return;
        routed = Terraria.Program.LaunchParameters.ContainsKey("-vnavroutecheck");
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavwingsonly")) trial = 1;
        string root = Path.GetFullPath(Terraria.Program.LaunchParameters["-tmlsavedirectory"]);
        if (!root.StartsWith("/tmp/") && !root.StartsWith("/private/tmp/")) throw new InvalidOperationException("temporary_root_required");
        On_Main.Update += Update;
        IL_Main.DoUpdate += Background;
    }

    static void Background(ILContext il)
    {
        var cursor = new ILCursor(il);
        if (!cursor.TryGotoNext(MoveType.After, instruction => instruction.MatchStsfld<Main>("hasFocus")))
            throw new InvalidOperationException("background_pause_boundary_missing");
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavbackgroundcheck"))
            cursor.EmitDelegate<Action>(() => Main.hasFocus = false);
        if (!cursor.TryGotoNext(MoveType.After, instruction => instruction.MatchLdsfld<Main>("hasFocus")))
            throw new InvalidOperationException("background_pause_boundary_missing");
        cursor.EmitDelegate<Func<bool, bool>>(_ => true);
    }

    void Update(On_Main.orig_Update original, Main game, GameTime time)
    {
        original(game, time);
        if (!opened && ++menuTicks % 60 == 1) Console.WriteLine("VNAV_DEMO_MENU " + Main.menuMode);
        if (opened || Main.menuMode != 0) return;
        opened = true;
        if (Terraria.Social.SocialAPI.Cloud != null) Terraria.Social.SocialAPI.Cloud.EnabledByDefault = false;
        WorldGen.clearWorld();
        WorldFile.GetAllMetadata(Path.Combine(Main.SavePath, "Worlds", "Demo.wld"), false).SetAsActive();
        var file = PlayerFileData.CreateAndSave(new Player { name = "Vanilla movement demo" });
        if (file.IsCloudSave || !file.Path.StartsWith(Main.SavePath, StringComparison.Ordinal)) throw new InvalidOperationException("player_not_isolated");
        Player.LoadPlayer(file.Path, false).SetAsActive(); Main.autoPause = false;
        WorldGen.playWorld();
    }

    void Scene()
    {
        Main.dayTime = true; Main.time = 27000; Main.worldSurface = 440; Main.rockLayer = 600;
        var player = Main.LocalPlayer;
        var defaults = new Player { active = true, whoAmI = Main.myPlayer, statLife = 400, statLifeMax = 400, ConsumedLifeCrystals = 15 };
        VanillaFrame.RestoreFields(player, VanillaFrame.CaptureFields(defaults));
        foreach (var item in player.inventory.Concat(player.armor).Concat(player.miscEquips)) item.SetDefaults(ItemID.None);
        int slot = 3;
        foreach (int item in trial == 0 ? new[] { ItemID.SandstorminaBottle } : new[] { ItemID.AngelWings }) player.armor[slot++].SetDefaults(item);
        if (routed)
        {
            player.inventory[1].SetDefaults(ItemID.MoltenPickaxe);
            player.inventory[2].SetDefaults(ItemID.WoodPlatform); player.inventory[2].stack = 500;
        }
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        foreach (var item in Main.item) item.active = false;
        for (int x = 455; x < 575; x++)
        for (int y = 310; y < 445; y++)
        {
            var tile = Main.tile[x, y]; tile.ClearEverything();
            bool boundary = x is 455 or 574 || y is 310 or 444;
            if (boundary || y >= 400 || x is >= 509 and < 540 && y >= 388)
            { tile.HasTile = true; tile.TileType = boundary ? TileID.Stone : TileID.Dirt; }
        }
        if (routed && trial == 1)
        {
            for (int x = 456; x < 574; x++)
            for (int y = 388; y < 400; y++)
            {
                var tile = Main.tile[x, y]; tile.ClearEverything();
                if (y == 390 || x is >= 509 and < 512 && y >= 390) { tile.HasTile = true; tile.TileType = TileID.Dirt; }
            }
        }
        player.position = new Vector2(500 * 16, 400 * 16 - player.height);
        barrierTiles = Enumerable.Range(456, 118).Sum(x => Enumerable.Range(390, 10).Count(y => Main.tile[x, y].HasTile));
        player.fallStart = (int)(player.position.Y / 16); player.velocity = Vector2.Zero;
        ticks = 0; warmup = 8; started = false; shot = false; prepared = true; trace = new JsonArray();
        Main.NewText(trial == 0 ? "演示1：只用沙暴瓶登上12格高台" : "演示2：翅膀登上同一个高台", Color.LightGreen);
    }

    public override void PostUpdateEverything()
    {
        if (!enabled || Main.gameMenu) return;
        try
        {
            if (descent) { Descent(); return; }
            if (!prepared) { Scene(); return; }
            if (warmup-- > 0) return;
            var demo = VanillaNavigationDemo.Instance;
            if (!started)
            {
                string[] args = routed ? new[] { "route", "522", trial == 0 ? "387" : "399" } : new[] { "522", "388" };
                ModContent.GetInstance<VanillaNavigationCommand>().Action(caller, "/vnav", args);
                if (!(routed ? RecedingNav.Active : VanillaNavigationDemo.Active)) throw new InvalidOperationException(caller.Last);
                started = true;
            }
            ticks++;
            var sample = VanillaPrediction.Observe(Main.LocalPlayer);
            sample["status"] = demo.PlannerStatus;
            sample["rejection"] = demo.PlannerRejection;
            sample["mismatch"] = demo.PlannerMismatch;
            sample["accepted"] = demo.PlanAccepted; sample["rejected"] = demo.PlanRejected;
            sample["recede"] = RecedingNav.Active; sample["macro"] = StateSpacePlanner.ExecRunning;
            sample["last_stop"] = RecedingNav.LastStop;
            if (demo.PlannerResult is JsonObject plan)
            {
                sample["selected"] = plan["selected"]?.DeepClone();
                sample["roundtrip_ms"] = plan["roundtrip_ms"]?.DeepClone();
                sample["compute_ms"] = plan["request_compute_ms"]?.DeepClone();
            }
            trace.Add(sample);
            if (ticks % 30 == 0) File.WriteAllText(Path.Combine(Main.SavePath, $"trace-{trial}.json"), trace.ToJsonString());
            if (routed ? RecedingNav.Active : VanillaNavigationDemo.Active)
            {
                if (ticks > 9000) throw new TimeoutException(demo.Status);
                return;
            }
            if (!(routed ? RecedingNav.LastStop == "done" : demo.Reached)) throw new InvalidOperationException(demo.Status + ":" + RecedingNav.LastStop);
            if (routed && trial == 1)
            {
                int remaining = Enumerable.Range(456, 118).Sum(x => Enumerable.Range(390, 10).Count(y => Main.tile[x, y].HasTile));
                if (remaining >= barrierTiles) throw new InvalidOperationException("enclosure_was_not_mined");
                int lastMacro = -1;
                for (int i = 0; i < trace.Count; i++) if (trace[i]["macro"].GetValue<bool>()) lastMacro = i;
                if (lastMacro < 0 || !trace.Skip(lastMacro + 1).Any(frame => frame["accepted"].GetValue<int>() > 0))
                    throw new InvalidOperationException("engine_did_not_resume_after_mining");
            }
            if (!routed && (demo.RoutePoints < 2 || !shot)) throw new InvalidOperationException("route_was_not_drawn");
            results.Add(new JsonObject { ["equipment"] = trial == 0 ? "sandstorm" : "wings", ["ticks"] = ticks,
                ["reached"] = true, ["hp"] = Main.LocalPlayer.statLife, ["route_points"] = demo.RoutePoints });
            File.WriteAllText(Path.Combine(Main.SavePath, "demo-result.json"), results.ToJsonString());
            Console.WriteLine($"VNAV_DEMO_CASE_PASS trial={trial} ticks={ticks}");
            if (++trial < 2) { prepared = false; return; }
            ModContent.GetInstance<VanillaNavigationCommand>().Action(caller, "/vnav stop", new[] { "stop" });
            if (VanillaNavigationDemo.Active || AxisLock.Held(Ax.Move) == "vanilla-demo") throw new InvalidOperationException("stop_did_not_release_control");
            Console.WriteLine("VNAV_DEMO_COMPLETE cases=2");
            enabled = false;
            if (!Terraria.Program.LaunchParameters.ContainsKey("-vnavshow")) Environment.Exit(0);
            Main.NewText("演示已通过。可以输入 /vnav X Y 继续尝试；/vnav stop 停止。", Color.LightGreen);
        }
        catch (Exception error)
        {
            controlTrace?.Flush();
            Console.WriteLine("VNAV_DEMO_FAILED " + error);
            enabled = false; VanillaNavigationDemo.Instance.Stop("演示失败：" + error.Message);
            if (!Terraria.Program.LaunchParameters.ContainsKey("-vnavshow")) Environment.Exit(1);
            Main.NewText("演示失败，窗口已保留：" + error.Message, Color.OrangeRed);
        }
    }

    void Descent()
    {
        var player = Main.LocalPlayer;
        if (!prepared)
        {
            controlTrace = File.CreateText(Path.Combine(Main.SavePath, "control-trace.csv"));
            controlTrace.WriteLine("tick,leg,x,y,vx,vy,actual_keys,scheduled_keys,queued,plan,index,terminal,goal_x,goal_y,plan_goal_x,anchor_x,engine,macro,rejection,hp,dead,fall_start");
            player.ConsumedLifeCrystals = 15; player.statLife = player.statLifeMax = 400;
            foreach (var item in player.inventory.Concat(player.armor).Concat(player.miscEquips)) item.SetDefaults(ItemID.None);
            player.armor[0].SetDefaults(ItemID.MoltenHelmet); player.armor[1].SetDefaults(ItemID.MoltenBreastplate); player.armor[2].SetDefaults(ItemID.MoltenGreaves);
            int slot = 3;
            foreach (int item in new[] { ItemID.SandstorminaBottle, ItemID.NeptunesShell, ItemID.CobaltShield, ItemID.HermesBoots, ItemID.BandofRegeneration }) player.armor[slot++].SetDefaults(item);
            player.inventory[0].SetDefaults(ItemID.Starfury);
            player.inventory[1].SetDefaults(ItemID.MoltenPickaxe);
            player.inventory[2].SetDefaults(ItemID.WoodPlatform); player.inventory[2].stack = 999;
            player.inventory[3].SetDefaults(ItemID.Torch); player.inventory[3].stack = 999;
            player.inventory[4].SetDefaults(ItemID.HealingPotion); player.inventory[4].stack = 50;
            player.inventory[5].SetDefaults(ItemID.Cobweb); player.inventory[5].stack = 999;
            prepared = true; warmup = 10; return;
        }
        if (warmup-- > 0) return;
        if (!started)
        {
            string[] args;
            if (continuity)
            {
                int x = Main.spawnTileX + offsets[continuityLeg], y = Math.Max(10, Main.spawnTileY - 60);
                while (y < Main.maxTilesY - 200 && !Predicates.IsGround(x, y)) y++;
                args = new[] { "route", x.ToString(), (y - 1).ToString() };
            }
            else
            {
                if (descentGoals == null) descentGoals = System.Threading.Tasks.Task.Run(HttpServerSystem.NavigationDescentGoals);
                if (!descentGoals.IsCompleted) return;
                var goals = descentGoals.GetAwaiter().GetResult();
                int x = goals["entrance"]["x"].GetValue<int>(), y = goals["entrance"]["y"].GetValue<int>();
                player.Teleport(new Vector2(x * 16 + 8 - player.width / 2, (y + 1) * 16 - player.height));
                player.velocity = Vector2.Zero; player.fallStart = (int)(player.position.Y / 16);
                args = new[] { "route", goals["hell_x"].ToString(), goals["hell_y"].ToString() };
                Console.WriteLine("VNAV_DESCENT_GOALS " + goals.ToJsonString());
                if (Terraria.Program.LaunchParameters.ContainsKey("-vnavmanualcheck"))
                {
                    VanillaNavBridge.Enabled = true;
                    if (Terraria.Program.LaunchParameters.ContainsKey("-vnavmanualstart"))
                        ModContent.GetInstance<VanillaNavigationCommand>().Action(caller, "/vnav", args);
                    enabled = false;
                    Main.NewText("手动测试：已关闭刷怪并添加光芒效果。DragonLens 已启用。", Color.LightGreen);
                    Main.NewText($"自动寻路：/vnav route {args[1]} {args[2]}；停止：/vnav stop", Color.LightGreen);
                    Console.WriteLine("VNAV_MANUAL_READY " + goals.ToJsonString());
                    return;
                }
            }
            ModContent.GetInstance<VanillaNavigationCommand>().Action(caller, "/vnav", args);
            if (Terraria.Program.LaunchParameters.ContainsKey("-vnavoldnavcheck")) VanillaNavBridge.Enabled = false;
            if (!RecedingNav.Active) throw new InvalidOperationException(caller.Last);
            started = true;
        }
        ticks++;
        if (VanillaNavBridge.TerrainOwned && VanillaNavigationDemo.Active)
            throw new InvalidOperationException("movement_stole_terrain_ownership");
        var sample = VanillaPrediction.Observe(player);
        var control = VanillaNavigationDemo.Instance.LastControl;
        controlTrace.WriteLine(FormattableString.Invariant($"{ticks},{continuityLeg},{player.position.X},{player.position.Y},{player.velocity.X},{player.velocity.Y},{sample["keys"]},{control.Keys},{control.Queued},{control.Plan},{control.Index},{control.Terminal},{control.Goal.X},{control.Goal.Y},{control.PlanGoal.X},{control.Anchor.X},{VanillaNavigationDemo.Active},{StateSpacePlanner.ExecRunning},{VanillaNavigationDemo.Instance.PlannerRejection},{player.statLife},{player.dead},{player.fallStart}"));
        sample["phase"] = continuity ? "continuity" : "jungle_descent";
        sample["focused"] = Main.hasFocus;
        sample["nav_stop"] = RecedingNav.LastStop;
        sample["macro"] = StateSpacePlanner.ExecRunning;
        sample["engine"] = VanillaNavigationDemo.Active;
        sample["rejection"] = VanillaNavigationDemo.Instance.PlannerRejection;
        sample["mismatch"] = VanillaNavigationDemo.Instance.PlannerMismatch;
        sample["use_item"] = player.controlUseItem;
        sample["held_item"] = player.HeldItem.type;
        sample["hostiles"] = Main.npc.Count(npc => npc.active && !npc.friendly && npc.damage > 0);
        if (VanillaNavigationDemo.Instance.PlannerResult is JsonObject plan)
        {
            if (!ReferenceEquals(plan, recordedPlan))
            {
                recordedPlan = plan;
                string selected = plan["selected"]?.GetValue<string>() ?? "none";
                if (selected == "wait" || selected.StartsWith("retreat"))
                    File.WriteAllText(Path.Combine(Main.SavePath, $"fallback-plan-{ticks}.json"), plan.ToJsonString());
            }
            sample["compute_ms"] = plan["request_compute_ms"]?.DeepClone();
            sample["roundtrip_ms"] = plan["roundtrip_ms"]?.DeepClone();
            sample["selected"] = plan["selected"]?.DeepClone();
            sample["timing"] = plan["timing"]?.DeepClone();
        }
        if (ticks % 6 == 0) { trace.Add(sample); if (trace.Count > 600) trace.RemoveAt(0); }
        if (ticks % 120 == 0)
        {
            controlTrace.Flush();
            File.WriteAllText(Path.Combine(Main.SavePath, "descent-trace.json"), trace.ToJsonString());
            Console.WriteLine($"VNAV_DESCENT_PROGRESS tick={ticks} cell=({player.Center.X / 16:0},{player.Bottom.Y / 16:0}) hp={player.statLife} phase={VanillaDescentRun.Instance.Status} nav={RecedingNav.LastStop} engine={VanillaNavigationDemo.Instance.Status}");
        }
        if (player.dead) throw new InvalidOperationException("navigation_player_dead");
        if (RecedingNav.Active && ticks < 54000) return;
        if (RecedingNav.LastStop != "done") throw new InvalidOperationException(RecedingNav.LastStop ?? "navigation_timeout");
        if (continuity)
        {
            results.Add(new JsonObject { ["offset"] = offsets[continuityLeg], ["tick"] = ticks,
                ["hp"] = player.statLife, ["x"] = player.Center.X / 16, ["y"] = player.Bottom.Y / 16 });
            File.WriteAllText(Path.Combine(Main.SavePath, "continuity-legs.json"), results.ToJsonString());
            Console.WriteLine($"VNAV_CONTINUITY_LEG_PASS offset={offsets[continuityLeg]} tick={ticks}");
            if (++continuityLeg < offsets.Length) { started = false; return; }
        }
        File.WriteAllText(Path.Combine(Main.SavePath, "descent-result.json"), new JsonObject { ["done"] = true,
            ["ticks"] = ticks, ["hp"] = player.statLife, ["x"] = player.Center.X / 16, ["y"] = player.Bottom.Y / 16,
            ["underworld"] = Main.UnderworldLayer, ["legs"] = results.DeepClone() }.ToJsonString());
        Console.WriteLine(continuity ? "VNAV_CONTINUITY_COMPLETE" : "VNAV_DESCENT_COMPLETE"); enabled = false;
        controlTrace.Flush();
        if (!Terraria.Program.LaunchParameters.ContainsKey("-vnavshow")) Environment.Exit(0);
    }

    public override void PostDrawInterface(SpriteBatch batch)
    {
        if (!enabled || !started || shot || !VanillaNavigationDemo.Active || VanillaNavigationDemo.Instance.RoutePoints < 2) return;
        var device = Main.instance.GraphicsDevice;
        int width = device.PresentationParameters.BackBufferWidth, height = device.PresentationParameters.BackBufferHeight;
        var pixels = new Color[width * height]; device.GetBackBufferData(pixels);
        using var texture = new Texture2D(device, width, height); texture.SetData(pixels);
        using var stream = File.Create(Path.Combine(Main.SavePath, $"demo-{trial}.png")); texture.SaveAsPng(stream, width, height);
        shot = true;
    }

    public override void Unload()
    {
        controlTrace?.Dispose();
        if (!opened) return;
        On_Main.Update -= Update; IL_Main.DoUpdate -= Background;
    }
}
