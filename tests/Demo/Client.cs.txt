public sealed class FieldTimingCheck : ModSystem
{
    public override void Load()
    {
        if (Main.dedServ && Terraria.Program.LaunchParameters.ContainsKey("-fieldtimingcheck"))
            Main.OnEnginePreload += Run;
    }
    public override void Unload() => Main.OnEnginePreload -= Run;
    static void Run()
    {
        try
        {
            var fixture = JsonNode.Parse(File.ReadAllText(Terraria.Program.LaunchParameters["-fieldtimingcheck"]));
            var player = VanillaFrame.RestoreInWorker(fixture["request"]["snapshot"].AsObject());
            Main.myPlayer = player.whoAmI;
            var start = VanillaLocalPlanner.FootCell(player.position, player.width, player.height);
            var snapshot = fixture["request"]["snapshot"].AsObject();
            var area = snapshot["area"].AsArray();
            int walkChecks = 0;
            double decisionMs = 0;
            int decisions = 0;
            for (int y = area[1].GetValue<int>() + 4; y < area[1].GetValue<int>() + area[3].GetValue<int>() - 4 && walkChecks < 6; y++)
            for (int x = area[0].GetValue<int>() + 12; x < area[0].GetValue<int>() + area[2].GetValue<int>() - 12 && walkChecks < 6; x++)
            {
                player.position = new Vector2(x * 16, y * 16 - player.height);
                player.velocity = Vector2.Zero; player.jump = 0; player.releaseJump = true;
                var goal = player.position + new Vector2(160, 0);
                if (!VanillaContactRegion.TryWalk(player, goal, out _)) continue;
                var flags = System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.NonPublic;
                var startupFields = new[] { "_fieldReady", "_goalWx", "_goalWy", "_mode", "_startedAt", "_startupWalkReported" }
                    .Select(name => typeof(RecedingNav).GetField(name, flags)).ToArray();
                var saved = startupFields.Select(field => field.GetValue(null)).ToArray();
                bool oldRegions = VanillaNavBridge.RegionsEnabled;
                try
                {
                    startupFields[0].SetValue(null, false);
                    startupFields[1].SetValue(null, x + 600);
                    startupFields[2].SetValue(null, y - 1);
                    startupFields[3].SetValue(null, RecedingNav.Mode.Snap);
                    startupFields[4].SetValue(null, System.Diagnostics.Stopwatch.GetTimestamp());
                    startupFields[5].SetValue(null, false);
                    VanillaNavBridge.RegionsEnabled = true; RecedingNav.Active = true;
                    player.controlRight = false;
                    RecedingNav.Tick();
                    if (!player.controlRight) throw new InvalidOperationException("startup_waited_for_global_field");
                }
                finally
                {
                    RecedingNav.Active = false; VanillaNavBridge.RegionsEnabled = oldRegions;
                    AxisLock.Release("recede");
                    for (int i = 0; i < startupFields.Length; i++) startupFields[i].SetValue(null, saved[i]);
                    player.controlRight = false;
                }
                int hp = player.statLife;
                var origin = player.position;
                int requests = 0, moving = 0;
                var pending = new System.Threading.Tasks.TaskCompletionSource<JsonObject>();
                Func<JsonObject, System.Threading.CancellationToken, System.Threading.Tasks.Task<JsonObject>> send = (request, token) =>
                { requests++; token.Register(() => pending.TrySetCanceled(token)); return pending.Task; };
                var type = typeof(VanillaNavigationDemo).Assembly.GetType("TerraBlind.VanillaRegionNavigation", true);
                var obsolete = new System.Threading.Tasks.TaskCompletionSource<JsonObject>();
                Func<JsonObject, System.Threading.CancellationToken, System.Threading.Tasks.Task<JsonObject>> delayed = (request, token) =>
                { token.Register(() => obsolete.TrySetCanceled(token)); return obsolete.Task; };
                using (var switcher = (IDisposable)Activator.CreateInstance(type, delayed))
                {
                    type.GetMethod("Tick").Invoke(switcher, new object[] { player, 0L, goal - new Vector2(0, 160) });
                    type.GetMethod("Tick").Invoke(switcher, new object[] { player, 1L, goal });
                    var resumed = (VanillaPrediction.Input)type.GetMethod("InputAt").Invoke(switcher, new object[] { 2L });
                    if (!obsolete.Task.IsCanceled || resumed.Keys == VanillaKeys.None)
                        throw new InvalidOperationException("pending_plan_blocked_safe_walk");
                }
                using var controller = (IDisposable)Activator.CreateInstance(type, send);
                for (int frame = 0; frame < 45; frame++)
                {
                    var timer = System.Diagnostics.Stopwatch.StartNew();
                    bool walkable = VanillaContactRegion.TryWalk(player, goal, out _);
                    type.GetMethod("Tick").Invoke(controller, new object[] { player, (long)frame, goal });
                    var input = (VanillaPrediction.Input)type.GetMethod("InputAt").Invoke(controller, new object[] { (long)frame + 1 });
                    decisionMs += timer.Elapsed.TotalMilliseconds; decisions++;
                    if (walkable && requests == 0 && input.Keys == VanillaKeys.None)
                        throw new InvalidOperationException("quick_walk_controller_gap");
                    if (input.Keys != VanillaKeys.None) moving++;
                    VanillaPrediction.Advance(player, input.Keys);
                    if (player.statLife < hp || player.dead || !VanillaContactRegion.Supported(player.position, player.width, player.height))
                        throw new InvalidOperationException("quick_walk_left_safe_ground");
                }
                if (requests > 1 || moving == 0) throw new InvalidOperationException("quick_walk_request_storm");
                if (player.position.X <= origin.X) throw new InvalidOperationException("quick_walk_no_progress");
                walkChecks++;
                player = VanillaFrame.RestoreInWorker(snapshot);
            }
            if (walkChecks == 0) throw new InvalidOperationException("no_real_map_walk_samples");
            Console.WriteLine($"QUICK_WALK_PASS samples={walkChecks} decisions={decisions} decision_ms={decisionMs:0.###}");
            player = VanillaFrame.RestoreInWorker(snapshot);
            var rows = new JsonArray();
            foreach (int distance in new[] { 60, 600 })
            foreach (bool wide in new[] { false, true })
            for (int repeat = 0; repeat < 2; repeat++)
            {
                int gx = start.x + (start.x + distance < Main.maxTilesX - 10 ? distance : -distance);
                var timer = System.Diagnostics.Stopwatch.StartNew();
                var field = MazeWand.BuildField(gx, start.y, start.x, start.y, wide);
                double elapsed = timer.Elapsed.TotalMilliseconds;
                timer.Restart();
                var bounded = MazeWand.BuildField(gx, start.y, start.x, start.y, wide, stopAtStart: true);
                double boundedMs = timer.Elapsed.TotalMilliseconds;
                if (!bounded.TryGetValue(start, out int cost) || !field.TryGetValue(start, out int fullCost) || cost != fullCost)
                    throw new InvalidOperationException("bounded_field_cost_mismatch");
                var path = MazeWand.TraceFrom(bounded, start.x, start.y, gx, start.y);
                if (path.Count == 0 || path[^1] != (gx, start.y)) throw new InvalidOperationException("bounded_field_route_incomplete");
                var row = new JsonObject { ["distance"] = distance, ["wide"] = wide, ["repeat"] = repeat,
                    ["elapsed_ms"] = elapsed, ["cells"] = field.Count, ["start_reachable"] = field.ContainsKey(start),
                    ["bounded_ms"] = boundedMs, ["bounded_cells"] = bounded.Count, ["route_cells"] = path.Count };
                rows.Add(row); Console.WriteLine("FIELD_TIMING " + row.ToJsonString());
                if (!field.ContainsKey(start)) throw new InvalidOperationException("field_did_not_reach_start");
            }
            File.WriteAllText(Path.Combine(Main.SavePath, "field-timing.json"), rows.ToJsonString());
            Console.WriteLine("FIELD_TIMING_PASS");
            Environment.Exit(0);
        }
        catch (Exception error) { Console.WriteLine("FIELD_TIMING_FAIL " + error); Environment.Exit(1); }
    }
}

public sealed class RegionHandoffCheck : ModSystem
{
    public override void Load()
    {
        if (Main.dedServ && Terraria.Program.LaunchParameters.ContainsKey("-regionhandoffcheck"))
            Main.OnEnginePreload += Run;
    }
    public override void Unload() => Main.OnEnginePreload -= Run;

    static void Run()
    {
        object controller = null;
        VanillaWorkerConnection worker = null;
        try
        {
            var fixture = JsonNode.Parse(File.ReadAllText(Terraria.Program.LaunchParameters["-regionhandoffcheck"])).AsObject();
            var snapshot = fixture["snapshot"].AsObject();
            var results = fixture["results"].AsArray();
            string scenario = fixture["scenario"]?.GetValue<string>() ?? "early";
            if (scenario == "worker")
                worker = VanillaWorkerConnection.Start(ModLoader.GetMod("TerraBlind").GetFileBytes("Assets/VanillaWorker.bin"),
                    Path.Combine(Main.SavePath, "World.wld"), System.Threading.CancellationToken.None).GetAwaiter().GetResult();
            var packed = System.Runtime.InteropServices.MemoryMarshal.Cast<byte, int>(
                Convert.FromBase64String(fixture["guidance"].GetValue<string>()).AsSpan()).ToArray();
            var field = new System.Collections.Generic.Dictionary<(int, int), int>();
            for (int i = 0; i < packed.Length; i += 3) field[(packed[i], packed[i + 1])] = packed[i + 2];
            typeof(VanillaNavBridge).GetField("field", System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.NonPublic).SetValue(null, field);
            var player = VanillaFrame.RestoreInWorker(snapshot);
            var startCell = VanillaLocalPlanner.FootCell(player.position, player.width, player.height);
            int startCost = field[startCell];
            var type = typeof(VanillaNavigationDemo).Assembly.GetType("TerraBlind.VanillaRegionNavigation", true);
            var tickMethod = type.GetMethod("Tick");
            var inputMethod = type.GetMethod("InputAt");
            int requests = 0;
            var waiting = new System.Threading.Tasks.TaskCompletionSource<JsonObject>();
            var delayed = new System.Threading.Tasks.TaskCompletionSource<JsonObject>();
            async System.Threading.Tasks.Task<JsonObject> SendWorker(JsonObject request, System.Threading.CancellationToken token, int index)
            {
                File.WriteAllText(Path.Combine(Main.SavePath, $"region-request-{index}.json"), request.ToJsonString());
                var response = await worker.Send(request, token);
                File.WriteAllText(Path.Combine(Main.SavePath, $"region-response-{index}.json"), response.ToJsonString());
                return response;
            }
            Func<JsonObject, System.Threading.CancellationToken, System.Threading.Tasks.Task<JsonObject>> send = (request, token) =>
            {
                int index = requests++;
                if (worker != null) return SendWorker(request, token, index);
                if (scenario == "late" && index == 1)
                {
                    token.Register(() => delayed.TrySetCanceled(token));
                    return delayed.Task;
                }
                int available = scenario == "early" ? results.Count : scenario == "late" ? 1 : 2;
                if (index < available) return System.Threading.Tasks.Task.FromResult(results[index].DeepClone().AsObject());
                token.Register(() => waiting.TrySetCanceled(token));
                return waiting.Task;
            };
            controller = Activator.CreateInstance(type, send);
            var goal = player.position + new Vector2(0, 160);
            void Tick(long frame) => tickMethod.Invoke(controller, new object[] { player, frame, goal });
            Tick(0); Tick(0);
            int firstFrames = results[0]["controls"].AsArray().Count;
            int frames = scenario == "worker" ? 1800 : scenario == "early" ? results.Sum(result => result["controls"].AsArray().Count)
                : firstFrames + (scenario == "late" ? 5 : 0);
            int missing = 0, hp = player.statLife;
            int initialWaiting = 0, intersegmentWaiting = 0;
            int firstInputFrame = -1, longestWait = 0, waitRun = 0;
            int completedSegments = 0;
            var clock = System.Diagnostics.Stopwatch.StartNew();
            for (long frame = 1; frame <= frames; frame++)
            {
                if (!(bool)type.GetMethod("HasInput").Invoke(controller, new object[] { frame }))
                {
                    missing++;
                    longestWait = Math.Max(longestWait, ++waitRun);
                    if ((int)type.GetProperty("Accepted").GetValue(controller) == 0) initialWaiting++;
                    else intersegmentWaiting++;
                }
                else
                {
                    if (firstInputFrame < 0) firstInputFrame = (int)frame;
                    waitRun = 0;
                }
                var input = (VanillaPrediction.Input)inputMethod.Invoke(controller, new object[] { frame });
                VanillaPrediction.Advance(player, input.Keys, input.UseSlot, input.Target);
                if (scenario == "worker" && type.GetProperty("Result").GetValue(controller) is JsonObject current
                    && (int)type.GetMethod("Index").Invoke(controller, new object[] { frame }) == current["controls"].AsArray().Count - 1)
                {
                    if (!VanillaPrediction.RegionPlayback.MatchesAnchor(player, current["frames"].AsArray()[^1].AsObject(), out string mismatch))
                        throw new InvalidOperationException("handoff_endpoint_mismatch:" + mismatch);
                    completedSegments++;
                    if (completedSegments == 4) { frames = (int)frame; break; }
                }
                if (scenario == "terrain" && frame == firstFrames)
                {
                    var next = results[1]["frames"][0];
                    var tile = Main.tile[(int)Math.Floor(next["x"].GetValue<float>() / 16) - 1,
                        (int)Math.Floor(next["y"].GetValue<float>() / 16) - 1];
                    tile.HasTile = !tile.HasTile; tile.TileType = TileID.Stone;
                }
                if (scenario == "late" && frame == frames)
                {
                    if (delayed.Task.IsCanceled) throw new InvalidOperationException("late_request_cancelled_before_observation");
                    delayed.SetResult(results[1].DeepClone().AsObject());
                }
                Tick(frame);
                if (scenario == "worker")
                {
                    double remaining = frame * 1000d / 60 - clock.Elapsed.TotalMilliseconds;
                    if (remaining > 0) System.Threading.Thread.Sleep(TimeSpan.FromMilliseconds(remaining));
                }
            }
            int accepted = (int)type.GetProperty("Accepted").GetValue(controller);
            int rejected = (int)type.GetProperty("Rejected").GetValue(controller);
            int expectedAccepted = scenario == "worker" ? 4 : scenario == "early" ? results.Count : 1;
            int expectedRejected = scenario is "worker" or "early" ? 0 : 1;
            int expectedMissing = scenario == "worker" ? missing : scenario == "late" ? 5 : 0;
            var endCell = VanillaLocalPlanner.FootCell(player.position, player.width, player.height);
            bool hasEndCost = field.TryGetValue(endCell, out int endCost);
            bool passed = accepted == expectedAccepted && rejected == expectedRejected && missing == expectedMissing
                && !player.dead && player.statLife >= hp && hasEndCost && endCost < startCost
                && (scenario != "worker" || intersegmentWaiting == 0 && completedSegments == 4);
            ((IDisposable)controller).Dispose(); controller = null;
            if (scenario != "worker" && !waiting.Task.IsCanceled) throw new InvalidOperationException("pending_request_not_cancelled");
            worker?.Dispose(); worker = null;
            var report = new JsonObject { ["passed"] = passed, ["scenario"] = scenario, ["accepted"] = accepted, ["rejected"] = rejected,
                ["frames"] = frames, ["missing_input_frames"] = missing, ["pending_cancelled"] = scenario != "worker",
                ["initial_wait_frames"] = initialWaiting, ["intersegment_wait_frames"] = intersegmentWaiting,
                ["first_input_frame"] = firstInputFrame, ["longest_wait_frames"] = longestWait,
                ["start_hp"] = hp, ["end_hp"] = player.statLife, ["dead"] = player.dead,
                ["wall_ms"] = clock.Elapsed.TotalMilliseconds,
                ["completed_segments"] = scenario == "worker" ? completedSegments : scenario == "early" ? accepted : 1,
                ["start_cost"] = startCost, ["end_cost"] = endCost,
                ["end"] = VanillaPrediction.Observe(player, false) };
            File.WriteAllText(Path.Combine(Main.SavePath, "region-handoff-result.json"), report.ToJsonString());
            if (!passed) throw new InvalidOperationException("handoff_failed " + report.ToJsonString());
            Console.WriteLine("REGION_HANDOFF_PASS " + report.ToJsonString());
            Environment.Exit(0);
        }
        catch (Exception error)
        {
            (controller as IDisposable)?.Dispose();
            worker?.Dispose();
            Console.WriteLine("REGION_HANDOFF_FAIL " + error);
            Environment.Exit(1);
        }
    }
}

public sealed class DamageTrace : ModPlayer
{
    public override bool FreeDodge(Player.HurtInfo info)
        => Terraria.Program.LaunchParameters.ContainsKey("-vnavreadmeroute");

    public override void PostUpdate()
    {
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavreadmeroute"))
            Player.AddBuff(BuffID.Gills, 216000);
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavdescentcheck") && Player.FindBuffIndex(BuffID.Shine) < 0)
            Player.AddBuff(BuffID.Shine, 216000);
    }

    public override void OnHurt(Player.HurtInfo info)
    {
        if (!Terraria.Program.LaunchParameters.ContainsKey("-vnavdescentcheck")) return;
        var source = info.DamageSource;
        Console.WriteLine(FormattableString.Invariant($"VNAV_HURT tick={Main.GameUpdateCount} damage={info.Damage} hp_before={Player.statLife} x={Player.Center.X / 16} y={Player.Bottom.Y / 16} fall_start={Player.fallStart} other={source.SourceOtherIndex} npc={source.SourceNPCIndex} projectile={source.SourceProjectileType}"));
    }
}

public sealed class NoSpawns : GlobalNPC
{
    public override void OnSpawn(NPC npc, Terraria.DataStructures.IEntitySource source)
    {
        if ((!Terraria.Program.LaunchParameters.ContainsKey("-vnavrecordedgoal")
            && !Terraria.Program.LaunchParameters.ContainsKey("-vnavregionscheck")) || npc.friendly || npc.damage <= 0) return;
        Console.WriteLine($"VNAV_SUPPRESSED_SPAWN type={npc.type} source={source.GetType().Name}");
        npc.active = false;
    }

    public override void EditSpawnRate(Player player, ref int spawnRate, ref int maxSpawns)
    {
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavdemocheck")
            || Terraria.Program.LaunchParameters.ContainsKey("-vnavdescentcheck")) maxSpawns = 0;
    }
}

    static void FinishProcess(int exitCode)
    {
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavkeepopen")) return;
        VanillaNavigationDemo.Instance.OnWorldUnload();
        Environment.Exit(exitCode);
    }

    void WorkerLifecycle()
    {
        var demo = VanillaNavigationDemo.Instance;
        var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic;
        var startupField = typeof(VanillaNavigationDemo).GetField("startup", flags);
        var workerField = typeof(VanillaNavigationDemo).GetField("worker", flags);
        var worker = workerField.GetValue(demo) as VanillaWorkerConnection;
        if (++ticks > 5400) throw new TimeoutException("worker_prewarm_timeout");
        if (ticks == 1)
        {
            prewarming = startupField.GetValue(demo);
            if (prewarming == null) throw new InvalidOperationException("world_prewarm_not_started");
            demo.Stop();
            if (!ReferenceEquals(prewarming, startupField.GetValue(demo)))
                throw new InvalidOperationException("stop_cancelled_prewarm");
        }
        if (worker == null) return;
        var start = typeof(VanillaNavigationDemo).GetMethod("StartNavigation", flags);
        for (int task = 0; task < 2; task++)
        {
            start.Invoke(demo, new object[] { Main.LocalPlayer.position });
            if (!demo.PlannerReady || !ReferenceEquals(worker, workerField.GetValue(demo)))
                throw new InvalidOperationException("task_did_not_reuse_worker");
            demo.Stop();
            if (!worker.Alive) throw new InvalidOperationException("stop_closed_worker");
        }
        demo.OnWorldUnload();
        if (worker.Alive || workerField.GetValue(demo) != null || startupField.GetValue(demo) != null)
            throw new InvalidOperationException("world_unload_leaked_worker");
        Console.WriteLine("VNAV_WORKER_COMPLETE tasks=2 starts=1 pending_stop=preserved unload=closed");
        enabled = false;
        Environment.Exit(0);
    }

    static void RestoreBaseline(Player player, string directory)
    {
        var meta = JsonNode.Parse(File.ReadAllText(Path.Combine(directory, "metadata.json")));
        if (meta["world_width"].GetValue<int>() != Main.maxTilesX || meta["world_height"].GetValue<int>() != Main.maxTilesY)
            throw new InvalidOperationException("baseline_world_dimensions_differ");
        using (var reader = new BinaryReader(File.OpenRead(Path.Combine(directory, "initial-terrain.bin"))))
        {
            if (reader.ReadInt32() != 1) throw new InvalidOperationException("baseline_terrain_version");
            int x0 = reader.ReadInt32(), y0 = reader.ReadInt32(), width = reader.ReadInt32(), height = reader.ReadInt32();
            for (int y = y0; y < y0 + height; y++)
            for (int x = x0; x < x0 + width; x++)
            {
                var tile = Main.tile[x, y];
                tile.HasTile = reader.ReadBoolean(); tile.TileType = reader.ReadUInt16(); tile.WallType = reader.ReadUInt16();
                tile.TileFrameX = reader.ReadInt16(); tile.TileFrameY = reader.ReadInt16();
                tile.Slope = (SlopeType)reader.ReadByte();
                tile.IsHalfBlock = reader.ReadBoolean(); tile.IsActuated = reader.ReadBoolean();
                tile.LiquidType = reader.ReadInt32(); tile.LiquidAmount = reader.ReadByte();
            }
        }
        void RestoreItems(Item[] items, JsonArray values)
        {
            for (int i = 0; i < items.Length; i++)
            {
                items[i].SetDefaults(values[i][0].GetValue<int>());
                int prefix = values[i][1].GetValue<int>();
                if (prefix != 0) items[i].Prefix(prefix);
                items[i].stack = values[i][2].GetValue<int>();
            }
        }
        RestoreItems(player.inventory, meta["inventory"].AsArray());
        RestoreItems(player.armor, meta["armor"].AsArray());
        RestoreItems(player.miscEquips, meta["misc_equips"].AsArray());
        var state = meta["start"];
        player.position = new Vector2(state["x"].GetValue<float>(), state["y"].GetValue<float>());
        player.velocity = new Vector2(state["vx"].GetValue<float>(), state["vy"].GetValue<float>());
        player.statLife = state["hp"].GetValue<int>(); player.gravDir = state["grav_dir"].GetValue<float>();
        player.wingTime = state["wing_time"].GetValue<float>(); player.jump = state["jump"].GetValue<int>();
        player.fallStart = state["fall_start"].GetValue<int>(); player.releaseJump = state["release_jump"].GetValue<bool>();
        for (int i = 0; i < player.ExtraJumps.Length; i++)
        {
            object jump = player.ExtraJumps[i];
            VanillaFrame.RestoreFields(jump, state["extra_jumps"][i].AsObject());
            player.ExtraJumps[i] = (Terraria.DataStructures.ExtraJumpState)jump;
        }
        for (int i = 0; i < player.buffType.Length; i++)
        { player.buffType[i] = state["buff_types"][i].GetValue<int>(); player.buffTime[i] = state["buff_times"][i].GetValue<int>(); }
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        foreach (var item in Main.item) item.active = false;
    }

    static void RestoreSnapshot(Player player, string snapshotPath)
    {
        void RestoreItems(Item[] items, JsonArray values)
        {
            for (int i = 0; i < items.Length; i++)
            {
                items[i].SetDefaults(values[i][0].GetValue<int>());
                int prefix = values[i][1].GetValue<int>();
                if (prefix != 0) items[i].Prefix(prefix);
                items[i].stack = values[i][2].GetValue<int>();
            }
        }
        {
            var capture = JsonNode.Parse(File.ReadAllText(snapshotPath));
            var snapshot = (capture["request"] ?? capture)["snapshot"].AsObject();
            if (snapshot["engine"].GetValue<string>() != VanillaFrame.EngineId
                || snapshot["world_width"].GetValue<int>() != Main.maxTilesX
                || snapshot["world_height"].GetValue<int>() != Main.maxTilesY)
                throw new InvalidOperationException("snapshot_engine_or_world_mismatch");
            int who = player.whoAmI;
            VanillaFrame.RestoreFields(player, snapshot["player"].AsObject());
            player.whoAmI = who;
            RestoreItems(player.inventory, snapshot["inventory"].AsArray());

            RestoreItems(player.miscEquips, snapshot["misc_equips"].AsArray());

            int width = snapshot["area"][2].GetValue<int>();
            var bytes = Convert.FromBase64String(snapshot["tiles"].GetValue<string>());
            var tiles = new int[bytes.Length / sizeof(int)]; Buffer.BlockCopy(bytes, 0, tiles, 0, bytes.Length);
            for (int i = 0; i < tiles.Length / 10; i++)
            {
                var tile = Main.tile[x0 + i % width, y0 + i / width]; int offset = i * 10;
                tile.ClearEverything(); tile.HasTile = tiles[offset] != 0;
                tile.TileType = (ushort)tiles[offset + 1]; tile.WallType = (ushort)tiles[offset + 2];
                tile.TileFrameX = (short)tiles[offset + 3]; tile.TileFrameY = (short)tiles[offset + 4];
                tile.Slope = (SlopeType)tiles[offset + 5]; tile.IsHalfBlock = tiles[offset + 6] != 0;
                tile.IsActuated = tiles[offset + 7] != 0; tile.LiquidType = tiles[offset + 8]; tile.LiquidAmount = (byte)tiles[offset + 9];
            }

        }
        foreach (var npc in Main.npc) npc.active = false;
        foreach (var projectile in Main.projectile) projectile.active = false;
        foreach (var item in Main.item) item.active = false;
    }

    void CaptureStall(Player player)
    {
        if (Vector2.DistanceSquared(player.position, snapshotAnchor) > 32 * 32)

    string TimingPhase(Player player, VanillaNavigationDemo demo)
    {
        if (!VanillaNavBridge.Enabled)
        {
            bool moving = Vector2.DistanceSquared(player.position, previousPosition) > .0625f;
            if (MineCoordinator.IsActive) return moving ? "legacy_moving_mining" : "legacy_mining";
            if (SkillExecutor.IsActive || BridgeBuilder.IsRunning) return "legacy_building";
            if (moving) return "legacy_moving";
            return StateSpacePlanner.ExecRunning ? "legacy_executing" : "legacy_waiting";
        }
        string stage = VanillaNavBridge.Stage;
        if (stage is not "Moving" and not "Approaching") return stage;
        if (!VanillaNavigationDemo.Active) return "field_setup";
        if (!demo.PlannerReady) return "worker_startup";
        if (!demo.LastControl.Queued) return "queue_empty";
        if (Vector2.DistanceSquared(player.position, previousPosition) > .0625f) return "moving";
        return demo.LastControl.Keys == 0 ? "planned_wait" : "blocked_input";
    }

    void RecordPhase(string next, Player player)
    {
        if (next == timingPhase) return;
        double now = routeClock.Elapsed.TotalMilliseconds;
        if (timingPhase != null)
        {
            double duration = now - phaseStart;
            timingTotals.TryGetValue(timingPhase, out double total);
            timingTotals[timingPhase] = total + duration;
            timingSegments.Add(new JsonObject { ["phase"] = timingPhase, ["start_ms"] = phaseStart,
                ["end_ms"] = now, ["duration_ms"] = duration, ["from_x"] = phasePosition.X / 16,
                ["from_y"] = phasePosition.Y / 16, ["to_x"] = player.Center.X / 16,
                ["to_y"] = player.Bottom.Y / 16,
                ["end_plan"] = VanillaNavigationDemo.Instance.LastControl.Plan,
                ["end_rejection"] = VanillaNavigationDemo.Instance.PlannerRejection });
        }
        timingPhase = next; phaseStart = now; phasePosition = new Vector2(player.Center.X, player.Bottom.Y);
    }

    void SaveTiming(bool done, string reason)
    {
        if (!routeClock.IsRunning) return;
        RecordPhase(null, Main.LocalPlayer);
        routeClock.Stop();
        var totals = new JsonObject();
        foreach (var pair in timingTotals) totals[pair.Key] = pair.Value;
        var activity = new JsonObject();
        foreach (var pair in activityTotals) activity[pair.Key] = pair.Value;
        string report = new JsonObject {
            ["done"] = done, ["reason"] = reason, ["wall_ms"] = routeClock.Elapsed.TotalMilliseconds,
            ["game_frames"] = ticks,
            ["goal_x"] = continuity ? null : routeGoalX, ["goal_y"] = continuity ? null : routeGoalY,
            ["phase_ms"] = totals, ["activity_ms"] = activity, ["segments"] = timingSegments.DeepClone(),
            ["plans"] = planTimings.DeepClone() }.ToJsonString();
        File.WriteAllText(Path.Combine(Main.SavePath, "route-timing.json"), report);
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavbaseline") && NavigationRecording.LastDirectory != null)
            File.WriteAllText(Path.Combine(NavigationRecording.LastDirectory, "route-timing.json"), report);
    }

    static void Background(ILContext il)
    {
        var cursor = new ILCursor(il);
        if (!cursor.TryGotoNext(MoveType.After, instruction => instruction.MatchStsfld<Main>("hasFocus")))
            throw new InvalidOperationException("background_pause_boundary_missing");
        if (Terraria.Program.LaunchParameters.ContainsKey("-vnavbackgroundcheck"))
            cursor.EmitDelegate<Action>(() => Main.hasFocus = false);
        if (!cursor.TryGotoNext(MoveType.After, instruction => instruction.MatchLdsfld<Main>("hasFocus")))
            throw new InvalidOperationException("background_pause_boundary_missing");
        cursor.EmitDelegate<Func<bool, bool>>(_ => true);
    }

                { var t = Main.tile[cells[i].tx, y]; t.HasTile = false; t.TileType = types[i]; }
            }

                    throw new InvalidOperationException("preview_leaked_thread");
                using (new PhysicsSimulator.PlatformPreview(new[] { (x, y - 1) }))
                    if (!PhysicsSimulator.PreviewPlatform(x, y - 1)) throw new InvalidOperationException("preview_nested_missing");
                if (PhysicsSimulator.PreviewPlatform(x, y - 1)) throw new InvalidOperationException("preview_nested_leaked");
                var s = start;
                for (int f = 0; f < native.Count; f++)
                {
                    s = PhysicsSimulator.Step(s, input, ph);
                    var n = native[f];
                    if (Math.Abs(s.Px - n.Px) > .01f || Math.Abs(s.Py - n.Py) > .01f

                }
            }
            if (PhysicsSimulator.HasPreviewPlatforms) throw new InvalidOperationException("preview_scope_leaked");

        }

    }

    Point JourneyApproach(Player player)
    {
        var target = journeyStops[continuityLeg];
        if (target.Kind is "main" or "main_return" or "surface") return new Point(target.X, target.Y);
        var tile = Main.tile[target.X, target.Y];
        bool present = tile.HasTile && (target.Kind == "crystal" ? tile.TileType == TileID.Heart
            : target.Kind == "chest" ? TileID.Sets.BasicChest[tile.TileType]
            : tile.TileType == TileID.JunglePlants && tile.TileFrameX == 144);
        if (!present) throw new InvalidOperationException("journey_resource_missing:" + target.Kind);
        return new Point(target.X, target.Y);
    }

    bool JourneyInteraction(Player player)
    {
        var target = journeyStops[continuityLeg];
        if (target.Kind is "main" or "main_return" or "surface") return true;
        if (target.Kind == "chest")
        {
            int chest = Chest.FindChest(target.X, target.Y);
            if (chest < 0 || !player.IsInInteractionRangeToMultiTileHitbox(target.X, target.Y))
                throw new InvalidOperationException("journey_chest_out_of_range");
            player.OpenChest(target.X, target.Y, chest);
            if (player.chest != chest) throw new InvalidOperationException("journey_chest_not_opened");
            Console.WriteLine($"VNAV_JOURNEY_INTERACTION kind=chest chest={chest} tile=({target.X},{target.Y})");
            player.chest = -1;
            return true;
        }
        int count = player.inventory.Where(item => item.type == target.Item).Sum(item => item.stack);
        if (!journeyActionStarted)
        {
            player.selectedItem = 1;
            if (!Reach.CanMine(player, target.X, target.Y)) throw new InvalidOperationException("journey_resource_out_of_range");
            journeyItemBefore = count; journeyActionAt = Main.GameUpdateCount; journeyActionStarted = true;
            ItemUseCoordinator.Start(new ItemUseRequest { TargetWx = target.X, TargetWy = target.Y, Slot = 1, Strict = true });
        }
        if (count > journeyItemBefore)
        {
            ItemUseCoordinator.Stop();
            Console.WriteLine($"VNAV_JOURNEY_INTERACTION kind={target.Kind} item={target.Item} before={journeyItemBefore} after={count}");
            return true;
        }
        if (Main.GameUpdateCount - journeyActionAt > 300)
        {
            File.WriteAllText(Path.Combine(Main.SavePath, "journey-interaction-failure.json"), new JsonObject {
                ["kind"] = target.Kind, ["item"] = target.Item,
                ["before"] = journeyItemBefore, ["after"] = count,
                ["outcome"] = ItemUseCoordinator.Outcome, ["reason"] = ItemUseCoordinator.Reason,
                ["active"] = ItemUseCoordinator.IsActive, ["focused"] = Main.hasFocus,
                ["reachable"] = Reach.CanMine(player, target.X, target.Y),
                ["target_present"] = Main.tile[target.X, target.Y].HasTile,
                ["player"] = VanillaPrediction.Observe(player) }.ToJsonString());
            ItemUseCoordinator.Stop();
            throw new InvalidOperationException("journey_resource_not_collected:" + target.Kind);
        }
        return false;
    }

    public override void PostDrawInterface(SpriteBatch batch)
    {
        if (!enabled || !started || !VanillaNavigationDemo.Active || VanillaNavigationDemo.Instance.RoutePoints < 2) return;
        bool hasRegion = VanillaNavigationDemo.Instance.PlannerResult?["goal_contact_region"] is JsonArray;
        if (shot && (!hasRegion || regionShot)) return;
        var device = Main.instance.GraphicsDevice;
        int width = device.PresentationParameters.BackBufferWidth, height = device.PresentationParameters.BackBufferHeight;
        var pixels = new Color[width * height]; device.GetBackBufferData(pixels);
        using var texture = new Texture2D(device, width, height); texture.SetData(pixels);
        if (!shot)
        {
            using var stream = File.Create(Path.Combine(Main.SavePath, $"demo-{trial}.png")); texture.SaveAsPng(stream, width, height);
        }
        if (hasRegion && !regionShot)
        {
            using var stream = File.Create(Path.Combine(Main.SavePath, "demo-contact-region.png")); texture.SaveAsPng(stream, width, height);
            regionShot = true;
        }
        shot = true;
    }
